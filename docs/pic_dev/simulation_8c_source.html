9/**
10 * @brief Copies the current time step's solution fields into history vectors
11 * (e.g., U(t_n) -> U_o, U_o -> U_rm1) for the next time step's calculations.
12 *
13 * This function is critical for multi-step time integration schemes (like BDF2)
14 * used by the legacy solver. It must be called at the end of every time step,
15 * after the new solution has been fully computed.
16 *
17 * The order of operations is important to avoid overwriting data prematurely.
18 *
19 * @param user The UserCtx for a single block. The function modifies the history
20 * vectors (Ucont_o, Ucont_rm1, etc.) within this context.
21 * @return PetscErrorCode 0 on success.
22 */
23PetscErrorCode UpdateSolverHistoryVectors(UserCtx *user)
24{
25 PetscErrorCode ierr;
26 SimCtx *simCtx = user->simCtx; // Access global settings if needed
27
28 PetscFunctionBeginUser;
29 LOG_ALLOW(LOCAL, LOG_DEBUG, "Rank %d, Block %d: Updating solver history vectors.\n",
30 simCtx->rank, user->_this);
31
32 // --- Primary Contravariant Velocity History ---
33 // The order is critical here.
34 // 1. First, move the n-1 state (Ucont_o) to the n-2 slot (Ucont_rm1).
35 ierr = VecCopy(user->Ucont_o, user->Ucont_rm1); CHKERRQ(ierr);
36 // 2. Then, move the new n state (Ucont) to the n-1 slot (Ucont_o).
37 ierr = VecCopy(user->Ucont, user->Ucont_o); CHKERRQ(ierr);
38
39 LOG_ALLOW(LOCAL,LOG_DEBUG, "Rank %d, Block %d, Ucont history updated.\n",simCtx->rank,user->_this);
40
41 // --- Update History for Other Fields ---
42 // These are typically only needed at the n-1 state.
43 ierr = VecCopy(user->Ucat, user->Ucat_o); CHKERRQ(ierr);
44 ierr = VecCopy(user->P, user->P_o); CHKERRQ(ierr);
45 LOG_ALLOW(LOCAL,LOG_DEBUG, "Rank %d, Block %d, Ucat & P history updated.\n",simCtx->rank,user->_this);
46
47 if (simCtx->immersed) {
48 ierr = VecCopy(user->Nvert, user->Nvert_o); CHKERRQ(ierr);
49 }
50
51 // --- Update History for Turbulence Models (if active) ---
52 if (simCtx->rans) {
53 ierr = VecCopy(user->K_Omega, user->K_Omega_o); CHKERRQ(ierr);
54 }
55
56 // --- Synchronize Local Ghost Regions for the new history vectors ---
57 // This is essential so that stencils in the next time step's calculations
58 // have correct values from neighboring processes.
59 ierr = DMGlobalToLocalBegin(user->fda, user->Ucont_o, INSERT_VALUES, user->lUcont_o); CHKERRQ(ierr);
60 ierr = DMGlobalToLocalEnd(user->fda, user->Ucont_o, INSERT_VALUES, user->lUcont_o); CHKERRQ(ierr);
61
62 ierr = DMGlobalToLocalBegin(user->fda, user->Ucont_rm1, INSERT_VALUES, user->lUcont_rm1); CHKERRQ(ierr);
63 ierr = DMGlobalToLocalEnd(user->fda, user->Ucont_rm1, INSERT_VALUES, user->lUcont_rm1); CHKERRQ(ierr);
64
65 if (simCtx->immersed) {
66 ierr = DMGlobalToLocalBegin(user->da, user->Nvert_o, INSERT_VALUES, user->lNvert_o); CHKERRQ(ierr);
67 ierr = DMGlobalToLocalEnd(user->da, user->Nvert_o, INSERT_VALUES, user->lNvert_o); CHKERRQ(ierr);
68 }
69
70 if (simCtx->rans) {
71 ierr = DMGlobalToLocalBegin(user->fda2, user->K_Omega_o, INSERT_VALUES, user->lK_Omega_o); CHKERRQ(ierr);
72 ierr = DMGlobalToLocalEnd(user->fda2, user->K_Omega_o, INSERT_VALUES, user->lK_Omega_o); CHKERRQ(ierr);
73 }
74
75 PetscFunctionReturn(0);
76}

80/**
81 * @brief Finalizes the simulation setup at t=0, ensuring a consistent state before time marching.
82 *
83 * This function is called from main() after the initial Eulerian and Lagrangian states have been

89 * 3. Preparing for the first step: Scatters particle data back to the grid.
90 * 4. Writing the initial output for step 0.

92 * @param simCtx Pointer to the main simulation context structure.

210/**
211 * @brief Performs post-load/post-init consistency checks for a restarted simulation.
212 *
213 * This function is called from main() ONLY when a restart is being performed
214 * (i.e., StartStep > 0). It inspects the particle restart mode to determine the
215 * correct finalization procedure for the Lagrangian swarm.
216 *
217 * - If particles were loaded from a file (`mode == "load"`), it verifies their
218 * locations within the grid to establish necessary runtime links.
219 * - If new particles were initialized into the restarted flow (`mode == "init"`),
220 * it runs the full `PerformInitialSetup` sequence to migrate, locate, and
221 * couple the new particles with the existing fluid state.
222 *
223 * @param simCtx The main simulation context.
224 * @return PetscErrorCode 0 on success.
225 */
226PetscErrorCode FinalizeRestartState(SimCtx *simCtx)
227{
228 PetscErrorCode ierr;
229 UserCtx *user = simCtx->usermg.mgctx[simCtx->usermg.mglevels - 1].user;
230
231 PetscFunctionBeginUser;
232
233 LOG_ALLOW(GLOBAL, LOG_INFO, "--- Finalizing RESTART from state (step=%d, t=%.4f) ---\n", simCtx->StartStep, simCtx->ti);
234
235 // This function only needs to handle the particle finalization logic.
236 // The Eulerian state is assumed to be fully loaded and consistent at this point.
237 if (simCtx->np > 0) {
238
239 // Use the particle restart mode to decide the workflow.
240 if (strcmp(simCtx->particleRestartMode, "load") == 0) {
241 // PARTICLES WERE LOADED: The state is complete, but we must verify
242 // the loaded CellIDs and build the in-memory grid-to-particle links.
243 LOG_ALLOW(GLOBAL, LOG_INFO, "Particle Mode 'load': Verifying particle locations and building grid links...\n");
244 ierr = PerformLoadedParticleSetup(simCtx); CHKERRQ(ierr);
245
246 } else { // Mode must be "init"
247 // PARTICLES WERE RE-INITIALIZED: They need to be fully settled and coupled
248 // to the surrounding (restarted) fluid state.
249 LOG_ALLOW(GLOBAL, LOG_INFO, "Particle Mode 'init': Running full initial setup for new particles in restarted flow.\n");
250 ierr = PerformInitializedParticleSetup(simCtx); CHKERRQ(ierr);
251 }
252 } else {
253 LOG_ALLOW(GLOBAL, LOG_INFO, "No particles in simulation, restart finalization is complete.\n");
254
255 // Write the initial eulerian fields (this is done in PerformInitialSetup if particles exist.)
256 for(PetscInt bi = 0; bi < simCtx->block_number; bi ++){
257 ierr = WriteSimulationFields(&user[bi]); CHKERRQ(ierr);
258 }
259 }
260
261 LOG_ALLOW(GLOBAL, LOG_INFO, "--- Restart state successfully finalized. --\n");
262
263 PetscFunctionReturn(0);
264}

268/**
269 * @brief Executes the main time-marching loop for the coupled Euler-Lagrange simulation.
270 *
271 * This function orchestrates the advancement of the simulation from the configured
272 * StartStep to the final step. It does NOT perform the initial t=0 setup, as that
273 * is handled by InitializeEulerianState and PerformInitialSetup in main().
274 *
275 * For each timestep, it performs the following sequence:
276 * 1. **Pre-Solver Actions:** Updates time-dependent boundary conditions (e.g., fluxin)
277 * and resets particle states for the new step.
278 * 2. **Eulerian Solve:** Calls the refactored legacy Flow_Solver to advance the
279 * entire fluid field by one time step.
280 * 3. **Lagrangian Update:** Executes the full particle workflow: advection based
281 * on the previous step's velocity, followed by settling (location/migration)
282 * in the new grid, and finally interpolation of the new fluid velocity.
283 * 4. **Two-Way Coupling:** Scatters particle data back to the grid to act as source
284 * terms for the subsequent time step.
285 * 5. **History & I/O:** Updates the solver's history vectors and writes output files
286 * at the specified frequency.
287 *
288 * @param user Array of UserCtx structures for the finest grid level.
289 * @return PetscErrorCode 0 on success, non-zero on failure.
290 */
291PetscErrorCode AdvanceSimulation(SimCtx *simCtx)
292{
293 PetscErrorCode ierr;
294 // Get the master context from the first block. All blocks share it.
295 UserCtx *user = simCtx->usermg.mgctx[simCtx->usermg.mglevels-1].user;
296
297 // Retrieve control parameters from SimCtx for clarity
298 const PetscInt StartStep = simCtx->StartStep;
299 const PetscInt StepsToRun = simCtx->StepsToRun;
300 const PetscInt OutputFreq = simCtx->OutputFreq;
301 const PetscReal dt = simCtx->dt;
302
303 // Variables for particle removal statistics
304 PetscInt removed_local_ob, removed_global_ob;
305 PetscInt removed_local_lost, removed_global_lost;
306
307 PetscFunctionBeginUser;
308 PROFILE_FUNCTION_BEGIN;
309 LOG_ALLOW(GLOBAL, LOG_INFO, "Starting main time-marching loop: %d steps from step %d (t=%.4f), dt=%.4f\n",
310 StepsToRun, StartStep, simCtx->StartTime, dt);
311
312 // --- Main Time-Marching Loop ---
313 for (PetscInt step = StartStep; step < StartStep + StepsToRun; step++) {
314
315 // =================================================================
316 // 1. PRE-STEP SETUP
317 // =================================================================
318
319 // Update simulation time and step counters in the master context
320 simCtx->step = step + 1;
321 simCtx->ti += simCtx->dt; //simCtx->StartTime + step * simCtx->dt;
322
323 LOG_ALLOW(GLOBAL, LOG_INFO, "--- Advancing Step %d (To t=%.4f) ---\n", simCtx->step, simCtx->ti);
324
325 // Update any time-dependent boundary conditions (e.g., pulsating inlet)
326 // if (simCtx->inletprofile == 3) {
327 // LOG_ALLOW(GLOBAL, LOG_DEBUG, "Updating time-dependent inlet flux.\n");
328 // fluxin(&user[0]); // Assumes block 0 drives the global flux value
329 // }
330
331 // For particles, reset their status to prepare for the new advection/location cycle
332 if (simCtx->np > 0) {
333 LOG_ALLOW(GLOBAL, LOG_DEBUG, "Resetting all particle statuses to NEEDS_LOCATION.\n");
334 ierr = ResetAllParticleStatuses(user); CHKERRQ(ierr);
335 }
336
337 // =================================================================
338 // 2. EULERIAN SOLVER STEP
339 // =================================================================
340
341 // Call the refactored, high-level legacy solver. This single function
342 // advances the entire multi-block fluid field from t_n to t_{n+1}.
343 LOG_ALLOW(GLOBAL, LOG_INFO, "Updating Eulerian Field ...\n");
344
345 ierr = Flow_Solver(simCtx); CHKERRQ(ierr);
346
347 LOG_ALLOW(GLOBAL, LOG_INFO, "Eulerian Field solved ...\n");
348 // =================================================================
349 // 3. LAGRANGIAN PARTICLE STEP
350 // =================================================================
351
352 if (simCtx->np > 0) {
353 LOG_ALLOW(GLOBAL, LOG_INFO, "Updating Lagrangian particle system...\n");
354
355 // a. Advect particles using the velocity interpolated from the *previous* step.
356 // P(t_{n+1}) = P(t_n) + V_p(t_n) * dt
357 ierr = UpdateAllParticlePositions(user); CHKERRQ(ierr);
358
359 // b. Settle all particles: find their new host cells and migrate them across ranks.
360 ierr = LocateAllParticlesInGrid_TEST(user, simCtx->bboxlist); CHKERRQ(ierr);
361
362 // c. Remove any particles that are now lost or out of the global domain.
363 ierr = CheckAndRemoveLostParticles(user, &removed_local_lost, &removed_global_lost); CHKERRQ(ierr);
364 ierr = CheckAndRemoveOutOfBoundsParticles(user, &removed_local_ob, &removed_global_ob, simCtx->bboxlist); CHKERRQ(ierr);
365 if (removed_global_lost + removed_global_ob > 0) {
366 LOG_ALLOW(GLOBAL, LOG_INFO, "Removed %d particles globally this step.\n", removed_global_lost + removed_global_ob);
367 }
368
369 // d. Interpolate the NEW fluid velocity (just computed by Flow_Solver) onto the
370 // particles' new positions. This gives them V_p(t_{n+1}) for the *next* advection step.
371 ierr = InterpolateAllFieldsToSwarm(user); CHKERRQ(ierr);
372
373 // e. (For Two-Way Coupling) Scatter particle data back to the grid to act as a source term.
374 ierr = CalculateParticleCountPerCell(user); CHKERRQ(ierr);
375 ierr = ScatterAllParticleFieldsToEulerFields(user); CHKERRQ(ierr);
376 }
377
378 // =================================================================
379 // 4. UPDATE HISTORY & I/O
380 // =================================================================
381
382 // Copy the newly computed fields (Ucont, P, etc.) to the history vectors
383 // (_o, _rm1) to prepare for the next time step.
384 for (PetscInt bi = 0; bi < simCtx->block_number; bi++) {
385 ierr = UpdateSolverHistoryVectors(&user[bi]); CHKERRQ(ierr);
386 }
387
388 // Handle periodic file output
389 if (OutputFreq > 0 && (step + 1) % OutputFreq == 0) {
390 LOG_ALLOW(GLOBAL, LOG_INFO, "Writing output for step %d.\n",simCtx->step);
391 for (PetscInt bi = 0; bi < simCtx->block_number; bi++) {
392 ierr = WriteSimulationFields(&user[bi]); CHKERRQ(ierr);
393 }
394 if (simCtx->np > 0) {
395 ierr = WriteAllSwarmFields(user); CHKERRQ(ierr);
396 if(get_log_level() >= LOG_INFO){
397 LOG_PARTICLE_FIELDS(user,simCtx->LoggingFrequency);
398 }
399 }
400 }
401
402 // Update Progress Bar
403 if(simCtx->rank == 0) {
404 PrintProgressBar(step,StartStep,StepsToRun,simCtx->ti);
405 if(get_log_level()>LOG_ERROR) PetscPrintf(PETSC_COMM_SELF,"\n");
406 }
407 } // --- End of Time-Marching Loop ---
408
409 // After the loop, print the 100% complete bar on rank 0 and add a newline
410 // to ensure subsequent terminal output starts on a fresh line.
411 if (simCtx->rank == 0 && StepsToRun > 0) {
412 PrintProgressBar(StartStep + StepsToRun - 1, StartStep, StepsToRun, simCtx->ti);
413 PetscPrintf(PETSC_COMM_SELF, "\n");
414 fflush(stdout);
415 }
416
417 LOG_ALLOW(GLOBAL, LOG_INFO, "Time marching completed. Final time t=%.4f.\n", simCtx->ti);
418 PROFILE_FUNCTION_END;
419 PetscFunctionReturn(0);
420}
