18{
19 PetscErrorCode ierr;
20 char *pipeline_copy, *token;
21
22 PetscFunctionBeginUser;
23 LOG_ALLOW(GLOBAL, LOG_INFO, "--- Starting Data Transformation Pipeline ---\n");
24
25 // Make a writable copy of the pipeline string for strtok
26 ierr = PetscStrallocpy(pps->process_pipeline, &pipeline_copy); CHKERRQ(ierr);
27
28 token = strtok(pipeline_copy, ",");
29 while (token) {
30 TrimWhitespace(token); // Use the helper from postprocess_config.c
31 LOG_ALLOW(GLOBAL, LOG_INFO, "Executing pipeline step: %s\n", token);
32
33 if (strcmp(token, "ComputeQCriterion") == 0) {
34 ierr = ComputeQCriterion(user); CHKERRQ(ierr);
35 }
36 // Add more kernels here
37 // A more advanced parser could handle arguments like NodalAverage(P->P_nodal)
38 else if (strcmp(token, "NodalAverage_P") == 0) { // Simple name for now
39 ierr = ComputeNodalAverage(user, "P", "P_nodal"); CHKERRQ(ierr);
40 }
41 else {
42 LOG_ALLOW(GLOBAL, LOG_WARNING, "Unknown processing kernel '%s' requested. Skipping.\n", token);
43 }
44
45 token = strtok(NULL, ",");
46 }
47
48 ierr = PetscFree(pipeline_copy); CHKERRQ(ierr);
49 LOG_ALLOW(GLOBAL, LOG_INFO, "--- Data Transformation Pipeline Complete ---\n");
50 PetscFunctionReturn(0);
51}
