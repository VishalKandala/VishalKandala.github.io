99/**
100 * @brief Computes the Q-Criterion, a scalar value identifying vortex cores.
101 *
102 * This function is self-contained. It ensures all its required input data
103 * (Ucat and grid metrics) have up-to-date ghost values before proceeding with
104 * the calculation. The result is stored in the global vector user->Qcrit.

106 * @param user The UserCtx, providing access to all necessary data.
107 * @return PetscErrorCode

122 ierr = UpdateLocalGhosts(user, "Csi"); CHKERRQ(ierr);
123 ierr = UpdateLocalGhosts(user, "Eta"); CHKERRQ(ierr);
124 ierr = UpdateLocalGhosts(user, "Zet"); CHKERRQ(ierr);
125 ierr = UpdateLocalGhosts(user, "Aj"); CHKERRQ(ierr);
126 ierr = UpdateLocalGhosts(user, "Nvert"); CHKERRQ(ierr);
127

129 ierr = DMDAGetLocalInfo(user->da, &info); CHKERRQ(ierr);
130
131 ierr = DMDAVecGetArrayRead(user->fda, user->lUcat, (void*)&lucat); CHKERRQ(ierr);
132 ierr = DMDAVecGetArrayRead(user->fda, user->lCsi, (void*)&lcsi); CHKERRQ(ierr);
133 ierr = DMDAVecGetArrayRead(user->fda, user->lEta, (void*)&leta); CHKERRQ(ierr);
134 ierr = DMDAVecGetArrayRead(user->fda, user->lZet, (void*)&lzet); CHKERRQ(ierr);

136 ierr = DMDAVecGetArrayRead(user->da, user->lNvert, (void*)&lnvert); CHKERRQ(ierr);
137 ierr = DMDAVecGetArray(user->da, user->Qcrit, (void*)&gq); CHKERRQ(ierr);

139 // --- 3. Define Loop Bounds for INTERIOR Cells ---
140 PetscInt i_start = (info.xs == 0) ? 1 : info.xs;
141 PetscInt i_end = (info.xs + info.xm == info.mx) ? info.mx - 1 : info.xs + info.xm;
142 PetscInt j_start = (info.ys == 0) ? 1 : info.ys;
143 PetscInt j_end = (info.ys + info.ym == info.my) ? info.my - 1 : info.ys + info.ym;
144 PetscInt k_start = (info.zs == 0) ? 1 : info.zs;
145 PetscInt k_end = (info.zs + info.zm == info.mz) ? info.mz - 1 : info.zs + info.zm;
146
147 // --- 4. Main Computation Loop ---
148 for (PetscInt k = k_start; k < k_end; k++) {
149 for (PetscInt j = j_start; j < j_end; j++) {
150 for (PetscInt i = i_start; i < i_end; i++) {
151
152 // Calculate velocity derivatives in computational space (central differences)
153 PetscReal uc = 0.5 * (lucat[k][j][i+1].x - lucat[k][j][i-1].x);
154 PetscReal vc = 0.5 * (lucat[k][j][i+1].y - lucat[k][j][i-1].y);
155 PetscReal wc = 0.5 * (lucat[k][j][i+1].z - lucat[k][j][i-1].z);
156
157 PetscReal ue = 0.5 * (lucat[k][j+1][i].x - lucat[k][j-1][i].x);
158 PetscReal ve = 0.5 * (lucat[k][j+1][i].y - lucat[k][j-1][i].y);
159 PetscReal we = 0.5 * (lucat[k][j+1][i].z - lucat[k][j-1][i].z);
160
161 PetscReal uz = 0.5 * (lucat[k+1][j][i].x - lucat[k-1][j][i].x);
162 PetscReal vz = 0.5 * (lucat[k+1][j][i].y - lucat[k-1][j][i].y);
163 PetscReal wz = 0.5 * (lucat[k+1][j][i].z - lucat[k-1][j][i].z);
164
165 // Average metrics to the cell center
166 PetscReal csi1 = 0.5 * (lcsi[k][j][i].x + lcsi[k][j][i-1].x) * laj[k][j][i];
167 PetscReal csi2 = 0.5 * (lcsi[k][j][i].y + lcsi[k][j][i-1].y) * laj[k][j][i];
168 PetscReal csi3 = 0.5 * (lcsi[k][j][i].z + lcsi[k][j][i-1].z) * laj[k][j][i];
169
170 PetscReal eta1 = 0.5 * (leta[k][j][i].x + leta[k][j-1][i].x) * laj[k][j][i];
171 PetscReal eta2 = 0.5 * (leta[k][j][i].y + leta[k][j-1][i].y) * laj[k][j][i];
172 PetscReal eta3 = 0.5 * (leta[k][j][i].z + leta[k][j-1][i].z) * laj[k][j][i];
173
174 PetscReal zet1 = 0.5 * (lzet[k][j][i].x + lzet[k-1][j][i].x) * laj[k][j][i];
175 PetscReal zet2 = 0.5 * (lzet[k][j][i].y + lzet[k-1][j][i].y) * laj[k][j][i];
176 PetscReal zet3 = 0.5 * (lzet[k][j][i].z + lzet[k-1][j][i].z) * laj[k][j][i];
177
178 // Calculate velocity gradient tensor components d_ij = du_i/dx_j
179 PetscReal d11 = uc * csi1 + ue * eta1 + uz * zet1;
180 PetscReal d12 = uc * csi2 + ue * eta2 + uz * zet2;
181 PetscReal d13 = uc * csi3 + ue * eta3 + uz * zet3;
182
183 PetscReal d21 = vc * csi1 + ve * eta1 + vz * zet1;
184 PetscReal d22 = vc * csi2 + ve * eta2 + vz * zet2;
185 PetscReal d23 = vc * csi3 + ve * eta3 + vz * zet3;
186
187 PetscReal d31 = wc * csi1 + we * eta1 + wz * zet1;
188 PetscReal d32 = wc * csi2 + we * eta2 + wz * zet2;
189 PetscReal d33 = wc * csi3 + we * eta3 + wz * zet3;
190
191 // Strain-Rate Tensor S_ij = 0.5 * (d_ij + d_ji)
192 PetscReal s11 = d11;
193 PetscReal s12 = 0.5 * (d12 + d21);
194 PetscReal s13 = 0.5 * (d13 + d31);
195 PetscReal s22 = d22;
196 PetscReal s23 = 0.5 * (d23 + d32);
197 PetscReal s33 = d33;
198
199 // Vorticity Tensor Omega_ij = 0.5 * (d_ij - d_ji)
200 PetscReal w12 = 0.5 * (d12 - d21);
201 PetscReal w13 = 0.5 * (d13 - d31);
202 PetscReal w23 = 0.5 * (d23 - d32);
203
204 // Squared norms of the tensors
205 PetscReal s_norm_sq = s11*s11 + s22*s22 + s33*s33 + 2.0*(s12*s12 + s13*s13 + s23*s23);
206 PetscReal w_norm_sq = 2.0 * (w12*w12 + w13*w13 + w23*w23);
207
208 gq[k][j][i] = 0.5 * (w_norm_sq - s_norm_sq);
209
210 if (lnvert[k][j][i] > 0.1) {
211 gq[k][j][i] = 0.0;
212 }
213 }
214 }
215 }
216
217 // --- 5. Restore arrays ---
218 ierr = DMDAVecRestoreArrayRead(user->fda, user->lUcat, (void*)&lucat); CHKERRQ(ierr);
219 ierr = DMDAVecRestoreArrayRead(user->fda, user->lCsi, (void*)&lcsi); CHKERRQ(ierr);
220 ierr = DMDAVecRestoreArrayRead(user->fda, user->lEta, (void*)&leta); CHKERRQ(ierr);
221 ierr = DMDAVecRestoreArrayRead(user->fda, user->lZet, (void*)&lzet); CHKERRQ(ierr);
222 ierr = DMDAVecRestoreArrayRead(user->da, user->lAj, (void*)&laj); CHKERRQ(ierr);
223 ierr = DMDAVecRestoreArrayRead(user->da, user->lNvert, (void*)&lnvert); CHKERRQ(ierr);
224 ierr = DMDAVecRestoreArray(user->da, user->Qcrit, (void*)&gq); CHKERRQ(ierr);
225
226 PetscFunctionReturn(0);
