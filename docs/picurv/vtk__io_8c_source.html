19 * @brief Write the XML header for a VTK structured-grid (`.vts`) file.
20 */
21static PetscErrorCode WriteVTSXMLHeader(FILE *fp, const VTKMetaData *meta, PetscInt *boffset)
22{
23 PROFILE_HELPER_BEGIN;
24 const char *byte_order = "LittleEndian";
25 const char *precision = "Float64";
26
27 const PetscInt header = (PetscInt)sizeof(uint64_t);
28
29 fprintf(fp, "<?xml version=\"1.0\"?>\n");
30 fprintf(fp, "<VTKFile type=\"StructuredGrid\" version=\"1.0\" byte_order=\"%s\" header_type=\"UInt64\">\n", byte_order);
31 fprintf(fp, " <StructuredGrid WholeExtent=\"0 %" PetscInt_FMT " 0 %" PetscInt_FMT " 0 %" PetscInt_FMT "\">\n", meta->mx - 1, meta->my - 1, meta->mz - 1);
32 fprintf(fp, " <Piece Extent=\"0 %" PetscInt_FMT " 0 %" PetscInt_FMT " 0 %" PetscInt_FMT "\">\n", meta->mx - 1, meta->my - 1, meta->mz - 1);
33
34 fprintf(fp, " <Points>\n");
35 fprintf(fp, " <DataArray type=\"%s\" Name=\"Position\" NumberOfComponents=\"3\" format=\"appended\" offset=\"%" PetscInt_FMT "\" />\n",
36 precision, *boffset);
37 *boffset += header + 3 * meta->npoints * (PetscInt)sizeof(PetscScalar);
38 fprintf(fp, " </Points>\n");
39
40 if (meta->num_point_data_fields > 0) {
41 fprintf(fp, " <PointData>\n");
42 for (PetscInt i = 0; i < meta->num_point_data_fields; i++) {
43 const VTKFieldInfo* field = &meta->point_data_fields[i];
44 fprintf(fp, " <DataArray type=\"%s\" Name=\"%s\" NumberOfComponents=\"%" PetscInt_FMT "\" format=\"appended\" offset=\"%" PetscInt_FMT "\" />\n",
45 precision, field->name, field->num_components, *boffset);
46 *boffset += header + field->num_components * meta->npoints * (PetscInt)sizeof(PetscScalar);
47 }
48 fprintf(fp, " </PointData>\n");
49 }
50
51 fprintf(fp, " </Piece>\n");
52 fprintf(fp, " </StructuredGrid>\n");
53 fprintf(fp, " <AppendedData encoding=\"raw\">\n_");

55 PROFILE_HELPER_END;

127/**
128 * @brief Copy this rank's part of the output grid out of a DMDA global vector.
129 * @details The output grid is the first `extent` nodes on each axis (the last DA node is a
130 * dummy). This rank's part is its owned box clipped to that grid, copied point by
131 * point (i fastest) with the vector's components interleaved.
132 */
133static PetscErrorCode CopyOwnedOutputBox(Vec vec, const PetscInt extent[3], PetscInt start[3], PetscInt size[3],
134 PetscScalar **out)
135{
136 DM dm = NULL;
137 DMDALocalInfo info;
138 const PetscScalar *values = NULL;
139 PetscInt p = 0;
140
141 PetscFunctionBeginUser;
142 PROFILE_FUNCTION_BEGIN;
143 PetscCall(VecGetDM(vec, &dm));
144 PetscCheck(dm != NULL, PETSC_COMM_SELF, PETSC_ERR_ARG_WRONG, "VTK output needs a vector attached to a DMDA.");
145 PetscCall(DMDAGetLocalInfo(dm, &info));
146 const PetscInt owned_start[3] = {info.xs, info.ys, info.zs}, owned[3] = {info.xm, info.ym, info.zm};
147 for (int d = 0; d < 3; ++d) {
148 start[d] = owned_start[d];
149 size[d] = PetscMax(0, PetscMin(owned_start[d] + owned[d], extent[d]) - owned_start[d]);
150 }
151 PetscCall(PetscMalloc1(info.dof * size[0] * size[1] * size[2], out));
152 PetscCall(VecGetArrayRead(vec, &values));
153 for (PetscInt k = 0; k < size[2]; ++k)
154 for (PetscInt j = 0; j < size[1]; ++j)
155 for (PetscInt i = 0; i < size[0]; ++i)
156 for (PetscInt c = 0; c < info.dof; ++c)
157 (*out)[p++] = values[((k * owned[1] + j) * owned[0] + i) * info.dof + c];
158 PetscCall(VecRestoreArrayRead(vec, &values));
159 PROFILE_FUNCTION_END;
160 PetscFunctionReturn(0);
161}

165/**
166 * @brief Implementation of \ref CreateVTKFileFromMetadata().
167 * @details Full API contract (arguments, ownership, side effects) is documented with
168 * the header declaration in `include/vtk_io.h`.
169 * @see CreateVTKFileFromMetadata()
170 */
171
172PetscErrorCode CreateVTKFileFromMetadata(const char *filename, const VTKMetaData *meta, MPI_Comm comm)
173{
174 static const char footer[] = "\n </AppendedData>\n</VTKFile>\n";
175 const PetscBool structured = (PetscBool)(meta->fileType == VTK_STRUCTURED);
176 char tmp_filename[PETSC_MAX_PATH_LEN];
177 PetscMPIInt rank;
178 long long header_bytes = 0; /* -1: rank 0 could not write the header */
179 MPI_File fh;
180 MPI_Offset position;
181 PetscInt blocks = 0, local_points = 1, boffset = 0;
182 struct VTKBlock { PetscInt components; MPI_Datatype type; size_t size; const void *data; } block[MAX_POINT_DATA_FIELDS + 3];
183
184 PetscFunctionBeginUser;
185 PROFILE_FUNCTION_BEGIN;
186 PetscCheck(structured || meta->fileType == VTK_POLYDATA, comm, PETSC_ERR_ARG_WRONG, "Unknown VTK file type.");
187 PetscCallMPI(MPI_Comm_rank(comm, &rank));
188 PetscCall(PetscSNPrintf(tmp_filename, sizeof(tmp_filename), "%s.tmp", filename));
189
190 /* The appended blocks, in the order the header declares them. */
191 block[blocks++] = (struct VTKBlock){3, MPIU_SCALAR, sizeof(PetscScalar), meta->coords};
192 for (PetscInt f = 0; f < meta->num_point_data_fields; ++f)
193 block[blocks++] = (struct VTKBlock){meta->point_data_fields[f].num_components, MPIU_SCALAR, sizeof(PetscScalar),
194 meta->point_data_fields[f].data};
195 if (!structured) {
196 block[blocks++] = (struct VTKBlock){1, MPIU_INT, sizeof(PetscInt), meta->connectivity};
197 block[blocks++] = (struct VTKBlock){1, MPIU_INT, sizeof(PetscInt), meta->offsets};
198 }
199 for (int d = 0; d < (structured ? 3 : 1); ++d) local_points *= meta->local_size[d];
200
201 if (rank == 0) {
202 FILE *fp = fopen(tmp_filename, "wb");
203 if (!fp || WriteVTKFileHeader(fp, meta, &boffset)) header_bytes = -1;
204 else header_bytes = (long long)ftell(fp);
205 if (fp && fclose(fp)) header_bytes = -1;
206 }
207 PetscCallMPI(MPI_Bcast(&header_bytes, 1, MPI_LONG_LONG, 0, comm));
208 PetscCheck(header_bytes >= 0, comm, PETSC_ERR_FILE_OPEN, "Could not write the VTK header of '%s'.", tmp_filename);
209
210 /* Rank 0 writes each block's byte count; every rank writes its own points into the block. */
211 PetscCallMPI(MPI_File_open(comm, tmp_filename, MPI_MODE_WRONLY, MPI_INFO_NULL, &fh));
212 position = (MPI_Offset)header_bytes;
213 for (PetscInt b = 0; b < blocks; ++b) {
214 const uint64_t bytes = (uint64_t)block[b].components * (uint64_t)meta->npoints * block[b].size;
215 MPI_Datatype point, view;
216 PetscMPIInt count;
217
218 PetscCallMPI(MPI_File_set_view(fh, 0, MPI_BYTE, MPI_BYTE, "native", MPI_INFO_NULL));
219 if (rank == 0) PetscCallMPI(MPI_File_write_at(fh, position, &bytes, (int)sizeof(bytes), MPI_BYTE, MPI_STATUS_IGNORE));
220 position += (MPI_Offset)sizeof(bytes);
221
222 PetscCallMPI(MPI_Type_contiguous((PetscMPIInt)block[b].components, block[b].type, &point));
223 PetscCallMPI(MPI_Type_commit(&point));
224 view = point;
225 if (structured && local_points > 0) {
226 const PetscMPIInt sizes[3] = {(PetscMPIInt)meta->mz, (PetscMPIInt)meta->my, (PetscMPIInt)meta->mx};
227 const PetscMPIInt subsizes[3] = {(PetscMPIInt)meta->local_size[2], (PetscMPIInt)meta->local_size[1], (PetscMPIInt)meta->local_size[0]};
228 const PetscMPIInt starts[3] = {(PetscMPIInt)meta->local_start[2], (PetscMPIInt)meta->local_start[1], (PetscMPIInt)meta->local_start[0]};
229 PetscCallMPI(MPI_Type_create_subarray(3, sizes, subsizes, starts, MPI_ORDER_C, point, &view));
230 PetscCallMPI(MPI_Type_commit(&view));
231 }
232 PetscCallMPI(MPI_File_set_view(fh, position, point, view, "native", MPI_INFO_NULL));
233 PetscCall(PetscMPIIntCast(local_points, &count));
234 if (structured) PetscCallMPI(MPI_File_write_all(fh, block[b].data, count, point, MPI_STATUS_IGNORE));
235 else PetscCallMPI(MPI_File_write_at_all(fh, (MPI_Offset)meta->local_start[0], block[b].data, count, point, MPI_STATUS_IGNORE));
236 if (view != point) PetscCallMPI(MPI_Type_free(&view));
237 PetscCallMPI(MPI_Type_free(&point));
238 position += (MPI_Offset)bytes;
239 }
240 PetscCallMPI(MPI_File_set_view(fh, 0, MPI_BYTE, MPI_BYTE, "native", MPI_INFO_NULL));
241 if (rank == 0) PetscCallMPI(MPI_File_write_at(fh, position, footer, (int)strlen(footer), MPI_BYTE, MPI_STATUS_IGNORE));
242 PetscCallMPI(MPI_File_close(&fh));
243 if (rank == 0) PetscCheck(rename(tmp_filename, filename) == 0, PETSC_COMM_SELF, PETSC_ERR_FILE_WRITE,
244 "Could not move '%s' to '%s'.", tmp_filename, filename);
245 LOG_ALLOW(GLOBAL, LOG_INFO, "Wrote VTK file '%s'.\n", filename);
246 PROFILE_FUNCTION_END;
247 PetscFunctionReturn(0);
248}

378/**
379 * @brief Internal helper implementation: `PrepareOutputParticleData()`.
380 * @details Local to this translation unit.
381 */
382PetscErrorCode PrepareOutputParticleData(UserCtx* user, PostProcessParams* pps, VTKMetaData* meta, PetscInt* p_n_total)
383{
384 const PetscInt stride = pps->particle_output_freq > 0 ? pps->particle_output_freq : 1;
385 PetscInt nlocal = 0, first = 0, before = 0, kept = 0, bs = 0;
386 PetscDataType type;
387 PetscReal *position = NULL;
388 char *fields_copy = NULL, *field_name = NULL;
389
390 PetscFunctionBeginUser;
391 PROFILE_HELPER_BEGIN;
392
393 /* Particles are numbered in rank order; output keeps every stride-th one, so this rank
394 keeps its particles from local index `first` on, written from output point `before`. */
395 PetscCall(DMSwarmGetLocalSize(user->swarm, &nlocal));
396 PetscCallMPI(MPI_Exscan(&nlocal, &before, 1, MPIU_INT, MPI_SUM, PETSC_COMM_WORLD));
397 if (user->simCtx->rank == 0) before = 0;
398 PetscCall(DMSwarmGetSize(user->swarm, p_n_total));
399 if (*p_n_total == 0) { PROFILE_HELPER_END; PetscFunctionReturn(0); }
400 first = (stride - before % stride) % stride;
401 kept = first < nlocal ? (nlocal - 1 - first) / stride + 1 : 0;
402
403 meta->fileType = VTK_POLYDATA;
404 meta->npoints = (*p_n_total - 1) / stride + 1;
405 meta->local_start[0] = (before + first) / stride;
406 meta->local_size[0] = kept;
407 LOG_ALLOW(GLOBAL, LOG_DEBUG, "Subsampling %" PetscInt_FMT " particles with stride %" PetscInt_FMT " -> %" PetscInt_FMT " output particles.\n",
408 *p_n_total, stride, meta->npoints);
409
410 PetscCall(DMSwarmGetField(user->swarm, ParticleFieldName(PARTICLE_FIELD_ID_POSITION), &bs, &type, (void **)&position));
411 PetscCheck(type == PETSC_REAL && bs == 3, PETSC_COMM_SELF, PETSC_ERR_ARG_WRONG,
412 "Particle position field must be 3 PETSC_REAL components, not %" PetscInt_FMT " of %s.", bs, PetscDataTypes[type]);
413 PetscCall(PetscMalloc1(3 * kept, &meta->coords));
414 for (PetscInt i = 0; i < kept; ++i)
415 for (int d = 0; d < 3; ++d) meta->coords[3 * i + d] = position[3 * (first + i * stride) + d];
416 PetscCall(DMSwarmRestoreField(user->swarm, ParticleFieldName(PARTICLE_FIELD_ID_POSITION), &bs, &type, (void **)&position));
417
418 PetscCall(PetscStrallocpy(pps->particle_fields, &fields_copy));
419 for (field_name = strtok(fields_copy, ","); field_name && meta->num_point_data_fields < MAX_POINT_DATA_FIELDS;
420 field_name = strtok(NULL, ",")) {
421 DM swarm_to_use = user->swarm;
422 const char *internal_name = field_name;
423 PetscErrorCode check_ierr;
424 PetscInt field_local = 0, field_total = 0;
425 void *source = NULL;
426
427 TrimWhitespace(field_name);
428 if (!*field_name || strcasecmp(field_name, "position") == 0) continue;
429
430 /* A field registered on the post-processing swarm comes from there, otherwise from the solver swarm. */
431 PetscCall(PetscPushErrorHandler(PetscIgnoreErrorHandler, NULL));
432 check_ierr = DMSwarmGetField(user->post_swarm, field_name, NULL, NULL, NULL);
433 PetscCall(PetscPopErrorHandler());
434 if (!check_ierr) {
435 swarm_to_use = user->post_swarm;
436 PetscCall(DMSwarmRestoreField(user->post_swarm, field_name, NULL, NULL, NULL));
437 LOG_ALLOW(GLOBAL, LOG_DEBUG, "Field '%s' will be sourced from the post-processing swarm.\n", field_name);
438 } else {
439 LOG_ALLOW(GLOBAL, LOG_DEBUG, "Field '%s' will be sourced from the main solver swarm.\n", field_name);
440 if (strcasecmp(field_name, "pid") == 0) internal_name = "DMSwarm_pid";
441 else if (strcasecmp(field_name, "CellID") == 0) internal_name = "DMSwarm_CellID";
442 else if (strcasecmp(field_name, "Migration Status") == 0) internal_name = "DMSwarm_location_status";
443 }
444 PetscCall(DMSwarmGetLocalSize(swarm_to_use, &field_local));
445 PetscCall(DMSwarmGetSize(swarm_to_use, &field_total));
446 if (field_total != *p_n_total) {
447 LOG_ALLOW(GLOBAL, LOG_WARNING, "Field '%s' has %" PetscInt_FMT " particles, but expected %" PetscInt_FMT ". Skipping.\n",
448 field_name, field_total, *p_n_total);
449 continue;
450 }
451 PetscCheck(field_local == nlocal, PETSC_COMM_SELF, PETSC_ERR_ARG_SIZ,
452 "Field '%s' holds %" PetscInt_FMT " particles on this rank, the solver swarm %" PetscInt_FMT ".",
453 field_name, field_local, nlocal);
454
455 VTKFieldInfo *current = &meta->point_data_fields[meta->num_point_data_fields];
456 PetscCall(DMSwarmGetField(swarm_to_use, internal_name, &bs, &type, &source));
