357/**
358 * @brief Returns the preserved provisional frozen-momentum point block.
359 *
360 * The arithmetic in this function is carried forward unchanged for a separate
361 * numerical audit. `A` retains its six existing directional face groups.
362 */
363static void FrozenMomentumJacobian_PointBlock(const SimCtx *simCtx,
364 const Cmpnts ***ucont, const Cmpnts ***csi, const Cmpnts ***eta,
365 const Cmpnts ***zet, const PetscReal ***aj, PetscInt i, PetscInt j,
366 PetscInt k, PetscScalar block[9])
367{
368 const PetscReal dtc = MomentumBDFCoefficient((SimCtx *)simCtx) / simCtx->dt;
369 const PetscReal nu = simCtx->ren > 0.0 ? 1.0 / simCtx->ren : 0.0;
370 const PetscReal AJip = aj[k][j][i];
371 const PetscReal AJjp = aj[k][j][i];
372 const PetscReal AJkp = aj[k][j][i];
373 const PetscReal g11ip = csi[k][j][i].x * csi[k][j][i].x +
374 csi[k][j][i].y * csi[k][j][i].y +
375 csi[k][j][i].z * csi[k][j][i].z;
376 const PetscReal g22jp = eta[k][j][i].x * eta[k][j][i].x +
377 eta[k][j][i].y * eta[k][j][i].y +
378 eta[k][j][i].z * eta[k][j][i].z;
379 const PetscReal g33kp = zet[k][j][i].x * zet[k][j][i].x +
380 zet[k][j][i].y * zet[k][j][i].y +
381 zet[k][j][i].z * zet[k][j][i].z;
382 const PetscReal U1ip = ucont[k][j][i].y, U2ip = ucont[k][j][i].z;
383 const PetscReal U0jp = ucont[k][j][i].x, U2jp = ucont[k][j][i].z;
384 const PetscReal U0kp = ucont[k][j][i].x, U1kp = ucont[k][j][i].y;
385 PetscReal A[6][4] = {{0.0}};
386 PetscReal Su, Sv, Sw, nui, nuj, nuk;
387
388 /* Preserved center-only bookkeeping: six directional face groups. */
389 A[0][0] = 0.5 * AJip * ucont[k][j][i].x;
390 A[1][0] = 0.5 * AJip * ucont[k][j][i - 1].x;
391 A[2][1] = 0.5 * AJjp * ucont[k][j][i].y;
392 A[3][1] = 0.5 * AJjp * ucont[k][j - 1][i].y;
393 A[4][2] = 0.5 * AJkp * ucont[k][j][i].z;
394 A[5][2] = 0.5 * AJkp * ucont[k - 1][j][i].z;
395 Su = A[0][0] - A[1][0];
396 Sv = A[2][1] - A[3][1];
397 Sw = A[4][2] - A[5][2];
398 nui = 2.0 * nu * AJip * g11ip;
399 nuj = 2.0 * nu * AJjp * g22jp;
400 nuk = 2.0 * nu * AJkp * g33kp;
401
402 /* Preserve all existing signs and coefficient formulas verbatim. */
403 block[0] = dtc + nui + Su; block[1] = 0.5 * AJip * U1ip; block[2] = 0.5 * AJip * U2ip;
404 block[3] = 0.5 * AJjp * U0jp; block[4] = dtc + nuj + Sv; block[5] = 0.5 * AJjp * U2jp;
405 block[6] = 0.5 * AJkp * U0kp; block[7] = 0.5 * AJkp * U1kp; block[8] = dtc + nuk + Sw;
406}

475#define __FUNCT__ "MomentumPreconditionerEngine_ApplyConstraintRows"
476/** @brief Inserts all common fixed, homogeneous, and periodic-duplicate rows. */
477static PetscErrorCode MomentumPreconditionerEngine_ApplyConstraintRows(
478 UserCtx *user, Mat preconditioning_matrix)
479{
480 DMDALocalInfo info = user->info;
481
482 PetscFunctionBeginUser;
483 for (PetscInt k = info.zs; k < info.zs + info.zm; ++k) {
484 for (PetscInt j = info.ys; j < info.ys + info.ym; ++j) {
485 for (PetscInt i = info.xs; i < info.xs + info.xm; ++i) {
486 for (PetscInt component = 0; component < 3; ++component) {
487 PetscInt ri, rj, rk;
488 MomentumNewtonKrylovRowType type = MomentumNewtonKrylov_ClassifyRow(
489 user, i, j, k, component, &ri, &rj, &rk);
490 if (type != MOM_NK_ROW_PHYSICAL) {
491 const PetscScalar one = 1.0, minus_one = -1.0;
492 AO ao = NULL;
493 PetscInt global_row = 3 * ((k * info.my + j) * info.mx + i) + component;
494 PetscCall(DMDAGetAO(user->fda, &ao));
495 PetscCall(AOApplicationToPetsc(ao, 1, &global_row));
496 PetscCall(MatSetValue(preconditioning_matrix, global_row, global_row,
497 one, INSERT_VALUES));
498 if (type == MOM_NK_ROW_PERIODIC_DUPLICATE) {
499 PetscInt global_col = 3 * ((rk * info.my + rj) * info.mx + ri) + component;
500 PetscCall(AOApplicationToPetsc(ao, 1, &global_col));
501 PetscCall(MatSetValue(preconditioning_matrix, global_row, global_col,
502 minus_one, INSERT_VALUES));
503 }
504 }
505 }
506 }
507 }
508 }
509 PetscFunctionReturn(PETSC_SUCCESS);
510}

677#undef __FUNCT__
678#define __FUNCT__ "MomentumNewtonKrylov_ClassifyRow"
679/**
680 * @brief Classifies one stored staggered component row and its periodic representative.
681 *
682 * Negative nonperiodic planes and positive dummy planes are zeroed by the legacy
683 * residual treatment. Only a face-normal row actually written by its boundary
684 * handler is conditioned; the remaining zeroed rows use F=X. Periodic endpoint
685 * rows use the same wrapped representatives as SynchronizePeriodicStaggeredFields().
686 * At periodic/nonperiodic intersections the periodic equation owns rows that a
687 * shrunken boundary-handler loop does not condition.
688 *
689 * @param user Block context containing boundary metadata and global dimensions.
690 * @param i Global i index.
691 * @param j Global j index.
692 * @param k Global k index.
693 * @param component Stored component, 0=x, 1=y, 2=z.
694 * @param ri Returned periodic representative i index.
695 * @param rj Returned periodic representative j index.
696 * @param rk Returned periodic representative k index.
697 * @return Exact version-one row category.
698 */
699static MomentumNewtonKrylovRowType MomentumNewtonKrylov_ClassifyRow(
700 UserCtx *user, PetscInt i, PetscInt j, PetscInt k, PetscInt component,
701 PetscInt *ri, PetscInt *rj, PetscInt *rk)
702{
703 const PetscInt mx = user->info.mx, my = user->info.my, mz = user->info.mz;
704 const PetscInt coord[3] = {i, j, k};
705 const PetscInt size[3] = {mx, my, mz};
706 const BCFace neg_face[3] = {BC_FACE_NEG_X, BC_FACE_NEG_Y, BC_FACE_NEG_Z};
707 PetscBool periodic[3], periodic_duplicate = PETSC_FALSE;
708 PetscBool residual_zeroed = PETSC_FALSE, conditioned = PETSC_FALSE;
709
710 *ri = i; *rj = j; *rk = k;
711 for (PetscInt axis = 0; axis < 3; ++axis) {
712 periodic[axis] = (PetscBool)(
713 user->boundary_faces[neg_face[axis]].mathematical_type == PERIODIC);
714 if (periodic[axis] && coord[axis] == 0) {
715 periodic_duplicate = PETSC_TRUE;
716 if (axis == 0) *ri = -2;
717 else if (axis == 1) *rj = -2;
718 else *rk = -2;
719 }
720 if (periodic[axis] && coord[axis] == size[axis] - 1) {
721 periodic_duplicate = PETSC_TRUE;
722 if (axis == 0) *ri = mx + 1;
723 else if (axis == 1) *rj = my + 1;
724 else *rk = mz + 1;
725 }
726
727 if (!periodic[axis] && coord[axis] == 0) residual_zeroed = PETSC_TRUE;
728 if (coord[axis] == size[axis] - 1) residual_zeroed = PETSC_TRUE;
729 if (!periodic[axis] && coord[axis] == size[axis] - 2 && component == axis)
730 residual_zeroed = PETSC_TRUE;
731 }
732
733 if (!periodic[component] &&
734 (coord[component] == 0 || coord[component] == size[component] - 2)) {
735 PetscBool tangential_interior = PETSC_TRUE;
736 for (PetscInt axis = 0; axis < 3; ++axis) {
737 if (axis == component) continue;
738 if (coord[axis] < 1 || coord[axis] > size[axis] - 2)
739 tangential_interior = PETSC_FALSE;
740 }
741 conditioned = tangential_interior;
742 }
743
744 if (conditioned) return MOM_NK_ROW_FIXED_CONDITIONED;
745 if (periodic_duplicate) return MOM_NK_ROW_PERIODIC_DUPLICATE;
746 if (residual_zeroed) return MOM_NK_ROW_FIXED_HOMOGENEOUS;
747 return MOM_NK_ROW_PHYSICAL;
748}

750#undef __FUNCT__
751#define __FUNCT__ "MomentumNewtonKrylov_ApplyConstraints"
752/**
753 * @brief Replaces every non-independent residual row with an explicit equation.
754 *
755 * Conditioned face-normal rows use F=X-Uconditioned, unconditioned legacy
756 * dummy/tangential rows use F=X, and periodic duplicates use Fdup=Xdup-Xrep.
757 * These equations prevent the zero Jacobian rows produced by simply retaining
758 * EnforceRHSBoundaryConditions() zeros in a matrix-free Newton operator. Immersed,
759 * masked, TwoD, and interface rows are rejected before this callback is installed.
760 * @param ctx Active solve context.
761 * @param X Unconditioned PETSc trial state.
762 * @param F Residual vector to update in place.
763 * @return PetscErrorCode 0 on success.
764 */
765static PetscErrorCode MomentumNewtonKrylov_ApplyConstraints(MomentumNewtonKrylovContext *ctx,
766 Vec X, Vec F)
767{
768 UserCtx *user = ctx->user;
769 DMDALocalInfo info = user->info;
770 Vec local_x = NULL;
771 Cmpnts ***x = NULL, ***conditioned = NULL, ***f = NULL, ***lx = NULL;
772 const PetscInt xs = info.xs, xe = info.xs + info.xm;
773 const PetscInt ys = info.ys, ye = info.ys + info.ym;
774 const PetscInt zs = info.zs, ze = info.zs + info.zm;
775
776 PetscFunctionBeginUser;
777 PetscCall(DMGetLocalVector(user->fda, &local_x));
778 PetscCall(DMGlobalToLocalBegin(user->fda, X, INSERT_VALUES, local_x));
779 PetscCall(DMGlobalToLocalEnd(user->fda, X, INSERT_VALUES, local_x));
780 PetscCall(DMDAVecGetArrayRead(user->fda, X, &x));
781 PetscCall(DMDAVecGetArrayRead(user->fda, user->Ucont, &conditioned));
782 PetscCall(DMDAVecGetArray(user->fda, F, &f));
783 PetscCall(DMDAVecGetArrayRead(user->fda, local_x, &lx));
784
785 for (PetscInt k = zs; k < ze; ++k) {
786 for (PetscInt j = ys; j < ye; ++j) {
787 for (PetscInt i = xs; i < xe; ++i) {
788 PetscScalar *fv = &f[k][j][i].x;
789 const PetscScalar *xv = &x[k][j][i].x;
790 const PetscScalar *cv = &conditioned[k][j][i].x;
791
792 for (PetscInt component = 0; component < 3; ++component) {
793 PetscInt ri, rj, rk;
794 MomentumNewtonKrylovRowType row = MomentumNewtonKrylov_ClassifyRow(
795 user, i, j, k, component, &ri, &rj, &rk);
796 const PetscScalar *rv = &lx[rk][rj][ri].x;
797
798 if (row == MOM_NK_ROW_FIXED_CONDITIONED) fv[component] = xv[component] - cv[component];
799 else if (row == MOM_NK_ROW_FIXED_HOMOGENEOUS) fv[component] = xv[component];
800 else if (row == MOM_NK_ROW_PERIODIC_DUPLICATE) fv[component] = xv[component] - rv[component];
801 }
802 }
803 }
804 }
805
806 PetscCall(DMDAVecRestoreArrayRead(user->fda, local_x, &lx));
807 PetscCall(DMDAVecRestoreArray(user->fda, F, &f));
808 PetscCall(DMDAVecRestoreArrayRead(user->fda, user->Ucont, &conditioned));
809 PetscCall(DMDAVecRestoreArrayRead(user->fda, X, &x));
810 PetscCall(DMRestoreLocalVector(user->fda, &local_x));
811 PetscFunctionReturn(PETSC_SUCCESS);
812}

814#undef __FUNCT__
815#define __FUNCT__ "MomentumNewtonKrylov_FormResidual"
816/**
817 * @brief Adapts a PETSc trial vector to the existing momentum residual path.
818 *
819 * A matrix-free SNES residual must be a deterministic function of the trial
820 * vector X alone: F(X) may not depend on any state left by a previous residual
821 * or MFFD evaluation, or finite-difference Jacobian actions become inconsistent.
822 * To honor that contract this callback fully derives the Cartesian velocity
823 * state (Ucat/lUcat) from X before the first boundary sweep -- see the inline
824 * comment below for why ApplyBoundaryConditions()'s own internal reconstruction
825 * is not sufficient for the first outlet pass.
826 *
827 * State invariants:
828 * - On entry, X is the only input that determines the result; user->Ucont,
829 * user->Ucat and their local ghosts are treated as scratch and are fully
830 * overwritten from X.
831 * - Supported handlers may overwrite flux totals and other diagnostics on
832 * every call, but those values must not affect a later call at the same X;
833 * the deterministic seed guarantees this.
834 * - No histories, pressure, viscosity, or controller state advance here.
835 *
836 * Side effects: overwrites user->Ucont/lUcont, user->Ucat/lUcat, user->Rhs, the
837 * boundary Ubcs targets, and boundary flux/area diagnostics; writes F.
838 *
839 * @param snes Calling nonlinear solver.
840 * @param X Trial solution (read-only).
841 * @param F Residual output.
842 * @param vctx Pointer to MomentumNewtonKrylovContext.
843 * @return PetscErrorCode 0 on success.
844 */
845static PetscErrorCode MomentumNewtonKrylov_FormResidual(SNES snes, Vec X, Vec F, void *vctx)
846{
847 MomentumNewtonKrylovContext *ctx = (MomentumNewtonKrylovContext *)vctx;
848 UserCtx *user = ctx->user;
849 const char *staggered_fields[] = {"Ucont"};
850 const char *cell_fields[] = {"Ucat"};
851
852 PetscFunctionBeginUser;
853 (void)snes;
854 PetscCall(VecCopy(X, user->Ucont));
855 PetscCall(SynchronizePeriodicStaggeredFields(user, 1, staggered_fields));
856
857 /* Deterministic pre-boundary Cartesian seed. Establish the full Ucat/lUcat
858 * state from the current X before any boundary handler runs:
859 *
860 * X -> Ucont/lUcont -> Ucat -> periodic Ucat -> lUcat -> boundaries
861 *
862 * Why this is required, and why it is NOT redundant with the reconstruction
863 * already performed inside ApplyBoundaryConditions():
864 *
865 * 1. A matrix-free SNES residual must be a deterministic function of X. If
866 * the Cartesian state is left over from a previous residual/MFFD call,
867 * F(X) becomes history dependent and the finite-difference Jacobian
868 * action Jv = (F(X+hv)-F(X))/h is invalidated.
869 * 2. The conservation-outlet handler reads lUcat during the FIRST boundary
870 * sweep (it measures the uncorrected outflow and builds the outlet
871 * profile from the Cartesian field). Without this seed it would read the
872 * stale lUcat from the preceding evaluation.
873 * 3. ApplyBoundaryConditions() does reconstruct Ucat/lUcat, but only AFTER
874 * each handler sweep (Contra2Cart runs after BoundarySystem_ExecuteStep
875 * within every pass). Those internal updates therefore prepare passes 2
876 * and 3 -- they cannot prepare the very first outlet read of pass 1.
877 * 4. Contra2Cart() rebuilds the global Ucat interior from the current
878 * lUcont, but it does not by itself refresh lUcat (nor lUcont; that was
879 * done by the SynchronizePeriodicStaggeredFields call above).
880 * 5. SynchronizePeriodicCellFields("Ucat") must run before the ghost
881 * scatter so periodic duplicate planes are finalized consistently (it is
882 * a no-op when no direction is periodic, as on the straight duct).
883 * 6. UpdateLocalGhosts("Ucat") is required because the outlet handler reads
884 * lUcat -- the local ghosted vector -- not merely the global Ucat.
885 *
886 * Do NOT "simplify" this to a bare Contra2Cart(user), and do NOT delete it
887 * as apparently redundant with ApplyBoundaryConditions(): the three internal
888 * boundary passes remain necessary (they refresh the Cartesian state after
889 * each boundary correction), but only this sequence makes pass 1's input a
890 * deterministic function of X. */
891 PetscCall(Contra2Cart(user));
892 PetscCall(SynchronizePeriodicCellFields(user, 1, cell_fields));
893 PetscCall(UpdateLocalGhosts(user, "Ucat"));
894
895 PetscCall(ApplyBoundaryConditions(user));
896 PetscCall(ComputeTotalResidual(user));
897 PetscCall(VecCopy(user->Rhs, F));
898 PetscCall(VecScale(F, -1.0));
899 PetscCall(MomentumNewtonKrylov_ApplyConstraints(ctx, X, F));
900 PetscFunctionReturn(PETSC_SUCCESS);
901}

905/**
906 * @brief Runs one per-call matrix-free Newton--Krylov momentum solve.
907 * @param user Single-block momentum context.
908 * @param ibm Must be NULL in version one.
909 * @param fsi Must be NULL in version one.
910 * @return 0 on convergence or an error after restoring the canonical entry state.
911 */
912PetscErrorCode MomentumSolver_NewtonKrylov(UserCtx *user, IBMNodes *ibm, FSInfo *fsi)
913{
914 PetscErrorCode ierr = PETSC_SUCCESS, cleanup_ierr;
915 SimCtx *simCtx;
916 SNES snes = NULL;
917 Vec solution = NULL, entry_backup = NULL;
918 KSP ksp = NULL;
919 PC pc = NULL;
920 MomentumPreconditionerDescription preconditioner_description = {0};
921 PetscBool restore_entry = PETSC_FALSE;
922 PetscBool rhs_created = PETSC_FALSE;
923 PetscBool solve_started = PETSC_FALSE;
924 PetscBool committed = PETSC_FALSE;
925 SNESConvergedReason reason = SNES_CONVERGED_ITERATING;
926 PetscInt nonlinear_its = 0, function_evals = 0, linear_its = 0;
927 PetscReal final_norm = PETSC_MAX_REAL;
928 MomentumNewtonKrylovContext ctx = {0};
929 const char *staggered_fields[] = {"Ucont"};
930
931 PetscFunctionBeginUser;
932 PetscCall(MomentumNewtonKrylov_Validate(user));
933 PetscCheck(ibm == NULL && fsi == NULL, PETSC_COMM_WORLD, PETSC_ERR_SUP,
934 "Newton Krylov version one does not accept IBM or FSI objects.");
935 PetscCheck(user->Rhs == NULL, PETSC_COMM_SELF, PETSC_ERR_ARG_WRONGSTATE,
936 "Newton Krylov requires UserCtx::Rhs to be unallocated on entry.");
937 simCtx = user->simCtx;
938
939 ierr = VecDuplicate(user->Ucont, &solution); if (ierr) goto cleanup;
940 ierr = VecDuplicate(user->Ucont, &entry_backup); if (ierr) goto cleanup;
941 ierr = VecDuplicate(user->Ucont, &user->Rhs); if (ierr) goto cleanup;
942 rhs_created = PETSC_TRUE;
943 ierr = SNESCreate(PetscObjectComm((PetscObject)user->Ucont), &snes); if (ierr) goto cleanup;
944
945 ierr = SynchronizePeriodicStaggeredFields(user, 1, staggered_fields); if (ierr) goto cleanup;
946 ierr = ApplyBoundaryConditions(user); if (ierr) goto cleanup;
947 ierr = VecCopy(user->Ucont, entry_backup); if (ierr) goto cleanup;
948 restore_entry = PETSC_TRUE;
949 ierr = VecCopy(user->Ucont, solution); if (ierr) goto cleanup;
950
951 ctx.user = user;
952 ierr = MomentumNewtonKrylov_ReadLinearizationConfig(
953 &ctx.jacobian, &preconditioner_description); if (ierr) goto cleanup;
954 ierr = SNESSetOptionsPrefix(snes, "mom_nk_"); if (ierr) goto cleanup;
955 ierr = SNESSetType(snes, SNESNEWTONLS); if (ierr) goto cleanup;
956 ierr = SNESSetDM(snes, user->fda); if (ierr) goto cleanup;
