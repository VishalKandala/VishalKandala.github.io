143 * @param user Single-block momentum context to validate.

146static PetscErrorCode MomentumNewtonKrylov_Validate(UserCtx *user)
147{
148 SimCtx *simCtx;
149 PetscReal mask_max = 0.0;
150
151 PetscFunctionBeginUser;
152 PetscCheck(user != NULL, PETSC_COMM_SELF, PETSC_ERR_ARG_NULL,
153 "Newton Krylov requires a non-NULL UserCtx.");
154 simCtx = user->simCtx;
155 PetscCheck(simCtx != NULL, PETSC_COMM_SELF, PETSC_ERR_ARG_NULL,
156 "Newton Krylov requires UserCtx::simCtx.");
157 PetscCheck(simCtx->block_number == 1, PETSC_COMM_WORLD, PETSC_ERR_SUP,
158 "Newton Krylov version one supports exactly one block (got %d).",
159 simCtx->block_number);
160 PetscCheck(!simCtx->immersed, PETSC_COMM_WORLD, PETSC_ERR_SUP,
161 "Newton Krylov version one does not support immersed boundaries.");
162 PetscCheck(!simCtx->movefsi && !simCtx->rotatefsi, PETSC_COMM_WORLD, PETSC_ERR_SUP,
163 "Newton Krylov version one does not support moving or rotating bodies/FSI.");
164 PetscCheck(!simCtx->moveframe && !simCtx->rotateframe, PETSC_COMM_WORLD, PETSC_ERR_SUP,
165 "Newton Krylov version one does not support moving or rotating reference frames.");
166 PetscCheck(!simCtx->rans, PETSC_COMM_WORLD, PETSC_ERR_SUP,
167 "Newton Krylov version one does not support RANS.");
168 PetscCheck(!simCtx->clark, PETSC_COMM_WORLD, PETSC_ERR_SUP,
169 "Newton Krylov version one does not support the Clark model.");
170 PetscCheck(!simCtx->TwoD, PETSC_COMM_WORLD, PETSC_ERR_SUP,
171 "Newton Krylov version one does not support TwoD component masking.");
172 PetscCheck(!simCtx->wallfunction, PETSC_COMM_WORLD, PETSC_ERR_SUP,
173 "Newton Krylov version one does not support wall functions.");
174 PetscCheck(simCtx->StartStep == 0, PETSC_COMM_WORLD, PETSC_ERR_SUP,
175 "Newton Krylov version one supports fresh starts only (StartStep must be zero)." );
176
177 for (PetscInt face = 0; face < 6; ++face) {
178 const BoundaryFaceConfig *cfg = &user->boundary_faces[face];
179 PetscBool supported = PETSC_FALSE;
180
181 switch (cfg->handler_type) {
182 case BC_HANDLER_WALL_NOSLIP:
183 supported = (PetscBool)(cfg->mathematical_type == WALL);
184 break;
185 case BC_HANDLER_INLET_CONSTANT_VELOCITY:
186 case BC_HANDLER_INLET_PARABOLIC:
187 case BC_HANDLER_INLET_PROFILE_FROM_FILE:
188 supported = (PetscBool)(cfg->mathematical_type == INLET);
189 break;
190 case BC_HANDLER_OUTLET_CONSERVATION:
191 supported = (PetscBool)(cfg->mathematical_type == OUTLET);
192 break;
193 case BC_HANDLER_PERIODIC_GEOMETRIC:
194 supported = (PetscBool)(cfg->mathematical_type == PERIODIC);
195 break;
196 default:
197 supported = PETSC_FALSE;
198 break;
199 }
200 PetscCheck(supported, PETSC_COMM_WORLD, PETSC_ERR_SUP,
201 "Newton Krylov version one does not support boundary face %d with mathematical type %d and handler %d.",
202 face, (PetscInt)cfg->mathematical_type, (PetscInt)cfg->handler_type);
203 }
204
205 PetscCheck((user->boundary_faces[BC_FACE_NEG_X].mathematical_type == PERIODIC) ==
206 (user->boundary_faces[BC_FACE_POS_X].mathematical_type == PERIODIC),
207 PETSC_COMM_WORLD, PETSC_ERR_SUP, "Newton Krylov requires paired x-periodic faces.");
208 PetscCheck((user->boundary_faces[BC_FACE_NEG_Y].mathematical_type == PERIODIC) ==
209 (user->boundary_faces[BC_FACE_POS_Y].mathematical_type == PERIODIC),
210 PETSC_COMM_WORLD, PETSC_ERR_SUP, "Newton Krylov requires paired y-periodic faces.");
211 PetscCheck((user->boundary_faces[BC_FACE_NEG_Z].mathematical_type == PERIODIC) ==
212 (user->boundary_faces[BC_FACE_POS_Z].mathematical_type == PERIODIC),
213 PETSC_COMM_WORLD, PETSC_ERR_SUP, "Newton Krylov requires paired z-periodic faces.");
214
215 PetscCheck(user->Nvert != NULL, PETSC_COMM_SELF, PETSC_ERR_ARG_NULL,
216 "Newton Krylov requires the cell-mask vector Nvert.");
217 PetscCall(VecMax(user->Nvert, NULL, &mask_max));
218 PetscCheck(mask_max <= 0.1, PETSC_COMM_WORLD, PETSC_ERR_SUP,
219 "Newton Krylov version one does not define equations for masked solid cells (max Nvert=%g).",
220 (double)mask_max);
221 PetscFunctionReturn(PETSC_SUCCESS);
222}

226/**
227 * @brief Classifies one stored staggered component row and its periodic representative.
228 *
229 * Negative nonperiodic planes and positive dummy planes are zeroed by the legacy
230 * residual treatment. Only a face-normal row actually written by its boundary
231 * handler is conditioned; the remaining zeroed rows use F=X. Periodic endpoint
232 * rows use the same wrapped representatives as SynchronizePeriodicStaggeredFields().
233 * At periodic/nonperiodic intersections the periodic equation owns rows that a
234 * shrunken boundary-handler loop does not condition.
235 *
236 * @param user Block context containing boundary metadata and global dimensions.
237 * @param i Global i index.
238 * @param j Global j index.
239 * @param k Global k index.
240 * @param component Stored component, 0=x, 1=y, 2=z.
241 * @param ri Returned periodic representative i index.
242 * @param rj Returned periodic representative j index.
243 * @param rk Returned periodic representative k index.
244 * @return Exact version-one row category.
245 */
246static MomentumNewtonKrylovRowType MomentumNewtonKrylov_ClassifyRow(
247 UserCtx *user, PetscInt i, PetscInt j, PetscInt k, PetscInt component,
248 PetscInt *ri, PetscInt *rj, PetscInt *rk)
249{
250 const PetscInt mx = user->info.mx, my = user->info.my, mz = user->info.mz;
251 const PetscInt coord[3] = {i, j, k};
252 const PetscInt size[3] = {mx, my, mz};
253 const BCFace neg_face[3] = {BC_FACE_NEG_X, BC_FACE_NEG_Y, BC_FACE_NEG_Z};
254 PetscBool periodic[3], periodic_duplicate = PETSC_FALSE;
255 PetscBool residual_zeroed = PETSC_FALSE, conditioned = PETSC_FALSE;
256
257 *ri = i; *rj = j; *rk = k;
258 for (PetscInt axis = 0; axis < 3; ++axis) {
259 periodic[axis] = (PetscBool)(
260 user->boundary_faces[neg_face[axis]].mathematical_type == PERIODIC);
261 if (periodic[axis] && coord[axis] == 0) {
262 periodic_duplicate = PETSC_TRUE;
263 if (axis == 0) *ri = -2;
264 else if (axis == 1) *rj = -2;
265 else *rk = -2;
266 }
267 if (periodic[axis] && coord[axis] == size[axis] - 1) {
268 periodic_duplicate = PETSC_TRUE;
269 if (axis == 0) *ri = mx + 1;
270 else if (axis == 1) *rj = my + 1;
271 else *rk = mz + 1;
272 }
273
274 if (!periodic[axis] && coord[axis] == 0) residual_zeroed = PETSC_TRUE;
275 if (coord[axis] == size[axis] - 1) residual_zeroed = PETSC_TRUE;
276 if (!periodic[axis] && coord[axis] == size[axis] - 2 && component == axis)
277 residual_zeroed = PETSC_TRUE;
278 }
279
280 if (!periodic[component] &&
281 (coord[component] == 0 || coord[component] == size[component] - 2)) {
282 PetscBool tangential_interior = PETSC_TRUE;
283 for (PetscInt axis = 0; axis < 3; ++axis) {
284 if (axis == component) continue;
285 if (coord[axis] < 1 || coord[axis] > size[axis] - 2)
286 tangential_interior = PETSC_FALSE;
287 }
288 conditioned = tangential_interior;
289 }
290
291 if (conditioned) return MOM_NK_ROW_FIXED_CONDITIONED;
292 if (periodic_duplicate) return MOM_NK_ROW_PERIODIC_DUPLICATE;
293 if (residual_zeroed) return MOM_NK_ROW_FIXED_HOMOGENEOUS;
294 return MOM_NK_ROW_PHYSICAL;
295}

299/**
300 * @brief Replaces every non-independent residual row with an explicit equation.
301 *
302 * Conditioned face-normal rows use F=X-Uconditioned, unconditioned legacy
303 * dummy/tangential rows use F=X, and periodic duplicates use Fdup=Xdup-Xrep.
304 * These equations prevent the zero Jacobian rows produced by simply retaining
305 * EnforceRHSBoundaryConditions() zeros in a matrix-free Newton operator. Immersed,
306 * masked, TwoD, and interface rows are rejected before this callback is installed.
307 * @param ctx Active solve context.
308 * @param X Unconditioned PETSc trial state.
309 * @param F Residual vector to update in place.
310 * @return PetscErrorCode 0 on success.
311 */
312static PetscErrorCode MomentumNewtonKrylov_ApplyConstraints(MomentumNewtonKrylovContext *ctx,
313 Vec X, Vec F)
314{
315 UserCtx *user = ctx->user;
316 DMDALocalInfo info = user->info;
317 Vec local_x = NULL;
318 Cmpnts ***x = NULL, ***conditioned = NULL, ***f = NULL, ***lx = NULL;
319 const PetscInt xs = info.xs, xe = info.xs + info.xm;
320 const PetscInt ys = info.ys, ye = info.ys + info.ym;
321 const PetscInt zs = info.zs, ze = info.zs + info.zm;
322
323 PetscFunctionBeginUser;
324 PetscCall(DMGetLocalVector(user->fda, &local_x));
325 PetscCall(DMGlobalToLocalBegin(user->fda, X, INSERT_VALUES, local_x));
326 PetscCall(DMGlobalToLocalEnd(user->fda, X, INSERT_VALUES, local_x));
327 PetscCall(DMDAVecGetArrayRead(user->fda, X, &x));
328 PetscCall(DMDAVecGetArrayRead(user->fda, user->Ucont, &conditioned));
329 PetscCall(DMDAVecGetArray(user->fda, F, &f));
330 PetscCall(DMDAVecGetArrayRead(user->fda, local_x, &lx));
331
332 for (PetscInt k = zs; k < ze; ++k) {
333 for (PetscInt j = ys; j < ye; ++j) {
334 for (PetscInt i = xs; i < xe; ++i) {
335 PetscScalar *fv = &f[k][j][i].x;
336 const PetscScalar *xv = &x[k][j][i].x;
337 const PetscScalar *cv = &conditioned[k][j][i].x;
338
339 for (PetscInt component = 0; component < 3; ++component) {
340 PetscInt ri, rj, rk;
341 MomentumNewtonKrylovRowType row = MomentumNewtonKrylov_ClassifyRow(
342 user, i, j, k, component, &ri, &rj, &rk);
343 const PetscScalar *rv = &lx[rk][rj][ri].x;
344
345 if (row == MOM_NK_ROW_FIXED_CONDITIONED) fv[component] = xv[component] - cv[component];
346 else if (row == MOM_NK_ROW_FIXED_HOMOGENEOUS) fv[component] = xv[component];
347 else if (row == MOM_NK_ROW_PERIODIC_DUPLICATE) fv[component] = xv[component] - rv[component];
348 }
349 }
350 }
351 }
352
353 PetscCall(DMDAVecRestoreArrayRead(user->fda, local_x, &lx));
354 PetscCall(DMDAVecRestoreArray(user->fda, F, &f));
355 PetscCall(DMDAVecRestoreArrayRead(user->fda, user->Ucont, &conditioned));
356 PetscCall(DMDAVecRestoreArrayRead(user->fda, X, &x));
357 PetscCall(DMRestoreLocalVector(user->fda, &local_x));
358 PetscFunctionReturn(PETSC_SUCCESS);
359}

363/**
364 * @brief Adapts a PETSc trial vector to the existing momentum residual path.
365 *
366 * Supported handlers may overwrite flux totals and other diagnostics on every
367 * call, but repeatability tests require those values not to affect a later call
368 * at the same X. No histories, pressure, viscosity, or controller state advance
369 * is performed here.
370 * @param snes Calling nonlinear solver.
371 * @param X Trial solution (read-only).
372 * @param F Residual output.
373 * @param vctx Pointer to MomentumNewtonKrylovContext.
374 * @return PetscErrorCode 0 on success.
375 */
376static PetscErrorCode MomentumNewtonKrylov_FormResidual(SNES snes, Vec X, Vec F, void *vctx)
377{
378 MomentumNewtonKrylovContext *ctx = (MomentumNewtonKrylovContext *)vctx;
379 UserCtx *user = ctx->user;
380 const char *staggered_fields[] = {"Ucont"};
381
382 PetscFunctionBeginUser;
383 (void)snes;
384 PetscCall(VecCopy(X, user->Ucont));
385 PetscCall(SynchronizePeriodicStaggeredFields(user, 1, staggered_fields));
386 PetscCall(ApplyBoundaryConditions(user));
387 PetscCall(ComputeTotalResidual(user));
388 PetscCall(VecCopy(user->Rhs, F));
389 PetscCall(VecScale(F, -1.0));
390 PetscCall(MomentumNewtonKrylov_ApplyConstraints(ctx, X, F));
391 PetscFunctionReturn(PETSC_SUCCESS);
392}

399 * @param ibm Must be NULL in version one.

404{

406 SimCtx *simCtx;

408 Mat J = NULL;
409 Vec solution = NULL, entry_backup = NULL;
410 KSP ksp = NULL;
411 PC pc = NULL;
412 const char *pc_type = NULL;

415 PetscBool rhs_created = PETSC_FALSE;
416 PetscBool solve_started = PETSC_FALSE;
417 PetscBool committed = PETSC_FALSE;
418 SNESConvergedReason reason = SNES_CONVERGED_ITERATING;
419 PetscInt nonlinear_its = 0, function_evals = 0, linear_its = 0;

422 const char *staggered_fields[] = {"Ucont"};
423
424 PetscFunctionBeginUser;
425 PetscCall(MomentumNewtonKrylov_Validate(user));
426 PetscCheck(ibm == NULL && fsi == NULL, PETSC_COMM_WORLD, PETSC_ERR_SUP,
427 "Newton Krylov version one does not accept IBM or FSI objects.");
428 PetscCheck(user->Rhs == NULL, PETSC_COMM_SELF, PETSC_ERR_ARG_WRONGSTATE,
429 "Newton Krylov requires UserCtx::Rhs to be unallocated on entry.");

433 ierr = VecDuplicate(user->Ucont, &entry_backup); if (ierr) goto cleanup;
434 ierr = VecDuplicate(user->Ucont, &user->Rhs); if (ierr) goto cleanup;
435 rhs_created = PETSC_TRUE;

437
438 ierr = SynchronizePeriodicStaggeredFields(user, 1, staggered_fields); if (ierr) goto cleanup;
439 ierr = ApplyBoundaryConditions(user); if (ierr) goto cleanup;
440 ierr = VecCopy(user->Ucont, entry_backup); if (ierr) goto cleanup;
441 restore_entry = PETSC_TRUE;
442 ierr = VecCopy(user->Ucont, solution); if (ierr) goto cleanup;
443
444 ctx.user = user;
445 ierr = SNESSetOptionsPrefix(snes, "mom_nk_"); if (ierr) goto cleanup;
446 ierr = SNESSetType(snes, SNESNEWTONLS); if (ierr) goto cleanup;
447 ierr = SNESSetDM(snes, user->fda); if (ierr) goto cleanup;
448 ierr = SNESSetFunction(snes, NULL, MomentumNewtonKrylov_FormResidual, &ctx); if (ierr) goto cleanup;
449 ierr = MatCreateSNESMF(snes, &J); if (ierr) goto cleanup;
450 ierr = SNESSetJacobian(snes, J, J, MatMFFDComputeJacobian, NULL); if (ierr) goto cleanup;
451 ierr = SNESGetKSP(snes, &ksp); if (ierr) goto cleanup;
452 ierr = KSPSetType(ksp, KSPGMRES); if (ierr) goto cleanup;
453 ierr = KSPGetPC(ksp, &pc); if (ierr) goto cleanup;
454 ierr = PCSetType(pc, PCNONE); if (ierr) goto cleanup;
455 ierr = SNESSetFromOptions(snes); if (ierr) goto cleanup;
456 ierr = SNESMonitorSet(snes, MomentumNewtonKrylov_Monitor, &ctx, NULL); if (ierr) goto cleanup;
457 ierr = PCGetType(pc, &pc_type); if (ierr) goto cleanup;
458 ierr = PetscStrcmp(pc_type, PCNONE, &pc_is_none); if (ierr) goto cleanup;
459 if (!pc_is_none) {
460 LOG(GLOBAL, LOG_ERROR,
461 "Newton Krylov version one requires PCNONE; option processing selected PC type '%s'.\n",
462 pc_type ? pc_type : "(unset)");
463 ierr = PETSC_ERR_SUP;
464 goto cleanup;
465 }

467 MomentumNewtonKrylov_OpenHistory(&ctx);
468 solve_started = PETSC_TRUE;

470 if (ierr) goto cleanup;
471 ierr = SNESGetConvergedReason(snes, &reason); if (ierr) goto cleanup;
472 ierr = SNESGetIterationNumber(snes, &nonlinear_its); if (ierr) goto cleanup;
473 ierr = SNESGetNumberFunctionEvals(snes, &function_evals); if (ierr) goto cleanup;
474 ierr = SNESGetLinearSolveIterations(snes, &linear_its); if (ierr) goto cleanup;
475 ierr = SNESGetFunctionNorm(snes, &final_norm); if (ierr) goto cleanup;
476
477 if (reason > 0) {
478 ierr = VecCopy(solution, user->Ucont); if (ierr) goto cleanup;
479 simCtx->mom_last_converged = PETSC_TRUE;
480 } else {
481 ierr = VecCopy(entry_backup, user->Ucont); if (ierr) goto cleanup;
482 simCtx->mom_last_converged = PETSC_FALSE;

492 linear_its, (double)final_norm, reason > 0 ? "committed" : "rolled back");

497 SNES retained without replacing the primary error so a failed attempt is

505 }

508 if (!ierr) ierr = cleanup_ierr;
509 cleanup_ierr = SynchronizePeriodicStaggeredFields(user, 1, staggered_fields);
510 if (!ierr) ierr = cleanup_ierr;
511 cleanup_ierr = ApplyBoundaryConditions(user);
512 if (!ierr) ierr = cleanup_ierr;
513 simCtx->mom_last_converged = PETSC_FALSE;
514 committed = PETSC_FALSE;
515 }
516 if (ctx.history_file) {
517 (void)fclose(ctx.history_file);
518 ctx.history_file = NULL;
519 }
520 if (solve_started) {
521 MomentumNewtonKrylov_WriteSummary(&ctx, reason, nonlinear_its, function_evals,
522 linear_its, final_norm, committed);
523 }
524 if (rhs_created) {
525 cleanup_ierr = VecDestroy(&user->Rhs);
526 if (!ierr) ierr = cleanup_ierr;
527 }
528 cleanup_ierr = VecDestroy(&entry_backup); if (!ierr) ierr = cleanup_ierr;
529 cleanup_ierr = VecDestroy(&solution); if (!ierr) ierr = cleanup_ierr;
530 cleanup_ierr = MatDestroy(&J); if (!ierr) ierr = cleanup_ierr;
531 cleanup_ierr = SNESDestroy(&snes); if (!ierr) ierr = cleanup_ierr;
532 PetscFunctionReturn(ierr);
