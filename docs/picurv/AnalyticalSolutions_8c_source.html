1/**
2 * @file AnalyticalSolutions.c
3 * @brief Prescribes the Eulerian fields from the case's source instead of solving for them.
4 *
5 * @details With `eulerian_field_source: analytical`, the field is the initial condition's source,
6 * evaluated at the step's time through the operations every target of a source shares: fluxes at
7 * the faces (PicurvSourceAddFluxes()) and a bulk velocity (PicurvSourceCellBulkFactor()). A Cartesian
8 * source also sets `Ucat` and an optional pressure at the cell centres, exactly; a curvilinear one
9 * derives `Ucat` from its fluxes.
10 *
11 * The boundary values are this module's own `Ubcs`. After setting the state it runs the sequence the
12 * solved flow is finalized with (see `FinalizePostProjectionCellFields`): `UpdateDummyCells` on
13 * non-periodic faces, `SynchronizePeriodicCellFields` on periodic ones, `UpdateCornerNodes` for edges
14 * and corners, and the periodic synchronization again because corner averaging can overwrite
15 * periodic endpoints. It does not call that finalizer, which would refresh `Ubcs` from the boundary
16 * handlers.
17 */
18

29/** @brief The prescribed field: the case's source, and what it reuses from step to step. */
30struct AnalyticalField {
31 PicurvVelocitySource source; /* `-ic_source_*`. */
32 PicurvExpression *pressure; /* `-ic_source_pressure` [Pa], or NULL for zero pressure. */
33 PetscBool time_dependent;
34 PetscBool pointwise; /* Reads only x, y, z, t in the Cartesian frame, and varies in space. */
35 PetscBool evaluated;
36 PetscReal factor; /* From the bulk velocity. */
37 Vec *fraction; /* Per block. */
38 Vec *length;
39};

41/** @brief Read the source once, and work out what it depends on. Collective. */
42static PetscErrorCode AnalyticalFieldCreate(SimCtx *simCtx)
43{
44 UserCtx *user_finest = simCtx->usermg.mgctx[simCtx->usermg.mglevels - 1].user;
45 struct AnalyticalField *field;
46 char text[4096];
47 PetscBool found = PETSC_FALSE, uses;
48
49 PetscFunctionBeginUser;
50 PROFILE_HELPER_BEGIN;
51 PetscCall(PetscNew(&field));
52 PetscCall(PicurvVelocitySourceCreate("ic_source", PETSC_FALSE, PETSC_FALSE, &field->source));
53 PetscCall(PetscOptionsGetString(NULL, NULL, "-ic_source_pressure", text, sizeof(text), &found));
54 if (found) {
55 PetscCall(PicurvExpressionCompile(text, kPicurvSourceVariableNames, PICURV_SOURCE_VAR_COUNT, PETSC_FALSE, &field->pressure));
56 PetscCall(PicurvExpressionUsesVariable(field->pressure, PICURV_SOURCE_VAR_T, &uses));
57 field->time_dependent = uses;
58 }
59 field->time_dependent = (PetscBool)(field->time_dependent || field->source.time_dependent ||
60 field->source.bulk_time_dependent);
61 field->pointwise = (PetscBool)(field->source.frame == PICURV_SOURCE_FRAME_CARTESIAN && !field->source.bulk);
62 {
63 PetscBool varies = PETSC_FALSE;
64 for (PetscInt c = 0; c < field->source.count; ++c) {
65 for (PetscInt var = 0; var < PICURV_SOURCE_VAR_COUNT; ++var) {
66 PetscCall(PicurvExpressionUsesVariable(field->source.component[c], var, &uses));
67 if (!uses) continue;
68 if (var >= PICURV_SOURCE_VAR_XI) field->pointwise = PETSC_FALSE;
69 if (var <= PICURV_SOURCE_VAR_Z) varies = PETSC_TRUE;
70 }
71 }
72 field->pointwise = (PetscBool)(field->pointwise && varies);
73 }
74 field->factor = 1.0;
75 PetscCall(PetscCalloc2(simCtx->block_number, &field->fraction, simCtx->block_number, &field->length));
76 for (PetscInt bi = 0; bi < simCtx->block_number; ++bi) {
77 PetscCall(DMCreateGlobalVector(user_finest[bi].fda, &field->fraction[bi]));
78 PetscCall(DMCreateGlobalVector(user_finest[bi].fda, &field->length[bi]));
79 PetscCall(ComputeGridLineFractions(&user_finest[bi], field->fraction[bi], field->length[bi]));
80 }
81 simCtx->analyticalField = field;
82 PROFILE_HELPER_END;
83 PetscFunctionReturn(0);
84}

121/** @brief Set Ubcs on every physical boundary face to the source at the face centre. */
122static PetscErrorCode SetBoundaryVelocityFromSource(UserCtx *user, const struct AnalyticalField *field, PetscInt block,
123 PetscReal physical_time)
124{
125 const DMDALocalInfo *info = &user->info;
126 const PetscReal L_ref = user->simCtx->scaling.L_ref, U_ref = user->simCtx->scaling.U_ref;
127 const PetscInt start[3] = {info->xs, info->ys, info->zs};
128 const PetscInt end[3] = {info->xs + info->xm, info->ys + info->ym, info->zs + info->zm};
129 const PetscInt nodes[3] = {info->mx, info->my, info->mz};
130 const Cmpnts ***frac, ***len, ***csi, ***eta, ***zet, ***cent_face[3];
131 const PetscReal ***aj;
132 Cmpnts ***ubcs;
133 PetscReal values[PICURV_SOURCE_VAR_COUNT];
134
135 PetscFunctionBeginUser;
136 PROFILE_HELPER_BEGIN;
137 PetscCall(DMDAVecGetArrayRead(user->fda, field->fraction[block], &frac));
138 PetscCall(DMDAVecGetArrayRead(user->fda, field->length[block], &len));
139 PetscCall(DMDAVecGetArrayRead(user->fda, user->lCsi, &csi));
140 PetscCall(DMDAVecGetArrayRead(user->fda, user->lEta, &eta));
141 PetscCall(DMDAVecGetArrayRead(user->fda, user->lZet, &zet));
142 PetscCall(DMDAVecGetArrayRead(user->da, user->lAj, &aj));
143 PetscCall(DMDAVecGetArrayRead(user->fda, user->lCentx, &cent_face[0]));
144 PetscCall(DMDAVecGetArrayRead(user->fda, user->lCenty, &cent_face[1]));
145 PetscCall(DMDAVecGetArrayRead(user->fda, user->lCentz, &cent_face[2]));
146 PetscCall(DMDAVecGetArray(user->fda, user->Bcs.Ubcs, &ubcs));
147 values[PICURV_SOURCE_VAR_T] = physical_time;
148 for (PetscInt normal = 0; normal < 3; ++normal) {
149 const PetscInt a = (normal + 1) % 3, b = (normal + 2) % 3;
150 for (PetscInt side = 0; side < 2; ++side) {
151 const PetscInt layer = side ? end[normal] - 1 : start[normal];
152 if (side ? end[normal] != nodes[normal] : start[normal] != 0) continue;
153 for (PetscInt q = start[b]; q < end[b]; ++q)
154 for (PetscInt p = start[a]; p < end[a]; ++p) {
155 PetscInt f[3], c[3];
156 Cmpnts edges[3], v;
157 f[normal] = layer; f[a] = p; f[b] = q;
158 /* Grid-line data lives on interior cells; rows of dummies read the nearest one. */
159 c[normal] = side ? nodes[normal] - 2 : 1;
160 c[a] = PetscMax(PetscMin(p, nodes[a] - 2), 1); c[b] = PetscMax(PetscMin(q, nodes[b] - 2), 1);
161 PicurvSourceSetVariables(values, cent_face[normal][f[2]][f[1]][f[0]], frac[c[2]][c[1]][c[0]],
162 len[c[2]][c[1]][c[0]], L_ref);
163 values[PICURV_SOURCE_VAR_XI + normal] = side ? 1.0 : 0.0;
164 PetscCall(ComputeCellEdgeVectors(aj[c[2]][c[1]][c[0]], csi[c[2]][c[1]][c[0]], eta[c[2]][c[1]][c[0]],
165 zet[c[2]][c[1]][c[0]], edges));
166 PetscCall(PicurvVelocitySourceEvaluate(&field->source, values, edges, &v));
167 ubcs[f[2]][f[1]][f[0]].x = field->factor * v.x / U_ref;
168 ubcs[f[2]][f[1]][f[0]].y = field->factor * v.y / U_ref;
169 ubcs[f[2]][f[1]][f[0]].z = field->factor * v.z / U_ref;
170 }
171 }
172 }
173 PetscCall(DMDAVecRestoreArray(user->fda, user->Bcs.Ubcs, &ubcs));
174 PetscCall(DMDAVecRestoreArrayRead(user->fda, user->lCentz, &cent_face[2]));
175 PetscCall(DMDAVecRestoreArrayRead(user->fda, user->lCenty, &cent_face[1]));
176 PetscCall(DMDAVecRestoreArrayRead(user->fda, user->lCentx, &cent_face[0]));
177 PetscCall(DMDAVecRestoreArrayRead(user->da, user->lAj, &aj));
178 PetscCall(DMDAVecRestoreArrayRead(user->fda, user->lZet, &zet));
179 PetscCall(DMDAVecRestoreArrayRead(user->fda, user->lEta, &eta));
180 PetscCall(DMDAVecRestoreArrayRead(user->fda, user->lCsi, &csi));
181 PetscCall(DMDAVecRestoreArrayRead(user->fda, field->length[block], &len));
182 PetscCall(DMDAVecRestoreArrayRead(user->fda, field->fraction[block], &frac));
183 PROFILE_HELPER_END;
184 PetscFunctionReturn(0);
185}

187/** @brief Set a block's Ucat and P at every interior cell centre from the source. */
188static PetscErrorCode SetCellFieldsFromSource(UserCtx *user, const struct AnalyticalField *field, PetscInt block,
189 PetscReal physical_time)
190{
191 const DMDALocalInfo *info = &user->info;
192 const ScalingCtx *scaling = &user->simCtx->scaling;
193 const PetscInt xs = info->xs, xe = info->xs + info->xm, mx = info->mx;
194 const PetscInt ys = info->ys, ye = info->ys + info->ym, my = info->my;
195 const PetscInt zs = info->zs, ze = info->zs + info->zm, mz = info->mz;
196 const PetscBool set_velocity = (PetscBool)(field->source.frame != PICURV_SOURCE_FRAME_CURVILINEAR);
197 const Cmpnts ***cent, ***frac, ***len;
198 Cmpnts ***ucat;
199 PetscReal ***p, values[PICURV_SOURCE_VAR_COUNT];
200
201 PetscFunctionBeginUser;
202 PROFILE_HELPER_BEGIN;
203 PetscCall(DMDAVecGetArrayRead(user->fda, user->Cent, &cent));
204 PetscCall(DMDAVecGetArrayRead(user->fda, field->fraction[block], &frac));
205 PetscCall(DMDAVecGetArrayRead(user->fda, field->length[block], &len));
206 PetscCall(DMDAVecGetArray(user->fda, user->Ucat, &ucat));
207 PetscCall(DMDAVecGetArray(user->da, user->P, &p));
208 values[PICURV_SOURCE_VAR_T] = physical_time;
209 for (PetscInt k = (zs == 0) ? 1 : zs; k < ((ze == mz) ? mz - 1 : ze); ++k)
210 for (PetscInt j = (ys == 0) ? 1 : ys; j < ((ye == my) ? my - 1 : ye); ++j)
211 for (PetscInt i = (xs == 0) ? 1 : xs; i < ((xe == mx) ? mx - 1 : xe); ++i) {
212 PicurvSourceSetVariables(values, cent[k][j][i], frac[k][j][i], len[k][j][i], scaling->L_ref);
213 if (set_velocity) {
214 Cmpnts v;
215 PetscCall(PicurvVelocitySourceEvaluate(&field->source, values, NULL, &v));
216 ucat[k][j][i].x = field->factor * v.x / scaling->U_ref;
217 ucat[k][j][i].y = field->factor * v.y / scaling->U_ref;
218 ucat[k][j][i].z = field->factor * v.z / scaling->U_ref;
219 }
220 if (field->pressure) {
221 PetscReal value;
222 PetscCall(PicurvExpressionEvaluate(field->pressure, values, NULL, &value));
223 p[k][j][i] = value / scaling->P_ref;
224 } else {
225 p[k][j][i] = 0.0;
226 }
227 }
228 PetscCall(DMDAVecRestoreArray(user->da, user->P, &p));
229 PetscCall(DMDAVecRestoreArray(user->fda, user->Ucat, &ucat));
230 PetscCall(DMDAVecRestoreArrayRead(user->fda, field->length[block], &len));
231 PetscCall(DMDAVecRestoreArrayRead(user->fda, field->fraction[block], &frac));
232 PetscCall(DMDAVecRestoreArrayRead(user->fda, user->Cent, &cent));
233 PROFILE_HELPER_END;
234 PetscFunctionReturn(0);
235}

237/**
238 * @brief Implementation of \ref AnalyticalSolutionEngine().
239 * @details The field is the case's source, evaluated at the step's time. As on an inlet, a source
240 * that does not read `t` is evaluated once.
241 * @see AnalyticalSolutionEngine()
242 */
243PetscErrorCode AnalyticalSolutionEngine(SimCtx *simCtx)
244{
245 UserCtx *user_finest = simCtx->usermg.mgctx[simCtx->usermg.mglevels - 1].user;
246 struct AnalyticalField *field;
247 PetscReal physical_time;
248
249 PetscFunctionBeginUser;
250 PROFILE_FUNCTION_BEGIN;
251 if (!simCtx->analyticalField) PetscCall(AnalyticalFieldCreate(simCtx));
252 field = simCtx->analyticalField;
253 if (field->evaluated && !field->time_dependent) { PROFILE_FUNCTION_END; PetscFunctionReturn(0); }
254 PetscCall(PicurvPhysicalTime(simCtx, simCtx->ti, &physical_time));
255 for (PetscInt bi = 0; bi < simCtx->block_number; ++bi) {
256 UserCtx *user = &user_finest[bi];
257 const FieldId staggered_fields[] = {FIELD_ID_UCONT};
258 if (field->source.bulk) {
259 PetscCall(VecSet(user->Ucat, 0.0));
260 PetscCall(PicurvSourceCellBulkFactor(user, &field->source, field->fraction[bi], field->length[bi], physical_time, &field->factor));
261 }
262 PetscCall(VecSet(user->Ucont, 0.0));
263 PetscCall(PicurvSourceAddFluxes(user, &field->source, field->fraction[bi], field->length[bi], physical_time, field->factor, PETSC_TRUE));
264 PetscCall(SetBoundaryVelocityFromSource(user, field, bi, physical_time));
265 PetscCall(SetCellFieldsFromSource(user, field, bi, physical_time));
266 PetscCall(SynchronizePeriodicStaggeredFields(user, 1, staggered_fields));
267 PetscCall(UpdateLocalGhosts(user, FIELD_ID_UCONT));
268 if (field->source.frame == PICURV_SOURCE_FRAME_CURVILINEAR) PetscCall(Contra2Cart(user));
269 PetscCall(UpdateLocalGhosts(user, FIELD_ID_UCAT));
270 PetscCall(UpdateLocalGhosts(user, FIELD_ID_P));
271 PetscCall(UpdateDummyCells(user, FIELD_ID_UCAT));
272 PetscCall(UpdateDummyCells(user, FIELD_ID_P));
273 PetscCall(SynchronizePeriodicCellFields(user, 2, kAnalyticalCellFields));
274 PetscCall(UpdateCornerNodes(user, FIELD_ID_UCAT));
275 PetscCall(UpdateCornerNodes(user, FIELD_ID_P));
276 PetscCall(SynchronizePeriodicCellFields(user, 2, kAnalyticalCellFields));
277 PetscCall(UpdateLocalGhosts(user, FIELD_ID_UCAT));
278 PetscCall(UpdateLocalGhosts(user, FIELD_ID_P));
279 }
280 field->evaluated = PETSC_TRUE;
281 PROFILE_FUNCTION_END;
282 PetscFunctionReturn(0);
283}

285/**
286 * @brief Implementation of \ref SetAnalyticalSolutionForParticles().
287 * @details Only a pointwise source reaches here; see AnalyticalFieldSupportsInterpolationError().
288 * @see SetAnalyticalSolutionForParticles()
289 */
290PetscErrorCode SetAnalyticalSolutionForParticles(Vec tempVec, SimCtx *simCtx)
291{
292 const struct AnalyticalField *field = simCtx->analyticalField;
293 const ScalingCtx *scaling = &simCtx->scaling;
294 PetscInt nLocal;
295 PetscReal *data, values[PICURV_SOURCE_VAR_COUNT], physical_time;
296
297 PetscFunctionBeginUser;
298 PROFILE_HELPER_BEGIN;
299 PetscCheck(AnalyticalFieldSupportsInterpolationError(simCtx), PETSC_COMM_SELF, PETSC_ERR_ARG_WRONGSTATE,
300 "The analytical field cannot be evaluated at a particle: it needs a Cartesian source in x, y, z and t.");
301 PetscCall(PetscArrayzero(values, PICURV_SOURCE_VAR_COUNT));
302 PetscCall(PicurvPhysicalTime(simCtx, simCtx->ti, &physical_time));
303 values[PICURV_SOURCE_VAR_T] = physical_time;
304 PetscCall(VecGetLocalSize(tempVec, &nLocal));
305 PetscCall(VecGetArray(tempVec, &data));
306 /* Interleaved [x0,y0,z0, x1,y1,z1, ...], replaced by the velocity there. */
307 for (PetscInt i = 0; i < nLocal; i += 3) {
308 Cmpnts v;
309 values[PICURV_SOURCE_VAR_X] = data[i] * scaling->L_ref;
310 values[PICURV_SOURCE_VAR_Y] = data[i + 1] * scaling->L_ref;
311 values[PICURV_SOURCE_VAR_Z] = data[i + 2] * scaling->L_ref;
312 PetscCall(PicurvVelocitySourceEvaluate(&field->source, values, NULL, &v));
313 data[i] = v.x / scaling->U_ref;
314 data[i + 1] = v.y / scaling->U_ref;
315 data[i + 2] = v.z / scaling->U_ref;
316 }
317 PetscCall(VecRestoreArray(tempVec, &data));
318 PROFILE_HELPER_END;
319 PetscFunctionReturn(0);
320}
