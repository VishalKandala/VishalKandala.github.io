1/**
2 * @file AnalyticalSolutions.c
3 * @brief Prescribes the Eulerian fields from the case's source instead of solving for them.
4 *
5 * @details With `eulerian_field_source: analytical`, the field is the initial condition's source,
6 * evaluated at the step's time through the operations every target of a source shares: fluxes at
7 * the faces (PicurvSourceAddFluxes()) and a bulk velocity (PicurvSourceCellBulkFactor()). A Cartesian
8 * source also sets `Ucat` and an optional pressure at the cell centres, exactly; a curvilinear one
9 * derives `Ucat` from its fluxes.
10 *
11 * The boundary values are this module's own `Ubcs`, so after setting the state it runs the part of
12 * the solved flow's finalization that does not refresh them from the boundary handlers
13 * (FinalizeCellFieldsFromBoundaryValues()).
14 */

23/** @brief The prescribed field: the case's source, and what it reuses from step to step. */
24struct AnalyticalField {
25 PicurvVelocitySource source; /* `-ic_source_*`. */
26 PicurvExpression *pressure; /* `-ic_source_pressure` [Pa], or NULL for zero pressure. */
27 PetscBool time_dependent;
28 PetscBool pointwise; /* Reads only x, y, z, t in the Cartesian frame, and varies in space. */
29 PetscBool evaluated;
30 PetscReal factor; /* From the bulk velocity. */
31 Vec *fraction; /* Per block. */
32 Vec *length;
33 PicurvSourceFaceSlots (*faces)[6]; /* Per block, what the source reads on each face. */
34};

36/** @brief Read the source once, and work out what it depends on. Collective. */
37static PetscErrorCode AnalyticalFieldCreate(SimCtx *simCtx)
38{
39 UserCtx *user_finest = simCtx->usermg.mgctx[simCtx->usermg.mglevels - 1].user;
40 struct AnalyticalField *field;
41 char text[4096];
42 PetscBool found = PETSC_FALSE, uses;
43
44 PetscFunctionBeginUser;
45 PROFILE_HELPER_BEGIN;
46 PetscCall(PetscNew(&field));
47 PetscCall(PicurvVelocitySourceCreate("ic_source", PETSC_FALSE, PETSC_FALSE, &field->source));
48 PetscCall(PetscOptionsGetString(NULL, NULL, "-ic_source_pressure", text, sizeof(text), &found));
49 if (found) {
50 PetscCall(PicurvExpressionCompile(text, kPicurvSourceVariableNames, PICURV_SOURCE_VAR_CELL_COUNT, PETSC_FALSE, &field->pressure));
51 PetscCall(PicurvExpressionUsesVariable(field->pressure, PICURV_SOURCE_VAR_T, &uses));
52 field->time_dependent = uses;
53 }
54 field->time_dependent = (PetscBool)(field->time_dependent || field->source.time_dependent ||
55 field->source.bulk_time_dependent);
56 field->pointwise = (PetscBool)(field->source.frame == PICURV_SOURCE_FRAME_CARTESIAN && !field->source.bulk);
57 {
58 PetscBool varies = PETSC_FALSE;
59 for (PetscInt c = 0; c < field->source.count; ++c) {
60 for (PetscInt var = 0; var < PICURV_SOURCE_VAR_COUNT; ++var) {
61 PetscCall(PicurvExpressionUsesVariable(field->source.component[c], var, &uses));
62 if (!uses) continue;
63 if (var >= PICURV_SOURCE_VAR_XI) field->pointwise = PETSC_FALSE;
64 if (var <= PICURV_SOURCE_VAR_Z) varies = PETSC_TRUE;
65 }
66 }
67 field->pointwise = (PetscBool)(field->pointwise && varies);
68 }
69 field->factor = 1.0;
70 PetscCall(PetscCalloc3(simCtx->block_number, &field->fraction, simCtx->block_number, &field->length,
71 simCtx->block_number, &field->faces));
72 for (PetscInt bi = 0; bi < simCtx->block_number; ++bi) {
73 PetscCall(DMCreateGlobalVector(user_finest[bi].fda, &field->fraction[bi]));
74 PetscCall(DMCreateGlobalVector(user_finest[bi].fda, &field->length[bi]));
75 PetscCall(ComputeGridLineFractions(&user_finest[bi], field->fraction[bi], field->length[bi]));
76 for (PetscInt face = 0; face < 6; ++face) {
77 PetscCall(PicurvSourceFaceSlotsCreate(&user_finest[bi], (BCFace)face, field->fraction[bi], field->length[bi],
78 &field->faces[bi][face]));
79 }
80 }
81 simCtx->analyticalField = field;
82 PROFILE_HELPER_END;
83 PetscFunctionReturn(0);
84}

122/** @brief Set Ubcs on every physical boundary face to the source at the face centre. */
123static PetscErrorCode SetBoundaryVelocityFromSource(UserCtx *user, const struct AnalyticalField *field, PetscInt block,
124 PetscReal physical_time)
125{
126 const PetscReal U_ref = user->simCtx->scaling.U_ref;
127 Cmpnts ***ubcs;
128
129 PetscFunctionBeginUser;
130 PROFILE_HELPER_BEGIN;
131 PetscCall(DMDAVecGetArray(user->fda, user->Bcs.Ubcs, &ubcs));
132 for (PetscInt face = 0; face < 6; ++face) {
133 const PicurvSourceFaceSlots *slots = &field->faces[block][face];
134 for (PetscInt q = 0; q < slots->n[1]; ++q)
135 for (PetscInt p = 0; p < slots->n[0]; ++p) {
136 const PetscInt s = p + slots->n[0] * q;
137 PetscInt f[3];
138 Cmpnts v;
139 /* A + face's boundary value sits one past its flux, in the dummy layer. */
140 f[slots->normal] = slots->layer + face % 2; f[slots->a] = slots->lo[0] + p; f[slots->b] = slots->lo[1] + q;
141 slots->values[s][PICURV_SOURCE_VAR_T] = physical_time;
142 PetscCall(PicurvVelocitySourceEvaluate(&field->source, slots->values[s], slots->edges[s], &v));
143 ubcs[f[2]][f[1]][f[0]].x = field->factor * v.x / U_ref;
144 ubcs[f[2]][f[1]][f[0]].y = field->factor * v.y / U_ref;
145 ubcs[f[2]][f[1]][f[0]].z = field->factor * v.z / U_ref;
146 }
147 }
148 PetscCall(DMDAVecRestoreArray(user->fda, user->Bcs.Ubcs, &ubcs));
149 PROFILE_HELPER_END;
150 PetscFunctionReturn(0);
151}

153/** @brief Set a block's Ucat at every interior cell centre from the source. */
154static PetscErrorCode SetCellVelocityFromSource(UserCtx *user, const struct AnalyticalField *field, PetscInt block,
155 PetscReal physical_time)
156{
157 const DMDALocalInfo *info = &user->info;
158 const ScalingCtx *scaling = &user->simCtx->scaling;
159 const PetscInt xs = info->xs, xe = info->xs + info->xm, mx = info->mx;
160 const PetscInt ys = info->ys, ye = info->ys + info->ym, my = info->my;
161 const PetscInt zs = info->zs, ze = info->zs + info->zm, mz = info->mz;
162 const Cmpnts ***cent, ***frac, ***len;
163 Cmpnts ***ucat;
164 PetscReal values[PICURV_SOURCE_VAR_COUNT];
165
166 PetscFunctionBeginUser;
167 PROFILE_HELPER_BEGIN;
168 PetscCall(DMDAVecGetArrayRead(user->fda, user->Cent, &cent));
169 PetscCall(DMDAVecGetArrayRead(user->fda, field->fraction[block], &frac));
170 PetscCall(DMDAVecGetArrayRead(user->fda, field->length[block], &len));
171 PetscCall(DMDAVecGetArray(user->fda, user->Ucat, &ucat));
172 values[PICURV_SOURCE_VAR_T] = physical_time;
173 for (PetscInt k = (zs == 0) ? 1 : zs; k < ((ze == mz) ? mz - 1 : ze); ++k)
174 for (PetscInt j = (ys == 0) ? 1 : ys; j < ((ye == my) ? my - 1 : ye); ++j)
175 for (PetscInt i = (xs == 0) ? 1 : xs; i < ((xe == mx) ? mx - 1 : xe); ++i) {
176 Cmpnts v;
177 PicurvSourceSetVariables(values, cent[k][j][i], frac[k][j][i], len[k][j][i], scaling->L_ref);
178 PetscCall(PicurvVelocitySourceEvaluate(&field->source, values, NULL, &v));
179 ucat[k][j][i].x = field->factor * v.x / scaling->U_ref;
180 ucat[k][j][i].y = field->factor * v.y / scaling->U_ref;
181 ucat[k][j][i].z = field->factor * v.z / scaling->U_ref;
182 }
183 PetscCall(DMDAVecRestoreArray(user->fda, user->Ucat, &ucat));
184 PetscCall(DMDAVecRestoreArrayRead(user->fda, field->length[block], &len));
185 PetscCall(DMDAVecRestoreArrayRead(user->fda, field->fraction[block], &frac));
186 PetscCall(DMDAVecRestoreArrayRead(user->fda, user->Cent, &cent));
187 PROFILE_HELPER_END;
188 PetscFunctionReturn(0);
189}

191/**
192 * @brief Implementation of \ref AnalyticalSolutionEngine().
193 * @details The field is the case's source, evaluated at the step's time. As on an inlet, a source
194 * that does not read `t` is evaluated once.
195 * @see AnalyticalSolutionEngine()
196 */
197PetscErrorCode AnalyticalSolutionEngine(SimCtx *simCtx)
198{
199 UserCtx *user_finest = simCtx->usermg.mgctx[simCtx->usermg.mglevels - 1].user;
200 struct AnalyticalField *field;
201 PetscReal physical_time;
202
203 PetscFunctionBeginUser;
204 PROFILE_FUNCTION_BEGIN;
205 if (!simCtx->analyticalField) PetscCall(AnalyticalFieldCreate(simCtx));
206 field = simCtx->analyticalField;
207 if (field->evaluated && !field->time_dependent) { PROFILE_FUNCTION_END; PetscFunctionReturn(0); }
208 PetscCall(PicurvPhysicalTime(simCtx, simCtx->ti, &physical_time));
209 for (PetscInt bi = 0; bi < simCtx->block_number; ++bi) {
210 UserCtx *user = &user_finest[bi];
211 const FieldId staggered_fields[] = {FIELD_ID_UCONT};
212 if (field->source.bulk) {
213 PetscCall(VecSet(user->Ucat, 0.0));
214 PetscCall(PicurvSourceCellBulkFactor(user, &field->source, field->fraction[bi], field->length[bi], physical_time, &field->factor));
215 }
216 PetscCall(VecSet(user->Ucont, 0.0));
217 PetscCall(PicurvSourceAddFluxes(user, &field->source, field->fraction[bi], field->length[bi], physical_time, field->factor, PETSC_TRUE));
218 PetscCall(SetBoundaryVelocityFromSource(user, field, bi, physical_time));
219 if (field->source.frame != PICURV_SOURCE_FRAME_CURVILINEAR) PetscCall(SetCellVelocityFromSource(user, field, bi, physical_time));
220 if (field->pressure) PetscCall(PicurvSourceSetCellPressure(user, field->pressure, field->fraction[bi], field->length[bi], physical_time));
221 else PetscCall(VecSet(user->P, 0.0));
222 PetscCall(SynchronizePeriodicStaggeredFields(user, 1, staggered_fields));
223 PetscCall(UpdateLocalGhosts(user, FIELD_ID_UCONT));
224 if (field->source.frame == PICURV_SOURCE_FRAME_CURVILINEAR) PetscCall(Contra2Cart(user));
225 PetscCall(FinalizeCellFieldsFromBoundaryValues(user));
226 }
227 field->evaluated = PETSC_TRUE;
228 PROFILE_FUNCTION_END;
229 PetscFunctionReturn(0);
230}

232/**
233 * @brief Implementation of \ref SetAnalyticalSolutionForParticles().
234 * @details Only a pointwise source reaches here; see AnalyticalFieldSupportsInterpolationError().
235 * @see SetAnalyticalSolutionForParticles()
236 */
237PetscErrorCode SetAnalyticalSolutionForParticles(Vec tempVec, SimCtx *simCtx)
238{
239 const struct AnalyticalField *field = simCtx->analyticalField;
240 const ScalingCtx *scaling = &simCtx->scaling;
241 PetscInt nLocal;
242 PetscReal *data, values[PICURV_SOURCE_VAR_COUNT], physical_time;
243
244 PetscFunctionBeginUser;
245 PROFILE_HELPER_BEGIN;
246 PetscCheck(AnalyticalFieldSupportsInterpolationError(simCtx), PETSC_COMM_SELF, PETSC_ERR_ARG_WRONGSTATE,
247 "The analytical field cannot be evaluated at a particle: it needs a Cartesian source in x, y, z and t.");
248 PetscCall(PetscArrayzero(values, PICURV_SOURCE_VAR_COUNT));
249 PetscCall(PicurvPhysicalTime(simCtx, simCtx->ti, &physical_time));
250 values[PICURV_SOURCE_VAR_T] = physical_time;
251 PetscCall(VecGetLocalSize(tempVec, &nLocal));
252 PetscCall(VecGetArray(tempVec, &data));
253 /* Interleaved [x0,y0,z0, x1,y1,z1, ...], replaced by the velocity there. */
254 for (PetscInt i = 0; i < nLocal; i += 3) {
255 Cmpnts v;
256 values[PICURV_SOURCE_VAR_X] = data[i] * scaling->L_ref;
257 values[PICURV_SOURCE_VAR_Y] = data[i + 1] * scaling->L_ref;
258 values[PICURV_SOURCE_VAR_Z] = data[i + 2] * scaling->L_ref;
259 PetscCall(PicurvVelocitySourceEvaluate(&field->source, values, NULL, &v));
260 data[i] = v.x / scaling->U_ref;
261 data[i + 1] = v.y / scaling->U_ref;
262 data[i + 2] = v.z / scaling->U_ref;
263 }
264 PetscCall(VecRestoreArray(tempVec, &data));
265 PROFILE_HELPER_END;
266 PetscFunctionReturn(0);
267}
