1/**
2 * @file AnalyticalSolutions.c
3 * @brief Prescribes the Eulerian fields from the case's source instead of solving for them.
4 *
5 * @details With `eulerian_field_source: analytical`, the field is the initial condition's source,
6 * evaluated at the step's time through the operations every target of a source shares: fluxes at
7 * the faces (PicurvSourceAddFluxes()) and a bulk velocity (PicurvSourceCellBulkFactor()). A Cartesian
8 * source also sets `Ucat` and an optional pressure at the cell centres, exactly; a curvilinear one
9 * derives `Ucat` from its fluxes.
10 *
11 * The boundary values are this module's own `Ubcs`, so after setting the state it runs the part of
12 * the solved flow's finalization that does not refresh them from the boundary handlers
13 * (FinalizeCellFieldsFromBoundaryValues()).
14 */

23/** @brief The prescribed field: the case's source, and what it reuses from step to step. */
24struct AnalyticalField {
25 PicurvVelocitySource source; /* `-ic_source_*`. */
26 PicurvExpression *pressure; /* `-ic_source_pressure` [Pa], or NULL for zero pressure. */
27 PetscBool time_dependent;
28 PetscBool pointwise; /* Reads only x, y, z, t in the Cartesian frame, and varies in space. */
29 PetscBool evaluated;
30 PetscReal factor; /* From the bulk velocity. */
31 Vec *fraction; /* Per block. */
32 Vec *length;
33 PicurvSourceFaceSlots (*faces)[6]; /* Per block, what the source reads on each face. */
34};

36/** @brief Read the source once, and work out what it depends on. Collective. */
37static PetscErrorCode AnalyticalFieldCreate(SimCtx *simCtx)
38{
39 UserCtx *user_finest = simCtx->usermg.mgctx[simCtx->usermg.mglevels - 1].user;
40 struct AnalyticalField *field;
41 PetscBool uses;
42
43 PetscFunctionBeginUser;
44 PROFILE_HELPER_BEGIN;
45 PetscCall(PetscNew(&field));
46 PetscCall(PicurvVelocitySourceCreate("ic_source", PETSC_FALSE, PETSC_FALSE, PETSC_FALSE, &field->source));
47 PetscCall(PicurvSourcePressureCreate("ic_source", PETSC_FALSE, &field->pressure));
48 if (field->pressure) {
49 PetscCall(PicurvExpressionUsesVariable(field->pressure, PICURV_SOURCE_VAR_T, &uses));
50 field->time_dependent = uses;
51 }
52 field->time_dependent = (PetscBool)(field->time_dependent || field->source.time_dependent ||
53 field->source.bulk_time_dependent);
54 field->pointwise = (PetscBool)(field->source.frame == PICURV_SOURCE_FRAME_CARTESIAN && !field->source.bulk);
55 {
56 PetscBool varies = PETSC_FALSE;
57 for (PetscInt c = 0; c < field->source.count; ++c) {
58 for (PetscInt var = 0; var < PICURV_SOURCE_VAR_COUNT; ++var) {
59 PetscCall(PicurvExpressionUsesVariable(field->source.component[c], var, &uses));
60 if (!uses) continue;
61 if (var >= PICURV_SOURCE_VAR_XI) field->pointwise = PETSC_FALSE;
62 if (var <= PICURV_SOURCE_VAR_Z) varies = PETSC_TRUE;
63 }
64 }
65 field->pointwise = (PetscBool)(field->pointwise && varies);
66 }
67 field->factor = 1.0;
68 PetscCall(PetscCalloc3(simCtx->block_number, &field->fraction, simCtx->block_number, &field->length,
69 simCtx->block_number, &field->faces));
70 for (PetscInt bi = 0; bi < simCtx->block_number; ++bi) {
71 PetscCall(DMCreateGlobalVector(user_finest[bi].fda, &field->fraction[bi]));
72 PetscCall(DMCreateGlobalVector(user_finest[bi].fda, &field->length[bi]));
73 PetscCall(ComputeGridLineFractions(&user_finest[bi], field->fraction[bi], field->length[bi]));
74 for (PetscInt face = 0; face < 6; ++face) {
75 PetscCall(PicurvSourceFaceSlotsCreate(&user_finest[bi], (BCFace)face, field->fraction[bi], field->length[bi],
76 &field->faces[bi][face]));
77 }
78 }
79 simCtx->analyticalField = field;
80 PROFILE_HELPER_END;
81 PetscFunctionReturn(0);
82}

88PetscErrorCode AnalyticalFieldDestroy(SimCtx *simCtx)
89{
90 struct AnalyticalField *field = simCtx->analyticalField;
91
92 PetscFunctionBeginUser;
93 PROFILE_HELPER_BEGIN;
94 if (field) {
95 PetscCall(PicurvVelocitySourceDestroy(&field->source));
96 PetscCall(PicurvExpressionDestroy(&field->pressure));
97 for (PetscInt bi = 0; bi < simCtx->block_number; ++bi) {
98 PetscCall(VecDestroy(&field->fraction[bi]));
99 PetscCall(VecDestroy(&field->length[bi]));

101 }
102 PetscCall(PetscFree3(field->fraction, field->length, field->faces));
103 PetscCall(PetscFree(simCtx->analyticalField));
104 }
105 PROFILE_HELPER_END;
106 PetscFunctionReturn(0);
107}

121/** @brief Set Ubcs on every physical boundary face to the source at the face centre. */
122static PetscErrorCode SetBoundaryVelocityFromSource(UserCtx *user, const struct AnalyticalField *field, PetscInt block,
123 PetscReal physical_time)
124{
125 const PetscReal U_ref = user->simCtx->scaling.U_ref;
126 Cmpnts ***ubcs;
127
128 PetscFunctionBeginUser;
129 PROFILE_HELPER_BEGIN;
130 PetscCall(DMDAVecGetArray(user->fda, user->Bcs.Ubcs, &ubcs));
131 for (PetscInt face = 0; face < 6; ++face) {
132 const PicurvSourceFaceSlots *slots = &field->faces[block][face];
133 for (PetscInt q = 0; q < slots->n[1]; ++q)
134 for (PetscInt p = 0; p < slots->n[0]; ++p) {
135 const PetscInt s = p + slots->n[0] * q;
136 PetscInt f[3];
137 Cmpnts v;
138 /* A + face's boundary value sits one past its flux, in the dummy layer. */
139 f[slots->normal] = slots->layer + face % 2; f[slots->a] = slots->lo[0] + p; f[slots->b] = slots->lo[1] + q;
140 slots->values[s][PICURV_SOURCE_VAR_T] = physical_time;
141 PetscCall(PicurvVelocitySourceEvaluate(&field->source, slots->values[s], NULL, slots->edges[s], &v));
142 ubcs[f[2]][f[1]][f[0]].x = field->factor * v.x / U_ref;
143 ubcs[f[2]][f[1]][f[0]].y = field->factor * v.y / U_ref;
144 ubcs[f[2]][f[1]][f[0]].z = field->factor * v.z / U_ref;
145 }
146 }
147 PetscCall(DMDAVecRestoreArray(user->fda, user->Bcs.Ubcs, &ubcs));
148 PROFILE_HELPER_END;
149 PetscFunctionReturn(0);
150}

152/**
153 * @brief Implementation of \ref AnalyticalSolutionEngine().
154 * @details The field is the case's source, evaluated at the step's time. As on an inlet, a source
155 * that does not read `t` is evaluated once.
156 * @see AnalyticalSolutionEngine()
157 */
158PetscErrorCode AnalyticalSolutionEngine(SimCtx *simCtx)
159{
160 UserCtx *user_finest = simCtx->usermg.mgctx[simCtx->usermg.mglevels - 1].user;
161 struct AnalyticalField *field;
162 PetscReal physical_time;
163
164 PetscFunctionBeginUser;
165 PROFILE_FUNCTION_BEGIN;
166 if (!simCtx->analyticalField) PetscCall(AnalyticalFieldCreate(simCtx));
167 field = simCtx->analyticalField;
168 if (field->evaluated && !field->time_dependent) { PROFILE_FUNCTION_END; PetscFunctionReturn(0); }
169 PetscCall(PicurvPhysicalTime(simCtx, simCtx->ti, &physical_time));
170 for (PetscInt bi = 0; bi < simCtx->block_number; ++bi) {
171 UserCtx *user = &user_finest[bi];
172 const FieldId staggered_fields[] = {FIELD_ID_UCONT};
173 if (field->source.bulk) {
174 PetscCall(VecSet(user->Ucat, 0.0));
175 PetscCall(PicurvSourceCellBulkFactor(user, &field->source, field->fraction[bi], field->length[bi], physical_time, &field->factor));
176 }
177 PetscCall(VecSet(user->Ucont, 0.0));
178 PetscCall(PicurvSourceAddFluxes(user, &field->source, field->fraction[bi], field->length[bi], physical_time, field->factor, PETSC_TRUE));
179 PetscCall(SetBoundaryVelocityFromSource(user, field, bi, physical_time));
180 /* A curvilinear source's Ucat follows from its fluxes, below. */
181 PetscCall(PicurvSourceSetCellFields(user, (field->source.frame != PICURV_SOURCE_FRAME_CURVILINEAR) ? &field->source : NULL,
182 field->factor, field->pressure, field->fraction[bi], field->length[bi], physical_time));
183 if (!field->pressure) PetscCall(VecSet(user->P, 0.0));
184 PetscCall(SynchronizePeriodicStaggeredFields(user, 1, staggered_fields));
185 PetscCall(UpdateLocalGhosts(user, FIELD_ID_UCONT));
186 if (field->source.frame == PICURV_SOURCE_FRAME_CURVILINEAR) PetscCall(Contra2Cart(user));
187 PetscCall(FinalizeCellFieldsFromBoundaryValues(user));
188 }
189 field->evaluated = PETSC_TRUE;
190 PROFILE_FUNCTION_END;
191 PetscFunctionReturn(0);
192}

194/**
195 * @brief Implementation of \ref SetAnalyticalSolutionForParticles().
196 * @details Only a pointwise source reaches here; see AnalyticalFieldSupportsInterpolationError().
197 * @see SetAnalyticalSolutionForParticles()
198 */
199PetscErrorCode SetAnalyticalSolutionForParticles(Vec tempVec, SimCtx *simCtx)
200{
201 const struct AnalyticalField *field = simCtx->analyticalField;
202 const ScalingCtx *scaling = &simCtx->scaling;
203 PetscInt nLocal;
204 PetscReal *data, values[PICURV_SOURCE_VAR_COUNT], physical_time;
205
206 PetscFunctionBeginUser;
207 PROFILE_HELPER_BEGIN;
208 PetscCheck(AnalyticalFieldSupportsInterpolationError(simCtx), PETSC_COMM_SELF, PETSC_ERR_ARG_WRONGSTATE,
209 "The analytical field cannot be evaluated at a particle: it needs a Cartesian source in x, y, z and t.");
210 PetscCall(PetscArrayzero(values, PICURV_SOURCE_VAR_COUNT));
211 PetscCall(PicurvPhysicalTime(simCtx, simCtx->ti, &physical_time));
212 values[PICURV_SOURCE_VAR_T] = physical_time;
213 PetscCall(VecGetLocalSize(tempVec, &nLocal));
214 PetscCall(VecGetArray(tempVec, &data));
215 /* Interleaved [x0,y0,z0, x1,y1,z1, ...], replaced by the velocity there. */
216 for (PetscInt i = 0; i < nLocal; i += 3) {
217 Cmpnts v;
218 values[PICURV_SOURCE_VAR_X] = data[i] * scaling->L_ref;
219 values[PICURV_SOURCE_VAR_Y] = data[i + 1] * scaling->L_ref;
220 values[PICURV_SOURCE_VAR_Z] = data[i + 2] * scaling->L_ref;
221 PetscCall(PicurvVelocitySourceEvaluate(&field->source, values, NULL, NULL, &v));
222 data[i] = v.x / scaling->U_ref;
223 data[i + 1] = v.y / scaling->U_ref;
224 data[i + 2] = v.z / scaling->U_ref;
225 }
226 PetscCall(VecRestoreArray(tempVec, &data));
227 PROFILE_HELPER_END;
228 PetscFunctionReturn(0);
229}
