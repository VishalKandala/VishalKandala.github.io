217/**
218 * @brief Rescales the minimal fixture's unit metrics to cubic cells of edge h.
219 */
220static PetscErrorCode ScaleMinimalMetricsToSpacing(UserCtx *user, PetscReal h)
221{
222 Vec area_vectors[12] = {user->Csi, user->Eta, user->Zet, user->ICsi, user->IEta, user->IZet,
223 user->JCsi, user->JEta, user->JZet, user->KCsi, user->KEta, user->KZet};
224 Vec local_area[12] = {user->lCsi, user->lEta, user->lZet, user->lICsi, user->lIEta, user->lIZet,
225 user->lJCsi, user->lJEta, user->lJZet, user->lKCsi, user->lKEta, user->lKZet};
226 Vec jacobians[4] = {user->Aj, user->IAj, user->JAj, user->KAj};
227 Vec local_jac[4] = {user->lAj, user->lIAj, user->lJAj, user->lKAj};
228
229 PetscFunctionBeginUser;
230 for (PetscInt v = 0; v < 12; ++v) {
231 PetscCall(VecScale(area_vectors[v], h * h)); /* a face of a cube of edge h */
232 PetscCall(DMGlobalToLocalBegin(user->fda, area_vectors[v], INSERT_VALUES, local_area[v]));
233 PetscCall(DMGlobalToLocalEnd(user->fda, area_vectors[v], INSERT_VALUES, local_area[v]));
234 }
235 for (PetscInt v = 0; v < 4; ++v) {
236 PetscCall(VecSet(jacobians[v], 1.0 / (h * h * h)));
237 PetscCall(DMGlobalToLocalBegin(user->da, jacobians[v], INSERT_VALUES, local_jac[v]));
238 PetscCall(DMGlobalToLocalEnd(user->da, jacobians[v], INSERT_VALUES, local_jac[v]));
239 }
240 PetscFunctionReturn(0);
241}

243/**
244 * @brief Returns the Clark contribution over the molecular viscous one, per unit x, at one cell.
245 *
246 * @details The field is u_x = x^2 on cubic cells of edge h, so du/dx = 2x and the exact
247 * gradient-model stress tau_11 = (h^2/12)(du/dx)^2 varies along x. Both the
248 * model's and viscosity's contributions leave Viscous() through the same flux
249 * differences, so their ratio does not depend on how Viscous() normalizes its
250 * output.
251 */
252static PetscErrorCode ClarkToMolecularRatioPerX(PetscReal h, PetscInt probe, PetscReal *ratio_per_x)
253{
254 SimCtx *simCtx = NULL;
255 UserCtx *user = NULL;
256 Vec molecular = NULL, with_clark = NULL;
257 Cmpnts ***ucat = NULL, ***visc0 = NULL, ***visc1 = NULL;
258 const PetscInt n = 10, mid = n / 2;
259
260 PetscFunctionBeginUser;
261 PetscCall(PicurvCreateMinimalContexts(&simCtx, &user, n, n, n));
262 PetscCall(ScaleMinimalMetricsToSpacing(user, h));
263 simCtx->ren = 1.0;
264 simCtx->les = NO_LES_MODEL;
265 simCtx->rans = 0;
266
267 PetscCall(DMDAVecGetArray(user->fda, user->Ucat, &ucat));
268 for (PetscInt k = user->info.zs; k < user->info.zs + user->info.zm; ++k)
269 for (PetscInt j = user->info.ys; j < user->info.ys + user->info.ym; ++j)
270 for (PetscInt i = user->info.xs; i < user->info.xs + user->info.xm; ++i) {
271 const PetscReal x = (i - 0.5) * h; /* cell centre */
272 ucat[k][j][i].x = x * x; ucat[k][j][i].y = 0.0; ucat[k][j][i].z = 0.0;
273 }
274 PetscCall(DMDAVecRestoreArray(user->fda, user->Ucat, &ucat));
275 PetscCall(DMGlobalToLocalBegin(user->fda, user->Ucat, INSERT_VALUES, user->lUcat));
276 PetscCall(DMGlobalToLocalEnd(user->fda, user->Ucat, INSERT_VALUES, user->lUcat));
277 PetscCall(VecSet(user->lUcont, 0.0));
278
279 PetscCall(VecDuplicate(user->lUcont, &molecular));
280 PetscCall(VecDuplicate(user->lUcont, &with_clark));
281 simCtx->les_gradient_model = 0;
282 PetscCall(Viscous(user, user->lUcont, user->lUcat, molecular));
283 simCtx->les_gradient_model = 1;
284 PetscCall(Viscous(user, user->lUcont, user->lUcat, with_clark));
285
286 PetscCall(DMDAVecGetArrayRead(user->fda, molecular, &visc0));
287 PetscCall(DMDAVecGetArrayRead(user->fda, with_clark, &visc1));
288 {
289 const PetscReal v_molecular = visc0[mid][mid][probe].x;
290 const PetscReal v_clark = visc1[mid][mid][probe].x - v_molecular;
291 PetscCheck(PetscAbsReal(v_molecular) > 0.0, PETSC_COMM_SELF, PETSC_ERR_PLIB,
292 "The molecular viscous term vanished at the probe; the fixture is wrong.");
293 *ratio_per_x = (v_clark / v_molecular) / ((probe - 0.5) * h);
294 }
295 PetscCall(DMDAVecRestoreArrayRead(user->fda, with_clark, &visc1));
296 PetscCall(DMDAVecRestoreArrayRead(user->fda, molecular, &visc0));
297 PetscCall(VecDestroy(&with_clark));
298 PetscCall(VecDestroy(&molecular));
299 PetscCall(PicurvDestroyMinimalContexts(&simCtx, &user));
300 PetscFunctionReturn(0);
301}

303/**
304 * @brief Tests that the Clark gradient model scales as the square of the filter width.
305 *
306 * @details The gradient-model stress is tau_ij = (Delta_k^2/12) du_i/dx_k du_j/dx_k, so
307 * for a fixed physical field its force grows as Delta^2 relative to the
308 * molecular viscous force. Halving the spacing must therefore divide the
309 * Clark-to-molecular ratio at a given x by four. Fixtures with unit cells hide
310 * any error in the power of Delta, which is why the spacing is varied here.
311 */
312static PetscErrorCode TestClarkGradientModelScalesWithFilterWidthSquared(void)
313{
314 PetscReal coarse = 0.0, fine = 0.0;
315
316 PetscFunctionBeginUser;
317 /* Probe cells chosen so the two spacings give comparable x without being on a
318 boundary: cell 5 at h = 0.2 and cell 5 at h = 0.1. The ratio is per unit x. */
319 PetscCall(ClarkToMolecularRatioPerX(0.2, 5, &coarse));
320 PetscCall(ClarkToMolecularRatioPerX(0.1, 5, &fine));
321 PetscCall(PicurvAssertBool((PetscBool)(coarse < 0.0 && fine < 0.0),
322 "the gradient model removes momentum where du/dx grows along x"));
323 PetscCall(PicurvAssertRealNear(4.0, coarse / fine, 1.0e-8,
324 "halving the filter width divides the gradient-model force by four"));
325 /* Exact value: the molecular flux carries both halves of the symmetric stress,
326 2 nu du/dx per unit area, so its difference across a cell is 4 nu h^3 for
327 u = x^2; the model's is -(2/3) h^5 x. Their ratio per unit x is -h^2/(6 nu),
328 which pins the 1/12 coefficient as well as the power of Delta. */
329 PetscCall(PicurvAssertRealNear(-0.2 * 0.2 / 6.0, coarse, 1.0e-10,
330 "gradient-model force matches (Delta^2/12) du/dx du/dx at h = 0.2"));
331 PetscCall(PicurvAssertRealNear(-0.1 * 0.1 / 6.0, fine, 1.0e-10,
332 "gradient-model force matches (Delta^2/12) du/dx du/dx at h = 0.1"));
333 PetscFunctionReturn(0);
334}
