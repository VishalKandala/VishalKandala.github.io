34/** @brief Exercises every Eulerian transformation currently dispatched by the pipeline. */
35static PetscErrorCode TestEulerianPipelineDecompositionIndependent(void)
36{
37 SimCtx *simCtx = NULL;
38 UserCtx *user = NULL;
39 PostProcessParams *pps = NULL;
40 PetscReal ***pressure = NULL;
41 Cmpnts ***velocity = NULL;
42 const PetscReal ***pressure_result = NULL;
43 const PetscReal ***pressure_nodal = NULL;
44 const PetscReal ***qcrit = NULL;
45 const Cmpnts ***velocity_nodal = NULL;
46 PetscReal pressure_error = 0.0;
47 PetscReal pressure_nodal_error = 0.0;
48 PetscReal velocity_nodal_error = 0.0;
49 PetscReal qcrit_error = 0.0;
50 PetscInt local_physical_nodes = 0;
51 PetscInt global_physical_nodes = 0;
52 PetscInt local_q_points = 0;
53 PetscInt global_q_points = 0;
54
55 PetscFunctionBeginUser;
56 PetscCall(PicurvCreateMinimalContexts(&simCtx, &user, 8, 6, 4));
57 PetscCall(PetscCalloc1(1, &simCtx->pps));
58 pps = simCtx->pps;
59 pps->reference[0] = 1;
60 pps->reference[1] = 1;
61 pps->reference[2] = 1;
62 simCtx->scaling.L_ref = 1.0;
63 simCtx->scaling.U_ref = 1.0;
64 simCtx->scaling.rho_ref = 2.0;
65 simCtx->scaling.P_ref = 2.0;
66
67 PetscCall(DMDAVecGetArray(user->da, user->P, &pressure));
68 PetscCall(DMDAVecGetArray(user->fda, user->Ucat, &velocity));
69 for (PetscInt k = user->info.zs; k < user->info.zs + user->info.zm; ++k) {
70 for (PetscInt j = user->info.ys; j < user->info.ys + user->info.ym; ++j) {
71 for (PetscInt i = user->info.xs; i < user->info.xs + user->info.xm; ++i) {
72 pressure[k][j][i] = ScalarValue(i, j, k);
73 velocity[k][j][i] = (Cmpnts){1.0, 2.0, 3.0};
74 }
75 }
76 }
77 PetscCall(DMDAVecRestoreArray(user->fda, user->Ucat, &velocity));
78 PetscCall(DMDAVecRestoreArray(user->da, user->P, &pressure));
79 PetscCall(VecSet(user->P_nodal, kSentinel));
80 PetscCall(VecSet(user->Ucat_nodal, kSentinel));
81 PetscCall(VecSet(user->Qcrit, kSentinel));
82
83 /* A dimensionalizing post-processor scales pressure as it is loaded, by rho U^2 = 2,
84 * before the pipeline runs; normalization is linear, so the expected values are
85 * the same whichever of the two is applied first. */
86 PetscCall(DimensionalizeField(user, "P"));
87 PetscCall(PetscStrncpy(
88 pps->process_pipeline,
89 "NormalizeRelativeField:P;"
90 "CellToNodeAverage:P>P_nodal;CellToNodeAverage:Ucat>Ucat_nodal;"
91 "ComputeQCriterion",
92 sizeof(pps->process_pipeline)));
93 PetscCall(EulerianDataProcessingPipeline(user, pps));
94
95 PetscCall(DMDAVecGetArrayRead(user->da, user->P, &pressure_result));
96 PetscCall(DMDAVecGetArrayRead(user->da, user->P_nodal, &pressure_nodal));
97 PetscCall(DMDAVecGetArrayRead(user->fda, user->Ucat_nodal, &velocity_nodal));
98 PetscCall(DMDAVecGetArrayRead(user->da, user->Qcrit, &qcrit));
99 for (PetscInt k = user->info.zs; k < user->info.zs + user->info.zm; ++k) {
100 for (PetscInt j = user->info.ys; j < user->info.ys + user->info.ym; ++j) {
101 for (PetscInt i = user->info.xs; i < user->info.xs + user->info.xm; ++i) {
102 const PetscBool physical_node =
103 (PetscBool)(i < user->info.mx - 1 && j < user->info.my - 1 &&
104 k < user->info.mz - 1);
105 const PetscBool q_point =
106 (PetscBool)(i >= 1 && i < user->info.mx - 1 &&
107 j >= 1 && j < user->info.my - 1 &&
108 k >= 1 && k < user->info.mz - 1);
109 const PetscReal expected_pressure = 2.0 * (ScalarValue(i, j, k) - 111.0);
110
111 pressure_error = PetscMax(
112 pressure_error,
113 PetscAbsReal(pressure_result[k][j][i] - expected_pressure));
114 if (physical_node) {
115 const PetscReal expected_nodal =
116 2.0 * (ScalarValue(i, j, k) + 55.5 - 111.0);
117
118 pressure_nodal_error = PetscMax(
119 pressure_nodal_error,
120 PetscAbsReal(pressure_nodal[k][j][i] - expected_nodal));
121 velocity_nodal_error = PetscMax(
122 velocity_nodal_error,
123 PetscAbsReal(velocity_nodal[k][j][i].x - 1.0));
124 velocity_nodal_error = PetscMax(
125 velocity_nodal_error,
126 PetscAbsReal(velocity_nodal[k][j][i].y - 2.0));
127 velocity_nodal_error = PetscMax(
128 velocity_nodal_error,
129 PetscAbsReal(velocity_nodal[k][j][i].z - 3.0));
130 ++local_physical_nodes;
131 } else {
132 pressure_nodal_error = PetscMax(
133 pressure_nodal_error,
134 PetscAbsReal(pressure_nodal[k][j][i] - kSentinel));
135 velocity_nodal_error = PetscMax(
136 velocity_nodal_error,
137 PetscAbsReal(velocity_nodal[k][j][i].x - kSentinel));
138 velocity_nodal_error = PetscMax(
139 velocity_nodal_error,
140 PetscAbsReal(velocity_nodal[k][j][i].y - kSentinel));
141 velocity_nodal_error = PetscMax(
142 velocity_nodal_error,
143 PetscAbsReal(velocity_nodal[k][j][i].z - kSentinel));
144 }
145 qcrit_error = PetscMax(
146 qcrit_error,
147 PetscAbsReal(qcrit[k][j][i] - (q_point ? 0.0 : kSentinel)));
148 if (q_point) ++local_q_points;
149 }
150 }
151 }
152 PetscCall(DMDAVecRestoreArrayRead(user->da, user->Qcrit, &qcrit));
153 PetscCall(DMDAVecRestoreArrayRead(user->fda, user->Ucat_nodal, &velocity_nodal));
154 PetscCall(DMDAVecRestoreArrayRead(user->da, user->P_nodal, &pressure_nodal));
155 PetscCall(DMDAVecRestoreArrayRead(user->da, user->P, &pressure_result));
156
157 PetscCallMPI(MPI_Allreduce(&local_physical_nodes, &global_physical_nodes, 1,
158 MPIU_INT, MPI_SUM, PETSC_COMM_WORLD));
159 PetscCallMPI(MPI_Allreduce(&local_q_points, &global_q_points, 1,
160 MPIU_INT, MPI_SUM, PETSC_COMM_WORLD));
161 PetscCall(PicurvAssertIntEqual(user->IM * user->JM * user->KM,
162 global_physical_nodes,
163 "the Eulerian pipeline must visit every physical node exactly once"));
164 PetscCall(PicurvAssertIntEqual((user->IM - 1) * (user->JM - 1) * (user->KM - 1),
165 global_q_points,
166 "Q-criterion must visit every interior cell exactly once"));
167 PetscCall(AssertGlobalError(pressure_error, 1.0e-12,
168 "pressure normalization and dimensionalization are decomposition independent"));
169 PetscCall(AssertGlobalError(pressure_nodal_error, 1.0e-12,
170 "scalar nodal averaging covers rank interfaces and physical boundaries"));
171 PetscCall(AssertGlobalError(velocity_nodal_error, 1.0e-12,
172 "vector nodal averaging covers rank interfaces and physical boundaries"));
173 PetscCall(AssertGlobalError(qcrit_error, 1.0e-12,
174 "Q-criterion covers the complete distributed interior"));
175
176 PetscCall(PicurvDestroyMinimalContexts(&simCtx, &user));
177 PetscFunctionReturn(0);
178}

180/** @brief Verifies the production-created derived swarm mirrors source ownership. */
181static PetscErrorCode TestParticlePipelineMatchesSourceOwnership(void)
182{
183 SimCtx *simCtx = NULL;
184 UserCtx *user = NULL;
185 PostProcessParams pps;
186 PetscMPIInt rank = 0, size = 1;
187 PetscInt source_local = 0;
188 PetscInt source_global = 0;
189 PetscInt post_local = 0;
190 PetscInt post_global = 0;
191 PetscInt offset = 0;
192 PetscScalar (*velocity)[3] = NULL;
193 const PetscScalar *ske = NULL;
194 PetscReal local_error = 0.0;
195 const PetscInt total_particles = 7;
196
197 PetscFunctionBeginUser;
198 PetscCallMPI(MPI_Comm_rank(PETSC_COMM_WORLD, &rank));
199 PetscCallMPI(MPI_Comm_size(PETSC_COMM_WORLD, &size));
200 source_local = total_particles / size + (rank < total_particles % size ? 1 : 0);
201 PetscCall(PetscMemzero(&pps, sizeof(pps)));
202 PetscCall(PicurvCreateMinimalContexts(&simCtx, &user, 4, 4, 4));
203 PetscCall(PicurvCreateSwarmPair(user, source_local, "unused"));
204 PetscCall(DMDestroy(&user->post_swarm));
205 PetscCall(PetscStrncpy(pps.particle_pipeline, "ComputeSpecificKE:velocity>ske",
206 sizeof(pps.particle_pipeline)));
207 PetscCall(SetupPostProcessSwarm(user, &pps));
208
209 if (size > 1) {
210 PetscCallMPI(MPI_Exscan(&source_local, &offset, 1, MPIU_INT, MPI_SUM,
211 PETSC_COMM_WORLD));
212 if (rank == 0) offset = 0;
213 }
214 PetscCall(DMSwarmGetField(user->swarm, "velocity", NULL, NULL, (void **)&velocity));
215 for (PetscInt p = 0; p < source_local; ++p) {
216 const PetscReal tag = (PetscReal)(offset + p + 1);
217
218 velocity[p][0] = tag;
219 velocity[p][1] = 2.0;
220 velocity[p][2] = -1.0;
221 }
222 PetscCall(DMSwarmRestoreField(user->swarm, "velocity", NULL, NULL,
223 (void **)&velocity));
224
225 PetscCall(ParticleDataProcessingPipeline(user, &pps));
226 PetscCall(DMSwarmGetLocalSize(user->swarm, &source_local));
227 PetscCall(DMSwarmGetSize(user->swarm, &source_global));
228 PetscCall(DMSwarmGetLocalSize(user->post_swarm, &post_local));
229 PetscCall(DMSwarmGetSize(user->post_swarm, &post_global));
230 PetscCall(PicurvAssertIntEqual(source_local, post_local,
231 "derived particle fields must share source local ownership"));
232 PetscCall(PicurvAssertIntEqual(source_global, post_global,
233 "derived and source swarms must have the same global size"));
234
235 PetscCall(DMSwarmGetField(user->post_swarm, "ske", NULL, NULL, (void **)&ske));
236 for (PetscInt p = 0; p < source_local; ++p) {
237 const PetscReal tag = (PetscReal)(offset + p + 1);
238 const PetscReal expected = 0.5 * (tag * tag + 5.0);
239
240 local_error = PetscMax(local_error,
241 PetscAbsReal(PetscRealPart(ske[p]) - expected));
242 }
243 PetscCall(DMSwarmRestoreField(user->post_swarm, "ske", NULL, NULL, (void **)&ske));
244 PetscCall(AssertGlobalError(local_error, 1.0e-12,
245 "specific kinetic energy must match every source particle"));
246
247 PetscCall(PicurvDestroyMinimalContexts(&simCtx, &user));
248 PetscFunctionReturn(0);
249}

367/**
368 * @brief A wall-normal profile CSV holds each plane's pooled moments on any decomposition.
369 *
370 * One state of a field varying across every plane is merged over i and k. Each written
371 * row must carry the mean and RMS over that plane's points, computed here directly
372 * from the closed form, and the plane's mean cell-centre coordinate.
373 */
374static PetscErrorCode TestFieldStatisticsProfileMultiRank(void)
375{
376 SimCtx *simCtx = NULL;
377 UserCtx *user = NULL;
378 PicurvWindowDefinition definition;
379 PicurvWindowStorage storage;
380 SpatialTargetPlan plan;
381 const PetscBool over_ik[3] = {PETSC_TRUE, PETSC_FALSE, PETSC_TRUE};
382 PetscReal ***pressure = NULL, sums[3][64], reduced[3][64];
383 PetscMPIInt rank = 0;
384 char tmpdir[PETSC_MAX_PATH_LEN], prefix[PETSC_MAX_PATH_LEN + 16], path[PETSC_MAX_PATH_LEN + 64];
385
386 PetscFunctionBeginUser;
387 PetscCallMPI(MPI_Comm_rank(PETSC_COMM_WORLD, &rank));
388 PetscCall(PetscMemzero(&definition, sizeof(definition)));
389 PetscCall(PetscStrncpy(definition.name, "plane", sizeof(definition.name)));
390 definition.weighting = PICURV_WEIGHTING_SAMPLE;
391 definition.cadence_kind = PICURV_CADENCE_STEP;
392 definition.step_cadence = 1;
393 definition.field_count = 1;
394 definition.fields[0].field_id = FIELD_ID_P;
395 definition.fields[0].want_second = PETSC_TRUE;
396
397 PetscCall(PicurvCreateMinimalContexts(&simCtx, &user, 8, 6, 4));
398 PetscCall(VecSet(user->Nvert, 0.0));
399 PetscCall(VecSet(user->lNvert, 0.0));
400 PetscCall(PicurvPopulateUniformCellCenters(user));
401 PetscCall(PicurvWindowStorageCreate(user, &definition, &storage));
402 PetscCall(DMDAVecGetArray(user->da, user->P, &pressure));
403 for (PetscInt k = user->info.zs; k < user->info.zs + user->info.zm; ++k)
404 for (PetscInt j = user->info.ys; j < user->info.ys + user->info.ym; ++j)
405 for (PetscInt i = user->info.xs; i < user->info.xs + user->info.xm; ++i)
406 pressure[k][j][i] = ScalarValue(i, j, k);
407 PetscCall(DMDAVecRestoreArray(user->da, user->P, &pressure));
408 PetscCall(PicurvWindowAccumulate(user, &definition, &storage, 1.0));
409
410 /* The expected profile, straight from the closed form over each rank's points. */
411 PetscCall(SpatialTargetPlanCreate(user, FIELD_ID_P, PICURV_STATISTICS_MASK_FLUID, &plan));
412 PetscCall(PetscMemzero(sums, sizeof(sums)));
413 for (PetscInt k = plan.start[2]; k < plan.end[2]; ++k)
414 for (PetscInt j = plan.start[1]; j < plan.end[1]; ++j)
415 for (PetscInt i = plan.start[0]; i < plan.end[0]; ++i) {
416 sums[0][j] += 1.0;
417 sums[1][j] += ScalarValue(i, j, k);
418 sums[2][j] += PetscSqr(ScalarValue(i, j, k));
419 }
420 PetscCallMPI(MPI_Allreduce(sums, reduced, 3 * 64, MPIU_REAL, MPI_SUM, PETSC_COMM_WORLD));
421
422 if (rank == 0) PetscCall(PicurvMakeTempDir(tmpdir, sizeof(tmpdir)));
423 PetscCallMPI(MPI_Bcast(tmpdir, PETSC_MAX_PATH_LEN, MPI_CHAR, 0, PETSC_COMM_WORLD));
424 PetscCall(PetscSNPrintf(prefix, sizeof(prefix), "%s/Field", tmpdir));
425 PetscCall(ComputeWindowStatisticsProfile(user, &definition, &storage, "mean,rms", over_ik, prefix, 7));
426
427 if (rank == 0) {
428 char line[1024];
429 PetscInt rows = 0, expected_rows = 0;
430 FILE *csv = NULL;
431
432 PetscCall(PetscSNPrintf(path, sizeof(path), "%s_profile_plane_00007.csv", prefix));
433 csv = fopen(path, "r");
434 PetscCheck(csv != NULL, PETSC_COMM_SELF, PETSC_ERR_FILE_OPEN, "Cannot open %s", path);
435 PetscCheck(fgets(line, sizeof(line), csv) != NULL, PETSC_COMM_SELF, PETSC_ERR_FILE_READ, "Empty profile");
436 PetscCall(PicurvAssertBool((PetscBool)(strcmp(line, "j,x,y,z,plane_P_mean,plane_P_rms\n") == 0),
437 "the header names the retained index, coordinates and outputs"));
438 while (fgets(line, sizeof(line), csv)) {
439 int cell = -1;
440 double x, y, z, mean, rms;
441 PetscReal expected_mean, expected_rms;
442
443 PetscCheck(sscanf(line, "%d,%lf,%lf,%lf,%lf,%lf", &cell, &x, &y, &z, &mean, &rms) == 6,
444 PETSC_COMM_SELF, PETSC_ERR_FILE_READ, "Malformed profile row: %s", line);
445 expected_mean = reduced[1][cell + 1] / reduced[0][cell + 1];
446 expected_rms = PetscSqrtReal(reduced[2][cell + 1] / reduced[0][cell + 1] - PetscSqr(expected_mean));
447 PetscCall(PicurvAssertRealNear(expected_mean, mean, 1.0e-9, "the row mean is the plane mean"));
448 PetscCall(PicurvAssertRealNear(expected_rms, rms, 1.0e-9, "the row RMS is the plane's pooled RMS"));
449 PetscCall(PicurvAssertRealNear((cell + 0.5) / (PetscReal)user->JM, y, 1.0e-9,
450 "the row coordinate is the plane's cell centre"));
451 rows++;
452 }
453 fclose(csv);
454 for (PetscInt j = 0; j < 64; ++j) if (reduced[0][j] > 0.0) expected_rows++;
455 PetscCall(PicurvAssertIntEqual(expected_rows, rows, "one row per sampled plane"));
456 PetscCall(PicurvRemoveTempDir(tmpdir));
457 }
458
459 PetscCall(PicurvWindowStorageDestroy(&storage));
460 PetscCall(PicurvDestroyMinimalContexts(&simCtx, &user));
461 PetscFunctionReturn(0);
462}

464/**

477 PicurvWindowStorage storage;
478 SpatialTargetPlan plan;

480 PetscReal sums[2][64], reduced[2][64];
481 PetscMPIInt rank = 0;
482 char tmpdir[PETSC_MAX_PATH_LEN], prefix[PETSC_MAX_PATH_LEN + 16], path[PETSC_MAX_PATH_LEN + 64];
483
484 PetscFunctionBeginUser;

551 for (PetscInt j = 0; j < 64; ++j) if (reduced[0][j] > 0.0) expected_rows++;
552 PetscCall(PicurvAssertIntEqual(expected_rows, rows, "one row per sampled plane"));
553 PetscCall(PicurvRemoveTempDir(tmpdir));
554 }
555
556 PetscCall(PicurvWindowStorageDestroy(&storage));
557 PetscCall(PicurvDestroyMinimalContexts(&simCtx, &user));
558 PetscFunctionReturn(0);
