22/** @brief Builds a definition requesting Ucat and P with second moments. */
23static PicurvWindowDefinition AccDefinition(PetscBool want_second)
24{
25 PicurvWindowDefinition d;
26 memset(&d, 0, sizeof(d));
27 strncpy(d.name, "acc", PICURV_WINDOW_NAME_LENGTH - 1);
28 d.weighting = PICURV_WEIGHTING_SAMPLE;
29 d.cadence_kind = PICURV_CADENCE_STEP;
30 d.step_cadence = 1;
31 d.field_count = 2;
32 d.fields[0].field_id = FIELD_ID_UCAT; d.fields[0].want_second = want_second;
33 d.fields[1].field_id = FIELD_ID_P; d.fields[1].want_second = want_second;
34 return d;
35}

37/** @brief Sets Ucat to a uniform vector and P to a uniform scalar. */
38static PetscErrorCode SetUniform(UserCtx *user, PetscReal x, PetscReal y, PetscReal z, PetscReal p)
39{
40 Cmpnts ***ucat = NULL;
41 PetscReal ***pp = NULL;
42 const DMDALocalInfo info = user->info;
43
44 PetscFunctionBeginUser;
45 PetscCall(DMDAVecGetArray(user->fda, user->Ucat, &ucat));
46 PetscCall(DMDAVecGetArray(user->da, user->P, &pp));
47 for (PetscInt k = info.zs; k < info.zs + info.zm; ++k)
48 for (PetscInt j = info.ys; j < info.ys + info.ym; ++j)
49 for (PetscInt i = info.xs; i < info.xs + info.xm; ++i) {
50 ucat[k][j][i].x = x; ucat[k][j][i].y = y; ucat[k][j][i].z = z;
51 pp[k][j][i] = p;
52 }
53 PetscCall(DMDAVecRestoreArray(user->da, user->P, &pp));
54 PetscCall(DMDAVecRestoreArray(user->fda, user->Ucat, &ucat));
55 PetscFunctionReturn(0);
56}

145/** @brief Three accepted states reproduce the analytically known moments at every point. */
146static PetscErrorCode TestKnownAccumulationAcrossField(void)
147{
148 SimCtx *simCtx = NULL;
149 UserCtx *user = NULL;
150 PicurvWindowStorage storage;
151 PicurvWindowDefinition d = AccDefinition(PETSC_TRUE);
152 const PetscReal series[3][3] = {{1.0, 2.0, 3.0}, {3.0, 6.0, 5.0}, {5.0, 4.0, 7.0}};
153 const PetscReal scalars[3] = {1.0, 2.0, 6.0};
154 const PetscReal expected_product[6] = {8.0, 4.0, 8.0, 8.0, 4.0, 8.0};
155 const char *labels[6] = {"xx", "xy", "xz", "yy", "yz", "zz"};
156 Cmpnts ***mean_vec = NULL;
157 PetscReal value = 0.0;
158 char context[128];
159
160 PetscFunctionBeginUser;
161 PetscCall(PicurvCreateMinimalContexts(&simCtx, &user, ACC_N, ACC_N, ACC_N));
162 PetscCall(VecSet(user->Nvert, 0.0));
163 PetscCall(PicurvWindowStorageCreate(user, &d, &storage));
164
165 for (PetscInt s = 0; s < 3; ++s) {
166 PetscCall(SetUniform(user, series[s][0], series[s][1], series[s][2], scalars[s]));
167 PetscCall(PicurvWindowAccumulate(user, &d, &storage, 1.0));
168 }
169
170 /* Occupancy: every fluid point saw all three states. */
171 PetscCall(ReadScalarAt(user, storage.count, 2, 2, 2, &value));
172 PetscCall(PicurvAssertRealNear(3.0, value, 1.0e-12, "per-point sample count"));
173 PetscCall(ReadScalarAt(user, storage.weight, 2, 2, 2, &value));
174 PetscCall(PicurvAssertRealNear(3.0, value, 1.0e-12, "per-point total weight"));
175 PetscCall(ReadScalarAt(user, storage.weight_sq, 2, 2, 2, &value));
176 PetscCall(PicurvAssertRealNear(3.0, value, 1.0e-12, "per-point squared-weight sum"));
177
178 /* Vector mean. */
179 PetscCall(DMDAVecGetArrayRead(user->fda, storage.mean[0], &mean_vec));
180 PetscCall(PicurvAssertRealNear(3.0, mean_vec[2][2][2].x, 1.0e-12, "Ucat mean x"));
181 PetscCall(PicurvAssertRealNear(4.0, mean_vec[2][2][2].y, 1.0e-12, "Ucat mean y"));
182 PetscCall(PicurvAssertRealNear(5.0, mean_vec[2][2][2].z, 1.0e-12, "Ucat mean z"));
183 PetscCall(DMDAVecRestoreArrayRead(user->fda, storage.mean[0], &mean_vec));
184
185 /* All six symmetric centered products, in fixed catalog order. */
186 for (PetscInt c = 0; c < 6; ++c) {
187 PetscCall(ReadComponentAt(user, storage.m2[0], 6, c, 2, 2, 2, &value));
188 PetscCall(PetscSNPrintf(context, sizeof(context), "Ucat centered product %s", labels[c]));
189 PetscCall(PicurvAssertRealNear(expected_product[c], value, 1.0e-11, context));
190 }
191
192 /* Scalar mean and second moment. */
193 PetscCall(ReadScalarAt(user, storage.mean[1], 2, 2, 2, &value));
194 PetscCall(PicurvAssertRealNear(3.0, value, 1.0e-12, "P mean"));
195 PetscCall(ReadScalarAt(user, storage.m2[1], 2, 2, 2, &value));
196 PetscCall(PicurvAssertRealNear(14.0, value, 1.0e-11, "P centered second moment"));
197
198 PetscCall(PicurvWindowStorageDestroy(&storage));
199 PetscCall(PicurvDestroyMinimalContexts(&simCtx, &user));
200 PetscFunctionReturn(0);
201}

209 * has to surface it.

212{

288static PetscErrorCode TestCovarianceAccumulation(void)
289{
290 SimCtx *simCtx = NULL;
291 UserCtx *user = NULL;
292 PicurvWindowStorage storage;
293 PicurvWindowDefinition d = AccDefinition(PETSC_TRUE);
294 const PetscReal series[3][3] = {{1.0, 2.0, 3.0}, {3.0, 6.0, 5.0}, {5.0, 4.0, 7.0}};
295 const PetscReal scalars[3] = {1.0, 2.0, 6.0};
296 /* Means are (3,4,5) and 3, so the centered sums are
297 * x: (-2)(-2)+(0)(-1)+(2)(3) = 10, y: (-2)(-2)+(2)(-1)+(0)(3) = 2,
298 * z: (-2)(-2)+(0)(-1)+(2)(3) = 10. */

302 char context[128];
303
304 PetscFunctionBeginUser;
305 d.covariance_count = 1;
306 d.covariances[0].first = FIELD_ID_UCAT;
307 d.covariances[0].second = FIELD_ID_P;
308
309 PetscCall(PicurvCreateMinimalContexts(&simCtx, &user, ACC_N, ACC_N, ACC_N));
310 PetscCall(VecSet(user->Nvert, 0.0));
311 PetscCall(PicurvWindowStorageCreate(user, &d, &storage));
312 {
313 PetscInt bs = 0;
314 PetscCall(VecGetBlockSize(storage.cm[0], &bs));
315 PetscCall(PicurvAssertIntEqual(3, bs, "a vector-scalar pair is one three-component vector"));
316 }
317
318 for (PetscInt s = 0; s < 3; ++s) {
319 PetscCall(SetUniform(user, series[s][0], series[s][1], series[s][2], scalars[s]));
320 PetscCall(PicurvWindowAccumulate(user, &d, &storage, 1.0));
321 }
322
323 for (PetscInt c = 0; c < 3; ++c) {
324 PetscCall(ReadComponentAt(user, storage.cm[0], 3, c, 2, 2, 2, &value));

326 PetscCall(PicurvAssertRealNear(expected[c], value, 1.0e-11, context));

331 PetscCall(PicurvAssertRealNear(3.0, value, 1.0e-12, "P mean is unaffected by the covariance pass"));

333 PetscCall(PicurvAssertRealNear(8.0, value, 1.0e-11, "Ucat xx product is unaffected by the covariance pass"));
334
335 PetscCall(PicurvWindowStorageDestroy(&storage));
336 PetscCall(PicurvDestroyMinimalContexts(&simCtx, &user));
337 PetscFunctionReturn(0);

411/**
412 * @brief Derived quantities reproduce the analytic values the accumulated state implies.
413 *
414 * The series is the one the moment suite verifies by hand, so a failure here isolates
415 * the normalization and component selection rather than the accumulation: with three
416 * unit-weight samples, `R_ij = C_ij / 3`, and the diagonal of the Ucat tensor is 8/3
417 * in each direction, giving TKE = 4 and an RMS of sqrt(8/3) per component.
418 */
419static PetscErrorCode TestDerivedQuantities(void)
420{
421 SimCtx *simCtx = NULL;
422 UserCtx *user = NULL;
423 PicurvWindowStorage storage;
424 PicurvWindowDefinition d = AccDefinition(PETSC_TRUE);
425 const PetscReal series[3][3] = {{1.0, 2.0, 3.0}, {3.0, 6.0, 5.0}, {5.0, 4.0, 7.0}};
426 const PetscReal scalars[3] = {1.0, 2.0, 6.0};
427 Vec scalar_target = NULL, vector_target = NULL;
428 PetscInt count = 0;
429 PetscReal value = 0.0;
430
431 PetscFunctionBeginUser;
432 d.covariance_count = 1;
433 d.covariances[0].first = FIELD_ID_UCAT;
434 d.covariances[0].second = FIELD_ID_P;
435
436 PetscCall(PicurvCreateMinimalContexts(&simCtx, &user, ACC_N, ACC_N, ACC_N));
437 PetscCall(VecSet(user->Nvert, 0.0));
438 PetscCall(PicurvWindowStorageCreate(user, &d, &storage));
439 for (PetscInt s = 0; s < 3; ++s) {
440 PetscCall(SetUniform(user, series[s][0], series[s][1], series[s][2], scalars[s]));
441 PetscCall(PicurvWindowAccumulate(user, &d, &storage, 1.0));
442 }
443 PetscCall(DMCreateGlobalVector(user->da, &scalar_target));
444 PetscCall(DMCreateGlobalVector(user->fda, &vector_target));
445
446 /* Requesting every output resolves each kind against what the window carries. */
447 PetscCall(PicurvWindowDerivedCount(&d, &storage, "mean,reynolds_stress,rms,tke,flux", &count));
448 /* 2 means + (6 Ucat + 1 P) stresses + (3 Ucat + 1 P) RMS + 1 TKE + 1 flux. */
449 PetscCall(PicurvAssertIntEqual(15, count, "every requested output is enumerated"));
450
451 for (PetscInt index = 0; index < count; ++index) {
452 PicurvDerivedField field;
453 PetscBool is_tke = PETSC_FALSE, is_rxx = PETSC_FALSE, is_rmsx = PETSC_FALSE;
454 PetscBool is_flux = PETSC_FALSE, is_pvar = PETSC_FALSE;
455
456 PetscCall(PicurvWindowDerive(user, &d, &storage, "mean,reynolds_stress,rms,tke,flux",
457 index, scalar_target, vector_target, &field));
458 PetscCall(PetscStrcmp(field.name, "acc_Ucat_tke", &is_tke));
459 PetscCall(PetscStrcmp(field.name, "acc_Ucat_R_xx", &is_rxx));
460 PetscCall(PetscStrcmp(field.name, "acc_Ucat_rmsx", &is_rmsx));
461 PetscCall(PetscStrcmp(field.name, "acc_Ucat_P_flux", &is_flux));
462 PetscCall(PetscStrcmp(field.name, "acc_P_variance", &is_pvar));
463
464 if (is_rxx) {
465 PetscCall(ReadScalarAt(user, scalar_target, 2, 2, 2, &value));

568/**
569 * @brief The spatial mean divides by sampled points, not by the whole vector.
570 *
571 * A derived field is zero outside the target domain and wherever the mask never
572 * admitted a point. Those zeros are absences, and averaging over them silently
573 * scales the answer down by the fraction of the vector the window never covered —
574 * a wrong number that still looks plausible.
575 */
576static PetscErrorCode TestSpatialMeanExcludesUnsampledPoints(void)
577{
578 SimCtx *simCtx = NULL;
579 UserCtx *user = NULL;
580 PicurvWindowStorage storage;
581 PicurvWindowDefinition d = AccDefinition(PETSC_FALSE);
582 SpatialTargetPlan plan;
583 PetscReal ***nvert = NULL;
584 PetscReal mean = 0.0;
585 PetscReal whole_vector_mean = 0.0;
586 PetscReal sum = 0.0;
587 PetscInt vector_size = 0;
588 PetscInt targeted = 0;
589
590 PetscFunctionBeginUser;
591 PetscCall(PicurvCreateMinimalContexts(&simCtx, &user, ACC_N, ACC_N, ACC_N));
592 PetscCall(VecSet(user->Nvert, 0.0));
593 PetscCall(PicurvWindowStorageCreate(user, &d, &storage));
594 PetscCall(SetUniform(user, 1.0, 2.0, 3.0, 4.0));
595 PetscCall(PicurvWindowAccumulate(user, &d, &storage, 1.0));
596
597 /* A field holding exactly one over every sampled point: its masked mean must be
598 * one, whatever fraction of the vector the target domain occupies. */
599 PetscCall(VecZeroEntries(user->PostScalar));
600 PetscCall(SpatialTargetPlanCreate(user, FIELD_ID_P, PICURV_STATISTICS_MASK_FLUID, &plan));
601 PetscCall(SpatialTargetPlanGlobalPointCount(&plan, PETSC_COMM_WORLD, &targeted));
602 {
603 PetscReal ***values = NULL;
604
605 PetscCall(DMDAVecGetArray(user->da, user->PostScalar, &values));
606 for (PetscInt k = plan.start[2]; k < plan.end[2]; ++k)
607 for (PetscInt j = plan.start[1]; j < plan.end[1]; ++j)
608 for (PetscInt i = plan.start[0]; i < plan.end[0]; ++i) values[k][j][i] = 1.0;
609 PetscCall(DMDAVecRestoreArray(user->da, user->PostScalar, &values));
610 }
611
612 PetscCall(PicurvWindowSpatialMean(user, &d, &storage, user->PostScalar, &mean));
613 PetscCall(PicurvAssertRealNear(1.0, mean, 1.0e-12,
614 "the mean over sampled points ignores untargeted entries"));
615
616 /* The whole-vector average is genuinely different, which is what makes the
617 * distinction worth asserting rather than assuming. */
618 PetscCall(VecSum(user->PostScalar, &sum));
619 PetscCall(VecGetSize(user->PostScalar, &vector_size));
620 whole_vector_mean = sum / (PetscReal)vector_size;
621 PetscCall(PicurvAssertBool((PetscBool)(vector_size > targeted),
622 "the vector is larger than the targeted domain"));
623 PetscCall(PicurvAssertBool((PetscBool)(whole_vector_mean < 0.9),
624 "a whole-vector average would understate the result"));
625
626 /* Blanking a point removes it from both the sum and the count, so a uniform
627 * field still averages to its own value rather than being diluted. */
628 PetscCall(DMDAVecGetArray(user->da, user->Nvert, &nvert));
629 nvert[3][3][3] = 1.0;
630 PetscCall(DMDAVecRestoreArray(user->da, user->Nvert, &nvert));
631 PetscCall(PicurvWindowStorageDestroy(&storage));
632 PetscCall(PicurvWindowStorageCreate(user, &d, &storage));
633 PetscCall(PicurvWindowAccumulate(user, &d, &storage, 1.0));
634 PetscCall(PicurvWindowSpatialMean(user, &d, &storage, user->PostScalar, &mean));
635 PetscCall(PicurvAssertRealNear(1.0, mean, 1.0e-12,
636 "a never-sampled point is excluded from the average"));
637
638 PetscCall(PicurvWindowStorageDestroy(&storage));
639 PetscCall(PicurvDestroyMinimalContexts(&simCtx, &user));
640 PetscFunctionReturn(0);
641}

985/** @brief Closed-form test field: a scalar varying in every direction and over samples. */
986static PetscReal ReductionScalar(PetscInt i, PetscInt j, PetscInt k, PetscInt s)
987{
988 return 0.3 * i - 0.2 * k * k + 0.1 * i * j + (s ? 0.5 * (i + 2 * k) : -0.25 * j);

991/**
992 * @brief A homogeneous merge equals the moments of the slot's pooled sample set.
993 *
994 * Two states of a field varying across the averaged plane are accumulated, then the
995 * state is merged over i and k. Each j slot must hold the mean and variance of all of
996 * its points' samples pooled, which differs from the average of the point variances
997 * by the spread of the point means. The merged state also derives through the
998 * unchanged derivation.
999 */
1000static PetscErrorCode TestReductionEqualsPooledMoments(void)
1001{
1002 SimCtx *simCtx = NULL;
1003 UserCtx *user = NULL;
1004 PicurvWindowStorage storage, reduced;
1005 PicurvWindowDefinition d = AccDefinition(PETSC_TRUE);
1006 const PetscBool over_ik[3] = {PETSC_TRUE, PETSC_FALSE, PETSC_TRUE};
1007 SpatialTargetPlan plan;
1008 PicurvDerivedField derived;
1009 PetscReal value = 0.0;
1010
1011 PetscFunctionBeginUser;
1012 d.field_count = 1;
1013 d.fields[0].field_id = FIELD_ID_P;
1014 PetscCall(PicurvCreateMinimalContexts(&simCtx, &user, ACC_N, ACC_N, ACC_N));
1015 PetscCall(VecSet(user->Nvert, 0.0));
1016 PetscCall(VecSet(user->lNvert, 0.0));
1017 PetscCall(PicurvWindowStorageCreate(user, &d, &storage));
1018 PetscCall(PicurvWindowStorageCreate(user, &d, &reduced));
1019 for (PetscInt s = 0; s < 2; ++s) {
1020 PetscReal ***p = NULL;
1021
1022 PetscCall(DMDAVecGetArray(user->da, user->P, &p));
1023 for (PetscInt k = user->info.zs; k < user->info.zs + user->info.zm; ++k)
1024 for (PetscInt j = user->info.ys; j < user->info.ys + user->info.ym; ++j)
1025 for (PetscInt i = user->info.xs; i < user->info.xs + user->info.xm; ++i)
1026 p[k][j][i] = ReductionScalar(i, j, k, s);
1027 PetscCall(DMDAVecRestoreArray(user->da, user->P, &p));
1028 PetscCall(PicurvWindowAccumulate(user, &d, &storage, 1.0));
1029 }
1030 PetscCall(PicurvWindowStorageReduce(user, &d, &storage, over_ik, &reduced));
1031 PetscCall(SpatialTargetPlanCreate(user, FIELD_ID_P, PICURV_STATISTICS_MASK_FLUID, &plan));
1032
1033 for (PetscInt j = plan.start[1]; j < plan.end[1]; ++j) {
1034 PetscReal sum = 0.0, sum_sq = 0.0, n = 0.0, mean = 0.0;
1035 char context[128];
1036
1037 /* A single rank owns the whole plane, so the plan spans its physical points. */
1038 for (PetscInt k = plan.start[2]; k < plan.end[2]; ++k)
1039 for (PetscInt i = plan.start[0]; i < plan.end[0]; ++i)
1040 for (PetscInt s = 0; s < 2; ++s) { sum += ReductionScalar(i, j, k, s); n += 1.0; }
1041 mean = sum / n;
1042 for (PetscInt k = plan.start[2]; k < plan.end[2]; ++k)
1043 for (PetscInt i = plan.start[0]; i < plan.end[0]; ++i)
1044 for (PetscInt s = 0; s < 2; ++s) sum_sq += PetscSqr(ReductionScalar(i, j, k, s) - mean);
1045
1046 PetscCall(ReadScalarAt(user, reduced.mean[0], 3, j, 2, &value));
1047 PetscCall(PetscSNPrintf(context, sizeof(context), "pooled mean of slot j=%d", (int)j));
1048 PetscCall(PicurvAssertRealNear(mean, value, 1.0e-12, context));
1049 PetscCall(ReadScalarAt(user, reduced.m2[0], 1, j, 3, &value));
1050 PetscCall(PetscSNPrintf(context, sizeof(context), "pooled variance of slot j=%d", (int)j));
1051 PetscCall(PicurvAssertRealNear(sum_sq / n, value, 1.0e-12, context));
1052 PetscCall(ReadScalarAt(user, reduced.weight, 2, j, 2, &value));
1053 PetscCall(PicurvAssertRealNear(1.0, value, 0.0, "a sampled slot carries unit weight"));
1054 if (j == 2) {
1055 PetscCall(PicurvWindowDerive(user, &d, &reduced, "rms", 0, user->PostScalar, user->PostVector, &derived));
1056 PetscCall(ReadScalarAt(user, user->PostScalar, 2, j, 1, &value));
1057 PetscCall(PicurvAssertRealNear(PetscSqrtReal(sum_sq / n), value, 1.0e-12,
1058 "the merged state derives through the unchanged derivation"));
1059 }
1060 }
1061
1062 PetscCall(PicurvWindowStorageDestroy(&reduced));
1063 PetscCall(PicurvWindowStorageDestroy(&storage));
1064 PetscCall(PicurvDestroyMinimalContexts(&simCtx, &user));
1065 PetscFunctionReturn(0);
1066}

1141
1142 PetscCall(PicurvWindowStorageDestroy(&storage));
1143 PetscCall(PicurvDestroyMinimalContexts(&simCtx, &user));
1144 PetscFunctionReturn(0);
