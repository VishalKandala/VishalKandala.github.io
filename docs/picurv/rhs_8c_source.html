6/** @brief Owned-cell bounds of the residual: interior cells, excluding the dummy layers. */
7static inline void RHSInteriorBounds(const DMDALocalInfo *info, PetscInt lo[3], PetscInt hi[3])
8{
9 const PetscInt m[3] = {info->mx, info->my, info->mz};
10 const PetscInt start[3] = {info->xs, info->ys, info->zs};
11 const PetscInt end[3] = {info->xs + info->xm, info->ys + info->ym, info->zs + info->zm};

13 for (PetscInt a = 0; a < 3; a++) {
14 lo[a] = (start[a] == 0) ? 1 : start[a];

37/**
38 * @brief QUICK flux through the face between cell @p c and `c + e_n`, upwinded by the sign of
39 * @p ucon. @p m1 and @p p2 are the offsets used for the far upwind cells, -1 and +2 in
40 * the interior; a one-sided face substitutes 0 or +1 for the side it cannot reach.
41 */
42static inline Cmpnts RHSQuickFlux(const Cmpnts ***ucat, const PetscInt c[3], PetscInt n,
43 PetscReal ucon, PetscInt m1, PetscInt p2)
44{
45 const PetscReal coef = 0.125;
46 const PetscReal up = ucon + fabs(ucon), um = ucon - fabs(ucon);
47 const Cmpnts u0 = RHSVelocityAt(ucat, c, n, 0), u1 = RHSVelocityAt(ucat, c, n, 1);
48 const Cmpnts um1 = RHSVelocityAt(ucat, c, n, m1), up2 = RHSVelocityAt(ucat, c, n, p2);
49 Cmpnts f;
50
51 f.x = um * (coef * (-up2.x -2. * u1.x +3. * u0.x) +u1.x) +
52 up * (coef * (-um1.x -2. * u0.x +3. * u1.x) +u0.x);
53 f.y = um * (coef * (-up2.y -2. * u1.y +3. * u0.y) +u1.y) +
54 up * (coef * (-um1.y -2. * u0.y +3. * u1.y) +u0.y);
55 f.z = um * (coef * (-up2.z -2. * u1.z +3. * u0.z) +u1.z) +
56 up * (coef * (-um1.z -2. * u0.z +3. * u1.z) +u0.z);
57 return f;
58}

60/** @brief Central flux through the face between cell @p c and `c + e_n`. */
61static inline Cmpnts RHSCentralFlux(const Cmpnts ***ucat, const PetscInt c[3], PetscInt n, PetscReal ucon)
62{
63 const Cmpnts u0 = RHSVelocityAt(ucat, c, n, 0), u1 = RHSVelocityAt(ucat, c, n, 1);
64 Cmpnts f;
65
66 f.x = ucon * ( u0.x + u1.x );

72/**
73 * @brief Convective flux of Cartesian momentum through the face between cell @p c and
74 * `c + e_n`, carried by half the contravariant flux on that face.
75 *
76 * Interior faces use the central flux when @p central (central differencing or LES) and QUICK
77 * otherwise; with central differencing the two end faces of an axis are central as well. A face
78 * at the low end of an axis or behind a solid cell is QUICK one-sided toward +n, and one at the
79 * high end or ahead of a solid cell one-sided toward -n, except across a periodic seam, which
80 * keeps the full stencil. A face meeting none of these carries no flux.
81 */
82static inline Cmpnts RHSConvectiveFaceFlux(const Cmpnts ***ucat, const PetscReal ***nvert,
83 const PetscInt c[3], PetscInt n, PetscReal ucon,
84 PetscInt m, PetscBool periodic, PetscBool central)
85{
86 const PetscInt s = c[n];
87 const PetscReal nv_0 = nvert[c[2]][c[1]][c[0]];
88 const PetscReal nv_p = nvert[c[2] + (n == 2)][c[1] + (n == 1)][c[0] + (n == 0)];
89 const Cmpnts zero = {0.0, 0.0, 0.0};
90 /* The cell below exists in the local array everywhere but at s = 0 on a bounded axis. */
91 const PetscBool has_m = (PetscBool)(s > 0 || periodic);
92 const PetscReal nv_m = has_m ? nvert[c[2] - (n == 2)][c[1] - (n == 1)][c[0] - (n == 0)] : 0.0;
93 const PetscBool full_seam = (PetscBool)(periodic && nv_m < 0.1 && nv_p < 0.1);
94
95 if (s > 0 && s < m - 2 && RHSConvectsAsFluid(nv_p) && RHSConvectsAsFluid(nv_m)) {
96 return central ? RHSCentralFlux(ucat, c, n, ucon) : RHSQuickFlux(ucat, c, n, ucon, -1, 2);
97 }
98 if (central && (s == 0 || s == m - 2) && RHSConvectsAsFluid(nv_p) && RHSConvectsAsFluid(nv_0)) {
99 return RHSCentralFlux(ucat, c, n, ucon);
100 }
101 if (s < m - 2 && (s == 0 || nv_m > 0.1)) {
102 return RHSQuickFlux(ucat, c, n, ucon, (full_seam && s == 0) ? -1 : 0, 2);
103 }
104 if (s > 0 && (s == m - 2 || nv_p > 0.1)) {
105 return RHSQuickFlux(ucat, c, n, ucon, -1, (full_seam && s == m - 2) ? 2 : 1);
106 }
107 return zero;
108}

110/**
111 * @brief Implementation of \ref ComputeConvection().
112 * @details Full API contract (arguments, ownership, side effects) is documented with
113 * the header declaration in `include/rhs.h`.
114 * @see ComputeConvection()
115 */
116PetscErrorCode ComputeConvection(UserCtx *user, Vec Ucont, Vec Ucat, Vec Conv)
117{
118 const SimCtx *simCtx = user->simCtx;
119 const DMDALocalInfo *info = &user->info;
120 const PetscInt m[3] = {info->mx, info->my, info->mz};
121 const PetscBool central = (PetscBool)(simCtx->les || simCtx->central);
122 PetscBool periodic[3];
123 PetscInt lo[3], hi[3];
124 Vec Fp[3];
125 Cmpnts ***fp[3], ***conv;
126 const Cmpnts ***ucont, ***ucat;
127 const PetscReal ***nvert;
128
129 PetscFunctionBeginUser;
130 PROFILE_FUNCTION_BEGIN;
131 RHSInteriorBounds(info, lo, hi);
132 PoissonOperator_PeriodicAxes(user, periodic);
133 PetscCall(PreparePeriodicQuickStencilFields(user, Ucat, user->lNvert));
134 PetscCall(VecSet(Conv, 0.0));
135 /* Face fluxes per axis, from fda's pool and zeroed as fresh vectors would be: faces that
136 carry no flux must read as zero. */
137 for (PetscInt n = 0; n < 3; n++) {
138 PetscCall(DMGetLocalVector(user->fda, &Fp[n]));
139 PetscCall(VecZeroEntries(Fp[n]));
140 PetscCall(DMDAVecGetArray(user->fda, Fp[n], &fp[n]));
141 }
142 PetscCall(DMDAVecGetArrayRead(user->fda, Ucont, (void *)&ucont));
143 PetscCall(DMDAVecGetArrayRead(user->fda, Ucat, (void *)&ucat));
144 PetscCall(DMDAVecGetArrayRead(user->da, user->lNvert, (void *)&nvert));
145 PetscCall(DMDAVecGetArray(user->fda, Conv, &conv));
146
147 /* Fluxes on every face of the owned cells: along n, from the face below the first cell. */
148 for (PetscInt n = 0; n < 3; n++) {
149 PetscInt flo[3] = {lo[0], lo[1], lo[2]};
150
151 flo[n] -= 1;
152 for (PetscInt k = flo[2]; k < hi[2]; k++) {
153 for (PetscInt j = flo[1]; j < hi[1]; j++) {
154 for (PetscInt i = flo[0]; i < hi[0]; i++) {
155 const PetscInt c[3] = {i, j, k};
156 const PetscReal ucon = (&ucont[k][j][i].x)[n] * 0.5;
157
158 fp[n][k][j][i] = RHSConvectiveFaceFlux(ucat, (const PetscReal ***)nvert, c, n, ucon,
159 m[n], periodic[n], central);
160 }
161 }
162 }
163 }
164
165 /* Convective term in Cartesian components: the net flux out of each cell. */
166 for (PetscInt k = lo[2]; k < hi[2]; k++) {
167 for (PetscInt j = lo[1]; j < hi[1]; j++) {
168 for (PetscInt i = lo[0]; i < hi[0]; i++) {
169 conv[k][j][i].x =
170 fp[0][k][j][i].x - fp[0][k][j][i-1].x +
171 fp[1][k][j][i].x - fp[1][k][j-1][i].x +
172 fp[2][k][j][i].x - fp[2][k-1][j][i].x;
173 conv[k][j][i].y =
174 fp[0][k][j][i].y - fp[0][k][j][i-1].y +
175 fp[1][k][j][i].y - fp[1][k][j-1][i].y +
176 fp[2][k][j][i].y - fp[2][k-1][j][i].y;
177 conv[k][j][i].z =
178 fp[0][k][j][i].z - fp[0][k][j][i-1].z +
179 fp[1][k][j][i].z - fp[1][k][j-1][i].z +
180 fp[2][k][j][i].z - fp[2][k-1][j][i].z;
181 }
182 }
183 }
184
185 PetscCall(DMDAVecRestoreArray(user->fda, Conv, &conv));
186 PetscCall(DMDAVecRestoreArrayRead(user->da, user->lNvert, (void *)&nvert));
187 PetscCall(DMDAVecRestoreArrayRead(user->fda, Ucat, (void *)&ucat));
188 PetscCall(DMDAVecRestoreArrayRead(user->fda, Ucont, (void *)&ucont));
189 for (PetscInt n = 0; n < 3; n++) {
190 PetscCall(DMDAVecRestoreArray(user->fda, Fp[n], &fp[n]));
191 PetscCall(DMRestoreLocalVector(user->fda, &Fp[n]));
192 }
193 LOG_ALLOW(GLOBAL, LOG_DEBUG, "Convective term calculated .\n");
194 PROFILE_FUNCTION_END;
195 PetscFunctionReturn(0);
196}

219/** @brief `(u(c+hi e_b+e_n) + u(c+hi e_b) - u(c+lo e_b+e_n) - u(c+lo e_b)) * weight`. */
220static PICURV_ALWAYS_INLINE Cmpnts RHSTransverseVelocityDifference(const Cmpnts ***ucat, const PetscInt c[3],
221 PetscInt n, PetscInt b, PetscInt lo,
222 PetscInt hi, PetscReal weight)
223{
224 const Cmpnts a1 = RHSVelocityAtBN(ucat, c, b, hi, n, 1), a0 = RHSVelocityAtBN(ucat, c, b, hi, n, 0);
225 const Cmpnts b1 = RHSVelocityAtBN(ucat, c, b, lo, n, 1), b0 = RHSVelocityAtBN(ucat, c, b, lo, n, 0);
226 Cmpnts d;
227
228 d.x = (a1.x + a0.x - b1.x - b0.x) * weight;
229 d.y = (a1.y + a0.y - b1.y - b0.y) * weight;
230 d.z = (a1.z + a0.z - b1.z - b0.z) * weight;
231 return d;
232}

234/**
235 * @brief Velocity difference along transverse axis @p b on the face between cell @p c and
236 * `c + e_n`: the average over the two cells beside the face, one-sided away from a
237 * blanked solid cell. Each branch has constant offsets once @p n and @p b are constants.
238 */
239static PICURV_ALWAYS_INLINE Cmpnts RHSTransverseFaceDifference(const Cmpnts ***ucat, const PetscReal ***nvert,
240 const PetscInt c[3], PetscInt n, PetscInt b)
241{
242 if (RHSIsBlankedSolid(RHSNvertAtBN(nvert, c, b, 1, n, 0)) || RHSIsBlankedSolid(RHSNvertAtBN(nvert, c, b, 1, n, 1)))
243 return RHSTransverseVelocityDifference(ucat, c, n, b, -1, 0, 0.5);
244 if (RHSIsBlankedSolid(RHSNvertAtBN(nvert, c, b, -1, n, 0)) || RHSIsBlankedSolid(RHSNvertAtBN(nvert, c, b, -1, n, 1)))
245 return RHSTransverseVelocityDifference(ucat, c, n, b, 0, 1, 0.5);
246 return RHSTransverseVelocityDifference(ucat, c, n, b, -1, 1, 0.25);
247}

253static PICURV_ALWAYS_INLINE void RHSFaceVelocityDifferences(const Cmpnts ***ucat, const PetscReal ***nvert,

263/**
264 * @brief Viscous stress through the face between cell @p c and `c + e_n`, in Cartesian
265 * components, from the face differences @p d and the face metrics.
266 *
267 * With F_b the contravariant base vectors on the face and N = F_n its normal, the flux of each
268 * velocity component U is `(sum_b (F_b . N) dU/dxi_b + sum_c (sum_b dU_c/dxi_b F_b) . e_U N_c)
269 * aj`: the full stress, gradient plus transpose. The eddy viscosity @p nu_t carries it when
270 * LES is on, and the molecular viscosity @p nu is added; with the gradient (Clark) model the
271 * tensor `sum_b d_b d_b^T` contracted with N, over 12, is subtracted.
272 */
273static PICURV_ALWAYS_INLINE Cmpnts RHSViscousFaceFlux(const PoissonFaceMetrics *metrics, const Cmpnts d[3],
274 const PetscInt c[3], PetscInt n, PetscReal nu,
275 PetscReal nu_t, PetscBool les, PetscBool clark)
276{
277 const Cmpnts F0 = metrics->metric[n][0][c[2]][c[1]][c[0]];
278 const Cmpnts F1 = metrics->metric[n][1][c[2]][c[1]][c[0]];
279 const Cmpnts F2 = metrics->metric[n][2][c[2]][c[1]][c[0]];
280 const Cmpnts N = (n == 0) ? F0 : (n == 1) ? F1 : F2;
281 const PetscReal ajc = metrics->aj[n][c[2]][c[1]][c[0]];
282 const PetscReal g0 = F0.x * N.x + F0.y * N.y + F0.z * N.z;
283 const PetscReal g1 = F1.x * N.x + F1.y * N.y + F1.z * N.z;
284 const PetscReal g2 = F2.x * N.x + F2.y * N.y + F2.z * N.z;
285 /* r_<velocity component><Cartesian axis> = sum_b dU/dxi_b F_b[axis] */
286 const PetscReal r11 = d[0].x * F0.x + d[1].x * F1.x + d[2].x * F2.x;
287 const PetscReal r21 = d[0].y * F0.x + d[1].y * F1.x + d[2].y * F2.x;
288 const PetscReal r31 = d[0].z * F0.x + d[1].z * F1.x + d[2].z * F2.x;
289 const PetscReal r12 = d[0].x * F0.y + d[1].x * F1.y + d[2].x * F2.y;
290 const PetscReal r22 = d[0].y * F0.y + d[1].y * F1.y + d[2].y * F2.y;
291 const PetscReal r32 = d[0].z * F0.y + d[1].z * F1.y + d[2].z * F2.y;
292 const PetscReal r13 = d[0].x * F0.z + d[1].x * F1.z + d[2].x * F2.z;
293 const PetscReal r23 = d[0].y * F0.z + d[1].y * F1.z + d[2].y * F2.z;
294 const PetscReal r33 = d[0].z * F0.z + d[1].z * F1.z + d[2].z * F2.z;
295 const PetscReal ex = g0 * d[0].x + g1 * d[1].x + g2 * d[2].x + r11 * N.x + r21 * N.y + r31 * N.z;
296 const PetscReal ey = g0 * d[0].y + g1 * d[1].y + g2 * d[2].y + r12 * N.x + r22 * N.y + r32 * N.z;
297 const PetscReal ez = g0 * d[0].z + g1 * d[1].z + g2 * d[2].z + r13 * N.x + r23 * N.y + r33 * N.z;
298 Cmpnts f = {0.0, 0.0, 0.0};
299
300 if (les) {
301 f.x = ex * ajc * (nu_t);
302 f.y = ey * ajc * (nu_t);
303 f.z = ez * ajc * (nu_t);
304 }
305 f.x += ex * ajc * (nu);
306 f.y += ey * ajc * (nu);
307 f.z += ez * ajc * (nu);
308 if (clark) {
309 /* d_b is the velocity difference between neighbouring cells along direction b, so it
310 already carries that direction's spacing: sum_b d_b d_b^T is Delta_k^2 du_i/dx_k
311 du_j/dx_k as it stands, and no further cell extent belongs in it. */
312 const SymTensor gradient_tensor =
313 SymTensorCombine(1.0, SymTensorSelfOuter(d[0]), 1.0,
314 SymTensorCombine(1.0, SymTensorSelfOuter(d[1]),
315 1.0, SymTensorSelfOuter(d[2])));
316 const Cmpnts gradient_flux = SymTensorTimesVector(gradient_tensor, N);
317
318 f.x -= gradient_flux.x / 12.;
319 f.y -= gradient_flux.y / 12.;
320 f.z -= gradient_flux.z / 12.;
321 }
322 return f;
323}

325/**
326 * @brief Viscous stresses on every face of orientation @p n of the owned cells, from the face
327 * below the first cell; @p n is a constant at each call, so each expands to its own loop.
328 */
329static PICURV_ALWAYS_INLINE void RHSViscousFluxesAlong(const UserCtx *user, const PoissonFaceMetrics *metrics,
330 const Cmpnts ***ucat, const PetscReal ***nvert,
331 const PetscReal ***lnu_t, const PetscReal ***lnu_wall,
332 const PetscInt lo[3], const PetscInt hi[3],
333 const PetscInt m[3], PetscInt n, PetscBool les,
334 PetscBool clark, Cmpnts ***fp)
335{
336 const PetscBool wall_lo = BoundaryFaceHasWallShear(&user->boundary_faces[2 * n]);
337 const PetscBool wall_hi = BoundaryFaceHasWallShear(&user->boundary_faces[2 * n + 1]);
338 const PetscReal ren = user->simCtx->ren;
339 PetscInt flo[3] = {lo[0], lo[1], lo[2]};
340 PetscInt across[3] = {0, 0, 0};
341
342 flo[n] -= 1;
343 across[n] = 1;
344 for (PetscInt k = flo[2]; k < hi[2]; k++) {
345 for (PetscInt j = flo[1]; j < hi[1]; j++) {
346 for (PetscInt i = flo[0]; i < hi[0]; i++) {
347 const PetscInt c[3] = {i, j, k};
348 double nu = 1./ren, nu_t = 0;
349 Cmpnts d[3];
350
351 if (les) {
352 nu_t = 0.5 * (lnu_t[k][j][i] + lnu_t[k + across[2]][j + across[1]][i + across[0]]);
353 /* Zero suits a wall-resolved run, where the subgrid stress vanishes at the
354 wall; with a wall model this face is where the modelled stress enters. */
355 if ((wall_lo && c[n] == 0) || (wall_hi && c[n] == m[n] - 2)) {
356 PetscInt w[3] = {i, j, k};
357 w[n] = (c[n] == 0) ? 1 : m[n] - 2;
358 nu_t = lnu_wall ? lnu_wall[w[2]][w[1]][w[0]] : 0.0;
359 }
360 }
361 RHSFaceVelocityDifferences(ucat, nvert, c, n, d);
362 fp[k][j][i] = RHSViscousFaceFlux(metrics, d, c, n, nu, nu_t, les, clark);
363 }
364 }
365 }
366}

368/**
369 * @brief Implementation of \ref ComputeViscous().
370 * @details Full API contract (arguments, ownership, side effects) is documented with
371 * the header declaration in `include/rhs.h`.
372 * @see ComputeViscous()
373 */
374PetscErrorCode ComputeViscous(UserCtx *user, Vec Ucont, Vec Ucat, Vec Visc)
375{
376 const SimCtx *simCtx = user->simCtx;
377 const DMDALocalInfo *info = &user->info;
378 const PetscInt m[3] = {info->mx, info->my, info->mz};
379 const PetscBool les = (PetscBool)(simCtx->les != 0);
380 const PetscBool clark = (PetscBool)(simCtx->les_gradient_model != 0);
381 PetscInt lo[3], hi[3];
382 PoissonFaceMetrics metrics;
383 Vec Fp[3];
384 Cmpnts ***fp[3], ***visc;
385 const Cmpnts ***ucat;
386 const PetscReal ***nvert, ***lnu_t = NULL, ***lnu_wall = NULL;
387
388 PetscFunctionBeginUser;
389 PROFILE_FUNCTION_BEGIN;
390 (void)Ucont;
391 RHSInteriorBounds(info, lo, hi);
392 PetscCall(VecSet(Visc, 0.0));
393 for (PetscInt n = 0; n < 3; n++) {
394 PetscCall(DMGetLocalVector(user->fda, &Fp[n]));
395 PetscCall(VecZeroEntries(Fp[n]));
396 PetscCall(DMDAVecGetArray(user->fda, Fp[n], &fp[n]));
397 }
398 PetscCall(PoissonOperator_GetFaceMetrics(user, &metrics));
399 PetscCall(DMDAVecGetArrayRead(user->fda, Ucat, (void *)&ucat));
400 PetscCall(DMDAVecGetArrayRead(user->da, user->lNvert, (void *)&nvert));
401 if (les) PetscCall(DMDAVecGetArrayRead(user->da, user->lNu_t, (void *)&lnu_t));
402 /* A wall model replaces the subgrid viscosity at its own face rather than adding to it, so
403 the flux there carries the stress the model computed instead of the molecular fraction. */
404 if (simCtx->wallfunction) PetscCall(DMDAVecGetArrayRead(user->da, user->lNu_Wall, (void *)&lnu_wall));

410 RHSViscousFluxesAlong(user, &metrics, ucat, nvert, lnu_t, lnu_wall, lo, hi, m, 2, les, clark, fp[2]);

413 for (PetscInt k = lo[2]; k < hi[2]; k++) {
414 for (PetscInt j = lo[1]; j < hi[1]; j++) {
415 for (PetscInt i = lo[0]; i < hi[0]; i++) {
416 visc[k][j][i].x =
417 (fp[0][k][j][i].x - fp[0][k][j][i-1].x +
418 fp[1][k][j][i].x - fp[1][k][j-1][i].x +
419 fp[2][k][j][i].x - fp[2][k-1][j][i].x);
420 visc[k][j][i].y =
421 (fp[0][k][j][i].y - fp[0][k][j][i-1].y +
422 fp[1][k][j][i].y - fp[1][k][j-1][i].y +
423 fp[2][k][j][i].y - fp[2][k-1][j][i].y);
424 visc[k][j][i].z =
425 (fp[0][k][j][i].z - fp[0][k][j][i-1].z +
426 fp[1][k][j][i].z - fp[1][k][j-1][i].z +
427 fp[2][k][j][i].z - fp[2][k-1][j][i].z);
428 }
429 }
430 }
431
432 PetscCall(DMDAVecRestoreArray(user->fda, Visc, &visc));
433 if (lnu_wall) PetscCall(DMDAVecRestoreArrayRead(user->da, user->lNu_Wall, (void *)&lnu_wall));
434 if (lnu_t) PetscCall(DMDAVecRestoreArrayRead(user->da, user->lNu_t, (void *)&lnu_t));
435 PetscCall(DMDAVecRestoreArrayRead(user->da, user->lNvert, (void *)&nvert));
436 PetscCall(DMDAVecRestoreArrayRead(user->fda, Ucat, (void *)&ucat));
437 PetscCall(PoissonOperator_RestoreFaceMetrics(user, &metrics));
438 for (PetscInt n = 0; n < 3; n++) {
439 PetscCall(DMDAVecRestoreArray(user->fda, Fp[n], &fp[n]));
440 PetscCall(DMRestoreLocalVector(user->fda, &Fp[n]));
441 }
442 LOG_ALLOW(GLOBAL, LOG_DEBUG, "Viscous terms calculated .\n");
443 PROFILE_FUNCTION_END;
444 PetscFunctionReturn(0);

475/**
476 * @brief Contravariant cell-centre components of the spatial terms, Rct = aj F_n . (Visc - Conv),
477 * with each base vector F_n averaged from its two faces to the cell centre.
478 */
479static PetscErrorCode RHSContravariantSpatialTerms(UserCtx *user, Vec Conv, Vec Visc, Vec Rct)
480{
481 PetscInt lo[3], hi[3];
482 const Cmpnts ***csi, ***eta, ***zet, ***conv, ***visc;
483 const PetscReal ***aj;
484 Cmpnts ***rct;
485
486 PetscFunctionBeginUser;
487 PROFILE_FUNCTION_BEGIN;
488 RHSInteriorBounds(&user->info, lo, hi);
489 PetscCall(DMDAVecGetArrayRead(user->fda, user->lCsi, (void *)&csi));
490 PetscCall(DMDAVecGetArrayRead(user->fda, user->lEta, (void *)&eta));
491 PetscCall(DMDAVecGetArrayRead(user->fda, user->lZet, (void *)&zet));
492 PetscCall(DMDAVecGetArrayRead(user->da, user->lAj, (void *)&aj));
493 PetscCall(DMDAVecGetArrayRead(user->fda, Conv, (void *)&conv));
494 PetscCall(DMDAVecGetArrayRead(user->fda, Visc, (void *)&visc));
495 PetscCall(DMDAVecGetArray(user->fda, Rct, &rct));
496 for (PetscInt k = lo[2]; k < hi[2]; k++) {
497 for (PetscInt j = lo[1]; j < hi[1]; j++) {
498 for (PetscInt i = lo[0]; i < hi[0]; i++) {
499 const Cmpnts rc = {visc[k][j][i].x - conv[k][j][i].x,
500 visc[k][j][i].y - conv[k][j][i].y,
501 visc[k][j][i].z - conv[k][j][i].z};
502 rct[k][j][i].x = aj[k][j][i] *
503 (0.5 * (csi[k][j][i].x + csi[k][j][i-1].x) * rc.x +
504 0.5 * (csi[k][j][i].y + csi[k][j][i-1].y) * rc.y +
505 0.5 * (csi[k][j][i].z + csi[k][j][i-1].z) * rc.z);
506 rct[k][j][i].y = aj[k][j][i] *
507 (0.5 * (eta[k][j][i].x + eta[k][j-1][i].x) * rc.x +
508 0.5 * (eta[k][j][i].y + eta[k][j-1][i].y) * rc.y +
509 0.5 * (eta[k][j][i].z + eta[k][j-1][i].z) * rc.z);
510 rct[k][j][i].z = aj[k][j][i] *
511 (0.5 * (zet[k][j][i].x + zet[k-1][j][i].x) * rc.x +
512 0.5 * (zet[k][j][i].y + zet[k-1][j][i].y) * rc.y +
513 0.5 * (zet[k][j][i].z + zet[k-1][j][i].z) * rc.z);
514 }
515 }
516 }
517 PetscCall(DMDAVecRestoreArray(user->fda, Rct, &rct));
518 PetscCall(DMDAVecRestoreArrayRead(user->fda, Visc, (void *)&visc));
519 PetscCall(DMDAVecRestoreArrayRead(user->fda, Conv, (void *)&conv));
520 PetscCall(DMDAVecRestoreArrayRead(user->da, user->lAj, (void *)&aj));
521 PetscCall(DMDAVecRestoreArrayRead(user->fda, user->lZet, (void *)&zet));
522 PetscCall(DMDAVecRestoreArrayRead(user->fda, user->lEta, (void *)&eta));
523 PetscCall(DMDAVecRestoreArrayRead(user->fda, user->lCsi, (void *)&csi));
524 PROFILE_FUNCTION_END;
525 PetscFunctionReturn(0);
526}

528/** @brief Face residual of component @p n at cell @p c: the two-cell average of @p rct less
529 * the pressure-gradient flux through the face. */
530static PICURV_ALWAYS_INLINE PetscReal RHSFaceTerm(const PoissonFaceMetrics *metrics, const PetscReal ***nvert,
531 const PetscReal ***p, const Cmpnts ***rct, const PetscInt c[3],
532 PetscInt n, const PetscInt m[3], const PetscBool periodic[3])
533{
534 const PetscReal *own = &rct[c[2]][c[1]][c[0]].x;
535 const PetscReal *next = &rct[c[2] + (n == 2)][c[1] + (n == 1)][c[0] + (n == 0)].x;
536
537 return 0.5 * (own[n] + next[n]) - PoissonOperator_FaceGradientFlux(metrics, nvert, p, c, n, m, periodic);
538}

540/**
541 * @brief Face residual from the ghosted cell-centre terms: the average of @p Rct over the two
542 * cells beside each face, less the pressure-gradient flux through it.
543 *
544 * The flux is the projection's face gradient (PoissonOperator_FaceGradientFlux()) applied to
545 * `P`. Every interior face is set, and on a periodic axis also the seam face at index 0.
546 */
547static PetscErrorCode RHSFaceTermsWithPressureGradient(UserCtx *user, Vec Rct, Vec Rhs)
548{
549 const DMDALocalInfo *info = &user->info;
550 const PetscInt m[3] = {info->mx, info->my, info->mz};
551 const PetscInt start[3] = {info->xs, info->ys, info->zs};
552 PetscBool periodic[3];
553 PetscInt interior_lo[3], interior_hi[3];
554 PoissonFaceMetrics metrics;
555 const PetscReal ***nvert, ***p;
556 const Cmpnts ***rct;
557 Cmpnts ***rhs;
558
559 PetscFunctionBeginUser;
560 PROFILE_FUNCTION_BEGIN;
561 RHSInteriorBounds(info, interior_lo, interior_hi);
562 PoissonOperator_PeriodicAxes(user, periodic);
563 PetscCall(PoissonOperator_GetFaceMetrics(user, &metrics));
564 PetscCall(DMDAVecGetArrayRead(user->da, user->lNvert, (void *)&nvert));
565 PetscCall(DMDAVecGetArrayRead(user->da, user->lP, (void *)&p));
566 PetscCall(DMDAVecGetArrayRead(user->fda, Rct, (void *)&rct));
567 PetscCall(DMDAVecGetArray(user->fda, Rhs, &rhs));
568 /* One pass over the owned cells sets all three face components; with each direction a
569 constant, every call below compiles to its own straight-line stencil. */
570 for (PetscInt k = interior_lo[2]; k < interior_hi[2]; k++) {
571 for (PetscInt j = interior_lo[1]; j < interior_hi[1]; j++) {
572 for (PetscInt i = interior_lo[0]; i < interior_hi[0]; i++) {
573 const PetscInt c[3] = {i, j, k};
574
575 rhs[k][j][i].x = RHSFaceTerm(&metrics, nvert, p, rct, c, 0, m, periodic);
576 rhs[k][j][i].y = RHSFaceTerm(&metrics, nvert, p, rct, c, 1, m, periodic);
577 rhs[k][j][i].z = RHSFaceTerm(&metrics, nvert, p, rct, c, 2, m, periodic);
578 }
579 }
580 }
581 /* On a periodic axis the seam face at index 0 also carries an unknown. */
582 if (periodic[0] && start[0] == 0) {
583 for (PetscInt k = interior_lo[2]; k < interior_hi[2]; k++)
584 for (PetscInt j = interior_lo[1]; j < interior_hi[1]; j++) {
585 const PetscInt c[3] = {0, j, k};
586 rhs[k][j][0].x = RHSFaceTerm(&metrics, nvert, p, rct, c, 0, m, periodic);
587 }
588 }
589 if (periodic[1] && start[1] == 0) {
590 for (PetscInt k = interior_lo[2]; k < interior_hi[2]; k++)
591 for (PetscInt i = interior_lo[0]; i < interior_hi[0]; i++) {
592 const PetscInt c[3] = {i, 0, k};
593 rhs[k][0][i].y = RHSFaceTerm(&metrics, nvert, p, rct, c, 1, m, periodic);
594 }
595 }
596 if (periodic[2] && start[2] == 0) {
597 for (PetscInt j = interior_lo[1]; j < interior_hi[1]; j++)
598 for (PetscInt i = interior_lo[0]; i < interior_hi[0]; i++) {
599 const PetscInt c[3] = {i, j, 0};
600 rhs[0][j][i].z = RHSFaceTerm(&metrics, nvert, p, rct, c, 2, m, periodic);
601 }
602 }
603 PetscCall(DMDAVecRestoreArray(user->fda, Rhs, &rhs));
604 PetscCall(DMDAVecRestoreArrayRead(user->fda, Rct, (void *)&rct));
605 PetscCall(DMDAVecRestoreArrayRead(user->da, user->lP, (void *)&p));
606 PetscCall(DMDAVecRestoreArrayRead(user->da, user->lNvert, (void *)&nvert));
607 PetscCall(PoissonOperator_RestoreFaceMetrics(user, &metrics));
608 PROFILE_FUNCTION_END;
609 PetscFunctionReturn(0);
610}

649/**
650 * @brief Implementation of \ref ComputeRHS().
651 * @details Full API contract (arguments, ownership, side effects) is documented with
652 * the header declaration in `include/rhs.h`.
653 * @see ComputeRHS()
654 */
655PetscErrorCode ComputeRHS(UserCtx *user, Vec Rhs)
656{
657 const FieldId cell_fields[] = {FIELD_ID_UCAT};
658 Vec Conv, Visc, Rct;
659
660 PetscFunctionBeginUser;
661 PROFILE_FUNCTION_BEGIN;
662 LOG_ALLOW(LOCAL, LOG_DEBUG, "Rank %d, Block %" PetscInt_FMT ": Computing RHS (FormFunction1)...\n",
663 user->simCtx->rank, user->_this);
664 /* Work vectors from fda's pool, zeroed as fresh vectors would be: the residual is evaluated
665 many times per step, and allocating them each time costs more than the zeroing. */
666 PetscCall(DMGetLocalVector(user->fda, &Conv)); PetscCall(VecZeroEntries(Conv));
667 PetscCall(DMGetLocalVector(user->fda, &Visc)); PetscCall(VecZeroEntries(Visc));
668 PetscCall(DMGetLocalVector(user->fda, &Rct)); PetscCall(VecZeroEntries(Rct));
669
670 /* Cartesian velocity, with its periodic copies and ghosts, from the contravariant flux. */
671 PetscCall(Contra2Cart(user));
672 PetscCall(SynchronizePeriodicCellFields(user, 1, cell_fields));
673 PetscCall(UpdateLocalGhosts(user, FIELD_ID_UCAT));
674
675 /* Spatial terms in Cartesian components at cell centres, then contravariant. */
676 LOG_ALLOW(LOCAL, LOG_DEBUG, " Calculating convective terms...\n");
677 PetscCall(ComputeConvection(user, user->lUcont, user->lUcat, Conv));
678 if (!user->simCtx->invicid) {
679 LOG_ALLOW(LOCAL, LOG_DEBUG, " Calculating viscous terms...\n");
680 PetscCall(ComputeViscous(user, user->lUcont, user->lUcat, Visc));
681 }
682 LOG_ALLOW(LOCAL, LOG_DEBUG, " Converting Cartesian RHS to Contravariant RHS...\n");
683 PetscCall(RHSContravariantSpatialTerms(user, Conv, Visc, Rct));
684 PetscCall(ComputeBodyForces(user, Rct));
685 PetscCall(SynchronizePeriodicLocalStaggeredField(user, Rct));
686
687 /* Face residual: averaged spatial terms less the pressure gradient, then inactive rows. */
688 LOG_ALLOW(LOCAL, LOG_DEBUG, " Adding pressure gradient term to RHS...\n");
689 PetscCall(RHSFaceTermsWithPressureGradient(user, Rct, Rhs));
690 PetscCall(RHSZeroInactiveRows(user, Rhs));
691 LOG_ALLOW(GLOBAL, LOG_DEBUG, "Final cleanup and edge-cases complete .\n");
692
693 PetscCall(DMRestoreLocalVector(user->fda, &Rct));
694 PetscCall(DMRestoreLocalVector(user->fda, &Visc));
695 PetscCall(DMRestoreLocalVector(user->fda, &Conv));
696 LOG_ALLOW(LOCAL, LOG_DEBUG, "Rank %d, Block %" PetscInt_FMT ": RHS computation complete.\n",
697 user->simCtx->rank, user->_this);
698 PROFILE_FUNCTION_END;
699 PetscFunctionReturn(0);
700}
