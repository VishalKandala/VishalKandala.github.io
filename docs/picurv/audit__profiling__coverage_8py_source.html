1#!/usr/bin/env python3
2"""Enforce that every eligible C function in src/ is profiled and closes its timer on every return.
3
4Rules (docs/pages/09_Monitor_Reference.md, profiling):
5- every function defined in src/*.c that is not `static inline` and is not on the exclusion
6 list below begins exactly one timer, PROFILE_FUNCTION_BEGIN or PROFILE_HELPER_BEGIN;
7- every explicit `return` and `PetscFunctionReturn`/`PetscFunctionReturnVoid` after the timer's
8 BEGIN is immediately preceded by PROFILE_FUNCTION_END or PROFILE_HELPER_END, and a function whose
9 body can fall off its end ends with one; error returns inside SETERRQ/PetscCall/CHKERRQ are
10 not explicit returns and are exempt;
11- no file defines or reads `__FUNCT__`: timers, logging and PETSc's error traces all take the
12 name from `__func__`.
13
14A missed END leaves the function's PETSc event open, after which its calls are no longer recorded.
15"""

115def blank_comments_and_strings(text: str) -> str:
116 """!
117 @brief Replace comments, string and character literals with spaces, keeping offsets and newlines.
118 @param[in] text C source.
119 @return Text of the same length in which only code remains.
120 """
121 out = list(text)
122 i, n = 0, len(text)
123 while i < n:
124 c = text[i]
125 if text.startswith("//", i):
126 j = text.find("\n", i)
127 j = n if j < 0 else j
128 for k in range(i, j):
129 out[k] = " "
130 i = j
131 elif text.startswith("/*", i):
132 j = text.find("*/", i + 2)
133 j = n if j < 0 else j + 2
134 for k in range(i, j):
135 if out[k] != "\n":
136 out[k] = " "
137 i = j
138 elif c in "\"'":
139 j = i + 1
140 while j < n and text[j] != c:
141 j += 2 if text[j] == "\\" else 1
142 for k in range(i + 1, min(j, n)):
143 if out[k] != "\n":
144 out[k] = " "
145 i = j + 1
146 elif c == "#" and (i == 0 or text[i - 1] == "\n"):
147 j = i
148 while True: # Preprocessor line, with continuations
149 e = text.find("\n", j)
150 e = n if e < 0 else e
151 if e > 0 and text[e - 1] == "\\":
152 j = e + 1
153 continue
154 break
155 for k in range(i, e):
156 out[k] = " "
157 i = e
158 else:
159 i += 1
160 return "".join(out)

163def find_functions(code: str):
164 """!
165 @brief Locate top-level function definitions in comment- and string-blanked C code.
166 @param[in] code Output of blank_comments_and_strings().
167 @return List of (name, signature_start, body_open, body_close, is_static_inline).
168 """
169 functions = []
170 depth, i, n = 0, 0, len(code)
171 stmt_start = 0
172 while i < n:
173 c = code[i]
174 if c == "{":
175 if depth == 0:
176 head = code[stmt_start:i]
177 m = re.search(r"\b([A-Za-z_]\w*)\s*\‍(([^;{}]*)\‍)\s*$", head, re.S)
178 if m and m.group(1) not in CONTROL_WORDS and not re.search(r"[=]\s*$", head.split("(")[0]):
179 j, d = i + 1, 1
180 while j < n and d:
181 d += (code[j] == "{") - (code[j] == "}")
182 j += 1
183 sig = head[: m.start()]
184 words = set(re.findall(r"\b\w+\b", sig))
185 functions.append((m.group(1), stmt_start, i, j - 1, "static" in words and "inline" in words))
186 i, stmt_start = j, j
187 continue
188 depth += 1
189 elif c == "}":
190 depth -= 1
191 if depth == 0:
192 stmt_start = i + 1
193 elif c == ";" and depth == 0:
194 stmt_start = i + 1
195 i += 1
196 return functions

199def check_function(code: str, body_open: int, body_close: int):
200 """!
201 @brief Check one function body against the timer rules.
202 @param[in] code Blanked source.
203 @param[in] body_open Offset of the body's opening brace.
204 @param[in] body_close Offset of the body's closing brace.
205 @return List of problem strings with source offsets.
206 """
207 body = code[body_open + 1 : body_close]
208 problems = []
209 begins = len(BEGIN_RE.findall(body))
210 if begins != 1:
211 problems.append((body_open, f"has {begins} timer BEGIN macros, expected 1"))
212 return problems
213 timer_start = BEGIN_RE.search(body).start()
214 for m in RETURN_RE.finditer(body):
215 if m.start() < timer_start:
216 continue # An early return before the timer starts closes nothing
217 before = body[: m.start()].rstrip()
218 if not END_TAIL_RE.search(before):
219 problems.append((body_open + 1 + m.start(), "return not preceded by PROFILE_*_END"))
220 if not ends_in_return(body) and not END_TAIL_RE.search(body.rstrip()):
221 problems.append((body_close, "body can fall off its end without PROFILE_*_END"))
222 return problems

225def ends_in_return(body: str) -> bool:
226 """!
227 @brief Whether a function body's last top-level statement is a return.
228 @details A body ending in a block (`}`) or in any other statement can fall off its end.
229 @param[in] body Blanked function body, without its braces.
230 @return True when the final statement returns.
231 """
232 tail = body.rstrip()
233 if not tail.endswith(";"):
234 return False
235 depth, start = 0, 0
236 for i, c in enumerate(tail[:-1]):
237 if c in "([{":
238 depth += 1
239 elif c in ")]}":
240 depth -= 1
241 if c == "}" and depth == 0:
242 start = i + 1
243 elif c == ";" and depth == 0:
244 start = i + 1
245 return bool(RETURN_RE.match(tail[start:].lstrip()))

258def main() -> int:
259 """!
260 @brief Run the audit over src/*.c, include/*.h and tests/c, print problems.
261 @return 0 when clean, 1 otherwise.
262 """
263 problems = []
264 seen = set()
265 for path in SOURCES:
266 text = path.read_text(encoding="utf-8")
267 code = blank_comments_and_strings(text)
268 rel = path.relative_to(REPO_ROOT)
269 if str(rel) in EXCLUDED_FILES:
270 continue
271 for name, _, body_open, body_close, static_inline in find_functions(code):
272 seen.add(name)
273 if name in KERNELS and not static_inline and BEGIN_RE.search(code[body_open + 1 : body_close]):
274 problems.append(f"{rel}:{line_of(text, body_open)}: {name}: is a declared per-particle/per-cell kernel "
275 "but begins a timer; remove the timer or drop it from KERNELS")
276 continue
277 if static_inline or name in EXCLUDED or name in KERNELS:
278 continue
279 for offset, msg in check_function(code, body_open, body_close):
280 problems.append(f"{rel}:{line_of(text, offset)}: {name}: {msg}")
281 for path in sorted((REPO_ROOT / "src").glob("*.c")) + sorted((REPO_ROOT / "include").glob("*.h")) + sorted((REPO_ROOT / "tests" / "c").glob("*.[ch]")):
282 text = path.read_text(encoding="utf-8")
283 for m in FUNCT_RE.finditer(blank_comments_and_strings(text)):
284 problems.append(f"{path.relative_to(REPO_ROOT)}:{line_of(text, m.start())}: uses __FUNCT__; use __func__")
285 stale = sorted((set(EXCLUDED) | set(KERNELS)) - seen)
286 for name in stale:
287 problems.append(f"tests/tooling/audit_profiling_coverage.py: excluded function '{name}' no longer exists")
288 for p in problems:
289 print(p)
290 if problems:
291 print(f"\nFound {len(problems)} profiling coverage issue(s).", file=sys.stderr)
292 return 1
293 print("Profiling coverage audit passed: every eligible src/ function times itself and closes its timer.")
294 return 0
