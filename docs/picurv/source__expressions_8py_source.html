111def validate_expression(expression: str, names=EULERIAN_EXPRESSION_NAMES, random=False) -> None:
112 """!
113 @brief Reject anything outside the expression language the runtime also evaluates.
114 @details Python expression syntax restricted to decimal numbers, the given names, `pi`,
115 arithmetic, comparisons (chains included), `and`/`or`/`not`, and the
116 documented functions at their fixed arities. `uniform(n)`/`normal(n)` are
117 accepted only when @p random is true, with an optional stream `n` that must be
118 a non-negative integer literal.
119 @param[in] expression User expression.
120 @param[in] names Variables in scope, besides `pi`.
121 @param[in] random Whether per-particle random draws may be called.
122 @throws ValueError naming the first construct outside the language.
123 """
124 if not isinstance(expression, str) or not expression.strip():
125 raise ValueError("an expression must be a non-empty string.")
126 if "\n" in expression or "\r" in expression:
127 raise ValueError("an expression must fit on one line.")
128 try:
129 tree = ast.parse(expression, mode="eval")
130 tokens = list(tokenize.generate_tokens(io.StringIO(expression).readline))
131 except (SyntaxError, tokenize.TokenError) as exc:
132 raise ValueError(f"expression '{expression}' is not valid syntax: {exc}.") from exc
133 for token in tokens:
134 if token.type == tokenize.NUMBER and not _DECIMAL_LITERAL.match(token.string):
135 raise ValueError(f"'{token.string}' is not a decimal number; write numbers like 1, 0.5, or 2e-3.")
136 _check_expression_node(tree.body, frozenset(names) | {"pi"}, (), random)

139def _series_bound(node) -> int:
140 """!
141 @brief The integer a series bound denotes: an integer literal, possibly negated.
142 @param[in] node Bound node.
143 @return The bound.
144 @throws ValueError for anything else.
145 """
146 sign = 1
147 if isinstance(node, ast.UnaryOp) and isinstance(node.op, (ast.USub, ast.UAdd)):
148 sign, node = (-1 if isinstance(node.op, ast.USub) else 1), node.operand
149 if (not isinstance(node, ast.Constant) or type(node.value) not in (int, float)
150 or node.value != int(node.value)):
151 raise ValueError("series bounds must be integer numbers.")
152 return sign * int(node.value)

155def _check_expression_node(node, names, indices, random) -> None:
156 """!
157 @brief Check one node of a parsed expression, with the series indices enclosing it in scope.
158 @param[in] node AST node.
159 @param[in] names Variables in scope, `pi` included.
160 @param[in] indices Indices of the enclosing series, outermost first.
161 @param[in] random Whether per-particle random draws may be called.
162 @throws ValueError naming the first construct outside the language.
163 """
164 if not isinstance(node, ALLOWED_AST_NODES):
165 raise ValueError(f"unsupported expression syntax: {type(node).__name__}.")
166 if isinstance(node, ast.Constant) and type(node.value) not in (int, float):
167 raise ValueError(f"only numbers may appear as constants, not {node.value!r}.")
168 if isinstance(node, ast.Name):
169 if node.id not in names and node.id not in indices:
170 raise ValueError(f"unsupported expression name '{node.id}'.")
171 return
172 if isinstance(node, ast.Call):
173 name = node.func.id if isinstance(node.func, ast.Name) else None
174 if node.keywords:
175 raise ValueError("functions take positional arguments only.")
176 if name in RANDOM_FUNCTIONS and random:
177 if len(node.args) > 1:
178 raise ValueError(f"{name}() takes at most one stream argument.")
179 if node.args:
180 stream = node.args[0]
181 if (not isinstance(stream, ast.Constant) or type(stream.value) not in (int, float)
182 or stream.value < 0 or stream.value != int(stream.value)):
183 raise ValueError(f"the stream of {name}() must be a non-negative integer literal.")
184 return
185 if name not in FUNCTION_ARITY:
186 raise ValueError(f"'{name}' is not an available function.")
187 if len(node.args) != FUNCTION_ARITY[name]:
188 raise ValueError(f"{name}() takes {FUNCTION_ARITY[name]} argument(s).")
189 if name == "sum":
190 index, lower, upper, body = node.args
191 if not isinstance(index, ast.Name):
192 raise ValueError("a series sum(k, lo, hi, body) needs an index name first.")
193 if index.id in names or index.id in indices:
194 raise ValueError(f"the series index '{index.id}' reuses a name already in scope.")
195 if len(indices) == MAX_SERIES_DEPTH:
196 raise ValueError(f"series may nest at most {MAX_SERIES_DEPTH} deep.")
197 if _series_bound(upper) - _series_bound(lower) + 1 > MAX_SERIES_TERMS:
198 raise ValueError(f"a series may add at most {MAX_SERIES_TERMS} terms.")
199 _check_expression_node(body, names, indices + (index.id,), random)
200 return
201 for argument in node.args:
202 _check_expression_node(argument, names, indices, random)
203 return
204 for child in ast.iter_child_nodes(node):
205 _check_expression_node(child, names, indices, random)

208class _LogicalRewriter(ast.NodeTransformer):
209 """!
210 @brief Rewrite logical operators and chained comparisons into array operations.
211 @details `and`, `or`, `not` and a chain `a < b < c` have no elementwise meaning for
212 arrays, so they become `logical_and`, `logical_or`, `logical_not` calls, the
213 same 1-or-0 semantics the runtime evaluator gives them.
214 """
215
216 @staticmethod
217 def _call(name, args):
218 """!
219 @brief Build a call to one of the evaluation-only logical helpers.
220 @param[in] name Helper name.
221 @param[in] args Argument nodes.
222 @return Call node.
223 """
224 return ast.Call(func=ast.Name(id=name, ctx=ast.Load()), args=list(args), keywords=[])
225
226 def visit_Call(self, node):
227 """!
228 @brief Rewrite `sum(k, lo, hi, body)` as a loop over `lambda k: body`.
229 @param[in] node Call node.
230 @return Rewritten node.
231 """
232 self.generic_visit(node)
233 if not (isinstance(node.func, ast.Name) and node.func.id == "sum"):
234 return node
235 index, lower, upper, body = node.args
236 loop = ast.Lambda(args=ast.arguments(posonlyargs=[], args=[ast.arg(arg=index.id)], vararg=None,
237 kwonlyargs=[], kw_defaults=[], kwarg=None, defaults=[]),
238 body=body)
239 return self._call("series", [loop, lower, upper])
240
241 def visit_BoolOp(self, node):
242 """!
243 @brief Fold `a and b and c` into nested logical calls.
244 @param[in] node BoolOp node.
245 @return Rewritten node.
246 """
247 self.generic_visit(node)
248 name = "logical_and" if isinstance(node.op, ast.And) else "logical_or"
249 result = node.values[0]
250 for value in node.values[1:]:
251 result = self._call(name, [result, value])
252 return result
253
254 def visit_UnaryOp(self, node):
255 """!
256 @brief Rewrite `not a` as `logical_not(a)`.
257 @param[in] node UnaryOp node.
258 @return Rewritten node.
259 """
260 self.generic_visit(node)
261 if isinstance(node.op, ast.Not):
262 return self._call("logical_not", [node.operand])
263 return node
264
265 def visit_Compare(self, node):
266 """!
267 @brief Rewrite `a < b < c` as `logical_and(a < b, b < c)`.
268 @param[in] node Compare node.
269 @return Rewritten node.
270 """
271 self.generic_visit(node)
272 if len(node.ops) == 1:
273 return node
274 operands = [node.left] + node.comparators
275 pairs = [ast.Compare(left=copy.deepcopy(operands[i]), ops=[op], comparators=[copy.deepcopy(operands[i + 1])])
276 for i, op in enumerate(node.ops)]
277 result = pairs[0]
278 for pair in pairs[1:]:
279 result = self._call("logical_and", [result, pair])
280 return result

301def evaluate_values(expression: str, values: dict):
302 """!
303 @brief Evaluate an already validated expression over named arrays or scalars.
304 @param[in] expression Expression text.
305 @param[in] values Mapping of variable name to value or array.
306 @return Result as a float array, not yet checked for finiteness.
307 """
308 np = require_numpy()
309 tree = ast.fix_missing_locations(_LogicalRewriter().visit(ast.parse(expression, mode="eval")))
310 env = dict(_evaluated_functions())
311 env.update({"logical_and": np.logical_and, "logical_or": np.logical_or,
312 "logical_not": np.logical_not, "pi": math.pi,
313 "series": lambda body, lo, hi: sum((body(k) for k in range(int(lo), int(hi) + 1)), 0.0)})
314 for name in SOLVER_ONLY_FUNCTIONS:
315 env[name] = _solver_only(name)
316 env.update(values)
317 with np.errstate(all="ignore"):
318 # Globals, not locals: a series body is a lambda, which sees only the globals.
319 value = eval(compile(tree, "<ic-expression>", "eval"), dict(env, __builtins__={}))
320 return np.asarray(value, dtype=float)

428def substitute_params(expression: str, params: dict) -> str:
429 """!
430 @brief Replace declared parameter names in an expression by their values.
431 @param[in] expression Expression text.
432 @param[in] params Declared parameters.
433 @return Expression text with parameters inlined.
434 """
435 pieces, cursor = [], 0
436 for token in tokenize.generate_tokens(io.StringIO(expression).readline):
437 if token.type == tokenize.NAME and token.string in params:
438 start, end = token.start[1], token.end[1]
439 pieces.append(expression[cursor:start])
440 pieces.append(f"({float(params[token.string])!r})")
441 cursor = end
442 pieces.append(expression[cursor:])
443 return "".join(pieces)

461def lower_value(spec, names=PARTICLE_EXPRESSION_NAMES, random=True, where="value") -> str:
462 """!
463 @brief Turn a field value - number, expression, or structured regions - into one expression.
464 @details A mapping may declare `params` (named numbers usable in its expressions and
465 region parameters) and gives either `value` or `background` with `regions`.
466 Regions paint in order, a later region over an earlier one: a sharp region by
467 `where(d <= 0, v, before)`, a smoothed one by `before*(1 - s) + v*s`, with `d`
468 the signed distance to the region and `s` its edge profile over `edge.width`.
469 The result is validated like any expression.
470 @param[in] spec Number, expression string, or mapping.
471 @param[in] names Variables in scope.
472 @param[in] random Whether random draws may be called.
473 @param[in] where Configuration path, for diagnostics.
474 @return Validated expression text.
475 """
476 if isinstance(spec, bool):
477 raise ValueError(f"{where} must be a number, an expression, or a mapping.")
478 if isinstance(spec, (int, float)):
479 if not math.isfinite(spec):
480 raise ValueError(f"{where} must be finite.")
481 return repr(float(spec))
482 if isinstance(spec, str):
483 validate_at(spec, names, random, where)
484 return spec
485 if not isinstance(spec, dict):
486 raise ValueError(f"{where} must be a number, an expression, or a mapping.")
487 unknown = set(spec) - {"params", "value", "background", "regions"}
488 if unknown:
489 raise ValueError(f"{where} does not take {sorted(unknown)}; it takes params, value, background, regions.")
490 params = spec.get("params") or {}
491 if not isinstance(params, dict):
492 raise ValueError(f"{where}.params must be a mapping of names to numbers.")
493 reserved = set(names) | RESERVED_WORDS
494 for name, number in params.items():
495 if not isinstance(name, str) or not name.isidentifier() or name in reserved:
496 raise ValueError(f"{where}.params name '{name}' is not a free identifier.")
497 _number_text(number, {}, f"{where}.params.{name}")
498 scope = set(names) | set(params)

500 def expression_of(value, path):
501 """!
502 @brief Lower one value (number, `params` name, or expression) with parameters inlined.
503 @param[in] value Configured value.
504 @param[in] path Configuration path, for diagnostics.
505 @return Expression text.
506 """
507 if isinstance(value, str) and value not in params:
508 validate_at(value, scope, random, path)
509 return f"({substitute_params(value, params)})"
510 return _number_text(value, params, path)
511
512 if "value" in spec:
513 if "background" in spec or "regions" in spec:
514 raise ValueError(f"{where} gives either value, or background with regions.")
515 lowered = expression_of(spec["value"], f"{where}.value")
516 else:
517 regions = spec.get("regions")
518 if not isinstance(regions, list) or not regions:
519 raise ValueError(f"{where}.regions must be a non-empty list.")
520 lowered = expression_of(spec.get("background", 0.0), f"{where}.background")
521 for index, region in enumerate(regions):
522 path = f"{where}.regions[{index}]"
523 if not isinstance(region, dict) or "value" not in region:
524 raise ValueError(f"{path} must be a mapping with a shape and a value.")
525 distance = _region_distance(region, params, path)
526 value = expression_of(region["value"], f"{path}.value")
527 edge = region.get("edge")
528 if edge is None:
529 lowered = f"where({distance} <= 0, {value}, {lowered})"
530 continue
531 if not isinstance(edge, dict) or set(edge) != {"profile", "width"} or edge["profile"] not in EDGE_PROFILES:
532 raise ValueError(f"{path}.edge must give profile ({', '.join(EDGE_PROFILES)}) and width.")
533 width = _number_text(edge["width"], params, f"{path}.edge.width")
534 if float(width) <= 0.0:
535 raise ValueError(f"{path}.edge.width must be positive.")
536 ramp = f"maximum(0, minimum(1, 0.5 - ({distance})/{width}))"
537 blend = EDGE_PROFILES[edge["profile"]].format(s=ramp)
538 lowered = f"({lowered})*(1 - {blend}) + {value}*{blend}"
539 validate_expression(lowered, names, random)
540 return lowered
