1/**
2 * @file les.c
3 * @brief Implements the Large Eddy Simulation (LES) subgrid-scale closure.
4 *
5 * The closure is built from small kernels that each do one thing, so the physics is
6 * readable at the call sites and testable without a running solve:
7 *
8 * - symmetric-tensor algebra (`SymTensor*`), which knows nothing about turbulence;
9 * - strain rate and filter width, shared by every eddy-viscosity model;
10 * - the Germano identity pieces, `LeonardStress()` and `GermanoModelTensor()`;
11 * - the coefficient averaging, which is `PicurvSpatialRatioAverage()` from the shared
12 * spatial-target module: it turns the local dynamic model into the plane- or
13 * volume-averaged one without a separate code path per mode;
14 * - limiting and eddy-viscosity assembly.
15 *
16 * Two routines drive them. `ComputeSmagorinskyConstant()` runs the Germano-Lilly
17 * procedure for the dynamic model, and `ComputeEddyViscosityLES()` turns whichever
18 * coefficient is in force into the eddy viscosity the momentum equations add to the
19 * molecular one. `LogLESDiagnostics()` reports the coefficient's time history.
20 *
21 * The coefficient stored in `UserCtx::CS` is `C`, the factor multiplying
22 * `Delta^2 |S|`, which is `Cs^2` in the classical notation and is signed when
23 * backscatter is admitted. Conversion to the familiar `Cs` happens only where a
24 * human reads the number.
25 */

70/**
71 * @brief Marks the near-wall cells a wall model overwrites, so the procedure skips them.
72 *
73 * A wall model replaces the first interior cell's velocity with an analytic profile.
74 * The Germano identity evaluated there measures that imposed profile rather than
75 * resolved turbulence, so the cell is excluded from both contraction sums. This is not
76 * a modelling preference: the data at those cells is not a measurement, and averaging it
77 * in biases whatever set it belongs to. Under global averaging that set is the whole
78 * domain; under homogeneous averaging retaining the wall-normal direction it is the wall
79 * plane, whose value would otherwise be entirely imposed.
80 *
81 * A cell left with no data receives a zero coefficient, which is the right answer where
82 * a wall model is already supplying the stress.
83 *
84 * Only meaningful while a wall model is active; with none, nothing is imposed and every
85 * cell carries a real measurement.
86 */
87static PetscErrorCode BuildWallModelExclusionMask(UserCtx *user, Vec inclusion)
88{
89 const DMDALocalInfo info = user->info;
90 const BCFace negative[3] = {BC_FACE_NEG_X, BC_FACE_NEG_Y, BC_FACE_NEG_Z};
91 const BCFace positive[3] = {BC_FACE_POS_X, BC_FACE_POS_Y, BC_FACE_POS_Z};
92 const PetscInt extent[3] = {info.mx, info.my, info.mz};
93 PetscReal ***mask = NULL;
94 SpatialTargetPlan plan;
95
96 PetscFunctionBeginUser;
97 PetscCall(VecSet(inclusion, 1.0));
98 PetscCall(LESResolveDomain(user, &plan));
99 PetscCall(DMDAVecGetArray(user->da, inclusion, &mask));
100 for (PetscInt k = plan.start[2]; k < plan.end[2]; k++)
101 for (PetscInt j = plan.start[1]; j < plan.end[1]; j++)
102 for (PetscInt i = plan.start[0]; i < plan.end[0]; i++) {
103 const PetscInt coord[3] = {i, j, k};
104
105 for (PetscInt axis = 0; axis < 3; axis++) {
106 /* Index 1 and extent-2 are the first interior cells the wall model writes. */
107 if ((user->boundary_faces[negative[axis]].mathematical_type == WALL &&
108 coord[axis] == 1) ||
109 (user->boundary_faces[positive[axis]].mathematical_type == WALL &&
110 coord[axis] == extent[axis] - 2)) {
111 mask[k][j][i] = 0.0;
112 break;
113 }
114 }
115 }
116 PetscCall(DMDAVecRestoreArray(user->da, inclusion, &mask));
117 PetscFunctionReturn(0);
118}

260#define __FUNCT__ "StrainRateFromGradients"
261/**
262 * @brief Implementation of \ref StrainRateFromGradients().
263 * @details Full API contract is documented with the header declaration in
264 * `include/les.h`.
265 * @see StrainRateFromGradients()
266 */
267PetscErrorCode StrainRateFromGradients(Cmpnts dudx, Cmpnts dvdx, Cmpnts dwdx,
268 SymTensor *strain, PetscReal *magnitude)
269{
270 SymTensor s;
271
272 PetscFunctionBeginUser;
273
274 s.xx = dudx.x;
275 s.xy = 0.5 * (dudx.y + dvdx.x);
276 s.xz = 0.5 * (dudx.z + dwdx.x);
277 s.yy = dvdx.y;
278 s.yz = 0.5 * (dvdx.z + dwdx.y);
279 s.zz = dwdx.z;
280
281 if (strain) *strain = s;
282 // |S| = sqrt(2 S_ij S_ij), with the off-diagonal doubling handled by the contraction.
283 if (magnitude) *magnitude = PetscSqrtReal(2.0 * SymTensorNormSq(s));
284
285 PetscFunctionReturn(0);
286}

288#undef __FUNCT__
289#define __FUNCT__ "ComputeCellFilterWidth"
290/**
291 * @brief Implementation of \ref ComputeCellFilterWidth().
292 * @details Full API contract is documented with the header declaration in
293 * `include/les.h`.
294 * @see ComputeCellFilterWidth()
295 */
296PetscErrorCode ComputeCellFilterWidth(LESFilterWidthModel model, PetscReal aj,
297 Cmpnts csi, Cmpnts eta, Cmpnts zet, PetscReal *delta)
298{
299 double dx = 0.0, dy = 0.0, dz = 0.0;
300
301 PetscFunctionBeginUser;
302 PetscCheck(delta != NULL, PETSC_COMM_SELF, PETSC_ERR_ARG_NULL,
303 "Filter width destination cannot be NULL.");
304 PetscCheck(aj > 0.0, PETSC_COMM_SELF, PETSC_ERR_ARG_OUTOFRANGE,
305 "Cell Jacobian must be positive to define a filter width; received %g.", (double)aj);
306
307 if (model == LES_FILTER_WIDTH_CUBE_ROOT_VOLUME) {
308 // Cell volume alone: exact for a cube, and increasingly optimistic as the
309 // cell is stretched, because it averages the three extents geometrically.
310 *delta = PetscPowReal(1.0 / aj, 1.0 / 3.0);
311 PetscFunctionReturn(0);
312 }
313
314 PetscCall(ComputeCellCharacteristicLengthScale(aj, csi, eta, zet, &dx, &dy, &dz));
315
316 switch (model) {
317 case LES_FILTER_WIDTH_MAX_EDGE:
318 // The largest unresolved scale the cell can carry; conservative on stretched grids.
319 *delta = PetscMax(dx, PetscMax(dy, dz));
320 break;
321 case LES_FILTER_WIDTH_GEOMETRIC_MEAN:
322 default:
323 *delta = PetscPowReal(PetscMax(dx * dy * dz, LES_EPSILON), 1.0 / 3.0);
324 break;
325 }
326
327 PetscFunctionReturn(0);
328}

342SymTensor LeonardStress(Cmpnts velocity_filtered, SymTensor velocity_product_filtered)
343{
344 // L_ij = (u_i u_j)^ - u^_i u^_j : the stress carried between the two filter widths.

349#undef __FUNCT__
350#define __FUNCT__ "GermanoModelTensor"
351/**
352 * @brief Implementation of \ref GermanoModelTensor().
353 * @details Full API contract is documented with the header declaration in
354 * `include/les.h`.
355 * @see GermanoModelTensor()
356 */
357SymTensor GermanoModelTensor(PetscReal delta, PetscReal alpha,
358 PetscReal strain_magnitude_filtered,
359 SymTensor strain_filtered,
360 SymTensor strain_product_filtered)
361{
362 // Term one: the model evaluated on the test-filtered strain. Both factors were
363 // filtered separately, then multiplied.
364 const SymTensor test_scale = SymTensorCombine(alpha * strain_magnitude_filtered,
365 strain_filtered, 0.0, strain_filtered);
366
367 // Term two: the grid-scale model formed pointwise and then test-filtered. This is
368 // the filter of a product, and it is not equal to the product of the filtered
369 // factors above. That inequality carries the whole information content of the
370 // dynamic procedure.
371 const SymTensor grid_scale = strain_product_filtered;
372
373 const SymTensor model = SymTensorCombine(-2.0 * delta * delta, test_scale,
374 2.0 * delta * delta, grid_scale);
375
376 // Trace-free, so the Leonard stress's trace cannot enter the contraction through
377 // whatever discrete divergence the velocity field carries.
378 return SymTensorDeviator(model);
379}

394{
395 const LESConfig *config = &user->simCtx->les_config;
396 PetscInt selected = 0;
397 PetscBool periodic[3];
398
399 PetscFunctionBeginUser;
400 PetscCheck(direction != NULL, PETSC_COMM_SELF, PETSC_ERR_ARG_NULL,
401 "Averaging direction destination cannot be NULL.");
402 LESPeriodicAxes(user->simCtx, periodic);
403
404 for (PetscInt axis = 0; axis < 3; axis++) direction[axis] = PETSC_FALSE;
405
406 switch (config->averaging_mode) {

413 PetscFunctionReturn(0);

424 // homogeneous in every case the solver can currently express.

428 direction[axis] = PETSC_TRUE;
429 selected++;
430 }
431 }
432 }
433 PetscCheck(selected > 0, PETSC_COMM_SELF, PETSC_ERR_ARG_WRONGSTATE,
434 "Homogeneous LES averaging was requested, but this block declares no "
435 "periodic axis and the configuration names no direction. Name the "

437 PetscFunctionReturn(0);

444 * @brief Implementation of \ref ClipModelCoefficient().
445 * @details Full API contract is documented with the header declaration in
446 * `include/les.h`.
447 * @see ClipModelCoefficient()
448 */
449PetscReal ClipModelCoefficient(PetscReal coefficient, const LESConfig *config, PetscBool *limited)
450{
451 PetscReal result = coefficient;
452
453 switch (config->clip_mode) {
454 case LES_CLIP_NONE:
455 // Signed coefficient retained: backscatter survives, and stability is left to
456 // the total-viscosity floor rather than to the coefficient's sign.
457 break;

459 if (result < 0.0) result = 0.0;
460 break;
461 case LES_CLIP_CLAMP:

464 if (result > config->max_cs * config->max_cs) result = config->max_cs * config->max_cs;
465 break;
466 }
467
468 if (limited) *limited = (PetscBool)(result != coefficient);
469 return result;
470}

512/**
513 * @brief Maps a cell index to the index whose strain rate represents it.
514 *
515 * The strain rate must be known one cell beyond the computed range, because the
516 * test-filter stencil reaches there. At an interior index the answer is the index
517 * itself. At the two layout-boundary planes it is not, and for opposite reasons.
518 *
519 * On a periodic axis, index 0 and index `m-1` are duplicates of the last and first
520 * interior cells. A central difference evaluated at index 0 straddles the wrap and
521 * returns nonsense, so the strain is evaluated instead at ghost index `-2`, which
522 * addresses the same physical cell from the interior side. Index `m+1` plays the
523 * same role at the other end. These are the ghost indices the periodic cell
524 * synchronization itself copies from, and the DMDA carries the three ghost layers
525 * they need whenever any axis is periodic.
526 *
527 * On a non-periodic axis the two planes are the solver's boundary layer, where a
528 * central difference would reach outside the domain. The nearest interior cell's
529 * strain is used instead, which is a zeroth-order extrapolation into a layer that
530 * only ever contributes to a filter average.
531 */
532static PetscInt LESStrainSampleIndex(PetscInt index, PetscInt extent, PetscBool periodic)
533{
534 if (index == 0) return periodic ? -2 : 1;
535 if (index == extent - 1) return periodic ? extent + 1 : extent - 2;
536 return index;
537}

539/**
540 * @brief Fills the strain-rate scratch fields across the computed range and its halo.
541 */
542static PetscErrorCode ComputeStrainRateField(UserCtx *user, Cmpnts ***ucat,
543 Vec strain_diagonal, Vec strain_offdiagonal,
544 Vec strain_magnitude)
545{
546 DM da = user->da, fda = user->fda;
547 DMDALocalInfo info;
548 SpatialTargetPlan plan;
549 PetscBool periodic[3];
550 Cmpnts ***diagonal = NULL, ***offdiagonal = NULL;
551 PetscReal ***magnitude = NULL;
552
553 PetscFunctionBeginUser;
554
555 PetscCall(DMDAGetLocalInfo(da, &info));
556 PetscCall(LESResolveDomain(user, &plan));
557 LESPeriodicAxes(user->simCtx, periodic);
558
559 PetscCall(DMDAVecGetArray(fda, strain_diagonal, &diagonal));
560 PetscCall(DMDAVecGetArray(fda, strain_offdiagonal, &offdiagonal));
561 PetscCall(DMDAVecGetArray(da, strain_magnitude, &magnitude));
562
563 // One cell beyond the computed range in every direction, because that is how far
564 // the 3x3x3 test-filter stencil reaches from the outermost computed cell.
565 for (PetscInt k = plan.start[2] - 1; k <= plan.end[2]; k++)
566 for (PetscInt j = plan.start[1] - 1; j <= plan.end[1]; j++)
567 for (PetscInt i = plan.start[0] - 1; i <= plan.end[0]; i++) {
568 const PetscInt si = LESStrainSampleIndex(i, info.mx, periodic[0]);
569 const PetscInt sj = LESStrainSampleIndex(j, info.my, periodic[1]);
570 const PetscInt sk = LESStrainSampleIndex(k, info.mz, periodic[2]);
571
572 Cmpnts dudx, dvdx, dwdx;
573 SymTensor strain;
574 PetscReal strain_abs;
575
576 PetscCall(ComputeVectorFieldDerivatives(user, si, sj, sk, ucat, &dudx, &dvdx, &dwdx));
577 PetscCall(StrainRateFromGradients(dudx, dvdx, dwdx, &strain, &strain_abs));
578
579 diagonal[k][j][i].x = strain.xx;
580 diagonal[k][j][i].y = strain.yy;
581 diagonal[k][j][i].z = strain.zz;
582 offdiagonal[k][j][i].x = strain.xy;
583 offdiagonal[k][j][i].y = strain.xz;
584 offdiagonal[k][j][i].z = strain.yz;
585 magnitude[k][j][i] = strain_abs;
586 }
587
588 PetscCall(DMDAVecRestoreArray(fda, strain_diagonal, &diagonal));
589 PetscCall(DMDAVecRestoreArray(fda, strain_offdiagonal, &offdiagonal));
590 PetscCall(DMDAVecRestoreArray(da, strain_magnitude, &magnitude));
591
592 PetscFunctionReturn(0);
593}

667 // of the test-scale and grid-scale terms in the model tensor.
668 alpha = config->test_filter_width_ratio * config->test_filter_width_ratio;

670 PetscCall(VecDuplicate(user->lUcat, &lStrainDiag));
671 PetscCall(VecDuplicate(user->lUcat, &lStrainOff));
672 PetscCall(VecDuplicate(user->lNvert, &lStrainAbs));

741 // and it is not the same as multiplying the two filtered factors.

769 // averaging kernel free of any weighting policy of its own.

778 PetscCall(DMDAVecRestoreArray(da, lMM, &mm));
779 PetscCall(DMDAVecRestoreArray(fda, user->lUcat, &ucat));
780
781 // --- 3. Average the two contractions, then divide ---------------------------
782 // Both sums are formed before the division. Averaging the pointwise quotients
783 // instead would not be the least-squares coefficient Lilly's closure defines.
784 PetscCall(PicurvSpatialRatioAverage(user, &plan, lLM, lMM, lWallMask, average_direction,
785 PetscObjectComm((PetscObject)da), lCsq, NULL));
786
787 // --- 4. Limit the coefficient and record what limiting cost -----------------
788 PetscCall(DMDAVecGetArrayRead(da, lLM, &lm));
789 PetscCall(DMDAVecGetArrayRead(da, lMM, &mm));
790 PetscCall(DMDAVecGetArrayRead(da, lCsq, &csq));
791 PetscCall(DMDAVecGetArray(da, user->CS, &cs_out));
792
793 for (PetscInt k = plan.start[2]; k < plan.end[2]; k++)
794 for (PetscInt j = plan.start[1]; j < plan.end[1]; j++)
795 for (PetscInt i = plan.start[0]; i < plan.end[0]; i++) {
796 PetscBool limited = PETSC_FALSE;
797 PetscReal weight, raw;
798
799 if (!SpatialTargetPlanMaskAllows(&plan, nvert[k][j][i])) {
800 cs_out[k][j][i] = 0.0;
801 continue;
802 }
803
804 weight = 1.0 / aj[k][j][i];
805 raw = csq[k][j][i];
806
807 cs_out[k][j][i] = ClipModelCoefficient(raw, config, &limited);
808
809 // The two contraction fields already carry the cell volume from the loop
810 // above, so they are summed as they stand rather than weighted twice.
811 diagnostics.contraction_lm += lm[k][j][i];
812 diagnostics.contraction_mm += mm[k][j][i];
813 diagnostics.fluid_volume += weight;
814 if (raw < 0.0) diagnostics.backscatter_volume += weight;
815 if (limited) diagnostics.limited_volume += weight;
816 }
817
818 PetscCall(DMDAVecRestoreArrayRead(da, lLM, &lm));
819 PetscCall(DMDAVecRestoreArrayRead(da, lMM, &mm));
820 PetscCall(DMDAVecRestoreArrayRead(da, lCsq, &csq));
821 PetscCall(DMDAVecRestoreArray(da, user->CS, &cs_out));
822 PetscCall(DMDAVecRestoreArrayRead(fda, user->lCsi, &csi));
823 PetscCall(DMDAVecRestoreArrayRead(fda, user->lEta, &eta));
824 PetscCall(DMDAVecRestoreArrayRead(fda, user->lZet, &zet));
825 PetscCall(DMDAVecRestoreArrayRead(da, user->lNvert, &nvert));
826 PetscCall(DMDAVecRestoreArrayRead(da, user->lAj, &aj));
827
828 PetscCall(VecDestroy(&lStrainDiag));
829 PetscCall(VecDestroy(&lStrainOff));
830 PetscCall(VecDestroy(&lStrainAbs));
831 PetscCall(VecDestroy(&lLM));
832 PetscCall(VecDestroy(&lMM));
833 PetscCall(VecDestroy(&lCsq));
834 if (lWallMask) PetscCall(VecDestroy(&lWallMask));
835
836 user->les_diagnostics = diagnostics;
837 PetscCall(FinalizeSmagorinskyConstantField(user));
838
839 PROFILE_FUNCTION_END;
840 PetscFunctionReturn(0);

1031 // as a negative Cs rather than hidden behind a square root.

1045 (double)mean, (double)PetscSqrtReal(variance),
1046 (double)global_min_coefficient, (double)global_max[0],
1047 (double)(global_sum[3] / volume), (double)global_max[1],

1049 (double)(global_sum[4] / volume),
1050 (double)(global_sum[5] / volume), (double)(global_sum[6] / volume));
1051 PetscCheck(fclose(file) == 0, PETSC_COMM_SELF, PETSC_ERR_FILE_WRITE,
1052 "Unable to close the LES diagnostics file.");
1053
1054 LOG_ALLOW(GLOBAL, LOG_INFO,
1055 " LES coefficient: Cs_effective=%.4f, nu_t/nu (mean)=%.3e, backscatter=%.1f%%\n",
1056 (double)cs_effective, (double)(global_sum[3] / volume / molecular),
1057 (double)(100.0 * global_sum[5] / volume));
1058 }
1059
1060 PetscFunctionReturn(0);
