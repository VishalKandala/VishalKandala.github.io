7/**
8 * @file poisson.h
9 * @brief Pressure-Poisson projection: the multigrid pressure-correction solve, the
10 * pressure update, and the velocity correction that makes `Ucont` divergence-free.
11 *
12 * The discrete operator, the right-hand side, and the projection gradient are built from
13 * one face-gradient definition, so the Laplacian the solver inverts is exactly the
14 * divergence of the gradient the projection applies.
15 *
16 * Every masked operation reads the solid field from the `UserCtx` of the level it acts
17 * on (`lNvert`). Solid cells are excluded from the operator, the right-hand side, the
18 * projection, and the multigrid transfers; cells next to a solid face use one-sided
19 * transverse differences.
20 */
21
22/* ===========================================================================================
23 * Face-gradient operator, shared with the momentum residual's pressure gradient.
24 * poisson.c documents the face gradient; the per-face primitives are inline so the
25 * momentum residual's hot loop does not pay a call per face.
26 * =========================================================================================== */
27
28/** Cells whose `nvert` exceeds this value are solid. */
29#define POISSON_SOLID_THRESHOLD 0.1
30
31/**
32 * @brief Transverse difference used at one face.
33 *
34 * The difference is `weight * (sum of Phi on row hi - sum of Phi on row lo)`, where a row
35 * is the pair of cells on either side of the face, displaced by the given offset along the
36 * transverse axis.
37 */
38typedef struct {
39 PetscInt lo; /**< Transverse offset of the subtracted row pair. */
40 PetscInt hi; /**< Transverse offset of the added row pair. */
41 PetscReal weight; /**< 0.25 central, 0.5 one-sided, 0 when no fluid side remains. */
42} PoissonTransverseDifference;
43
44/** @brief Everything needed to evaluate the gradient flux through one face. */
45typedef struct {
46 PetscReal dot[3]; /**< F_b . F_n on the face, b = 0..2. */
47 PetscReal aj; /**< Inverse Jacobian on the face. */
48 PoissonTransverseDifference diff[3]; /**< Transverse differences; diff[n] is unused. */
49} PoissonFaceGradient;

57/**
58 * @brief Value of a cell-centred array at cell @p c displaced by @p d.
59 *
60 * @param[in] field Ghosted cell-centred array, indexed [k][j][i].
61 * @param[in] c Cell (i, j, k).
62 * @param[in] d Displacement (di, dj, dk).
63 * @return The value at `c + d`.
64 */
65static inline PetscReal PoissonOperator_At(const PetscReal ***field, const PetscInt c[3],
66 const PetscInt d[3])
67{
68 return field[c[2] + d[2]][c[1] + d[1]][c[0] + d[0]];
69}

71/**
72 * @brief Chooses the transverse difference along axis @p t at the face between cell
73 * @p c and `c + e_n`.
74 *
75 * The central difference averages the two cells beside the face over rows -1 and +1.
76 * When the +1 row is a non-periodic boundary layer or touches a solid cell, the difference
77 * falls back to rows -1 and 0, and symmetrically to rows 0 and +1; with neither side
78 * available the transverse term vanishes.
79 *
80 * @param[in] nvert Ghosted solid field.
81 * @param[in] c Cell on the low side of the face.
82 * @param[in] n Face orientation (0, 1, 2 for i, j, k).
83 * @param[in] t Transverse axis, not @p n.
84 * @param[in] m Grid sizes (mx, my, mz).
85 * @param[in] periodic Whether each axis is periodic.
86 * @return The rows and weight of the difference.
87 */
88static inline PoissonTransverseDifference PoissonOperator_TransverseDifference(
89 const PetscReal ***nvert, const PetscInt c[3], PetscInt n, PetscInt t,
90 const PetscInt m[3], const PetscBool periodic[3])
91{
92 PoissonTransverseDifference diff = {0, 0, 0.0};
93 PetscInt plus[3] = {0, 0, 0}, plus_n[3] = {0, 0, 0};
94 PetscInt minus[3] = {0, 0, 0}, minus_n[3] = {0, 0, 0};
95 const PetscInt s = c[t];
96
97 plus[t] = 1; plus_n[t] = 1; plus_n[n] = 1;
98 minus[t] = -1; minus_n[t] = -1; minus_n[n] = 1;
99 const PetscReal solid_plus = PoissonOperator_At(nvert, c, plus) + PoissonOperator_At(nvert, c, plus_n);
100 const PetscReal solid_minus = PoissonOperator_At(nvert, c, minus) + PoissonOperator_At(nvert, c, minus_n);
101
102 if ((s == m[t] - 2 && !periodic[t]) || solid_plus > POISSON_SOLID_THRESHOLD) {
103 if (solid_minus < POISSON_SOLID_THRESHOLD && (s != 1 || periodic[t])) {
104 diff.lo = -1; diff.hi = 0; diff.weight = 0.5;
105 }
106 } else if ((s == 1 && !periodic[t]) || solid_minus > POISSON_SOLID_THRESHOLD) {
107 if (solid_plus < POISSON_SOLID_THRESHOLD) {
108 diff.lo = 0; diff.hi = 1; diff.weight = 0.5;
109 }
110 } else {
111 diff.lo = -1; diff.hi = 1; diff.weight = 0.25;
112 }
113 return diff;
114}

116/**
117 * @brief Gathers the metric coefficients and transverse differences of the gradient flux
118 * through the face between cell @p c and `c + e_n`.
119 *
120 * @param[in] metrics Face metric arrays from PoissonOperator_GetFaceMetrics().
121 * @param[in] nvert Ghosted solid field.
122 * @param[in] c Cell on the low side of the face.
123 * @param[in] n Face orientation (0, 1, 2 for i, j, k).
124 * @param[in] m Grid sizes (mx, my, mz).
125 * @param[in] periodic Whether each axis is periodic.
126 * @return The face's metric products, inverse Jacobian and transverse differences.
127 */
128static inline PoissonFaceGradient PoissonOperator_FaceGradientStencil(
129 const PoissonFaceMetrics *metrics, const PetscReal ***nvert, const PetscInt c[3],
130 PetscInt n, const PetscInt m[3], const PetscBool periodic[3])
131{
132 PoissonFaceGradient face;
133 const Cmpnts normal = metrics->metric[n][n][c[2]][c[1]][c[0]];
134
135 face.aj = metrics->aj[n][c[2]][c[1]][c[0]];
136 for (PetscInt b = 0; b < 3; b++) {
137 const Cmpnts base = metrics->metric[n][b][c[2]][c[1]][c[0]];
138 face.dot[b] = base.x * normal.x + base.y * normal.y + base.z * normal.z;
139 if (b == n) {
140 face.diff[b].lo = 0; face.diff[b].hi = 0; face.diff[b].weight = 0.0;
141 } else {
142 face.diff[b] = PoissonOperator_TransverseDifference(nvert, c, n, b, m, periodic);
143 }
144 }
145 return face;
146}

148/**
149 * @brief Gradient flux of a cell-centred @p field through the face between cell @p c and
150 * `c + e_n`: `sum_b g_nb D_b(field)` of the face gradient defined in poisson.c.
151 *
152 * The projection applies it to `Phi`; the momentum residual applies it to `P`, so both use
153 * the one discrete gradient.
154 *
155 * @param[in] metrics Face metric arrays from PoissonOperator_GetFaceMetrics().
156 * @param[in] nvert Ghosted solid field.
157 * @param[in] field Ghosted cell-centred field whose gradient is taken.
158 * @param[in] c Cell on the low side of the face.
159 * @param[in] n Face orientation (0, 1, 2 for i, j, k).
160 * @param[in] m Grid sizes (mx, my, mz).
161 * @param[in] periodic Whether each axis is periodic.
162 * @return The gradient flux through the face.
163 */
164static inline PetscReal PoissonOperator_FaceGradientFlux(
165 const PoissonFaceMetrics *metrics, const PetscReal ***nvert, const PetscReal ***field,
166 const PetscInt c[3], PetscInt n, const PetscInt m[3], const PetscBool periodic[3])
167{
168 const PoissonFaceGradient face = PoissonOperator_FaceGradientStencil(metrics, nvert, c, n, m, periodic);
169 PetscInt across[3] = {0, 0, 0};
170 PetscReal difference[3];
171
172 across[n] = 1;
173 for (PetscInt b = 0; b < 3; b++) {
174 if (b == n) {
175 difference[b] = PoissonOperator_At(field, c, across) - field[c[2]][c[1]][c[0]];
176 continue;
177 }
178 const PoissonTransverseDifference diff = face.diff[b];
179 PetscInt hi_lower[3] = {0, 0, 0}, hi_upper[3] = {0, 0, 0};
180 PetscInt lo_lower[3] = {0, 0, 0}, lo_upper[3] = {0, 0, 0};
181 hi_lower[b] = diff.hi; hi_upper[b] = diff.hi; hi_upper[n] = 1;
182 lo_lower[b] = diff.lo; lo_upper[b] = diff.lo; lo_upper[n] = 1;
183 difference[b] = (diff.weight == 0.0) ? 0.0 :
184 (PoissonOperator_At(field, c, hi_lower) + PoissonOperator_At(field, c, hi_upper) -
185 PoissonOperator_At(field, c, lo_lower) - PoissonOperator_At(field, c, lo_upper)) * diff.weight;
186 }
187 return difference[0] * face.dot[0] * face.aj +
188 difference[1] * face.dot[1] * face.aj +
189 difference[2] * face.dot[2] * face.aj;
190}

218/**
219 * @brief Solves the pressure-correction equation for every block with geometric multigrid.
220 *
221 * On the first call for a block, assembles the operator on every multigrid level and
222 * builds the outer Krylov solver with its `PCMG` preconditioner, grid transfers, level
223 * smoothers, coarse solve, and null space. The solver is kept in the finest level's
224 * `UserCtx::ksp` and reused on every later call; it is destroyed with the context.
225 *
226 * Each call refreshes the ghosted contravariant flux, forms the right-hand side, solves
227 * for `Phi` on the finest level, and appends the iteration history to
228 * `Poisson_Solver_Convergence_History_Block_<bi>.log` in the run's log directory.
229 *
230 * Solver controls are read from PETSc options under the `ps_` prefix
231 * (`-ps_ksp_*`, `-ps_mg_levels_N_*`, `-ps_mg_coarse_*`) when the solver is built.
232 *
233 * @param[in,out] usermg Multigrid hierarchy; the finest level's `Phi` receives the solution.
234 * @return PETSc error code. A non-finite residual or a preconditioner that could not be
235 * built stops the run with PETSC_ERR_NOT_CONVERGED.
236 */
237extern PetscErrorCode PoissonSolver_Multigrid(UserMG *usermg);
