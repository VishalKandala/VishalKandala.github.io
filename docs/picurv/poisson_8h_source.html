7/**
8 * @file poisson.h
9 * @brief Pressure-Poisson projection: the multigrid pressure-correction solve, the
10 * pressure update, and the velocity correction that makes `Ucont` divergence-free.
11 *
12 * The discrete operator, the right-hand side, and the projection gradient are built from
13 * one face-gradient definition, so the Laplacian the solver inverts is exactly the
14 * divergence of the gradient the projection applies.
15 *
16 * Every masked operation reads the solid field from the `UserCtx` of the level it acts
17 * on (`lNvert`). Solid cells are excluded from the operator, the right-hand side, the
18 * projection, and the multigrid transfers; cells next to a solid face use one-sided
19 * transverse differences.
20 */
21
22/**
23 * @brief Solves the pressure-correction equation for every block with geometric multigrid.
24 *
25 * On the first call for a block, assembles the operator on every multigrid level and
26 * builds the outer Krylov solver with its `PCMG` preconditioner, grid transfers, level
27 * smoothers, coarse solve, and null space. The solver is kept in the finest level's
28 * `UserCtx::ksp` and reused on every later call; it is destroyed with the context.
29 *
30 * Each call refreshes the ghosted contravariant flux, forms the right-hand side, solves
31 * for `Phi` on the finest level, and appends the iteration history to
32 * `Poisson_Solver_Convergence_History_Block_<bi>.log` in the run's log directory.
33 *
34 * Solver controls are read from PETSc options under the `ps_` prefix
35 * (`-ps_ksp_*`, `-ps_mg_levels_N_*`, `-ps_mg_coarse_*`) when the solver is built.
36 *
37 * @param[in,out] usermg Multigrid hierarchy; the finest level's `Phi` receives the solution.
38 * @return PETSc error code. A non-finite residual or a preconditioner that could not be
39 * built stops the run with PETSC_ERR_NOT_CONVERGED.
40 */
41extern PetscErrorCode PoissonSolver_Multigrid(UserMG *usermg);

43/**
44 * @brief Assembles the pressure-correction operator on one multigrid level.
45 *
46 * Allocates `user->A` on the first call and reassembles its entries on later calls into
47 * the same nonzero structure. Fluid rows carry the 19-point curvilinear Laplacian;
48 * dummy rows on the domain boundary and solid rows are identities. Non-periodic faces
49 * are homogeneous Neumann; periodic faces wrap to the opposite interior layer.
50 *
51 * @param[in,out] user Level context supplying metrics, `lNvert`, and boundary types.
52 * @return PETSc error code.
53 */
54extern PetscErrorCode AssemblePoissonOperator(UserCtx *user);
