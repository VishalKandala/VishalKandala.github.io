7/**
8 * @file poisson.h
9 * @brief Pressure-Poisson projection: the multigrid pressure-correction solve, the
10 * pressure update, and the velocity correction that makes `Ucont` divergence-free.
11 *
12 * The discrete operator, the right-hand side, and the projection gradient are built from
13 * one face-gradient definition, so the Laplacian the solver inverts is exactly the
14 * divergence of the gradient the projection applies.
15 *
16 * Every masked operation reads the solid field from the `UserCtx` of the level it acts
17 * on (`lNvert`). Solid cells are excluded from the operator, the right-hand side, the
18 * projection, and the multigrid transfers; cells next to a solid face use one-sided
19 * transverse differences.
20 */
21
22/* ===========================================================================================
23 * Face-gradient operator, shared with the momentum residual's pressure gradient.
24 * poisson.c documents the face gradient; the per-face primitives are inline so the
25 * momentum residual's hot loop does not pay a call per face.
26 * =========================================================================================== */
27
28/** Cells whose `nvert` exceeds this value are solid. */
29#define POISSON_SOLID_THRESHOLD 0.1

31/**
32 * @brief Transverse difference used at one face.
33 *
34 * The difference is `weight * (sum of Phi on row hi - sum of Phi on row lo)`, where a row
35 * is the pair of cells on either side of the face, displaced by the given offset along the
36 * transverse axis.
37 */
38typedef struct {
39 PetscInt lo; /**< Transverse offset of the subtracted row pair. */
40 PetscInt hi; /**< Transverse offset of the added row pair. */
41 PetscReal weight; /**< 0.25 central, 0.5 one-sided, 0 when no fluid side remains. */
42} PoissonTransverseDifference;

45typedef struct {

47 PetscReal aj; /**< Inverse Jacobian on the face. */
48 PoissonTransverseDifference diff[3]; /**< Transverse differences; diff[n] is unused. */
49} PoissonFaceGradient;

57/**
58 * @brief Value of a cell-centred array at cell @p c displaced by @p d.
59 *
60 * @param[in] field Ghosted cell-centred array, indexed [k][j][i].

65static PICURV_ALWAYS_INLINE PetscReal PoissonOperator_At(const PetscReal ***field, const PetscInt c[3],
66 const PetscInt d[3])
67{
68 return field[c[2] + d[2]][c[1] + d[1]][c[0] + d[0]];

79/**
80 * @brief Value of a cell-centred array at `c + p e_b + q e_n`.
81 *
82 * With @p b, @p n, @p p and @p q constants every offset is a constant, so the per-face stencils
83 * below compile to straight-line loads.
84 *
85 * @param[in] field Ghosted cell-centred array, indexed [k][j][i].
86 * @param[in] c Cell (i, j, k).
87 * @param[in] b First axis.
88 * @param[in] p Displacement along @p b.
89 * @param[in] n Second axis.
90 * @param[in] q Displacement along @p n.
91 * @return The value at the displaced cell.
92 */
93static PICURV_ALWAYS_INLINE PetscReal PoissonOperator_AtBN(const PetscReal ***field, const PetscInt c[3],
94 PetscInt b, PetscInt p, PetscInt n, PetscInt q)
95{
96 return field[c[2] + (b == 2) * p + (n == 2) * q][c[1] + (b == 1) * p + (n == 1) * q][c[0] + (b == 0) * p + (n == 0) * q];
97}

99/**
100 * @brief Chooses the transverse stencil along axis @p t at the face between cell @p c and
101 * `c + e_n`.
102 *
103 * The central difference averages the two cells beside the face over rows -1 and +1.
104 * When the +1 row is a non-periodic boundary layer or touches a solid cell, the difference
105 * falls back to rows -1 and 0, and symmetrically to rows 0 and +1; with neither side
106 * available the transverse term vanishes.
107 *
108 * @param[in] nvert Ghosted solid field.
109 * @param[in] c Cell on the low side of the face.
110 * @param[in] n Face orientation (0, 1, 2 for i, j, k).
111 * @param[in] t Transverse axis, not @p n.
112 * @param[in] m Grid sizes (mx, my, mz).
113 * @param[in] periodic Whether each axis is periodic.
114 * @return The stencil to use.
115 */
116static PICURV_ALWAYS_INLINE PoissonTransverseKind PoissonOperator_TransverseKind(
117 const PetscReal ***nvert, const PetscInt c[3], PetscInt n, PetscInt t,
118 const PetscInt m[3], const PetscBool periodic[3])
119{
120 const PetscInt s = c[t];
121 const PetscReal solid_plus = PoissonOperator_AtBN(nvert, c, t, 1, n, 0) + PoissonOperator_AtBN(nvert, c, t, 1, n, 1);
122 const PetscReal solid_minus = PoissonOperator_AtBN(nvert, c, t, -1, n, 0) + PoissonOperator_AtBN(nvert, c, t, -1, n, 1);
123
124 if ((s == m[t] - 2 && !periodic[t]) || solid_plus > POISSON_SOLID_THRESHOLD) {
125 return (solid_minus < POISSON_SOLID_THRESHOLD && (s != 1 || periodic[t])) ? POISSON_TRANSVERSE_LOWER
126 : POISSON_TRANSVERSE_NONE;
127 }
128 if ((s == 1 && !periodic[t]) || solid_minus > POISSON_SOLID_THRESHOLD) {
129 return (solid_plus < POISSON_SOLID_THRESHOLD) ? POISSON_TRANSVERSE_UPPER : POISSON_TRANSVERSE_NONE;
130 }
131 return POISSON_TRANSVERSE_CENTRAL;
132}

134/**
135 * @brief The rows and weight of the transverse difference along axis @p t at the face between
136 * cell @p c and `c + e_n`, for operator assembly; PoissonOperator_TransverseKind() chooses.
137 *
138 * @param[in] nvert Ghosted solid field.
139 * @param[in] c Cell on the low side of the face.
140 * @param[in] n Face orientation (0, 1, 2 for i, j, k).
141 * @param[in] t Transverse axis, not @p n.
142 * @param[in] m Grid sizes (mx, my, mz).
143 * @param[in] periodic Whether each axis is periodic.
144 * @return The rows and weight of the difference.
145 */
146static PICURV_ALWAYS_INLINE PoissonTransverseDifference PoissonOperator_TransverseDifference(
147 const PetscReal ***nvert, const PetscInt c[3], PetscInt n, PetscInt t,
148 const PetscInt m[3], const PetscBool periodic[3])
149{
150 PoissonTransverseDifference diff = {0, 0, 0.0};
151
152 switch (PoissonOperator_TransverseKind(nvert, c, n, t, m, periodic)) {
153 case POISSON_TRANSVERSE_LOWER: diff.lo = -1; diff.hi = 0; diff.weight = 0.5; break;
154 case POISSON_TRANSVERSE_UPPER: diff.lo = 0; diff.hi = 1; diff.weight = 0.5; break;
155 case POISSON_TRANSVERSE_CENTRAL: diff.lo = -1; diff.hi = 1; diff.weight = 0.25; break;
156 case POISSON_TRANSVERSE_NONE: break;
157 }
158 return diff;
159}

161/**
162 * @brief Transverse difference of @p field along axis @p t at the face between cell @p c and
163 * `c + e_n`: `(sum of the row-pair hi - sum of the row-pair lo) * weight`.
164 *
165 * @param[in] nvert Ghosted solid field.
166 * @param[in] field Ghosted cell-centred field.
167 * @param[in] c Cell on the low side of the face.
168 * @param[in] n Face orientation (0, 1, 2 for i, j, k).
169 * @param[in] t Transverse axis, not @p n.
170 * @param[in] m Grid sizes (mx, my, mz).
171 * @param[in] periodic Whether each axis is periodic.
172 * @return The weighted difference, 0 when no side is available.
173 */
174static PICURV_ALWAYS_INLINE PetscReal PoissonOperator_TransverseFieldDifference(
175 const PetscReal ***nvert, const PetscReal ***field, const PetscInt c[3], PetscInt n, PetscInt t,
176 const PetscInt m[3], const PetscBool periodic[3])
177{
178 switch (PoissonOperator_TransverseKind(nvert, c, n, t, m, periodic)) {
179 case POISSON_TRANSVERSE_LOWER:
180 return (PoissonOperator_AtBN(field, c, t, 0, n, 0) + PoissonOperator_AtBN(field, c, t, 0, n, 1) -
181 PoissonOperator_AtBN(field, c, t, -1, n, 0) - PoissonOperator_AtBN(field, c, t, -1, n, 1)) * 0.5;
182 case POISSON_TRANSVERSE_UPPER:
183 return (PoissonOperator_AtBN(field, c, t, 1, n, 0) + PoissonOperator_AtBN(field, c, t, 1, n, 1) -
184 PoissonOperator_AtBN(field, c, t, 0, n, 0) - PoissonOperator_AtBN(field, c, t, 0, n, 1)) * 0.5;
185 case POISSON_TRANSVERSE_CENTRAL:
186 return (PoissonOperator_AtBN(field, c, t, 1, n, 0) + PoissonOperator_AtBN(field, c, t, 1, n, 1) -
187 PoissonOperator_AtBN(field, c, t, -1, n, 0) - PoissonOperator_AtBN(field, c, t, -1, n, 1)) * 0.25;
188 case POISSON_TRANSVERSE_NONE:
189 default:
190 return 0.0;
191 }
192}

194/**
195 * @brief Gathers the metric coefficients and transverse differences of the gradient flux
196 * through the face between cell @p c and `c + e_n`.
197 *
198 * @param[in] metrics Face metric arrays from PoissonOperator_GetFaceMetrics().
199 * @param[in] nvert Ghosted solid field.
200 * @param[in] c Cell on the low side of the face.
201 * @param[in] n Face orientation (0, 1, 2 for i, j, k).
202 * @param[in] m Grid sizes (mx, my, mz).
203 * @param[in] periodic Whether each axis is periodic.
204 * @return The face's metric products, inverse Jacobian and transverse differences.
205 */
206static PICURV_ALWAYS_INLINE PoissonFaceGradient PoissonOperator_FaceGradientStencil(
207 const PoissonFaceMetrics *metrics, const PetscReal ***nvert, const PetscInt c[3],
208 PetscInt n, const PetscInt m[3], const PetscBool periodic[3])
209{
210 PoissonFaceGradient face;
211 const Cmpnts normal = metrics->metric[n][n][c[2]][c[1]][c[0]];
212
213 face.aj = metrics->aj[n][c[2]][c[1]][c[0]];
214 for (PetscInt b = 0; b < 3; b++) {
215 const Cmpnts base = metrics->metric[n][b][c[2]][c[1]][c[0]];
216 face.dot[b] = base.x * normal.x + base.y * normal.y + base.z * normal.z;
217 if (b == n) {
218 face.diff[b].lo = 0; face.diff[b].hi = 0; face.diff[b].weight = 0.0;
219 } else {
220 face.diff[b] = PoissonOperator_TransverseDifference(nvert, c, n, b, m, periodic);
221 }
222 }
223 return face;
224}

226/**
227 * @brief Gradient flux of a cell-centred @p field through the face between cell @p c and
228 * `c + e_n`: `sum_b g_nb D_b(field)` of the face gradient defined in poisson.c.
229 *
230 * The projection applies it to `Phi`; the momentum residual applies it to `P`, so both use
231 * the one discrete gradient.
232 *
233 * @param[in] metrics Face metric arrays from PoissonOperator_GetFaceMetrics().
234 * @param[in] nvert Ghosted solid field.
235 * @param[in] field Ghosted cell-centred field whose gradient is taken.
236 * @param[in] c Cell on the low side of the face.
237 * @param[in] n Face orientation (0, 1, 2 for i, j, k).
238 * @param[in] m Grid sizes (mx, my, mz).
239 * @param[in] periodic Whether each axis is periodic.
240 * @return The gradient flux through the face.
241 */
242static PICURV_ALWAYS_INLINE PetscReal PoissonOperator_FaceGradientFlux(
243 const PoissonFaceMetrics *metrics, const PetscReal ***nvert, const PetscReal ***field,
244 const PetscInt c[3], PetscInt n, const PetscInt m[3], const PetscBool periodic[3])
245{
246 const Cmpnts normal = metrics->metric[n][n][c[2]][c[1]][c[0]];
247 const PetscReal aj = metrics->aj[n][c[2]][c[1]][c[0]];
248 PetscReal dot[3], difference[3];
249
250 for (PetscInt b = 0; b < 3; b++) {
251 const Cmpnts base = metrics->metric[n][b][c[2]][c[1]][c[0]];
252 dot[b] = base.x * normal.x + base.y * normal.y + base.z * normal.z;
253 }
254 difference[n] = PoissonOperator_AtBN(field, c, n, 0, n, 1) - field[c[2]][c[1]][c[0]];
255 difference[(n + 1) % 3] = PoissonOperator_TransverseFieldDifference(nvert, field, c, n, (n + 1) % 3, m, periodic);
256 difference[(n + 2) % 3] = PoissonOperator_TransverseFieldDifference(nvert, field, c, n, (n + 2) % 3, m, periodic);
257 return difference[0] * dot[0] * aj +
258 difference[1] * dot[1] * aj +
259 difference[2] * dot[2] * aj;
260}

288/**
289 * @brief Solves the pressure-correction equation for every block with geometric multigrid.
290 *
291 * On the first call for a block, assembles the operator on every multigrid level and
292 * builds the outer Krylov solver with its `PCMG` preconditioner, grid transfers, level
293 * smoothers, coarse solve, and null space. The solver is kept in the finest level's
294 * `UserCtx::ksp` and reused on every later call; it is destroyed with the context.
295 *
296 * Each call refreshes the ghosted contravariant flux, forms the right-hand side, solves
297 * for `Phi` on the finest level, and appends the iteration history to
298 * `Poisson_Solver_Convergence_History_Block_<bi>.log` in the run's log directory.
299 *
300 * Solver controls are read from PETSc options under the `ps_` prefix
301 * (`-ps_ksp_*`, `-ps_mg_levels_N_*`, `-ps_mg_coarse_*`) when the solver is built.
302 *
303 * @param[in,out] usermg Multigrid hierarchy; the finest level's `Phi` receives the solution.
304 * @return PETSc error code. A non-finite residual or a preconditioner that could not be
305 * built stops the run with PETSC_ERR_NOT_CONVERGED.
306 */
307extern PetscErrorCode PoissonSolver_Multigrid(UserMG *usermg);
