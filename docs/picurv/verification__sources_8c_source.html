105/**
106 * @brief Fills the Eulerian diffusivity field from the configured verification profile.
107 * @details See `include/verification_sources.h` for the public parameter and return contract.
108 */
109PetscErrorCode ApplyVerificationDiffusivityOverride(UserCtx *user)
110{
111 PetscErrorCode ierr;
112 SimCtx *simCtx = user->simCtx;
113 PetscReal ***diff_arr = NULL;
114 const Cmpnts ***cent = NULL;
115 DMDALocalInfo info = user->info;
116 PetscReal min_gamma = PETSC_MAX_REAL;
117
118 PetscFunctionBeginUser;
119 PROFILE_HELPER_BEGIN;
120
121 // Leave the production diffusivity untouched unless this verification mode is selected.
122 if (!VerificationDiffusivityOverrideActive(simCtx)) { PROFILE_HELPER_END; PetscFunctionReturn(0); }
123
124 ierr = DMDAVecGetArray(user->da, user->Diffusivity, &diff_arr); CHKERRQ(ierr);
125 ierr = DMDAVecGetArrayRead(user->fda, user->Cent, &cent); CHKERRQ(ierr);
126
127 for (PetscInt k = info.zs; k < info.zs + info.zm; ++k) {
128 for (PetscInt j = info.ys; j < info.ys + info.ym; ++j) {
129 for (PetscInt i = info.xs; i < info.xs + info.xm; ++i) {
130 PetscReal gamma;
131 ierr = VerificationFieldEvaluate(simCtx, &simCtx->verificationDiffusivity, cent[k][j][i], simCtx->ti, &gamma); CHKERRQ(ierr);
132 gamma /= simCtx->scaling.U_ref * simCtx->scaling.L_ref;
133 diff_arr[k][j][i] = gamma;
134 min_gamma = PetscMin(min_gamma, gamma);
135 }
136 }
137 }
138
139 ierr = DMDAVecRestoreArrayRead(user->fda, user->Cent, &cent); CHKERRQ(ierr);
140 ierr = DMDAVecRestoreArray(user->da, user->Diffusivity, &diff_arr); CHKERRQ(ierr);
141
142 {
143 PetscReal global_min_gamma = 0.0;
144 ierr = MPI_Allreduce(&min_gamma, &global_min_gamma, 1, MPIU_REAL, MPI_MIN, PETSC_COMM_WORLD); CHKERRMPI(ierr);
145 // Diffusivity must remain positive on every rank for the PDE operator to be valid.
146 if (global_min_gamma <= 0.0) {
147 SETERRQ(PETSC_COMM_WORLD, PETSC_ERR_ARG_WRONG,
148 "verification diffusivity override produced non-positive Gamma values (min=%g).",
149 (double)global_min_gamma);
150 }
151 }
152
153 {
154 const FieldId periodic_fields[] = {FIELD_ID_DIFFUSIVITY};
155 // Make the analytic field coherent across periodic seams before ghost updates.
156 ierr = SynchronizePeriodicCellFields(user, 1, periodic_fields); CHKERRQ(ierr);
157 ierr = UpdateLocalGhosts(user, FIELD_ID_DIFFUSIVITY); CHKERRQ(ierr);
158 }
159 LOG_ALLOW(GLOBAL, LOG_DEBUG, "Applied the verification diffusivity.\n");
160
161 PROFILE_HELPER_END;
162 PetscFunctionReturn(0);
163}

236{
237 PetscErrorCode ierr;

239 const Cmpnts ***cent = NULL;
240 DMDALocalInfo info;
241 PetscInt xs, xe, ys, ye, zs, ze, mx, my, mz;

243
244 PetscFunctionBeginUser;
245 PROFILE_HELPER_BEGIN;
246 if (!user) SETERRQ(PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "UserCtx cannot be NULL.");

249 info = user->info;
250 xs = info.xs; xe = info.xs + info.xm;
251 ys = info.ys; ye = info.ys + info.ym;
252 zs = info.zs; ze = info.zs + info.zm;
253 mx = info.mx; my = info.my; mz = info.mz;
254 lxs = (xs == 0) ? xs + 1 : xs; lxe = (xe == mx) ? xe - 1 : xe;
255 lys = (ys == 0) ? ys + 1 : ys; lye = (ye == my) ? ye - 1 : ye;

257
258 ierr = VecSet(targetVec, 0.0); CHKERRQ(ierr);
259 ierr = DMDAVecGetArray(user->da, targetVec, &target); CHKERRQ(ierr);
260 ierr = DMDAVecGetArrayRead(user->fda, user->Cent, &cent); CHKERRQ(ierr);
261
262 for (PetscInt k = lzs; k < lze; ++k) {
263 for (PetscInt j = lys; j < lye; ++j) {
264 for (PetscInt i = lxs; i < lxe; ++i) {
265 PetscReal value = 0.0;
266 ierr = VerificationFieldEvaluate(user->simCtx, &user->simCtx->verificationScalar, cent[k][j][i], user->simCtx->ti, &value); CHKERRQ(ierr);

269 }
270 }
