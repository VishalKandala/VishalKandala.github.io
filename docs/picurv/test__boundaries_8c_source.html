11/**
12 * @brief Appends one key/value pair to a linked list of boundary-condition parameters.
13 */
14static PetscErrorCode AppendBCParam(BC_Param **head, const char *key, const char *value)
15{
16 BC_Param *node = NULL;
17 BC_Param **cursor = head;
18
19 PetscFunctionBeginUser;
20 PetscCheck(head != NULL, PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "BC param list head cannot be NULL.");
21 PetscCall(PetscCalloc1(1, &node));
22 PetscCall(PetscStrallocpy(key, &node->key));
23 PetscCall(PetscStrallocpy(value, &node->value));
24 while (*cursor) {
25 cursor = &(*cursor)->next;
26 }
27 *cursor = node;
28 PetscFunctionReturn(0);
29}

1198/**
1199 * @brief Tests that the wall-function integration reproduces the model it dispatches to.
1200 *
1201 * The kernels are covered separately; what is unverified without this is the wiring
1202 * around them, where the plausible failures are silent. Feeding the reference and
1203 * boundary wall distances in the wrong order, taking the reference velocity from the
1204 * wrong cell, or normalising the face normal incorrectly all still produce a
1205 * plausible-looking corrected velocity. The check here is that the corrected
1206 * first-interior velocity is exactly what the log-law model predicts at that cell's
1207 * distance, for the friction velocity the integration itself stored.
1208 */
1209static PetscErrorCode TestApplyWallFunctionMatchesTheLogLawItDispatches(void)
1210{
1211 SimCtx *simCtx = NULL;
1212 UserCtx *user = NULL;
1213 Cmpnts ***ucat = NULL, ***ubcs = NULL;
1214 PetscReal ***utau = NULL;
1215 const PetscReal speed = 1.0;
1216 const PetscReal roughness = 1.0e-16; /* smooth wall */
1217 PetscReal kinematic_viscosity, distance_first, distance_second;
1218 PetscReal stored_utau, corrected_speed, expected_speed, reference_check;
1219 PetscReal untouched_before = 0.0, untouched_after = 0.0;
1220
1221 PetscFunctionBeginUser;
1222 PetscCall(PicurvCreateMinimalContexts(&simCtx, &user, 6, 6, 6));
1223 simCtx->ren = 1000.0;
1224 simCtx->wall_roughness_height = roughness;
1225 kinematic_viscosity = 1.0 / simCtx->ren;
1226
1227 PetscCall(SeedWallFrictionVelocityStorage(user));
1228 user->boundary_faces[BC_FACE_NEG_X].mathematical_type = WALL;
1229 PetscCall(SeedWallFunctionFixture(user, speed));
1230
1231 /* With unit Jacobians and a unit face area the model's distances are exact:
1232 the first cell centre sits half a cell from the wall, the second one and a
1233 half. Reproducing them here rather than reading them back is what makes a
1234 swapped-distance wiring error detectable. */
1235 distance_first = 0.5;
1236 distance_second = 2.0 * distance_first + 0.5;
1237
1238 /* Disabled: the routine must not touch the field at all. */
1239 simCtx->wallfunction = 0;
1240 PetscCall(DMDAVecGetArrayRead(user->fda, user->Ucat, (const Cmpnts ***)&ucat));
1241 untouched_before = ucat[2][2][1].y;
1242 PetscCall(DMDAVecRestoreArrayRead(user->fda, user->Ucat, (const Cmpnts ***)&ucat));
1243 PetscCall(ApplyWallFunction(user));
1244 PetscCall(DMDAVecGetArrayRead(user->fda, user->Ucat, (const Cmpnts ***)&ucat));
1245 untouched_after = ucat[2][2][1].y;
1246 PetscCall(DMDAVecRestoreArrayRead(user->fda, user->Ucat, (const Cmpnts ***)&ucat));
1247 PetscCall(PicurvAssertRealNear(untouched_before, untouched_after, 0.0,
1248 "a disabled wall function must leave the velocity untouched"));
1249
1250 /* Enabled: the first interior cell is corrected onto the modelled profile. */
1251 simCtx->wallfunction = 1;
1252 PetscCall(ApplyWallFunction(user));
1253
1254 PetscCall(DMDAVecGetArrayRead(user->da, user->Friction_Velocity, (const PetscReal ***)&utau));
1255 stored_utau = utau[2][2][1];
1256 PetscCall(DMDAVecRestoreArrayRead(user->da, user->Friction_Velocity, (const PetscReal ***)&utau));
1257 PetscCall(PicurvAssertBool((PetscBool)(stored_utau > 0.0),
1258 "the integration must store a positive friction velocity"));
1259
1260 /* The stored friction velocity must be the one that reproduces the *reference*
1261 cell's speed at the *reference* distance. Swapping the two distances passes the
1262 next check but fails this one. */
1263 reference_check = u_hydset_roughness(kinematic_viscosity, distance_second,
1264 stored_utau, roughness);
1265 PetscCall(PicurvAssertRealNear(speed, reference_check, 1.0e-6,
1266 "the friction velocity must invert the log law at the "
1267 "reference cell's distance"));
1268
1269 PetscCall(DMDAVecGetArrayRead(user->fda, user->Ucat, (const Cmpnts ***)&ucat));
1270 corrected_speed = ucat[2][2][1].y;
1271 PetscCall(DMDAVecRestoreArrayRead(user->fda, user->Ucat, (const Cmpnts ***)&ucat));
1272
1273 expected_speed = u_hydset_roughness(kinematic_viscosity, distance_first,
1274 stored_utau, roughness);
1275 PetscCall(PicurvAssertRealNear(expected_speed, corrected_speed, 1.0e-9,
1276 "the corrected velocity must be the modelled profile at "
1277 "the first interior cell's distance"));
1278 PetscCall(PicurvAssertBool((PetscBool)(corrected_speed < speed),
1279 "a log-law wall correction must slow the first interior cell "
1280 "relative to the reference cell"));
1281
1282 /* The ghost row carries no boundary velocity of its own. */
1283 PetscCall(DMDAVecGetArrayRead(user->fda, user->Bcs.Ubcs, (const Cmpnts ***)&ubcs));
1284 PetscCall(PicurvAssertRealNear(0.0, ubcs[2][2][0].y, 0.0,
1285 "the wall ghost row must carry a zero boundary velocity"));
1286 PetscCall(DMDAVecRestoreArrayRead(user->fda, user->Bcs.Ubcs, (const Cmpnts ***)&ubcs));
1287
1288 PetscCall(VecDestroy(&user->Friction_Velocity));
1289 PetscCall(VecDestroy(&user->lFriction_Velocity));
1290 PetscCall(PicurvDestroyMinimalContexts(&simCtx, &user));
1291 PetscFunctionReturn(0);
1292}

1385 * carried the same u_tau while still slowing the cell and still differing from the other
1386 * laws, which is everything the selector test above checks.

1394 WALL_FUNCTION_CABOT};

1396 "Werner-Wengle must respond to the flow it is given",
1397 "Cabot must respond to the flow it is given"};
1398 const PetscReal speeds[2] = {0.5, 2.0};
1399
1400 PetscFunctionBeginUser;
1401 PetscCall(PicurvCreateMinimalContexts(&simCtx, &user, 6, 6, 6));
1402 simCtx->ren = 1000.0;
1403 simCtx->wall_roughness_height = 1.0e-16;
1404 PetscCall(SeedWallFrictionVelocityStorage(user));
1405 user->boundary_faces[BC_FACE_NEG_X].mathematical_type = WALL;

1426 names[m]));

1430 PetscCall(VecDestroy(&user->lFriction_Velocity));

1435/**
1436 * @brief Tests that the friction-velocity storage matches what the field catalog declares.
1437 *
1438 * The wall model reaches its storage through a DM, and PETSc only rejects a mismatched
1439 * DM/vector pairing at the moment of access - inside a branch that runs only when wall
1440 * functions are enabled on a case that has a WALL face. That is how storage created from
1441 * a dof-3 DM and read through a dof-1 one stayed latent. Comparing the allocation against
1442 * the catalog descriptor turns that into a failure here.

1444static PetscErrorCode TestWallFrictionVelocityStorageMatchesTheCatalog(void)
1445{

1447 UserCtx *user = NULL;
1448 const FieldDescriptor *descriptor = NULL;
1449 FieldView view;
1450 PetscInt global_size = 0, local_size = 0;
1451 PetscInt xm = 0, ym = 0, zm = 0, gxm = 0, gym = 0, gzm = 0;
1452 PetscReal ***utau = NULL;
1453
1454 PetscFunctionBeginUser;
1455 PetscCall(PicurvCreateMinimalContexts(&simCtx, &user, 6, 6, 6));
1456 PetscCall(SeedWallFrictionVelocityStorage(user));
1457
1458 PetscCall(FieldGetDescriptor(FIELD_ID_U_TAU, &descriptor));
1459 PetscCall(PicurvAssertBool((PetscBool)(descriptor->dof == 1),
1460 "the friction velocity is one scalar per cell"));

1462 "a scalar cell field belongs on the dof-1 DM"));
1463
1464 PetscCall(FieldGetView(user, FIELD_ID_U_TAU, &view));
1465 PetscCall(PicurvAssertBool((PetscBool)(view.global_vec == user->Friction_Velocity &&

1470 PetscCall(DMDAGetCorners(user->da, NULL, NULL, NULL, &xm, &ym, &zm));

1477 "the local friction velocity must match the ghosted extent"));

1480 PetscCall(DMDAVecGetArray(user->da, user->Friction_Velocity, &utau));
1481 PetscCall(DMDAVecRestoreArray(user->da, user->Friction_Velocity, &utau));
1482

1489/**
1490 * @brief Tests that the wall pass records the statistics its diagnostics report.
1491 *
1492 * `y+` cannot be recovered downstream, so it is formed where the wall distance is still
1493 * in hand. This checks that what is accumulated there is the same quantity the reported
1494 * column claims to be, and that a pass over no wall face records nothing rather than
1495 * recording zeros.
1496 */
1497static PetscErrorCode TestWallModelDiagnosticsRecordTheCorrectedCells(void)
1498{
1499 SimCtx *simCtx = NULL;
1500 UserCtx *user = NULL;
1501 PetscReal kinematic_viscosity = 0.0;
1502 PetscReal expected_y_plus = 0.0;
1503 const PetscReal distance_first = 0.5;
1504
1505 PetscFunctionBeginUser;
1506 PetscCall(PicurvCreateMinimalContexts(&simCtx, &user, 6, 6, 6));
1507 simCtx->ren = 1000.0;
1508 simCtx->wall_roughness_height = 1.0e-16;
1509 kinematic_viscosity = 1.0 / simCtx->ren;
1510 PetscCall(SeedWallFrictionVelocityStorage(user));
1511 user->boundary_faces[BC_FACE_NEG_X].mathematical_type = WALL;
1512 PetscCall(SeedWallFunctionFixture(user, 1.0));
1513
1514 /* Disabled: nothing is corrected, so nothing is recorded. */
1515 simCtx->wallfunction = 0;
1516 PetscCall(ApplyWallFunction(user));
1517 PetscCall(PicurvAssertBool((PetscBool)(user->wall_diagnostics.cells == 0),
1518 "a disabled wall model must record no corrected cells"));
1519
1520 simCtx->wallfunction = 1;
1521 PetscCall(ApplyWallFunction(user));
1522 PetscCall(PicurvAssertBool((PetscBool)(user->wall_diagnostics.cells > 0),
1523 "an enabled wall model must record the cells it corrected"));
1524
1525 /* One face, so the sums are one cell's values and y+ is exactly u_tau * y / nu.
1526 Reproducing it from the stored friction velocity is what makes a wrong distance
1527 or a wrong viscosity in the accumulation visible. */
1528 expected_y_plus = user->wall_diagnostics.friction_velocity_sum * distance_first /
1529 kinematic_viscosity / (PetscReal)user->wall_diagnostics.cells;
1530 PetscCall(PicurvAssertRealNear(user->wall_diagnostics.y_plus_sum /
1531 (PetscReal)user->wall_diagnostics.cells,
1532 expected_y_plus, 1.0e-9,
1533 "y+ must be formed from the same distance the "
1534 "correction used"));
1535 PetscCall(PicurvAssertRealNear(user->wall_diagnostics.wall_distance_sum /
1536 (PetscReal)user->wall_diagnostics.cells,
1537 distance_first, 1.0e-9,
1538 "the recorded wall distance must be the first cell's"));
1539 PetscCall(PicurvAssertBool((PetscBool)(user->wall_diagnostics.friction_velocity_min <=
1540 user->wall_diagnostics.friction_velocity_max),
1541 "the recorded extrema must bracket each other"));
1542
1543 /* A second pass must describe that pass alone, not both. */
1544 {
1545 const PetscInt first_pass_cells = user->wall_diagnostics.cells;
1546
1547 PetscCall(ApplyWallFunction(user));
1548 PetscCall(PicurvAssertBool((PetscBool)(user->wall_diagnostics.cells == first_pass_cells),
1549 "each pass must replace the previous pass's sample set"));
1550 }
1551
1552 PetscCall(VecDestroy(&user->Friction_Velocity));
1553 PetscCall(VecDestroy(&user->lFriction_Velocity));
1554 PetscCall(PicurvDestroyMinimalContexts(&simCtx, &user));
1555 PetscFunctionReturn(0);
1556}
