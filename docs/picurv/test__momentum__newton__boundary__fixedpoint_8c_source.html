1/**
2 * @file test_momentum_newton_boundary_fixedpoint.c
3 * @brief Opt-in integration regression for the Newton--Krylov deterministic
4 * Cartesian-seed correction on the production-sized straight duct.
5 *
6 * The diagnostic phase established that the production Newton--Krylov residual
7 * must reconstruct Ucat/lUcat from the current SNES trial vector X before the
8 * first boundary sweep (see src/momentum_newton_krylov.c). That correction is
9 * now in production. This file locks the resulting guarantees in place using the
10 * *actual production* residual callback (included via a privately renamed copy of
11 * the production translation unit) -- it deliberately does NOT keep a duplicate
12 * test-local seeded callback, so it exercises exactly what ships:
13 *
14 * 1. residual purity at the reproduced physical step-2 state (immediate and
15 * after real PETSc MFFD products);
16 * 2. a complete production step-2 solve from the true projected step-1 state
17 * using PETSc's default classical Gram--Schmidt;
18 * 3. boundary-map compatibility: the converged three-pass solution also zeros
19 * the 24-pass outlet residual;
20 * 4. clean pressure projection of the converged momentum state.
21 *
22 * The heavy fixture keeps this target out of `make check`; the cheap
23 * seed-removal detector lives in unit-newton-krylov (default suite). Central
24 * checks run on one and four MPI ranks.
25 */

75{
76 Vec field[FP_FIELD_COUNT];
77 PetscFunctionBeginUser;

119/** @brief Computes collective global difference norms for global or rank-local vectors. */
120static PetscErrorCode CollectiveDifference(Vec a, Vec b, PetscReal *norm_inf, PetscReal *norm_2)
121{
122 Vec delta = NULL;
123 MPI_Comm comm;
124 PetscMPIInt comm_size, world_size;
125 PetscReal local_inf, local_2;
126 PetscFunctionBeginUser;
127 PetscCall(VecDuplicate(a, &delta));
128 PetscCall(VecWAXPY(delta, -1.0, a, b));
129 PetscCall(VecNorm(delta, NORM_INFINITY, &local_inf));
130 PetscCall(VecNorm(delta, NORM_2, &local_2));
131 comm = PetscObjectComm((PetscObject)a);
132 PetscCallMPI(MPI_Comm_size(comm, &comm_size));
133 PetscCallMPI(MPI_Comm_size(PETSC_COMM_WORLD, &world_size));
134 if (comm_size == 1 && world_size > 1) {
135 PetscReal square = local_2 * local_2;
136 PetscCallMPI(MPI_Allreduce(&local_inf, norm_inf, 1, MPIU_REAL, MPI_MAX, PETSC_COMM_WORLD));
137 PetscCallMPI(MPI_Allreduce(&square, norm_2, 1, MPIU_REAL, MPI_SUM, PETSC_COMM_WORLD));
138 *norm_2 = PetscSqrtReal(*norm_2);
139 } else {
140 *norm_inf = local_inf;
141 *norm_2 = local_2;
142 }
143 PetscCall(VecDestroy(&delta));
144 PetscFunctionReturn(PETSC_SUCCESS);
145}

147/** @brief Finds the MPI rank containing the largest local entry difference. */
148static PetscErrorCode DifferenceMaxRank(Vec a, Vec b, PetscMPIInt *max_rank)
149{
150 const PetscScalar *aa = NULL, *bb = NULL;
151 PetscInt nlocal;
152 PetscReal local_max = 0.0, *all_max = NULL;
153 PetscMPIInt size;
154 PetscFunctionBeginUser;
155 PetscCall(VecGetLocalSize(a, &nlocal));
156 PetscCall(VecGetArrayRead(a, &aa)); PetscCall(VecGetArrayRead(b, &bb));
157 for (PetscInt i = 0; i < nlocal; ++i) local_max = PetscMax(local_max, PetscAbsScalar(bb[i] - aa[i]));
158 PetscCall(VecRestoreArrayRead(b, &bb)); PetscCall(VecRestoreArrayRead(a, &aa));
159 PetscCallMPI(MPI_Comm_size(PETSC_COMM_WORLD, &size));
160 PetscCall(PetscMalloc1(size, &all_max));
161 PetscCallMPI(MPI_Allgather(&local_max, 1, MPIU_REAL, all_max, 1, MPIU_REAL, PETSC_COMM_WORLD));
162 *max_rank = 0;
163 for (PetscMPIInt r = 1; r < size; ++r) if (all_max[r] > all_max[*max_rank]) *max_rank = r;
164 PetscCall(PetscFree(all_max));
165 PetscFunctionReturn(PETSC_SUCCESS);
166}

172/**
173 * @brief Reconstructs Ucat/lUcat deterministically from X, mirroring the seed
174 * now performed by the production callback.
175 *
176 * Used only by the N-pass compatibility residual (Experiment 3), which needs a
177 * boundary-pass count other than the production default of one call (three
178 * passes). For a single call it reproduces the production callback exactly.
179 */
180static PetscErrorCode SeedCartesianFromX(UserCtx *user, Vec X)
181{
182 const char *staggered[] = {"Ucont"};
183 const char *cell[] = {"Ucat"};
184 PetscFunctionBeginUser;
185 PetscCall(VecCopy(X, user->Ucont));
186 PetscCall(SynchronizePeriodicStaggeredFields(user, 1, staggered));
187 PetscCall(Contra2Cart(user));
188 PetscCall(SynchronizePeriodicCellFields(user, 1, cell));
189 PetscCall(UpdateLocalGhosts(user, "Ucat"));
190 PetscFunctionReturn(PETSC_SUCCESS);
191}

193/**
194 * @brief Deterministic residual at X using an explicit number of boundary-condition
195 * calls (each three internal passes), for the boundary-map compatibility test.
196 */
197static PetscErrorCode ResidualNCalls(MomentumNewtonKrylovContext *ctx, Vec X,
198 PetscInt boundary_calls, Vec F)
199{
200 UserCtx *user = ctx->user;
201 PetscFunctionBeginUser;
202 PetscCall(SeedCartesianFromX(user, X));
203 for (PetscInt i = 0; i < boundary_calls; ++i) PetscCall(ApplyBoundaryConditions(user));
204 PetscCall(ComputeTotalResidual(user));
205 PetscCall(VecCopy(user->Rhs, F));
206 PetscCall(VecScale(F, -1.0));
207 PetscCall(MomentumNewtonKrylov_ApplyConstraints(ctx, X, F));
208 PetscFunctionReturn(PETSC_SUCCESS);
209}

236/** @brief Records per-Newton nonlinear norm, KSP iterations/reason, and accepted lambda. */
237static PetscErrorCode SolveMonitor(SNES snes, PetscInt it, PetscReal norm, void *vctx)
238{
239 SolveMonitorCtx *m = (SolveMonitorCtx *)vctx;
240 SNESLineSearch ls = NULL;
241 KSP ksp = NULL;
242 KSPConvergedReason kr = KSP_CONVERGED_ITERATING;
243 PetscReal lambda = -1.0;
244 PetscInt cum_ksp = 0, step_ksp = 0;
245 PetscFunctionBeginUser;
246 if (it == 0 && m->initial_norm) *m->initial_norm = norm;
247 PetscCall(SNESGetLinearSolveIterations(snes, &cum_ksp));
248 PetscCall(SNESGetKSP(snes, &ksp));
249 PetscCall(KSPGetConvergedReason(ksp, &kr));
250 PetscCall(KSPGetIterationNumber(ksp, &step_ksp));
251 PetscCall(SNESGetLineSearch(snes, &ls));
252 if (it > 0) PetscCall(SNESLineSearchGetLambda(ls, &lambda));
253 PetscCall(PetscPrintf(PETSC_COMM_WORLD,
254 "FULLSOLVE %s newton=%d nonlinear_norm=% .16e last_ksp_its=%d cumulative_ksp_its=%d "
255 "last_ksp_reason=%d accepted_lambda=% .6e\n",
256 m->label, (int)it, (double)norm, (int)step_ksp, (int)cum_ksp, (int)kr, (double)lambda));
257 PetscFunctionReturn(PETSC_SUCCESS);
258}

268/**
269 * @brief Creates a test-owned matrix-free SNES driven by the *production* residual callback.
270 * @param[in] user Test context supplying the velocity DMDA.
271 * @param[in] pctx Production-compatible residual context borrowed by the SNES.
272 * @param[in] use_mgs When false (production default), leaves GMRES orthogonalization at
273 * PETSc's default classical Gram--Schmidt -- the path the production solver uses.
274 * @param[out] snes Created nonlinear solver.
275 * @param[out] J Created matrix-free Jacobian operator.
276 * @param[out] ksp Borrowed linear solver configured by the created SNES.
277 * @return PetscErrorCode 0 on success.
278 */
279static PetscErrorCode CreateProductionSNES(UserCtx *user, MomentumNewtonKrylovContext *pctx,
280 PetscBool use_mgs, SNES *snes, Mat *J, KSP *ksp)
281{
282 PC pc = NULL;
283 PetscFunctionBeginUser;
284 PetscCall(SNESCreate(PETSC_COMM_WORLD, snes));
285 PetscCall(SNESSetOptionsPrefix(*snes, "mom_nk_"));
286 PetscCall(SNESSetType(*snes, SNESNEWTONLS));
287 PetscCall(SNESSetDM(*snes, user->fda));
288 PetscCall(SNESSetFunction(*snes, NULL, MomentumNewtonKrylov_FormResidual, pctx));
289 PetscCall(MatCreateSNESMF(*snes, J));
290 PetscCall(SNESSetJacobian(*snes, *J, *J, MatMFFDComputeJacobian, NULL));
291 PetscCall(SNESGetKSP(*snes, ksp));
292 PetscCall(KSPSetType(*ksp, KSPGMRES));
293 if (use_mgs) PetscCall(KSPGMRESSetOrthogonalization(*ksp, KSPGMRESModifiedGramSchmidtOrthogonalization));
294 PetscCall(KSPGMRESSetRestart(*ksp, 80));
295 PetscCall(KSPGetPC(*ksp, &pc));
296 PetscCall(PCSetType(pc, PCNONE));
297 PetscCall(SNESSetFromOptions(*snes));
298 PetscFunctionReturn(PETSC_SUCCESS);
299}

301/** @brief Runs one complete production-callback step-2 solve from the restored entry state. */
302static PetscErrorCode RunProductionFullSolve(UserCtx *user, const FpSnapshot *entry, Vec X0,
303 PetscBool use_mgs, const char *label, SolveResult *res)
304{
305 MomentumNewtonKrylovContext pctx = {user, NULL, PETSC_FALSE, 0.0};
306 SNES snes = NULL; Mat J = NULL; KSP ksp = NULL; Vec solution = NULL;
307 SolveMonitorCtx m;
308 PetscFunctionBeginUser;
309 memset(res, 0, sizeof(*res));
310 res->initial_norm = -1.0;
311 PetscCall(FpRestore(user, entry));
312 PetscCall(VecDuplicate(X0, &solution)); PetscCall(VecCopy(X0, solution));
313 PetscCall(CreateProductionSNES(user, &pctx, use_mgs, &snes, &J, &ksp));
314 m.label = label; m.initial_norm = &res->initial_norm;
315 PetscCall(SNESMonitorSet(snes, SolveMonitor, &m, NULL));
316 PetscCall(SNESSolve(snes, NULL, solution));
317 PetscCall(SNESGetConvergedReason(snes, &res->reason));
318 PetscCall(SNESGetIterationNumber(snes, &res->its));
319 PetscCall(SNESGetNumberFunctionEvals(snes, &res->fevals));
320 PetscCall(SNESGetLinearSolveIterations(snes, &res->ksp_total));
321 PetscCall(SNESGetFunctionNorm(snes, &res->final_norm));
322 res->converged = (PetscBool)(res->reason > 0);
323 PetscCall(VecDuplicate(solution, &res->Xfinal)); PetscCall(VecCopy(solution, res->Xfinal));
324 PetscCall(PetscPrintf(PETSC_COMM_WORLD,
325 "FULLSOLVE_SUMMARY %s reason=%s (%d) newton=%d fevals=%d ksp_total=%d orthogonalization=%s "
326 "initial_norm=% .16e final_norm=% .16e -> %s\n",
327 label, SNESConvergedReasons[res->reason], (int)res->reason, (int)res->its, (int)res->fevals,
328 (int)res->ksp_total, use_mgs ? "modified-GS" : "classical-GS(default)",
329 (double)res->initial_norm, (double)res->final_norm, res->converged ? "CONVERGED" : "FAILED"));
330 PetscCall(VecDestroy(&solution)); PetscCall(MatDestroy(&J)); PetscCall(SNESDestroy(&snes));
331 PetscFunctionReturn(PETSC_SUCCESS);
332}

351/**
352 * @brief Builds the production-sized duct, advances the true physical step 1, and
353 * exercises purity, full-solve convergence, boundary-map compatibility, and
354 * projection using the production Newton--Krylov callback and solver.
355 */
356static PetscErrorCode TestNewtonKrylovProductionRegression(void)
357{
358 SimCtx *simCtx = NULL;
359 UserCtx *user = NULL;
360 char tmpdir[PETSC_MAX_PATH_LEN] = "";
361 FpSnapshot entry = {0};
362 MomentumNewtonKrylovContext pctx;
363 Vec X0 = NULL, X1 = NULL, Xfinal = NULL;
364 Vec F0 = NULL, F1 = NULL, Fafter = NULL, v = NULL, jv = NULL;
365 Vec F3 = NULL, F24 = NULL;
366 SolveResult res_cgs = {0};
367 PetscReal norms[2] = {0.0, 0.0};
368 PetscReal imm_inf = 0.0, imm_2 = 0.0, mffd_inf = 0.0, mffd_2 = 0.0;
369 PetscReal f3_2 = 0.0, f24_2 = 0.0, f24_inf = 0.0, f24m3_inf = 0.0, f24m3_2 = 0.0;
370 PetscReal max_div = -1.0;
371 const char *fields[] = {"Ucont"};
372 const char *stag[] = {"Ucont"}; const char *cell[] = {"Ucat"};
373 const PetscReal purity_inf_tol = 1e-13, purity_2_tol = 1e-12;
374 const PetscReal compat_tol = 1e-8, div_tol = 1e-6;
375 PetscMPIInt world, mffd_rank = 0;
376 PetscErrorCode prod_ierr;
377
378 PetscFunctionBeginUser;
379 PetscCallMPI(MPI_Comm_size(PETSC_COMM_WORLD, &world));
380 PetscCall(PetscPrintf(PETSC_COMM_WORLD,
381 "\n########## NEWTON-KRYLOV PRODUCTION REGRESSION (ranks=%d) ##########\n", (int)world));
382 PetscCall(PicurvBuildMomentumPurityRuntimeContext(NULL, &simCtx, &user, tmpdir, sizeof(tmpdir)));
383 PetscCall(InitializeEulerianState(simCtx));
384 PetscCall(ConfigureNewtonOptions());
385
386 /* Complete physical step 1 (momentum + Poisson + projection + finalization)
387 * through the real run path, advance histories, then enter step 2. */
388 simCtx->step = 1; simCtx->ti = simCtx->dt;
389 PetscCall(FlowSolver(simCtx));
390 PetscCall(UpdateSolverHistoryVectors(user));
391 simCtx->step = 2; simCtx->ti = 2.0 * simCtx->dt;
392
393 PetscCall(VecDuplicate(user->Ucont, &user->Rhs));
394 pctx.user = user; pctx.history_file = NULL; pctx.have_initial_norm = PETSC_FALSE; pctx.initial_norm = 0.0;
395
396 /* Canonicalize the entry exactly as production MomentumSolver_NewtonKrylov does. */
397 PetscCall(SynchronizePeriodicStaggeredFields(user, 1, fields));
398 PetscCall(ApplyBoundaryConditions(user));
399 PetscCall(FpCapture(user, &entry));
400 PetscCall(VecDuplicate(user->Ucont, &X0)); PetscCall(VecCopy(user->Ucont, X0));
401
402 /* -------- 4.1 residual purity at the step-2 / Newton-1 state -------- */
403 PetscCall(PetscPrintf(PETSC_COMM_WORLD, "\n===== 4.1 production-callback residual purity =====\n"));
404 { /* Reach the accepted Newton-1 state X1 with the production callback + default CGS. */
405 SNES snes = NULL; Mat J = NULL; KSP ksp = NULL;
406 PetscCall(VecDuplicate(X0, &X1)); PetscCall(VecCopy(X0, X1));
407 PetscCall(CreateProductionSNES(user, &pctx, PETSC_FALSE, &snes, &J, &ksp));
408 PetscCall(SNESSetConvergenceTest(snes, StopAfterOne, norms, NULL));
409 PetscCall(SNESSolve(snes, NULL, X1));
410 PetscCall(MatDestroy(&J)); PetscCall(SNESDestroy(&snes));
411 }
412 PetscCall(VecDuplicate(X1, &F0)); PetscCall(VecDuplicate(X1, &F1));
413 PetscCall(VecDuplicate(X1, &Fafter)); PetscCall(VecDuplicate(X1, &v)); PetscCall(VecDuplicate(X1, &jv));
414
415 /* Immediate repeatability with hidden state poisoned between the two calls. */
416 PetscCall(MomentumNewtonKrylov_FormResidual(NULL, X1, F0, &pctx));
417 simCtx->FluxInSum = 1234.0; simCtx->FluxOutSum = -4321.0;
418 simCtx->FarFluxInSum = 77.0; simCtx->FarFluxOutSum = -88.0;
419 PetscCall(VecSet(user->Ucat, 7.0)); PetscCall(VecSet(user->lUcat, 7.0));
420 PetscCall(MomentumNewtonKrylov_FormResidual(NULL, X1, F1, &pctx));
421 PetscCall(CollectiveDifference(F0, F1, &imm_inf, &imm_2));
422 PetscCall(PetscPrintf(PETSC_COMM_WORLD,
423 "PURITY immediate inf=% .3e l2=% .3e (tol inf=%.0e l2=%.0e)\n",
424 (double)imm_inf, (double)imm_2, (double)purity_inf_tol, (double)purity_2_tol));
425
426 /* Post-MFFD repeatability: real PETSc MFFD products, then reevaluate. */
427 {
428 SNES snes = NULL; Mat J = NULL; KSP ksp = NULL;
429 PetscInt lo, hi; PetscScalar *a = NULL; PetscReal vn;
430 PetscCall(VecGetOwnershipRange(v, &lo, &hi));
431 PetscCall(VecGetArray(v, &a));
432 for (PetscInt i = lo; i < hi; ++i) a[i - lo] = PetscSinReal(0.17 * (PetscReal)(i + 1));
433 PetscCall(VecRestoreArray(v, &a));
434 PetscCall(VecNorm(v, NORM_2, &vn)); PetscCall(VecScale(v, 1.0 / vn));
435 PetscCall(CreateProductionSNES(user, &pctx, PETSC_FALSE, &snes, &J, &ksp));
436 PetscCall(MatMFFDSetBase(J, X1, F0));
437 for (PetscInt i = 0; i < 40; ++i) PetscCall(MatMult(J, v, jv));
438 PetscCall(MomentumNewtonKrylov_FormResidual(NULL, X1, Fafter, &pctx));
439 PetscCall(CollectiveDifference(F0, Fafter, &mffd_inf, &mffd_2));
440 PetscCall(DifferenceMaxRank(F0, Fafter, &mffd_rank));
441 PetscCall(MatDestroy(&J)); PetscCall(SNESDestroy(&snes));
442 }
443 PetscCall(PetscPrintf(PETSC_COMM_WORLD,
444 "PURITY post_mffd inf=% .3e l2=% .3e max_rank=%d\n",
445 (double)mffd_inf, (double)mffd_2, (int)mffd_rank));
446
447 /* -------- 4.2 full step-2 solve with default classical Gram-Schmidt -------- */
448 PetscCall(PetscPrintf(PETSC_COMM_WORLD, "\n===== 4.2 full production step-2 solve (default classical GS) =====\n"));
449 PetscCall(RunProductionFullSolve(user, &entry, X0, PETSC_FALSE, "CGS-default", &res_cgs));
450
451 /* Also drive the ACTUAL production solver wrapper (default GS) and take X_final
452 * from it. The wrapper requires user->Rhs unallocated on entry. */
453 PetscCall(FpRestore(user, &entry));
454 PetscCall(VecDestroy(&user->Rhs));
455 prod_ierr = MomentumSolver_NewtonKrylov_BoundaryFixedpointPrivateCopy(user, NULL, NULL);
456 PetscCall(VecDuplicate(user->Ucont, &user->Rhs));
457 PetscCall(VecDuplicate(user->Ucont, &Xfinal)); PetscCall(VecCopy(user->Ucont, Xfinal));
458 PetscCall(PetscPrintf(PETSC_COMM_WORLD,
459 "PROD_SOLVER ierr=%d mom_last_converged=%d\n",
460 (int)prod_ierr, (int)simCtx->mom_last_converged));
461 {
462 PetscReal dinf, d2; PetscMPIInt r;
463 PetscCall(CollectiveDifference(res_cgs.Xfinal, Xfinal, &dinf, &d2));
464 PetscCall(DifferenceMaxRank(res_cgs.Xfinal, Xfinal, &r));
465 PetscCall(PetscPrintf(PETSC_COMM_WORLD,
466 "PROD_SOLVER test_snes_vs_wrapper inf=% .3e l2=% .3e max_rank=%d\n",
467 (double)dinf, (double)d2, (int)r));
468 }
469
470 /* -------- 4.3 boundary-map compatibility at the converged solution -------- */
471 PetscCall(PetscPrintf(PETSC_COMM_WORLD, "\n===== 4.3 boundary-map compatibility (3 vs 24 passes) =====\n"));
472 PetscCall(VecDuplicate(Xfinal, &F3)); PetscCall(VecDuplicate(Xfinal, &F24));
473 PetscCall(FpRestore(user, &entry));
474 PetscCall(ResidualNCalls(&pctx, Xfinal, 1, F3)); /* 3 passes == production callback */
475 PetscCall(VecNorm(F3, NORM_2, &f3_2));
476 PetscCall(FpRestore(user, &entry));
477 PetscCall(ResidualNCalls(&pctx, Xfinal, 8, F24)); /* 24 passes */
478 PetscCall(VecNorm(F24, NORM_2, &f24_2));
479 PetscCall(VecNorm(F24, NORM_INFINITY, &f24_inf));
480 PetscCall(CollectiveDifference(F3, F24, &f24m3_inf, &f24m3_2));
481 PetscCall(PetscPrintf(PETSC_COMM_WORLD,
482 "COMPAT F3_l2=% .6e F24_l2=% .6e F24_inf=% .6e F24_minus_F3_l2=% .6e (tol %.0e)\n",
483 (double)f3_2, (double)f24_2, (double)f24_inf, (double)f24m3_2, (double)compat_tol));
484
485 /* -------- 4.4 projection compatibility -------- */
486 PetscCall(PetscPrintf(PETSC_COMM_WORLD, "\n===== 4.4 projection compatibility =====\n"));
487 PetscCall(FpRestore(user, &entry));
488 PetscCall(VecCopy(Xfinal, user->Ucont));
489 PetscCall(SynchronizePeriodicStaggeredFields(user, 1, stag));
490 PetscCall(Contra2Cart(user));
491 PetscCall(SynchronizePeriodicCellFields(user, 1, cell));
492 PetscCall(UpdateLocalGhosts(user, "Ucat"));
493 PetscCall(ApplyBoundaryConditions(user));
494 PetscCall(PoissonSolver_MG(&simCtx->usermg));
495 PetscCall(UpdatePressure(user));
496 PetscCall(Projection(user));
497 PetscCall(UpdateLocalGhosts(user, "P"));
498 PetscCall(ComputeDivergence(user));
499 max_div = simCtx->MaxDiv;
500 PetscCall(PetscPrintf(PETSC_COMM_WORLD,
501 "PROJECTION max_divergence=% .6e net_divergence_sum=% .6e (tol %.0e)\n",
502 (double)max_div, (double)simCtx->summationRHS, (double)div_tol));
503
504 /* -------- consolidated verdicts (printed before assertions) -------- */
505 PetscCall(PetscPrintf(PETSC_COMM_WORLD, "\nREGRESSION_VERDICTS (ranks=%d)\n", (int)world));
506 PetscCall(PetscPrintf(PETSC_COMM_WORLD, " PRODUCTION_PURITY: %s\n",
507 (imm_inf <= purity_inf_tol && imm_2 <= purity_2_tol &&
508 mffd_inf <= purity_inf_tol && mffd_2 <= purity_2_tol) ? "PASS" : "FAIL"));
509 PetscCall(PetscPrintf(PETSC_COMM_WORLD, " FULL_STEP2_DEFAULT_GS: %s\n",
510 res_cgs.converged ? "PASS" : "FAIL"));
511 PetscCall(PetscPrintf(PETSC_COMM_WORLD, " PRODUCTION_SOLVER_WRAPPER: %s\n",
512 (prod_ierr == 0 && simCtx->mom_last_converged) ? "PASS" : "FAIL"));
513 PetscCall(PetscPrintf(PETSC_COMM_WORLD, " THREE_PASS_24_PASS_COMPATIBLE: %s\n",
514 (f24_2 <= compat_tol) ? "PASS" : "FAIL"));
515 PetscCall(PetscPrintf(PETSC_COMM_WORLD, " PROJECTION_DIVERGENCE_FREE: %s\n",
516 (max_div <= div_tol) ? "PASS" : "FAIL"));
517
518 /* -------- assertions -------- */
519 PetscCheck(imm_inf <= purity_inf_tol && imm_2 <= purity_2_tol, PETSC_COMM_WORLD, PETSC_ERR_PLIB,
520 "production residual not immediately repeatable (inf=%g l2=%g)", (double)imm_inf, (double)imm_2);
521 PetscCheck(mffd_inf <= purity_inf_tol && mffd_2 <= purity_2_tol, PETSC_COMM_WORLD, PETSC_ERR_PLIB,
522 "production residual changed after MFFD products (inf=%g l2=%g)", (double)mffd_inf, (double)mffd_2);
523 PetscCheck(res_cgs.converged, PETSC_COMM_WORLD, PETSC_ERR_PLIB,
524 "full step-2 solve (default GS) did not converge (reason=%d)", (int)res_cgs.reason);
525 PetscCheck(res_cgs.reason != SNES_DIVERGED_LINE_SEARCH && res_cgs.reason != SNES_DIVERGED_LINEAR_SOLVE,
526 PETSC_COMM_WORLD, PETSC_ERR_PLIB, "full step-2 solve diverged via line search or linear solve (reason=%d)",
527 (int)res_cgs.reason);
528 PetscCheck(prod_ierr == 0 && simCtx->mom_last_converged, PETSC_COMM_WORLD, PETSC_ERR_PLIB,
529 "production Newton-Krylov solver wrapper did not converge (ierr=%d)", (int)prod_ierr);
530 PetscCheck(f24_2 <= compat_tol, PETSC_COMM_WORLD, PETSC_ERR_PLIB,
531 "converged three-pass solution does not satisfy the 24-pass residual (||F24||2=%g)", (double)f24_2);
532 PetscCheck(max_div <= div_tol, PETSC_COMM_WORLD, PETSC_ERR_PLIB,
533 "projection of the converged momentum state is not divergence free (max_div=%g)", (double)max_div);
534
535 PetscCall(VecDestroy(&F24)); PetscCall(VecDestroy(&F3));
536 PetscCall(VecDestroy(&Xfinal)); PetscCall(VecDestroy(&res_cgs.Xfinal));
537 PetscCall(VecDestroy(&jv)); PetscCall(VecDestroy(&v));
538 PetscCall(VecDestroy(&Fafter)); PetscCall(VecDestroy(&F1)); PetscCall(VecDestroy(&F0));
539 PetscCall(VecDestroy(&X1)); PetscCall(VecDestroy(&X0));
540 PetscCall(FpDestroy(&entry));
541 PetscCall(VecDestroy(&user->Rhs));
542 PetscCall(PicurvDestroyRuntimeContext(&simCtx));
543 PetscCall(PicurvRemoveTempDir(tmpdir));
544 PetscFunctionReturn(PETSC_SUCCESS);
545}
