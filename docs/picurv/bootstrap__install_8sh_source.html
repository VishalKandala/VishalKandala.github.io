92python_runtime_library_dirs() {
93 local python_bin="$1"
94 local resolved_python
95
96 resolved_python="$(command -v "${python_bin}" || printf '%s' "${python_bin}")"
97 "${python_bin}" - <<'PY'
98import os
99import sysconfig
100
101for key in ("LIBDIR", "LIBPL"):
102 value = sysconfig.get_config_var(key)
103 if value and os.path.isdir(value):
104 print(os.path.realpath(value))
105PY
106 if command -v ldd >/dev/null 2>&1 && [[ -x "${resolved_python}" ]]; then
107 ldd "${resolved_python}" 2>/dev/null | awk '
108 /=> \// { path = $3 }
109 /^[[:space:]]*\// { path = $1 }
110 path != "" {
111 sub(/\/[^\/]+$/, "", path)
112 print path
113 path = ""
114 }
115 '
116 fi
117}

136install_shell_hook() {
137 local rc_path="$1"
138 local hook_path="${REPO_ROOT}/etc/picurv.sh"
139 local begin_marker="# >>> PICurv shell setup >>>"
140 local end_marker="# <<< PICurv shell setup <<<"
141 local tmp_path
142
143 mkdir -p "$(dirname "${rc_path}")"
144 touch "${rc_path}"
145 tmp_path="$(mktemp "${rc_path}.picurv.XXXXXX")"
146 awk -v begin="${begin_marker}" -v end="${end_marker}" '
147 $0 == begin {skip = 1; next}
148 $0 == end {skip = 0; next}
149 skip != 1 {print}
150 ' "${rc_path}" > "${tmp_path}"
151 {
152 printf '%s\n' "${begin_marker}"
153 printf 'source %q\n' "${hook_path}"
154 printf '%s\n' "${end_marker}"
155 } >> "${tmp_path}"
156 mv "${tmp_path}" "${rc_path}"
157}
