40 * @brief Internal helper implementation: `ComputeTotalResidual()`.
41 * @details Local to this translation unit.
42 */
43static PetscErrorCode ComputeTotalResidual(UserCtx *user)
44{
45 PetscErrorCode ierr;
46 SimCtx *simCtx = user->simCtx;
47
48 // Extract Time Parameters from Context.
49 // BDF step selection now lives in MomentumUsesBDF2() (shared with the stability estimate).
50 const PetscReal dt = simCtx->dt;
51
52 PetscFunctionBeginUser;
53 PROFILE_FUNCTION_BEGIN;
54 // 1. Calculate Spatial Terms (stored in user->Rhs)
55 // Rhs = -Div(Flux) + Viscous + Source
56 ierr = ComputeRHS(user, user->Rhs); CHKERRQ(ierr);
57
58 // 2. Add Physical Time Derivative Terms (BDF Discretization)
59 // The equation solved is: dU/dtau = RHS_Spatial + RHS_Temporal
60
61 /* BDF order selected by the shared helper (single source of truth, shared with
62 the momentum stability estimate). Numerically identical to the previous
63 inlined predicate: a0 = 1.5 for BDF2, 1.0 for BDF1. */
64 if (MomentumUsesBDF2(simCtx)) {
65 // --- BDF2 (Second Order Backward Difference) ---
66 // Formula: (1.5*U^{n} - 2.0*U^{n-1} + 0.5*U^{n-2}) / dt = RHS_Spatial(U^{n})
67 // Moved to : -1.5/dt * U^{n} + 2.0/dt * U^{n-1} - 0.5/dt * U^{n-2} + RHS_Spatial(U^{n})
68 ierr = VecAXPY(user->Rhs, -COEF_TIME_ACCURACY/dt, user->Ucont);
69 ierr = VecAXPY(user->Rhs, +2.0/dt, user->Ucont_o);
70 ierr = VecAXPY(user->Rhs, -0.5/dt, user->Ucont_rm1);
71 } else {
72 // --- BDF1 (First Order / Euler Implicit) ---
73 // Formula: (U^{n} - U^{n-1}) / dt = RHS_Spatial(U^{n})
74 // Moved to RHS: -1.0/dt * U^{n} + 1.0/dt * U^{n-1} + RHS_Spatial(U^{n})
75 ierr = VecAXPY(user->Rhs, -1.0/dt, user->Ucont);
76 ierr = VecAXPY(user->Rhs, +1.0/dt, user->Ucont_o);
77 }
78
79 // 3. Enforce Boundary Conditions on the Residual
80 ierr = EnforceRHSBoundaryConditions(user); CHKERRQ(ierr);
81
82 PROFILE_FUNCTION_END;
83 PetscFunctionReturn(0);
84}

197/**
198 * @brief Compute a conservative global pseudo-time spectral radius estimate.
199 *
200 * The spectral radius lambda_max (units: 1/s) governs the pseudo-time stability limit
201 * for the explicit Jameson 4-stage RK smoother. The pseudo-time step is then:
202 *
203 * dtau = pseudo_cfl / lambda_max
204 *
205 * making pseudo_cfl a true dimensionless Courant number independent of the physical
206 * timestep dt. Stable range for 4-stage Jameson RK: pseudo_cfl in [0, ~2.83].
207 *
208 * Per owned cell (k,j,i) the convective spectral radius contribution is:
209 *
210 * lambda_cell = (|U_xi_flux| + |U_eta_flux| + |U_zeta_flux|) x (1/V)
211 *
212 * where:
213 * - U_xi_flux = ucont[k][j][i].x : volumetric flux through the xi-face [m3/s]
214 * - U_eta_flux = ucont[k][j][i].y : volumetric flux through the eta-face [m3/s]
215 * - U_zeta_flux = ucont[k][j][i].z : volumetric flux through the zeta-face [m3/s]
216 * - 1/V = lAj[k][j][i] : inverse cell volume (cell Jacobian) [1/m3]
217 *
218 * Product units: [m3/s * 1/m3] = [1/s] = spectral radius.
219 *
220 * Only the 'positive' face at index (k,j,i) is used per direction (no neighbor
221 * access), so no ghost-cell scatter is needed. lUcont is valid after the pre-loop
222 * SynchronizePeriodicStaggeredFields; lAj is set at grid initialisation and never
223 * changes. No DMGlobalToLocal calls are added.
224 *
225 * A BDF2 lower bound COEF_TIME_ACCURACY/dt is applied after the MPI reduction so
226 * that zero-flow startup (all ucont == 0) gives dtau = pseudo_cfl * dt / 1.5 rather
227 * than infinity.
228 *
229 * @param[in] user Array of UserCtx (one per block).
230 * @param[in] block_number Number of blocks.
231 * @param[in] dt Physical time step (for the BDF2 lower bound).
232 * @param[out] lambda_max_out Global spectral radius [1/s].
233 */
234static PetscErrorCode ComputeGlobalSpectralRadiusEstimate(
235 UserCtx *user, PetscInt block_number, PetscReal dt, PetscReal *lambda_max_out)
236{
237 PetscErrorCode ierr;
238 PetscReal local_max = 0.0;
239
240 PetscFunctionBeginUser;
241
242 for (PetscInt bi = 0; bi < block_number; bi++) {
243 DMDALocalInfo info = user[bi].info; /* pre-stored local ownership info */
244 Cmpnts ***ucont;
245 PetscReal ***aj;
246
247 /* Read-only access: lUcont (valid after SynchronizePeriodicStaggeredFields),
248 lAj (grid metric, unchanged after initialisation). */
249 ierr = DMDAVecGetArrayRead(user[bi].fda, user[bi].lUcont, &ucont); CHKERRQ(ierr);
250 ierr = DMDAVecGetArrayRead(user[bi].da, user[bi].lAj, &aj); CHKERRQ(ierr);
251
252 /* Loop over owned cells only — no ghost-cell indices accessed. */
253 for (PetscInt k = info.zs; k < info.zs + info.zm; k++) {
254 for (PetscInt j = info.ys; j < info.ys + info.ym; j++) {
255 for (PetscInt i = info.xs; i < info.xs + info.xm; i++) {
256 /* Sum absolute face-flux magnitudes (one face per coordinate direction).
257 * Using only the 'positive' face (index i/j/k, not i-1/j-1/k-1) avoids
258 * neighbor access. For smooth velocity fields the error is < 2x and the
259 * adaptive CFL controller handles any residual conservatism. */
260 PetscReal flux_sum = PetscAbsReal(ucont[k][j][i].x)
261 + PetscAbsReal(ucont[k][j][i].y)
262 + PetscAbsReal(ucont[k][j][i].z);
263
264 /* aj = 1/J = 1/cell_volume; lambda = flux/volume [1/s] */
265 PetscReal lambda = flux_sum * aj[k][j][i];
266 local_max = PetscMax(local_max, lambda);
267 }
268 }
269 }
270
271 ierr = DMDAVecRestoreArrayRead(user[bi].fda, user[bi].lUcont, &ucont); CHKERRQ(ierr);
272 ierr = DMDAVecRestoreArrayRead(user[bi].da, user[bi].lAj, &aj); CHKERRQ(ierr);
273 }
274
275 /* Reduce across all MPI ranks to get the global maximum. */
276 ierr = MPI_Allreduce(&local_max, lambda_max_out, 1, MPIU_REAL, MPI_MAX, PETSC_COMM_WORLD);
277 CHKERRQ(ierr);
278
279 /* BDF2 lower bound: if the velocity field is zero (startup, stagnant IC) lambda_max
280 would be 0, giving dtau = inf. Clamp to 1.5/dt so dtau = pseudo_cfl * dt/1.5 in
281 that degenerate case (conservative but finite). */
282 *lambda_max_out = PetscMax(*lambda_max_out, COEF_TIME_ACCURACY / dt);
283
284 PetscFunctionReturn(0);
285}

299/* nvert bands matching the physical operator:
300 * SKIP : cell excluded from the estimate (solid/blanked), nvert > 0.1.
301 * VISC1S : Viscous() one-sided cross-derivative trigger band, (0.5, 7).
302 * QUICKBLK: Convection() QUICK-modifier band, [0.1, 7] (mirrors NOT(<0.1 || >innerblank)). */
303#define MOM_SKIP_SOLID(v) ((v) > 0.1)
304#define MOM_VISC_ONESIDED(v) ((v) > 0.5 && (v) < 7.0)
305#define MOM_QUICK_BLOCKS(v) ((v) >= 0.1 && (v) <= 7.0)

307/* Read-only: does ANY of the six viscous faces entering cell (k,j,i) use an
308 * nvert-selected one-sided cross-derivative? Conservative solid-band check over
309 * the full 3x3x3 neighborhood that the per-face Viscous() predicates can reach.
310 * Indices i +/- 1, j +/- 1, k +/- 1 are always valid on the interior loop range. */
311static PetscBool MomCellUsesOneSidedViscousStencil(PetscReal ***nvert, PetscInt k, PetscInt j, PetscInt i)
312{
313 for (PetscInt dk = -1; dk <= 1; dk++)
314 for (PetscInt dj = -1; dj <= 1; dj++)
315 for (PetscInt di = -1; di <= 1; di++) {
316 if (!dk && !dj && !di) continue;
317 if (MOM_VISC_ONESIDED(nvert[k+dk][j+dj][i+di])) return PETSC_TRUE;
318 }
319 return PETSC_FALSE;
320}

335/* Read-only: is the QUICK reconstruction in one computational direction modified for
336 * cell (k,j,i)? The cell uses BOTH faces in that direction (index f and f-1), whose
337 * QUICK branches are gated by a non-periodic domain edge or by nvert blockers within
338 * the stencil reach (including the second neighbor f-2). The +/-2 reads are guarded by
339 * the ghost range [g0, g1) so they are safe for both periodic and non-periodic DMs. */
340static PetscBool MomQuickDirModified(PetscReal ***nvert, PetscInt a, PetscInt b, PetscInt c,
341 PetscInt idx, PetscInt m, PetscBool np0, PetscBool np1,
342 PetscInt g0, PetscInt g1, char dir)
343{
344 /* (a,b,c) are the fixed indices; idx is the moving index in direction `dir`. */
345 if (np0 && idx <= 1) return PETSC_TRUE; /* faces idx,idx-1 reach the negative edge */
346 if (np1 && idx >= m-2) return PETSC_TRUE; /* ... or the positive edge */
347 for (PetscInt d = -2; d <= 2; d++) {
348 if (d == 0) continue;
349 const PetscInt p = idx + d;
350 if (p < g0 || p >= g1) continue; /* outside local ghost range: treat as fluid */
351 PetscReal v;
352 if (dir == 'i') v = nvert[a][b][p];
353 else if (dir == 'j') v = nvert[a][p][c];
354 else v = nvert[p][b][c];
355 if (MOM_QUICK_BLOCKS(v)) return PETSC_TRUE;
356 }
357 return PETSC_FALSE;
358}

360/* Active-row mask (3 bits: xi=1, eta=2, zeta=4) mirroring the final RHS masking.
361 * A solid cell disables all rows; a positive solid neighbour or a positive non-periodic
362 * physical face disables the corresponding normal (staggered) row; TwoD disables the
363 * homogeneous direction. Returns 0 only when the location carries no active unknown. */
364static PetscInt MomCellActiveRows(PetscReal ***nvert, PetscInt k, PetscInt j, PetscInt i,
365 PetscInt mx, PetscInt my, PetscInt mz,
366 PetscBool np_x1, PetscBool np_y1, PetscBool np_z1, PetscInt twoD)
367{
368 if (MOM_SKIP_SOLID(nvert[k][j][i])) return 0; /* solid cell: all rows inactive */
369 PetscInt bits = 0x7;
370 if (MOM_SKIP_SOLID(nvert[k][j][i+1])) bits &= ~0x1; /* positive xi neighbour solid */
371 if (MOM_SKIP_SOLID(nvert[k][j+1][i])) bits &= ~0x2; /* positive eta neighbour solid */
372 if (MOM_SKIP_SOLID(nvert[k+1][j][i])) bits &= ~0x4; /* positive zeta neighbour solid */
373 if (np_x1 && i == mx-2) bits &= ~0x1; /* positive non-periodic xi face */
374 if (np_y1 && j == my-2) bits &= ~0x2;
375 if (np_z1 && k == mz-2) bits &= ~0x4;
376 if (twoD == 1) bits &= ~0x1; /* TwoD: homogeneous xi direction */
377 return bits;
378}

528 + fy * (PetscAbsReal(Uyp)+PetscAbsReal(Uym))
529 + fz * (PetscAbsReal(Uzp)+PetscAbsReal(Uzm)) );
530 const PetscReal lcC = lcB + 0.5 * Ajc * divU;

532 (the nonlinear zero-order Jacobian term delta_u . grad u).
533 Physical gradient: d/dx_dir = Ajc*(Csi.dir d/dxi + Eta.dir d/deta + Zet.dir d/dzeta). */

538 0.5*(ucat[k][j+1][i].y-ucat[k][j-1][i].y),
539 0.5*(ucat[k][j+1][i].z-ucat[k][j-1][i].z) };
540 const Cmpnts duz = { 0.5*(ucat[k+1][j][i].x-ucat[k-1][j][i].x),
541 0.5*(ucat[k+1][j][i].y-ucat[k-1][j][i].y),
542 0.5*(ucat[k+1][j][i].z-ucat[k-1][j][i].z) };
543 const Cmpnts C = csi[k][j][i], E = eta[k][j][i], Z = zet[k][j][i];

545 PetscAbsReal(Ajc*(C.x*duc.cmp + E.x*due.cmp + Z.x*duz.cmp)) + \
546 PetscAbsReal(Ajc*(C.y*duc.cmp + E.y*due.cmp + Z.y*duz.cmp)) + \

575 }

577 lv += PetscAbsReal(nuf) * Ajc * jaj[k][fj][i]
578 * MomFaceGabs(jeta[k][fj][i], jcsi[k][fj][i], jzet[k][fj][i]);
579 }
580 /* zeta+ (k), zeta- (k-1) */

583 PetscReal nuf = nu_mol;
584 if (has_nut) {

586 if ((wzn && fk==0) || (wzp && fk==mz-2)) nt = 0.0;
587 nuf += nt;
588 }
589 /* zeta-face normal is the zeta metric (kzet) -> pass as N. */
590 lv += PetscAbsReal(nuf) * Ajc * kaj[fk][j][i]
591 * MomFaceGabs(kzet[fk][j][i], kcsi[fk][j][i], keta[fk][j][i]);
592 }
593 lv *= 4.0; /* 2 (two-face row-sum) x 2 (longitudinal full-stress) */
594 }

597 applied ONCE per cell, after a complete 3x3x3 solid-band check. */
598 PetscInt one_sided = 0;

673 rep->lambda_v = dbuf[1];
674 rep->lambda = sel_global;
675 rep->lambda_B = glo3[0]; rep->lambda_C = glo3[1]; rep->lambda_D = glo3[2];
676 rep->ci = ibuf[0]; rep->cj = ibuf[1]; rep->ck = ibuf[2];
677 rep->cblock = ibuf[3]; rep->cclass = ibuf[4]; rep->one_sided = ibuf[5];
678 rep->active_cells = global_active;
679
680 /* Only a genuinely empty active set (all cells masked) may fall back to the temporal
681 term alone. A non-finite or non-positive estimate with active cells is a hard error. */
682 if (global_active == 0) {
683 rep->lambda = lambda_t;
684 } else {
685 PetscCheck(PetscIsNormalReal(rep->lambda) && rep->lambda > 0.0, PETSC_COMM_WORLD, PETSC_ERR_FP,
686 "momentum stability estimate is non-finite or non-positive with %" PetscInt_FMT
687 " active cells", global_active);
688 }
689
690 /* Dominant limiter at the controlling cell. */
691 if (rep->lambda_t >= rep->lambda_c && rep->lambda_t >= rep->lambda_v) rep->limiter = MOM_STAB_LIMITER_TIME;
692 else if (rep->lambda_c >= rep->lambda_v) rep->limiter = MOM_STAB_LIMITER_CONVECTION;
693 else rep->limiter = MOM_STAB_LIMITER_VISCOSITY;
694
695 PetscFunctionReturn(0);
696}

813
814 /* --- Spectral Radius Estimate (computed once per physical timestep) ---
815 * lambda_max [1/s] is the global maximum of (face_flux_sum * inverse_cell_volume)
816 * over all owned cells across all MPI ranks. See ComputeGlobalSpectralRadiusEstimate().
817 *
818 * dtau = pseudo_cfl / lambda_max (pseudo_cfl is now a true Courant number)
819 * dtau_min/max are derived from the user-specified min/max_pseudo_cfl bounds.
820 *
821 * The BDF2 lower bound inside the helper ensures lambda_max >= 1.5/dt, so dtau
822 * remains finite even when the velocity field is zero (startup / stagnant ICs). */
823 ierr = ComputeGlobalSpectralRadiusEstimate(user, block_number, dt, &lambda_max); CHKERRQ(ierr);
824 dtau_min = simCtx->min_pseudo_cfl / lambda_max;
825 dtau_max = simCtx->max_pseudo_cfl / lambda_max;
826
827 /* Initialise per-block pseudo-time step from the warm-start CFL (carried from last timestep). */
828 for (PetscInt bi = 0; bi < block_number; bi++) {
829 pseudo_dtau[bi] = cfl / lambda_max; /* dtau [physical time], NOT a fraction of dt */
830 }
831
832 /* --- SHADOW MODE: new operator-scaled stability estimate (diagnostic only) ---
833 * Computes the Workstream-A estimate alongside the legacy one and logs a compact
834 * comparison. The legacy estimate above continues to drive production dtau; the
835 * shadow estimate changes nothing. Enable with -mom_stability_shadow. */
836 {
837 PetscBool shadow = PETSC_FALSE;
838 ierr = PetscOptionsGetBool(NULL, NULL, "-mom_stability_shadow", &shadow, NULL); CHKERRQ(ierr);
839 if (shadow) {
840 MomStabilityReport rep;
841 ierr = ComputeMomentumStabilityEstimate(user, block_number, dt, MOM_STAB_CAND_C, &rep); CHKERRQ(ierr);
842 const char *lim = (rep.limiter==MOM_STAB_LIMITER_TIME) ? "time"
843 : (rep.limiter==MOM_STAB_LIMITER_CONVECTION) ? "convection" : "viscosity";
844 /* Detailed shadow comparison at DEBUG only (gated by level + function allowlist);
845 never printed unconditionally. Enable in tests via the logging controls. */
846 LOG_ALLOW(GLOBAL, LOG_DEBUG,
847 "Momentum scale [shadow]: legacy_dtau=%.4e new_dtau=%.4e ratio=%.4f limiter=%s | "
848 "lambda_legacy=%.4e lambda_new=%.4e (B=%.4e C=%.4e D=%.4e) | "
849 "lt=%.4e lc=%.4e lv=%.4e | cell=(%d,%d,%d) blk=%d class=%d onesided=%d\n",
850 cfl / lambda_max, cfl / rep.lambda, rep.lambda / lambda_max, lim,
851 lambda_max, rep.lambda, rep.lambda_B, rep.lambda_C, rep.lambda_D,
852 rep.lambda_t, rep.lambda_c, rep.lambda_v,
853 (int)rep.ci, (int)rep.cj, (int)rep.ck, (int)rep.cblock, (int)rep.cclass, (int)rep.one_sided);
854 }
855 }
856
857 LOG_ALLOW(GLOBAL, LOG_INFO,
858 "Dual-time solver: lambda_max=%.4e [1/s] dtau_init=%.4e dtau range [%.4e, %.4e] "
859 "CFL range [%.4f, %.4f] rejection_threshold=%.3f (EMA alpha=%.2f) "
860 "growth=%.3f reduction=%.3f max_accepted=%d.\n",
861 lambda_max, cfl / lambda_max, dtau_min, dtau_max,
862 simCtx->min_pseudo_cfl, simCtx->max_pseudo_cfl,
863 simCtx->mom_dt_jameson_residual_norm_noise_allowance_factor,
864 simCtx->mom_ratio_ema_alpha,
865 simCtx->pseudo_cfl_growth_factor, simCtx->pseudo_cfl_reduction_factor,
866 max_pseudo_steps);
867
868 // --- 2. Main Pseudo-Time Iteration Loop ---
869 pseudo_iter = 0;
870 PetscBool residual_convergence_enabled =
871 (PetscBool)(simCtx->mom_resid_atol > 0.0 || simCtx->mom_resid_rtol > 0.0);
872 PetscBool converged = PETSC_FALSE;
873 PetscBool last_trial_nonfinite = PETSC_FALSE;
874 /* Rejected iterations no longer consume the accepted-iteration budget.
875 A hard cap of 3x max_pseudo_steps guards against infinite rejection loops. */
876 const PetscInt max_total_attempts = max_pseudo_steps * 3;
877 while (!converged && accepted_iter < max_pseudo_steps && pseudo_iter < max_total_attempts)
878 {
879 pseudo_iter++;
880 force_restart = PETSC_FALSE;
881
882 // Every attempt starts from the last globally accepted state.
883 for (PetscInt bi = 0; bi < block_number; bi++) {
884 ierr = VecCopy(user[bi].pUcont, user[bi].Ucont); CHKERRQ(ierr);
885 ierr = VecCopy(pRhs[bi], user[bi].Rhs); CHKERRQ(ierr);
886 const char *staggered_fields[] = {"Ucont"};
887 ierr = SynchronizePeriodicStaggeredFields(&user[bi], 1, staggered_fields); CHKERRQ(ierr);
888 ierr = ApplyBoundaryConditions(&user[bi]); CHKERRQ(ierr);
889 }
890
891 for (PetscInt bi = 0; bi < block_number; bi++) {
892
893 // === 4-Stage Jameson RK Smoothing Loop ===
894 for (istage = 0; istage < 4; istage++) {
895
896 LOG_ALLOW(GLOBAL,LOG_TRACE," Pseudo-Iter: %d | RK-Stage: %d\n", pseudo_iter, istage);
897
898 /* RK Update: U_new = U_old + (dtau * alpha) * Residual
899 * pseudo_dtau[bi] IS dtau [physical time] — no further multiplication by dt.
900 * (In the old scheme dtau = pseudo_cfl * dt; here dtau = pseudo_cfl / lambda_max.) */
901 ierr = VecWAXPY(user[bi].Ucont,
902 pseudo_dtau[bi] * alfa[istage],
903 user[bi].Rhs,
904 user[bi].pUcont); CHKERRQ(ierr);
905
906 // Sync Ghosts & Re-apply BCs for intermediate stage
907 const char *staggered_fields[] = {"Ucont"};
908 ierr = SynchronizePeriodicStaggeredFields(&user[bi], 1, staggered_fields); CHKERRQ(ierr);
909
910 ierr = ApplyBoundaryConditions(&user[bi]); CHKERRQ(ierr);
911
912 // --- Re-calculate Total Residual for next stage ---
913 ierr = ComputeTotalResidual(&user[bi]); CHKERRQ(ierr);
914
915 } // End RK Stages
916
917 // Immersed boundary interpolation (if enabled)
918
919 // === Convergence Metrics Calculation ===
920
921 // Calculate dU = U_current - U_backup
922 ierr = VecWAXPY(user[bi].dUcont, -1.0, user[bi].pUcont, user[bi].Ucont); CHKERRQ(ierr);
923
924 // Compute Infinity Norms
925 ierr = VecNorm(user[bi].dUcont, NORM_INFINITY, &delta_sol_norm_curr[bi]); CHKERRQ(ierr);
926 ierr = VecNorm(user[bi].Rhs, NORM_INFINITY, &resid_norm_curr[bi]); CHKERRQ(ierr);
927
928 // Normalize relative metrics
929 if (pseudo_iter == 1) {
930 delta_sol_norm_init[bi] = delta_sol_norm_curr[bi];
931 delta_sol_rel_curr[bi] = 1.0;
932 resid_rel_curr[bi] = 1.0;
933 // resid_norm_init[bi] set correctly before the loop — do not overwrite
934 } else {
935 if (delta_sol_norm_init[bi] > 1.0e-10)
936 delta_sol_rel_curr[bi] = delta_sol_norm_curr[bi] / delta_sol_norm_init[bi];
937 else
938 delta_sol_rel_curr[bi] = 0.0;
939
940 if(resid_norm_init[bi] > 1.0e-10)
941 resid_rel_curr[bi] = resid_norm_curr[bi] / resid_norm_init[bi];
942 else
943 resid_rel_curr[bi] = 0.0;
944 }
945
946 /* Pre-compute per-block step-to-step trial ratio for later file logging. */
947 {
948 const PetscReal resid_floor_log = 1.0e-30;
949 if (resid_norm_prev[bi] > resid_floor_log)
950 trial_ratio_log[bi] = resid_norm_curr[bi] / resid_norm_prev[bi];
951 else if (resid_norm_curr[bi] <= resid_floor_log)
952 trial_ratio_log[bi] = 0.0;
953 else
954 trial_ratio_log[bi] = PETSC_MAX_REAL;
955 }
956 } // End loop over blocks
957
958 // --- Update Global Convergence Criteria ---
959 global_norm_delta = -1.0e20;
960 global_rel_delta = -1.0e20;
961 global_norm_resid = -1.0e20;
962 global_rel_resid = -1.0e20;
963
964 for (PetscInt bi = 0; bi < block_number; bi++) {
965 global_norm_delta = PetscMax(delta_sol_norm_curr[bi], global_norm_delta);
966 global_rel_delta = PetscMax(delta_sol_rel_curr[bi], global_rel_delta);
967 global_norm_resid = PetscMax(resid_norm_curr[bi], global_norm_resid);
968 global_rel_resid = PetscMax(resid_rel_curr[bi], global_rel_resid);
969 }
970 ierr = PetscTime(&te); CHKERRQ(ierr);
971 cput = te - ts;
972 LOG_ALLOW(GLOBAL, LOG_INFO, " Pseudo-Iter(k) %d: |dUk|=%e, |dUk|/|dU0| = %e, |Rk|/|R0| = %e, CPU=%.2fs\n",
973 pseudo_iter, global_norm_delta, global_rel_delta, global_rel_resid, cput);
974
975 // === Adaptive Pseudo-CFL Trial Acceptance and Rollback ===
976 const PetscReal resid_floor = 1.0e-30;
977 PetscReal global_trial_ratio = 0.0;
978 PetscBool global_nonfinite = PETSC_FALSE;
979 for (PetscInt bi = 0; bi < block_number; bi++) {
980 PetscReal ratio;
981 if (resid_norm_prev[bi] > resid_floor) {
982 ratio = resid_norm_curr[bi] / resid_norm_prev[bi];
983 } else if (resid_norm_curr[bi] <= resid_floor) {
984 ratio = 0.0;
985 } else {
986 ratio = PETSC_MAX_REAL;
987 }
988 global_trial_ratio = PetscMax(global_trial_ratio, ratio);
989 global_nonfinite = (PetscBool)(global_nonfinite ||
990 PetscIsInfOrNanReal(delta_sol_norm_curr[bi]) ||
991 PetscIsInfOrNanReal(resid_norm_curr[bi]) ||
992 PetscIsInfOrNanReal(ratio));
993 }
994
995 PetscReal reduced_trial_ratio;
996 PetscMPIInt local_nonfinite = global_nonfinite ? 1 : 0;
997 PetscMPIInt reduced_nonfinite = 0;
998 ierr = MPI_Allreduce(&global_trial_ratio, &reduced_trial_ratio, 1, MPIU_REAL, MPI_MAX, PETSC_COMM_WORLD); CHKERRQ(ierr);
999 ierr = MPI_Allreduce(&local_nonfinite, &reduced_nonfinite, 1, MPI_INT, MPI_LOR, PETSC_COMM_WORLD); CHKERRQ(ierr);
1000 global_trial_ratio = reduced_trial_ratio;
1001 global_nonfinite = reduced_nonfinite ? PETSC_TRUE : PETSC_FALSE;
1002 last_trial_nonfinite = global_nonfinite;
1003
1004 /* EMA-smooth the trial ratio to tolerate occasional non-monotonic residual bumps.
1005 Non-finite trials bypass smoothing and always trigger rollback. */
1006 if (!global_nonfinite) {
1007 smoothed_trial_ratio = simCtx->mom_ratio_ema_alpha * global_trial_ratio
1008 + (1.0 - simCtx->mom_ratio_ema_alpha) * smoothed_trial_ratio;
1009 }
1010 LOG_ALLOW(GLOBAL, LOG_DEBUG,
1011 " [k=%d] raw_ratio=%.4e smoothed_ratio=%.4e (threshold=%.3f) | "
1012 "|R_prev|=%.6e | |R_curr|=%.6e | CFL=%.6f\n",
1013 pseudo_iter, global_trial_ratio, smoothed_trial_ratio,
1014 simCtx->mom_dt_jameson_residual_norm_noise_allowance_factor,
1015 resid_norm_prev[0], resid_norm_curr[0], pseudo_dtau[0]);
1016
1017 if (simCtx->no_pseudo_cfl_backtrack) {
1018 /* Diagnostic mode: commit every finite trial; only non-finite triggers rollback. */
1019 force_restart = global_nonfinite;
1020 } else {
1021 force_restart = (PetscBool)(global_nonfinite ||
1022 smoothed_trial_ratio > simCtx->mom_dt_jameson_residual_norm_noise_allowance_factor);
1023 }
1024
1025 if (force_restart) {
1026 /* old_dtau: dtau used for this (rejected) trial [physical time].
1027 * next_dtau: reduced dtau for the retry, clamped to dtau_min.
1028 * Both are physical-time values; the effective CFL = dtau * lambda_max. */
1029 PetscReal old_dtau = pseudo_dtau[0];
1030 PetscReal next_dtau = PetscMax(dtau_min, old_dtau * simCtx->pseudo_cfl_reduction_factor);
1031 rejected_iter++;
1032 recovery_streak = 0;
1033 for (PetscInt bi = 0; bi < block_number; bi++) {
1034 ierr = VecCopy(user[bi].pUcont, user[bi].Ucont); CHKERRQ(ierr);
1035 ierr = VecCopy(pRhs[bi], user[bi].Rhs); CHKERRQ(ierr);
1036 const char *staggered_fields[] = {"Ucont"};
1037 ierr = SynchronizePeriodicStaggeredFields(&user[bi], 1, staggered_fields); CHKERRQ(ierr);
1038 ierr = ApplyBoundaryConditions(&user[bi]); CHKERRQ(ierr);
1039 pseudo_dtau[bi] = next_dtau;
1040 }
1041 LOG_ALLOW(GLOBAL, LOG_WARNING,
1042 " Trial %d REJECTED (raw_ratio=%.4e, smoothed=%.4e, nonfinite=%d); "
1043 "dtau %.4e -> %.4e (cfl_eff %.4f -> %.4f)%s\n",
1044 pseudo_iter, global_trial_ratio, smoothed_trial_ratio, (int)global_nonfinite,
1045 old_dtau, next_dtau, old_dtau * lambda_max, next_dtau * lambda_max,
1046 (old_dtau == next_dtau) ? " [AT FLOOR — no dtau reduction]" : "");
1047 /* Log rejected trial to file before rolling back. */
1048 if (!rank) {
1049 for (PetscInt bi = 0; bi < block_number; bi++) {
1050 FILE *f;
1051 char filen[PETSC_MAX_PATH_LEN + 128];
1052 ierr = PetscSNPrintf(filen, sizeof(filen),
1053 "%s/Momentum_Solver_Convergence_History_Block_%1d.log",
1054 simCtx->log_dir, bi); CHKERRQ(ierr);
1055 if (simCtx->step == simCtx->StartStep + 1 && pseudo_iter == 1 && !simCtx->continueMode)
1056 f = fopen(filen, "w");
1057 else
1058 f = fopen(filen, "a");
1059 if (simCtx->continueMode && simCtx->step == simCtx->StartStep + 1 && pseudo_iter == 1)
1060 PetscFPrintf(PETSC_COMM_WORLD, f, "# Continuation from step %" PetscInt_FMT "\n", simCtx->StartStep);
1061 /* Log format: dtau [physical time] + cfl_eff [dimensionless Courant number = dtau*lambda_max] */
1062 PetscFPrintf(PETSC_COMM_WORLD, f,
1063 "Step: %d | PseudoIter(k): %d | dtau: %.6e | cfl_eff: %.4f | |dUk|: %le | "
1064 "|dUk|/|dU0|: %le | |Rk|: %le | |Rk|/|R0|: %le | "
1065 "trial_ratio: %le | smoothed_ratio: %le | status: rejected | "
1066 "dtau_after: %.6e | cfl_eff_after: %.4f\n",
1067 (int)ti, (int)pseudo_iter, old_dtau, old_dtau * lambda_max,
1068 delta_sol_norm_curr[bi], delta_sol_rel_curr[bi],
1069 resid_norm_curr[bi], resid_rel_curr[bi],
1070 trial_ratio_log[bi], smoothed_trial_ratio,
1071 next_dtau, next_dtau * lambda_max);
1072 fclose(f);
1073 }
1074 }
1075 if (old_dtau <= dtau_min) {
1076 if (global_nonfinite) {
1077 SETERRQ(PETSC_COMM_WORLD, PETSC_ERR_CONV_FAILED,
1078 "Momentum solver produced a non-finite trial at minimum pseudo-CFL.");
1079 }
1080 /* Finite rejection at the dtau floor: reducing further is impossible.
1081 Break instead of retrying bit-identically until max_pseudo_steps. */
1082 LOG_ALLOW(GLOBAL, LOG_WARNING,
1083 " Trial %d REJECTED (ratio=%.4e) at minimum dtau (%.4e, cfl_eff=%.4f) "
1084 "with no further reduction possible; breaking retry loop.\n",
1085 pseudo_iter, global_trial_ratio, old_dtau, old_dtau * lambda_max);
1086 break;
1087 }
1088 continue;
1089 }
1090
1091 accepted_iter++;
1092 last_trial_nonfinite = PETSC_FALSE;
1093 last_accepted_resid = resid_norm_curr[0]; /* track for final summary */
1094 for (PetscInt bi = 0; bi < block_number; bi++) {
1095 ierr = VecCopy(user[bi].Ucont, user[bi].pUcont); CHKERRQ(ierr);
1096 ierr = VecCopy(user[bi].Rhs, pRhs[bi]); CHKERRQ(ierr);
1097 resid_norm_prev[bi] = resid_norm_curr[bi];
1098 delta_sol_norm_prev[bi] = delta_sol_norm_curr[bi];
1099 }
1100
1101 /* old_dtau: dtau used for this (accepted) trial.
1102 * next_dtau: dtau for the next trial, grown/reduced based on convergence rate.
1103 * Bounds (dtau_min, dtau_max) are derived from simCtx->min/max_pseudo_cfl / lambda_max. */
1104 PetscReal old_dtau = pseudo_dtau[0]; /* save before any update */
1105 PetscReal next_dtau = old_dtau;
1106 if (global_trial_ratio < 0.90) {
1107 /* Residual decreasing fast: grow dtau to accelerate convergence. */
1108 next_dtau *= simCtx->pseudo_cfl_growth_factor;
1109 recovery_streak = 0;
1110 } else if (global_trial_ratio <= 1.0) {
1111 /* Residual barely decreasing: grow only after 3 clean consecutive trials. */
1112 recovery_streak++;
1113 if (recovery_streak >= 3) {
1114 next_dtau *= simCtx->pseudo_cfl_growth_factor;
1115 recovery_streak = 0;
1116 }
1117 } else {
1118 /* Accepted but residual slightly grew (within EMA noise allowance): reduce dtau. */
1119 next_dtau *= simCtx->pseudo_cfl_reduction_factor;
1120 recovery_streak = 0;
1121 }
1122 next_dtau = PetscMin(next_dtau, dtau_max);
1123 next_dtau = PetscMax(next_dtau, dtau_min);
1124 for (PetscInt bi = 0; bi < block_number; bi++) pseudo_dtau[bi] = next_dtau;
1125
1126 LOG_ALLOW(GLOBAL, LOG_DEBUG,
1127 " Trial %d ACCEPTED (raw_ratio=%.4e, smoothed=%.4e); |dU|=%.6e | "
1128 "dtau %.4e -> %.4e (cfl_eff %.4f -> %.4f)\n",
1129 pseudo_iter, global_trial_ratio, smoothed_trial_ratio, global_norm_delta,
1130 old_dtau, next_dtau, old_dtau * lambda_max, next_dtau * lambda_max);
1131
1132 /* --- Post-decision file logging (one row per accepted trial) --- */
1133 if (!rank) {
1134 for (PetscInt bi = 0; bi < block_number; bi++) {
1135 FILE *f;
1136 char filen[PETSC_MAX_PATH_LEN + 128];
1137 ierr = PetscSNPrintf(filen, sizeof(filen),
1138 "%s/Momentum_Solver_Convergence_History_Block_%1d.log",
1139 simCtx->log_dir, bi); CHKERRQ(ierr);
1140 if (simCtx->step == simCtx->StartStep + 1 && pseudo_iter == 1 && !simCtx->continueMode)
1141 f = fopen(filen, "w");
1142 else
1143 f = fopen(filen, "a");
1144 if (simCtx->continueMode && simCtx->step == simCtx->StartStep + 1 && pseudo_iter == 1)
1145 PetscFPrintf(PETSC_COMM_WORLD, f, "# Continuation from step %" PetscInt_FMT "\n", simCtx->StartStep);
1146 /* Log format: dtau [physical time] + cfl_eff [dimensionless Courant number = dtau*lambda_max] */
1147 PetscFPrintf(PETSC_COMM_WORLD, f,
1148 "Step: %d | PseudoIter(k): %d | dtau: %.6e | cfl_eff: %.4f | |dUk|: %le | "
1149 "|dUk|/|dU0|: %le | |Rk|: %le | |Rk|/|R0|: %le | "
1150 "trial_ratio: %le | smoothed_ratio: %le | status: accepted | "
1151 "dtau_after: %.6e | cfl_eff_after: %.4f\n",
1152 (int)ti, (int)pseudo_iter, old_dtau, old_dtau * lambda_max,
1153 delta_sol_norm_curr[bi], delta_sol_rel_curr[bi],
1154 resid_norm_curr[bi], resid_rel_curr[bi],
1155 trial_ratio_log[bi], smoothed_trial_ratio,
1156 next_dtau, next_dtau * lambda_max);
1157 fclose(f);
1158 }
1159 }
1160
1161 PetscBool update_pass;
1162 PetscBool residual_pass = PETSC_TRUE;
1163 if (residual_convergence_enabled) {
1164 update_pass = (PetscBool)(global_norm_delta <= tol_abs_delta && global_rel_delta <= tol_rtol_delta);
1165 residual_pass = (PetscBool)(
1166 (simCtx->mom_resid_atol > 0.0 && global_norm_resid <= simCtx->mom_resid_atol) ||
1167 (simCtx->mom_resid_rtol > 0.0 && global_rel_resid <= simCtx->mom_resid_rtol));
1168 } else {
1169 update_pass = (PetscBool)(global_norm_delta <= tol_abs_delta && global_rel_delta <= tol_rtol_delta);
1170 }
1171 converged = (PetscBool)(update_pass && residual_pass);
1172
1173 if (block_number > 1) {
1174 // ierr = Block_Interface_U(user); CHKERRQ(ierr);
1175 }
1176 } // End while loop
1177
1178 if (last_trial_nonfinite) {
1179 SETERRQ(PETSC_COMM_WORLD, PETSC_ERR_CONV_FAILED,
1180 "Momentum solver exhausted its attempt limit while recovering from a non-finite trial.");
1181 }
1182
1183 /* Convert the final dtau back to a CFL number for the warm-start of the next physical timestep.
1184 * simCtx->pseudo_cfl stores a dimensionless Courant number (dtau * lambda_max); the next
1185 * timestep's lambda_max may differ as the flow evolves, so storing the CFL number (not dtau)
1186 * ensures the warm-start adapts correctly to the new spectral radius. */
1187 PetscReal next_dtau_start = pseudo_dtau[0];
1188 PetscReal next_cfl_warmstart = next_dtau_start * lambda_max; /* CFL = dtau * lambda_max */
1189
1190 if (!rank) {
1191 LOG_ALLOW(GLOBAL, LOG_INFO,
1192 " Step %d finished: dtau=%.4e cfl_eff=%.4f lambda_max=%.4e [1/s]. "
1193 "Next step warm-starts at cfl=%.4f.\n",
1194 simCtx->step, next_dtau_start, next_cfl_warmstart, lambda_max, next_cfl_warmstart);
1195 }
1196
1197 /* Store the CFL number (not dtau) so it remains meaningful across physical timesteps. */
1198 simCtx->pseudo_cfl = next_cfl_warmstart;
1199 simCtx->mom_last_lambda_max = lambda_max;
1200
1201 /* Expose the last accepted finite state. On converged or floor-break exits pUcont==Ucont
1202 already (VecCopy would be an identity), but ghosts may have gone stale; always re-sync
1203 and re-apply BCs. Only restore the solution vector itself when nothing was ever accepted. */
1204 if (accepted_iter == 0) {
1205 for (PetscInt bi = 0; bi < block_number; bi++) {
1206 ierr = VecCopy(user[bi].pUcont, user[bi].Ucont); CHKERRQ(ierr);
1207 ierr = VecCopy(pRhs[bi], user[bi].Rhs); CHKERRQ(ierr);
1208 }
1209 }
1210 for (PetscInt bi = 0; bi < block_number; bi++) {
1211 const char *staggered_fields[] = {"Ucont"};
1212 ierr = SynchronizePeriodicStaggeredFields(&user[bi], 1, staggered_fields); CHKERRQ(ierr);
1213 ierr = ApplyBoundaryConditions(&user[bi]); CHKERRQ(ierr);
1214 }
1215
1216 simCtx->mom_last_converged = converged;
1217 if (!converged) {
1218 PetscPrintf(PETSC_COMM_WORLD,
1219 "[WARNING] Momentum solver step %d: reached %d total attempts (%d accepted, %d rejected) "
1220 "without convergence; continuing from last accepted finite state.\n",
1221 (int)ti, pseudo_iter, accepted_iter, rejected_iter);
1222 }
1223 if (accepted_iter == 0) {
1224 PetscPrintf(PETSC_COMM_WORLD,
1225 "[WARNING] Momentum solver step %d: no pseudo-time trials were accepted; "
1226 "retaining physical-step entry state.\n", (int)ti);
1227 }
1228 LOG_ALLOW(GLOBAL, LOG_INFO,
1229 "Momentum solver finished: %d attempts (%d accepted, %d rejected) of %d max accepted / %d hard cap, "
1230 "converged=%s, last accepted |R|=%.6e, last accepted |dU|=%.6e, "
1231 "next_dtau=%.4e (cfl=%.4f).\n",
1232 pseudo_iter, accepted_iter, rejected_iter, max_pseudo_steps, max_total_attempts,
1233 converged ? "yes" : "no",
1234 (accepted_iter > 0) ? last_accepted_resid : resid_norm_init[0],
