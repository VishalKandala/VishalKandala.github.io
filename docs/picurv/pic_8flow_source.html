121def _extract_key_path(message: str) -> str:
122 """Best-effort key-path extraction from free-form validation messages."""
123 dotted = re.search(r"\b([A-Za-z_][A-Za-z0-9_]*(?:\.[A-Za-z0-9_\[\]-]+)+)\b", message)
124 if dotted:
125 return dotted.group(1)
126
127 bracketed = re.search(r"\b([A-Za-z_][A-Za-z0-9_]*\[[^\]]+\](?:\[[^\]]+\])*)\b", message)
128 if bracketed:
129 return bracketed.group(1)

131 quoted = re.findall(r"'([A-Za-z0-9_.\[\]-]+)'", message)
132 for token in quoted:

292def validate_and_nondimensionalize_picgrid(source_grid: str, dest_grid: str, L_ref: float, expected_nblk: int = None) -> dict:
293 """!
294 @brief Validates PICGRID payload and writes a non-dimensionalized copy.
295 @details Accepts files with or without leading "PICGRID" token. Output is always
296 written in canonical PICGRID format with header and per-block dims.
297 @param[in] source_grid Input grid file path.
298 @param[in] dest_grid Output grid file path.
299 @param[in] L_ref Reference length for non-dimensionalization.
300 @param[in] expected_nblk Optional expected block count.
301 @return Summary dictionary with nblk, dims, and total_nodes.
302 @throws ValueError on malformed grid.
303 """
304 if L_ref == 0.0:
305 raise ValueError("length_ref must be non-zero when processing grid coordinates.")
306 if not os.path.isfile(source_grid):
307 raise ValueError(f"Grid file not found: {source_grid}")
308
309 with open(source_grid, "r") as fin:
310 line_iter = _iter_nonempty_noncomment_lines(fin)
311 try:
312 _, first_token = next(line_iter)
313 except StopIteration:
314 raise ValueError(f"Grid file '{source_grid}' is empty.")
315
316 if first_token == "PICGRID":
317 try:
318 _, nblk_line = next(line_iter)
319 except StopIteration:
320 raise ValueError(f"Grid file '{source_grid}' missing block count after PICGRID header.")
321 else:
322 nblk_line = first_token
323
324 try:
325 nblk = int(nblk_line)
326 except ValueError:
327 raise ValueError(f"Invalid block count '{nblk_line}' in grid file '{source_grid}'.")
328 if nblk <= 0:
329 raise ValueError(f"Grid file '{source_grid}' has non-positive block count: {nblk}.")
330 if expected_nblk is not None and nblk != expected_nblk:
331 raise ValueError(
332 f"Grid file block count mismatch: case expects {expected_nblk}, grid contains {nblk}."
333 )
334
335 dims = []
336 for bi in range(nblk):
337 try:
338 lineno, dim_line = next(line_iter)
339 except StopIteration:
340 raise ValueError(f"Grid file '{source_grid}' missing dimensions for block {bi}.")
341 parts = dim_line.split()
342 if len(parts) != 3:
343 raise ValueError(
344 f"Invalid dimensions line at {source_grid}:{lineno}. Expected 3 integers, got: '{dim_line}'."
345 )
346 try:
347 im, jm, km = (int(parts[0]), int(parts[1]), int(parts[2]))
348 except ValueError:
349 raise ValueError(
350 f"Invalid dimensions line at {source_grid}:{lineno}. Non-integer values: '{dim_line}'."
351 )
352 if im <= 0 or jm <= 0 or km <= 0:
353 raise ValueError(
354 f"Invalid block dimensions at {source_grid}:{lineno}: ({im}, {jm}, {km}). Must be > 0."
355 )
356 dims.append((im, jm, km))
357
358 total_nodes_expected = sum(im * jm * km for (im, jm, km) in dims)
359 os.makedirs(os.path.dirname(dest_grid), exist_ok=True)
360 with open(dest_grid, "w") as fout:
361 fout.write("PICGRID\n")
362 fout.write(f"{nblk}\n")
363 for (im, jm, km) in dims:
364 fout.write(f"{im} {jm} {km}\n")
365
366 total_nodes_seen = 0
367 for lineno, coord_line in line_iter:
368 parts = coord_line.split()
369 if len(parts) != 3:
370 raise ValueError(
371 f"Invalid coordinate row at {source_grid}:{lineno}. Expected 3 floats, got: '{coord_line}'."
372 )
373 try:
374 x = float(parts[0]) / L_ref
375 y = float(parts[1]) / L_ref
376 z = float(parts[2]) / L_ref
377 except ValueError:
378 raise ValueError(
379 f"Invalid coordinate row at {source_grid}:{lineno}. Non-numeric values: '{coord_line}'."
380 )
381 total_nodes_seen += 1
382 if total_nodes_seen > total_nodes_expected:
383 raise ValueError(
384 f"Grid file '{source_grid}' has more coordinates ({total_nodes_seen}) than expected ({total_nodes_expected})."
385 )
386 fout.write(f"{x:.8e} {y:.8e} {z:.8e}\n")
387
388 if total_nodes_seen != total_nodes_expected:
389 raise ValueError(
390 f"Grid file '{source_grid}' has {total_nodes_seen} coordinates, expected {total_nodes_expected} from header."
391 )
392
393 return {"nblk": nblk, "dims": dims, "total_nodes": total_nodes_expected}

573 """

580 if L_ref == 0.0:
581 raise ValueError("properties.scaling.length_ref must be non-zero for non-dimensionalization.")

614 seen_faces[face] = idx
615
616 bc_type_raw = str(bc["type"]).strip()
617 bc_type = BC_TYPE_MAP.get(bc_type_raw.lower())
618 if bc_type is None:
619 raise ValueError(
620 f"Unsupported BC type '{bc_type_raw}' for face {face} in block {bi}. "
621 f"Supported: {sorted(set(BC_TYPE_MAP.values()))}."
622 )
623
624 handler = str(bc["handler"]).strip().lower()
625 handler_spec = BC_HANDLER_SPECS.get(handler)
626 if handler_spec is None:
627 raise ValueError(
628 f"Unsupported BC handler '{bc['handler']}' for face {face} in block {bi}. "
629 f"Supported now: {sorted(BC_HANDLER_SPECS.keys())}."
630 )
631 if bc_type not in handler_spec["types"]:
632 raise ValueError(
633 f"Invalid BC combination on block {bi}, face {face}: type '{bc_type}' cannot use handler '{handler}'."
634 )
635
636 params = bc.get("params", {})
637 if params is None:
638 params = {}
639 if not isinstance(params, dict):
640 raise ValueError(f"'params' for block {bi}, face {face} must be a mapping.")
641
642 # Reject legacy structured keys explicitly.
643 if "vector" in params or "velocity" in params:
644 raise ValueError(
645 f"Legacy params key ('vector'/'velocity') found on block {bi}, face {face}. "
646 "Use scalar keys 'vx', 'vy', 'vz'."
647 )
648
649 required = handler_spec["required_params"]
650 optional = handler_spec["optional_params"]
651 allowed = required | optional
652
653 missing = sorted(required - set(params.keys()))
654 if missing:
655 raise ValueError(
656 f"Missing required params for handler '{handler}' on block {bi}, face {face}: {missing}."
657 )
658 unknown = sorted(set(params.keys()) - allowed)
659 if unknown:
660 raise ValueError(
661 f"Unknown params for handler '{handler}' on block {bi}, face {face}: {unknown}. "
662 f"Allowed: {sorted(allowed)}."
663 )
664
665 converted_params = {}
666 for key, value in params.items():
667 if key in _NUMERIC_BC_PARAMS:
668 numeric = _to_float(value, f"boundary_conditions[{bi}][{idx}].params.{key}")
669 if key in {"vx", "vy", "vz", "v_max"}:
670 converted_params[key] = numeric / U_ref
671 elif key == "target_flux":
672 converted_params[key] = numeric / (U_ref * (L_ref ** 2))
673 elif key in _BOOL_BC_PARAMS:
674 converted_params[key] = _to_bool(value, f"boundary_conditions[{bi}][{idx}].params.{key}")
675 else:
676 # Defensive fallback; should not happen due unknown-key gate above.
677 converted_params[key] = value
678
679 prepared_block.append({
680 "face": face,
681 "type": bc_type,
682 "handler": handler,
683 "params": converted_params,
684 })
685
686 missing_faces = sorted(expected_faces - set(seen_faces.keys()))
687 if missing_faces:
688 raise ValueError(
689 f"boundary_conditions[{bi}] is incomplete. Missing faces: {missing_faces}. "
690 "Provide all six faces explicitly."
691 )
692
693 # Pairwise periodic consistency checks.
694 face_map = {entry["face"]: entry for entry in prepared_block}
695 for neg_face, pos_face in axis_pairs:
696 neg = face_map[neg_face]
697 pos = face_map[pos_face]
698 neg_periodic = (neg["type"] == "PERIODIC")
699 pos_periodic = (pos["type"] == "PERIODIC")
700 if neg_periodic != pos_periodic:
701 raise ValueError(
702 f"Inconsistent periodicity in block {bi}: {neg_face} and {pos_face} must both be PERIODIC or neither."
703 )
704
705 driven_handlers = {"constant_flux"}
706 if (neg["handler"] in driven_handlers) or (pos["handler"] in driven_handlers):
707 if neg["handler"] != pos["handler"]:
708 raise ValueError(
709 f"In block {bi}, driven periodic handlers on {neg_face}/{pos_face} must match exactly."
710 )
711 if not (neg_periodic and pos_periodic):
712 raise ValueError(
713 f"In block {bi}, driven periodic handler '{neg['handler']}' requires PERIODIC type on both faces."
714 )
715
716 prepared_blocks.append(prepared_block)
717
718 return prepared_blocks

1749def infer_unique_inlet_axis_from_prepared_bcs(prepared_blocks: list) -> str | None:
1750 """!
1751 @brief Infer the unique inlet axis across all blocks using C-side "primary inlet" ordering.
1752 @param[in] prepared_blocks Normalized BC blocks from `validate_and_prepare_boundary_conditions`.
1753 @return One of `"x"`, `"y"`, `"z"` if unique, `None` if no inlet exists.
1754 @throws ValueError if different blocks imply different inlet axes.
1755 """
1756 face_order = ("-Xi", "+Xi", "-Eta", "+Eta", "-Zeta", "+Zeta")
1757 face_axis = {
1758 "-Xi": "x", "+Xi": "x",
1759 "-Eta": "y", "+Eta": "y",
1760 "-Zeta": "z", "+Zeta": "z",
1761 }
1762
1763 inlet_axes = set()
1764 for block_bcs in prepared_blocks:
1765 face_map = {entry["face"]: entry for entry in block_bcs}
1766 for face in face_order:
1767 entry = face_map.get(face)
1768 if entry and entry["type"] == "INLET":
1769 inlet_axes.add(face_axis[face])
1770 break
1771
1772 if not inlet_axes:
1773 return None
1774 if len(inlet_axes) != 1:
1775 raise ValueError(
1776 "properties.initial_conditions.peak_velocity_physical requires all blocks to have a primary INLET "
1777 f"on the same axis. Found axes: {sorted(inlet_axes)}. Use u_physical/v_physical/w_physical instead."
1778 )
1779 return next(iter(inlet_axes))

1781def resolve_initial_velocity_components(initial_conditions: dict, finit_code: int, *, prepared_blocks: list | None = None) -> tuple[float, float, float]:
1782 """!
1783 @brief Resolve initial-condition velocity inputs to `(u, v, w)` physical components.
1784 @param[in] initial_conditions The `properties.initial_conditions` mapping from case.yml.
1785 @param[in] finit_code Normalized `-finit` integer code.
1786 @param[in] prepared_blocks Optional normalized BC blocks, required for scalar Poiseuille peak velocity.
1787 @return Tuple `(u, v, w)` in physical units.
1788 @throws KeyError if required explicit component keys are missing.
1789 @throws ValueError on invalid combinations or invalid scalar values.
1790 """
1791 peak_key = "peak_velocity_physical"
1792 has_peak_velocity = peak_key in initial_conditions
1793
1794 if has_peak_velocity and finit_code != 2:
1795 raise ValueError(
1796 "properties.initial_conditions.peak_velocity_physical is only valid when initial_conditions.mode is 'Poiseuille'."
1797 )
1798
1799 if finit_code == 2 and has_peak_velocity:
1800 component_keys = ("u_physical", "v_physical", "w_physical")
1801 if any(key in initial_conditions for key in component_keys):
1802 raise ValueError(
1803 "For Poiseuille mode, use either peak_velocity_physical or u_physical/v_physical/w_physical, not both."
1804 )
1805 try:
1806 peak_velocity = float(initial_conditions[peak_key])
1807 except (TypeError, ValueError) as exc:
1808 raise ValueError(
1809 f"Invalid value for properties.initial_conditions.{peak_key}: {initial_conditions[peak_key]!r}. "
1810 "Expected a numeric value."
1811 ) from exc
1812
1813 if prepared_blocks is None:
1814 raise ValueError(
1815 "properties.initial_conditions.peak_velocity_physical requires valid boundary_conditions so the inlet axis can be inferred."
1816 )
1817
1818 inlet_axis = infer_unique_inlet_axis_from_prepared_bcs(prepared_blocks)
1819 if inlet_axis is None:
1820 raise ValueError(
1821 "properties.initial_conditions.peak_velocity_physical requires at least one INLET face so the inlet axis can be inferred."
1822 )
1823 if inlet_axis == "x":
1824 return (peak_velocity, 0.0, 0.0)
1825 if inlet_axis == "y":
1826 return (0.0, peak_velocity, 0.0)
1827 return (0.0, 0.0, peak_velocity)
1828
1829 return parse_initial_velocity_components(initial_conditions, finit_code, require_explicit=(finit_code != 0))
