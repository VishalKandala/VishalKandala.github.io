1/**
2 * @file statistics_moments.h
3 * @brief Weighted centered-moment kernels for the field-statistics pipeline.
4 *
5 * These are pure numerical kernels: no configuration, no PETSc vectors, and no
6 * knowledge of windows, fields, or scheduling. They implement the accumulator
7 * contract described in @ref p58_update_sub.
8 *
9 * Centered state is used rather than raw sums because it retains everything the
10 * first and second moments need while staying numerically stable for
11 * high-mean/low-fluctuation signals. Raw sums are recoverable as
12 * `S_x = W*mu` and `Q_xx = M2 + W*mu^2`.
13 */
14

20/**
21 * @brief Weighted centered state for one scalar quantity at one point.
22 *
23 * `weight_sq` is retained because physical-time weighting produces unequal
24 * weights, so an effective sample size cannot be inferred from `count` alone.
25 */
26typedef struct {
27 PetscReal count; /**< Number of accepted samples. */
28 PetscReal weight; /**< Total weight W. */
29 PetscReal weight_sq; /**< Sum of squared weights W2. */
30 PetscReal mean; /**< Weighted mean mu. */
31 PetscReal m2; /**< Centered second-moment sum M2. */
32} PicurvMomentState;

34/**
35 * @brief Weighted centered co-moment state for one ordered pair of quantities.
36 *
37 * The two means are tracked inside the pair rather than referenced from two
38 * `PicurvMomentState` values, so a co-moment update is self-contained and cannot
39 * be corrupted by update ordering between the members.
40 */
41typedef struct {
42 PetscReal count; /**< Number of accepted samples. */
43 PetscReal weight; /**< Total weight W. */
44 PetscReal weight_sq; /**< Sum of squared weights W2. */
45 PetscReal mean_x; /**< Weighted mean of the first member. */
46 PetscReal mean_y; /**< Weighted mean of the second member. */
47 PetscReal cm; /**< Centered co-moment sum C. */
48} PicurvCoMomentState;

62/**
63 * @brief Applies one weighted sample to a scalar moment accumulator.
64 *
65 * Implements the stable weighted update
66 * `W' = W+w`, `d = x-mu`, `mu' = mu + (w/W')d`, `M2' = M2 + w*d*(x-mu')`.
67 *
68 * @param[in,out] state Accumulator to update.
69 * @param[in] value Sample value x.
70 * @param[in] weight Sample weight w; must be strictly positive.
71 * @return Zero on success, or `PETSC_ERR_ARG_OUTOFRANGE` for a non-positive weight.
72 */
73PetscErrorCode PicurvMomentStateUpdate(PicurvMomentState *state, PetscReal value, PetscReal weight);

75/**
76 * @brief Applies one weighted paired sample to a co-moment accumulator.
77 *
78 * Uses the pre-update mean of x and the post-update mean of y, which is the form
79 * that reduces exactly to the `M2` update when x and y are the same signal.
80 *
81 * @param[in,out] state Accumulator to update.
82 * @param[in] value_x Sample value of the first member.
83 * @param[in] value_y Sample value of the second member.
84 * @param[in] weight Sample weight w; must be strictly positive.
85 * @return Zero on success, or `PETSC_ERR_ARG_OUTOFRANGE` for a non-positive weight.
86 */
87PetscErrorCode PicurvCoMomentStateUpdate(PicurvCoMomentState *state,

90/**
91 * @brief Merges two independently accumulated scalar states.
92 *
93 * Implements the stable weighted parallel combination
94 * `d = mu_b - mu_a`, `W = W_a + W_b`, `mu = mu_a + (W_b/W)d`,
95 * `M2 = M2_a + M2_b + (W_a*W_b/W)d^2`.
96 * Merging an empty state is a no-op, so partitions that received no samples are
97 * safe to combine.
98 *
99 * @param[out] result Merged state; may alias @p a or @p b.
100 * @param[in] a First partition.
101 * @param[in] b Second partition.
102 * @return Zero on success, or `PETSC_ERR_ARG_NULL` when any argument is NULL.
103 */
104PetscErrorCode PicurvMomentStateMerge(PicurvMomentState *result,
105 const PicurvMomentState *a, const PicurvMomentState *b);

117/**
118 * @brief Returns the weighted variance `M2/W`, or zero when no weight accumulated.
119 *
120 * This is the population (weight-normalized) variance the postprocessing contract
121 * uses for `R_ii`. It is never negative for a state built only through
122 * `PicurvMomentStateUpdate`, but callers taking a square root must still clamp,
123 * because cancellation can drive `M2` slightly negative for degenerate inputs.
124 *
125 * @param[in] state Accumulator to read.

127 */
