92/**
93 * @brief Tests the runtime evaluator against every shared conformance case.
94 * @details `tests/fixtures/expression_conformance.txt` is also evaluated by
95 * `generators/ic.gen` in `tests/test_expression_language.py`, so the two
96 * implementations of the language cannot drift apart unnoticed.
97 */
98static PetscErrorCode TestExpressionConformance(void)
99{
100 FILE *file = fopen("tests/fixtures/expression_conformance.txt", "r");
101 char line[1024];
102 PetscInt cases = 0;
103
104 PetscFunctionBeginUser;
105 PetscCheck(file, PETSC_COMM_SELF, PETSC_ERR_FILE_OPEN, "Cannot open the expression conformance cases.");
106 while (fgets(line, sizeof(line), file)) {
107 char *fields[5];
108 PetscInt count = 0;
109 PicurvExpression *expression = NULL;
110 PetscReal values[3], result = 0.0;
111
112 if (line[0] == '#' || line[0] == '\n') continue;
113 for (char *cursor = line; count < 5; ++count) {
114 fields[count] = cursor;
115 char *separator = strchr(cursor, ';');
116 if (!separator) { ++count; break; }
117 *separator = '\0';
118 cursor = separator + 1;
119 }
120 PetscCheck(count == 5, PETSC_COMM_SELF, PETSC_ERR_FILE_UNEXPECTED, "Malformed conformance line.");
121 for (int v = 0; v < 3; ++v) values[v] = strtod(fields[v + 1], NULL);
122 PetscCall(PicurvExpressionCompile(fields[0], kConformanceNames, 3, PETSC_FALSE, &expression));
123 PetscCall(PicurvExpressionEvaluate(expression, values, NULL, &result));
124 PetscCall(PicurvAssertRealNear(strtod(fields[4], NULL), result, 1.0e-12, fields[0]));
125 PetscCall(PicurvExpressionDestroy(&expression));
126 ++cases;
127 }
128 fclose(file);
129 PetscCall(PicurvAssertBool((PetscBool)(cases >= 20), "the conformance file should hold its cases"));
130 PetscFunctionReturn(0);
131}

136static PetscErrorCode TestExpressionRejectsOutsideLanguage(void)
137{
138 const char *bad[] = {"0x1F", "1_000", "x // 2", "abs", "abs(1, 2)", "where(x, 1)", "foo",
139 "x +", "(x", "x y", "uniform(x)", "uniform(1.5)", "1.e", "not"};
140
141 PetscFunctionBeginUser;
142 for (size_t n = 0; n < sizeof(bad) / sizeof(bad[0]); ++n) {
143 PicurvExpression *expression = NULL;
144 PetscErrorCode refused;
145
146 PetscCall(PetscPushErrorHandler(PetscIgnoreErrorHandler, NULL));
147 refused = PicurvExpressionCompile(bad[n], kConformanceNames, 3, PETSC_TRUE, &expression);
148 PetscCall(PetscPopErrorHandler());
149 PetscCall(PicurvAssertIntEqual(PETSC_ERR_ARG_WRONG, refused, bad[n]));
150 }
151 {
152 PicurvExpression *expression = NULL;
153 PetscErrorCode refused;
154
155 PetscCall(PetscPushErrorHandler(PetscIgnoreErrorHandler, NULL));
156 refused = PicurvExpressionCompile("uniform()", kConformanceNames, 3, PETSC_FALSE, &expression);
157 PetscCall(PetscPopErrorHandler());
158 PetscCall(PicurvAssertIntEqual(PETSC_ERR_ARG_WRONG, refused, "draws need allow_random"));
159 }
160 PetscFunctionReturn(0);
161}

163/**
164 * @brief Tests that random draws are pure functions of their key.
165 * @details The same particle always draws the same number; a different particle, event,
166 * or field default stream draws independently; an explicit stream is shared
167 * across fields; and the normal draw has zero mean and unit variance.
168 */
169static PetscErrorCode TestExpressionDrawsArePureAndKeyed(void)
170{
171 PicurvExpression *uniform = NULL, *shared = NULL, *normal = NULL;
172 PicurvExpressionDrawKey key = {12345, 7, 0, 6};
173 PetscReal a, b, sum = 0.0, square = 0.0;
174 const PetscInt samples = 20000;
175
176 PetscFunctionBeginUser;
177 PetscCall(PicurvExpressionCompile("uniform()", kConformanceNames, 3, PETSC_TRUE, &uniform));
178 PetscCall(PicurvExpressionCompile("uniform(3)", kConformanceNames, 3, PETSC_TRUE, &shared));
179 PetscCall(PicurvExpressionCompile("normal()", kConformanceNames, 3, PETSC_TRUE, &normal));
180
181 PetscCall(PicurvExpressionEvaluate(uniform, NULL, &key, &a));
182 PetscCall(PicurvExpressionEvaluate(uniform, NULL, &key, &b));
183 PetscCall(PicurvAssertBool((PetscBool)(a == b && a >= 0.0 && a < 1.0), "a draw is pure and in [0, 1)"));
184 key.pid = 8;
185 PetscCall(PicurvExpressionEvaluate(uniform, NULL, &key, &b));
186 PetscCall(PicurvAssertBool((PetscBool)(a != b), "another particle draws independently"));
187 key.pid = 7;
188 key.salt = 1;
189 PetscCall(PicurvExpressionEvaluate(uniform, NULL, &key, &b));
190 PetscCall(PicurvAssertBool((PetscBool)(a != b), "another event draws independently"));
191 key.salt = 0;
192 key.field = 5;
193 PetscCall(PicurvExpressionEvaluate(uniform, NULL, &key, &b));
194 PetscCall(PicurvAssertBool((PetscBool)(a != b), "another field's default stream is independent"));
195 PetscCall(PicurvExpressionEvaluate(shared, NULL, &key, &a));
196 key.field = 6;
197 PetscCall(PicurvExpressionEvaluate(shared, NULL, &key, &b));
198 PetscCall(PicurvAssertBool((PetscBool)(a == b), "an explicit stream is shared across fields"));
199
200 for (PetscInt pid = 0; pid < samples; ++pid) {
201 key.pid = pid;
202 PetscCall(PicurvExpressionEvaluate(normal, NULL, &key, &a));
203 sum += a;
204 square += a * a;
205 }
206 PetscCall(PicurvAssertRealNear(0.0, sum / samples, 0.03, "normal draws have zero mean"));
207 PetscCall(PicurvAssertRealNear(1.0, square / samples, 0.04, "normal draws have unit variance"));
208
209 PetscCall(PicurvExpressionDestroy(&uniform));
210 PetscCall(PicurvExpressionDestroy(&shared));
211 PetscCall(PicurvExpressionDestroy(&normal));
212 PetscFunctionReturn(0);
213}
