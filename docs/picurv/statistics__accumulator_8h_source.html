1/**
2 * @file statistics_accumulator.h
3 * @brief Per-window PETSc accumulator storage and pointwise application.
4 *
5 * Holds the independent state each named window owns, and applies one accepted
6 * completed state to it, per @ref p58_state_sec.
7 *
8 * Storage is allocated once by the vector factory and released once at teardown.
9 * Application is strictly pointwise: it reads a source field value at an owned
10 * point and updates that point's accumulator, so it performs no halo exchange and
11 * allocates nothing.
12 */

20/**
21 * @brief Independent accumulator state for one window on one block.
22 *
23 * Per-point occupancy is tracked separately from the field moments because the
24 * fluid mask can move: a point contributes only to the states in which it was
25 * valid, so its own count and weight are what normalize its moments.
26 *
27 * Every product is one vector carrying all of its components, not one vector per
28 * component. A symmetric second-order tensor is a single object: splitting it would
29 * cost six memory streams in the per-step accumulation loop instead of one cache
30 * line, and six collective gathers per checkpoint instead of one. Component counts
31 * that neither `da` nor `fda` provides are carried by a DM mirroring the block
32 * decomposition at that degree of freedom.
33 */
34typedef struct PicurvWindowStorage {
35 PetscInt field_count; /**< Fields accumulated. */
36 PetscInt covariance_count; /**< Covariance pairs accumulated. */
37 Vec count; /**< Per-point accepted sample count. */
38 Vec weight; /**< Per-point valid weight. */
39 Vec weight_sq; /**< Per-point squared-weight sum. */
40 Vec *mean; /**< One per field, matching that field's layout. */
41 Vec *m2; /**< One per field; NULL when no second moment was requested. */
42 Vec *cm; /**< One per covariance pair. */
43} PicurvWindowStorage;

162/**
163 * @brief Derives one output field from centered accumulator state.
164 *
165 * Normalizes in exactly one place: `R_ij = C_ij / W`, `RMS_i = sqrt(R_ii)`,
166 * `k = (R_xx + R_yy + R_zz) / 2`, and a flux is the co-moment over the same weight.
167 * Each uses the moment kernels rather than repeating the division, so the online and
168 * offline halves of the pipeline cannot disagree about what centered state means.
169 *
170 * A variance that comes out slightly negative through floating-point cancellation is
171 * clamped only where a square root would otherwise fail, and only within
172 * `PICURV_STATISTICS_VARIANCE_FLOOR`. Stored state is never modified.
173 *
174 * Points the window never sampled are left at zero rather than divided by a zero
175 * weight; the valid-fraction range reports how much of the domain that covers.
176 *
177 * @param[in] user Block context supplying the target domain.
178 * @param[in] definition Window definition.
179 * @param[in] storage Accumulator state to read.
180 * @param[in] outputs Comma-separated output kinds.
181 * @param[in] index Derived field index in `[0, count)`.
182 * @param[out] scalar_target Scalar destination, used when the field has one component.
183 * @param[out] vector_target Vector destination, used when the field has three.
184 * @param[out] field Resolved name and component count of the derived field.
185 * @return Zero on success, or a PETSc error.
186 */
187PetscErrorCode PicurvWindowDerive(UserCtx *user, const PicurvWindowDefinition *definition,
188 const PicurvWindowStorage *storage, const char *outputs,
189 PetscInt index, Vec scalar_target, Vec vector_target,
190 PicurvDerivedField *field);

192/**
193 * @brief Reports the spatial mean of a derived field over the points a window sampled.
194 *
195 * The average is taken over targeted points that actually accumulated weight, not
196 * over the whole vector. A derived field is zero everywhere outside the target
197 * domain and at any point the mask never admitted, and those zeros are absences
198 * rather than measurements: including them would scale the answer down by the
199 * fraction of the vector the window never covered.
200 *
201 * Performs a collective reduction, so callers use it for reporting rather than per
202 * point.
203 *
204 * @param[in] user Block context supplying the target domain.
205 * @param[in] definition Window definition naming the accumulated fields.
206 * @param[in] storage Accumulator state supplying per-point occupancy.
207 * @param[in] field Derived field to average, on the cell-centred scalar DM.
208 * @param[out] mean Spatial mean; zero when the window sampled no point.
209 * @return Zero on success, or a PETSc error.
210 */
211PetscErrorCode PicurvWindowSpatialMean(UserCtx *user, const PicurvWindowDefinition *definition,
212 const PicurvWindowStorage *storage, Vec field,
213 PetscReal *mean);

215/**
216 * @brief Reports the range of per-point valid fraction across a window's domain.
217 *
218 * A point contributes only to the states in which the mask accepted it, so with a
219 * moving immersed body different points carry different sample counts. The ratio of
220 * a point's own count to the window's accepted-sample count is its valid fraction,
221 * and the range of that ratio is the window's mask-health indicator: a minimum of
222 * one means every point saw every state, and a minimum of zero means some point
223 * contributed nothing at all.
224 *
225 * Performs a collective reduction, so callers gate it on an already-active reporting
226 * path rather than calling it every step.
227 *
228 * @param[in] user Block context supplying the target domain and mask.
229 * @param[in] definition Window definition naming the accumulated fields.
230 * @param[in] storage Accumulator state to inspect.
231 * @param[in] sample_count Accepted states the window has recorded.
232 * @param[out] minimum Smallest valid fraction; one when no point is targeted.
233 * @param[out] maximum Largest valid fraction; zero when no point is targeted.
234 * @return Zero on success, or a PETSc error.
235 */
236PetscErrorCode PicurvWindowValidFractionRange(UserCtx *user, const PicurvWindowDefinition *definition,
237 const PicurvWindowStorage *storage, PetscInt sample_count,
238 PetscReal *minimum, PetscReal *maximum);
