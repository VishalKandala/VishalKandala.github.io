IBMNodes *  ibm,

Solves the momentum equations using dual-time Picard iteration with Jameson RK smoothing.

GLOSSARY & THEORETICAL BASIS

  1. METHODOLOGY: Dual-Time Stepping (Pseudo-Time Integration) We aim to solve the implicit BDF equation: R_spatial(U) + dU/dt_physical = 0. We do this by introducing a fictitious "Pseudo-Time" (tau) and iterating to steady state: dU/d(tau) = - [ R_spatial(U) + BDF_Terms(U) ] When dU/d(tau) -> 0, the physical time step is satisfied.
  2. ALGORITHM: Fixed-Point Iteration with Explicit Runge-Kutta This is technically a Fixed-Point iteration on the operator: U_new = U_old + pseudo_dtau * alfa_stage * Total_Residual(U_old) where pseudo_dtau = pseudo_cfl / lambda_max is the spectral-radius-based pseudo-time step (lambda_max = global max convective spectral radius). This makes pseudo_cfl a true dimensionless Courant number, independent of the physical time step dt. We use a 4-Stage Explicit RK scheme (Jameson-Schmidt-Turkel coeffs) to smooth errors.
  3. STABILITY: Adaptive Pseudo-CFL Trial Acceptance and Rollback If a pseudo-time trial causes excessive residual growth, the solver restores the

previous accepted state, reduces the global pseudo-CFL, and retries.

VARIABLE MAPPING

– Physics Variables (Legacy Names Kept) – ti : Physical Time Step Index. dt : Physical Time Step size (Delta t). st : Pseudo-Time Step size (Delta tau). alfa : Runge-Kutta stage coefficients {1/4, 1/3, 1/2, 1}. – Convergence & Solver Control (Renamed) – pseudo_iter : Counter for the inner dual-time loop. pseudo_dtau : Adaptive pseudo-time step [physical time], = pseudo_cfl / lambda_max. lambda_max : Global max convective spectral radius [1/s] from the current field. delta_sol_norm : The L_inf norm of the change in solution (dU).

resid_norm : The L_inf norm of the Total Residual (RHS).

Parameters
userPrimary UserCtx input for the operation.
ibmParameter ibm passed to MomentumSolver_DualTime_Picard_JamesonRK().
fsiParameter fsi passed to MomentumSolver_DualTime_Picard_JamesonRK().
Returns
PetscErrorCode 0 on success.
Note
Testing status: This solver is covered primarily through runtime smoke and orchestration tests. A smaller direct invariant-style positive-path harness remains part of the next-gap backlog.

528 ierr = MPI_Allreduce(&local_nonfinite, &reduced_nonfinite, 1, MPI_INT, MPI_LOR, PETSC_COMM_WORLD); CHKERRQ(ierr);

534 Non-finite trials bypass smoothing and always trigger rollback. */

537 + (1.0 - simCtx->mom_ratio_ema_alpha) * smoothed_trial_ratio;
538 }
539 LOG_ALLOW(GLOBAL, LOG_DEBUG,
540 " [k=%d] raw_ratio=%.4e smoothed_ratio=%.4e (threshold=%.3f) | "
541 "|R_prev|=%.6e | |R_curr|=%.6e | CFL=%.6f\n",
542 pseudo_iter, global_trial_ratio, smoothed_trial_ratio,
543 simCtx->mom_dt_jameson_residual_norm_noise_allowance_factor,
544 resid_norm_prev[0], resid_norm_curr[0], pseudo_dtau[0]);

546 if (simCtx->no_pseudo_cfl_backtrack) {
547 /* Diagnostic mode: commit every finite trial; only non-finite triggers rollback. */
548 force_restart = global_nonfinite;
549 } else {
550 force_restart = (PetscBool)(global_nonfinite ||
551 smoothed_trial_ratio > simCtx->mom_dt_jameson_residual_norm_noise_allowance_factor);
552 }

557 * Both are physical-time values; the effective CFL = dtau * lambda_max. */

560 rejected_iter++;

564 ierr = VecCopy(pRhs[bi], user[bi].Rhs); CHKERRQ(ierr);

567 ierr = ApplyBoundaryConditions(&user[bi]); CHKERRQ(ierr);

573 pseudo_iter, global_trial_ratio, smoothed_trial_ratio, (int)global_nonfinite,

599 trial_ratio_log[bi], smoothed_trial_ratio,
600 next_dtau, next_dtau * lambda_max);
601 fclose(f);
602 }
603 }
604 if (old_dtau <= dtau_min) {
605 if (global_nonfinite) {
606 SETERRQ(PETSC_COMM_WORLD, PETSC_ERR_CONV_FAILED,
607 "Momentum solver produced a non-finite trial at minimum pseudo-CFL.");
608 }
609 /* Finite rejection at the dtau floor: reducing further is impossible.
610 Break instead of retrying bit-identically until max_pseudo_steps. */
611 LOG_ALLOW(GLOBAL, LOG_WARNING,
612 " Trial %d REJECTED (ratio=%.4e) at minimum dtau (%.4e, cfl_eff=%.4f) "
613 "with no further reduction possible; breaking retry loop.\n",
614 pseudo_iter, global_trial_ratio, old_dtau, old_dtau * lambda_max);
615 break;
616 }
617 continue;
618 }

620 accepted_iter++;
621 last_trial_nonfinite = PETSC_FALSE;
622 last_accepted_resid = resid_norm_curr[0]; /* track for final summary */
623 for (PetscInt bi = 0; bi < block_number; bi++) {
624 ierr = VecCopy(user[bi].Ucont, user[bi].pUcont); CHKERRQ(ierr);
625 ierr = VecCopy(user[bi].Rhs, pRhs[bi]); CHKERRQ(ierr);
626 resid_norm_prev[bi] = resid_norm_curr[bi];
627 delta_sol_norm_prev[bi] = delta_sol_norm_curr[bi];
628 }

631 * next_dtau: dtau for the next trial, grown/reduced based on convergence rate.
632 * Bounds (dtau_min, dtau_max) are derived from simCtx->min/max_pseudo_cfl / lambda_max. */
633 PetscReal old_dtau = pseudo_dtau[0]; /* save before any update */
634 PetscReal next_dtau = old_dtau;
635 if (global_trial_ratio < 0.90) {
636 /* Residual decreasing fast: grow dtau to accelerate convergence. */
637 next_dtau *= simCtx->pseudo_cfl_growth_factor;
638 recovery_streak = 0;
639 } else if (global_trial_ratio <= 1.0) {
640 /* Residual barely decreasing: grow only after 3 clean consecutive trials. */
641 recovery_streak++;
642 if (recovery_streak >= 3) {
643 next_dtau *= simCtx->pseudo_cfl_growth_factor;
644 recovery_streak = 0;
645 }
646 } else {
647 /* Accepted but residual slightly grew (within EMA noise allowance): reduce dtau. */
648 next_dtau *= simCtx->pseudo_cfl_reduction_factor;
649 recovery_streak = 0;
650 }
651 next_dtau = PetscMin(next_dtau, dtau_max);
652 next_dtau = PetscMax(next_dtau, dtau_min);
653 for (PetscInt bi = 0; bi < block_number; bi++) pseudo_dtau[bi] = next_dtau;
654
655 LOG_ALLOW(GLOBAL, LOG_DEBUG,
656 " Trial %d ACCEPTED (raw_ratio=%.4e, smoothed=%.4e); |dU|=%.6e | "
657 "dtau %.4e -> %.4e (cfl_eff %.4f -> %.4f)\n",
658 pseudo_iter, global_trial_ratio, smoothed_trial_ratio, global_norm_delta,
659 old_dtau, next_dtau, old_dtau * lambda_max, next_dtau * lambda_max);

665 char filen[PETSC_MAX_PATH_LEN + 128];
666 ierr = PetscSNPrintf(filen, sizeof(filen),
667 "%s/Momentum_Solver_Convergence_History_Block_%1d.log",
668 simCtx->log_dir, bi); CHKERRQ(ierr);
669 if (simCtx->step == simCtx->StartStep + 1 && pseudo_iter == 1 && !simCtx->continueMode)
670 f = fopen(filen, "w");
671 else
672 f = fopen(filen, "a");
673 if (simCtx->continueMode && simCtx->step == simCtx->StartStep + 1 && pseudo_iter == 1)
674 PetscFPrintf(PETSC_COMM_WORLD, f, "# Continuation from step %" PetscInt_FMT "\n", simCtx->StartStep);
675 /* Log format: dtau [physical time] + cfl_eff [dimensionless Courant number = dtau*lambda_max] */
676 PetscFPrintf(PETSC_COMM_WORLD, f,
677 "Step: %d | PseudoIter(k): %d | dtau: %.6e | cfl_eff: %.4f | |dUk|: %le | "
678 "|dUk|/|dU0|: %le | |Rk|: %le | |Rk|/|R0|: %le | "
679 "trial_ratio: %le | smoothed_ratio: %le | status: accepted | "
680 "dtau_after: %.6e | cfl_eff_after: %.4f\n",
681 (int)ti, (int)pseudo_iter, old_dtau, old_dtau * lambda_max,
682 delta_sol_norm_curr[bi], delta_sol_rel_curr[bi],
683 resid_norm_curr[bi], resid_rel_curr[bi],
684 trial_ratio_log[bi], smoothed_trial_ratio,
685 next_dtau, next_dtau * lambda_max);
686 fclose(f);
687 }
688 }
689
690 PetscBool update_pass;
691 PetscBool residual_pass = PETSC_TRUE;
692 if (residual_convergence_enabled) {
693 update_pass = (PetscBool)(global_norm_delta <= tol_abs_delta && global_rel_delta <= tol_rtol_delta);
694 residual_pass = (PetscBool)(
695 (simCtx->mom_resid_atol > 0.0 && global_norm_resid <= simCtx->mom_resid_atol) ||
696 (simCtx->mom_resid_rtol > 0.0 && global_rel_resid <= simCtx->mom_resid_rtol));
697 } else {
698 update_pass = (PetscBool)(global_norm_delta <= tol_abs_delta && global_rel_delta <= tol_rtol_delta);
699 }
700 converged = (PetscBool)(update_pass && residual_pass);
701
702 if (block_number > 1) {
703 // ierr = Block_Interface_U(user); CHKERRQ(ierr);
704 }
705 } // End while loop
706
707 if (last_trial_nonfinite) {
708 SETERRQ(PETSC_COMM_WORLD, PETSC_ERR_CONV_FAILED,
709 "Momentum solver exhausted its attempt limit while recovering from a non-finite trial.");
710 }

712 /* Convert the final dtau back to a CFL number for the warm-start of the next physical timestep.
713 * simCtx->pseudo_cfl stores a dimensionless Courant number (dtau * lambda_max); the next
714 * timestep's lambda_max may differ as the flow evolves, so storing the CFL number (not dtau)
715 * ensures the warm-start adapts correctly to the new spectral radius. */
716 PetscReal next_dtau_start = pseudo_dtau[0];
717 PetscReal next_cfl_warmstart = next_dtau_start * lambda_max; /* CFL = dtau * lambda_max */
718

724 }

726 /* Store the CFL number (not dtau) so it remains meaningful across physical timesteps. */
727 simCtx->pseudo_cfl = next_cfl_warmstart;
728 simCtx->mom_last_lambda_max = lambda_max;
729
730 /* Expose the last accepted finite state. On converged or floor-break exits pUcont==Ucont
731 already (VecCopy would be an identity), but ghosts may have gone stale; always re-sync
732 and re-apply BCs. Only restore the solution vector itself when nothing was ever accepted. */
733 if (accepted_iter == 0) {
734 for (PetscInt bi = 0; bi < block_number; bi++) {
735 ierr = VecCopy(user[bi].pUcont, user[bi].Ucont); CHKERRQ(ierr);
736 ierr = VecCopy(pRhs[bi], user[bi].Rhs); CHKERRQ(ierr);
737 }
738 }

740 const char *staggered_fields[] = {"Ucont"};
741 ierr = SynchronizePeriodicStaggeredFields(&user[bi], 1, staggered_fields); CHKERRQ(ierr);
742 ierr = ApplyBoundaryConditions(&user[bi]); CHKERRQ(ierr);
743 }
744
745 simCtx->mom_last_converged = converged;
746 if (!converged) {
747 PetscPrintf(PETSC_COMM_WORLD,
748 "[WARNING] Momentum solver step %d: reached %d total attempts (%d accepted, %d rejected) "
749 "without convergence; continuing from last accepted finite state.\n",
750 (int)ti, pseudo_iter, accepted_iter, rejected_iter);
751 }
752 if (accepted_iter == 0) {
753 PetscPrintf(PETSC_COMM_WORLD,
754 "[WARNING] Momentum solver step %d: no pseudo-time trials were accepted; "
755 "retaining physical-step entry state.\n", (int)ti);
756 }
757 LOG_ALLOW(GLOBAL, LOG_INFO,
758 "Momentum solver finished: %d attempts (%d accepted, %d rejected) of %d max accepted / %d hard cap, "
759 "converged=%s, last accepted |R|=%.6e, last accepted |dU|=%.6e, "
760 "next_dtau=%.4e (cfl=%.4f).\n",
761 pseudo_iter, accepted_iter, rejected_iter, max_pseudo_steps, max_total_attempts,
762 converged ? "yes" : "no",
763 (accepted_iter > 0) ? last_accepted_resid : resid_norm_init[0],
764 (accepted_iter > 0) ? delta_sol_norm_prev[0] : 0.0,
765 next_dtau_start, next_cfl_warmstart);

767 // --- Final Cleanup ---
768 for (PetscInt bi = 0; bi < block_number; bi++) {
769 ierr = VecDestroy(&user[bi].Rhs); CHKERRQ(ierr);
770 ierr = VecDestroy(&user[bi].dUcont); CHKERRQ(ierr);
771 ierr = VecDestroy(&user[bi].pUcont); CHKERRQ(ierr);
772 ierr = VecDestroy(&pRhs[bi]); CHKERRQ(ierr);
773 }
774 ierr = PetscFree(pRhs); CHKERRQ(ierr);

780 ierr = PetscFree(resid_rel_curr); CHKERRQ(ierr);
781 ierr = PetscFree(trial_ratio_log); CHKERRQ(ierr);
