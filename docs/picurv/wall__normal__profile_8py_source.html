1#!/usr/bin/env python3
2"""!
3@file wall_normal_profile.py
4@brief Reduce a field-statistics window into wall-normal profiles for DNS comparison.
5
6@details
7The reduction over the homogeneous directions is the post-processor's. A recipe
8deriving the window with `field_statistics.homogeneous` set to the streamwise and
9spanwise directions and `outputs` including `mean` and `reynolds_stress` writes a
10profile CSV whose rows hold, per wall-normal cell, the plane mean velocity and the
11covariance over the plane's pooled samples, the per-cell temporal covariance plus
12the spread of the cell means. This script reads that profile back, folds the two
13walls onto one half-channel (the Reynolds shear stress changes sign under the
14reflection), and writes a CSV of
15
16 y, y+, U+, u'+, v'+, w'+, -<u'v'>+
17
18together with the log-law and viscous-sublayer reference curves. The window's
19effective time bounds come from the `checkpoint.meta` of the bundle the profile was
20derived from, and the wall positions from the run's grid.
21
22The friction velocity comes from one of:
23
24- `--body-force f`: u_tau = sqrt(f h), the exact mean force balance of a
25 body-force-driven channel;
26- `--driven-flow-csv`: the same balance with f the driving acceleration from the run's
27 `driven_flow.csv`, averaged over the window's effective bounds;
28- `--wall-model-csv`: the run's `wall_model.csv`, as sqrt(<u_tau^2>) over the
29 rows inside the window's effective bounds, which is the mean wall shear a
30 wall-modelled run actually applies (a wall-modelled LES resolves no wall
31 gradient to fit);
32- `--u-tau`: a value supplied directly;
33- otherwise sqrt(nu dU/dy) from the first cell, which is only meaningful when the
34 first cell sits in the viscous sublayer.
35
36The PICGRID reader is imported from `generators/spectra.gen` and the metadata parser
37from `picurv_cli`, rather than duplicated. The profile must be in solver units: a
38recipe with `dimensionalize` off, or a case with unit reference scales.
39
40Usage:
41
42@code
43 wall_normal_profile.py \\
44 --profile RUN/output/analysis/statistics/RECIPE/Field_profile_stationary_50000.csv \\
45 --checkpoint RUN/output/checkpoints/step_000000050000 --window stationary \\
46 --grid RUN/inputs/grid/grid.run --wall-axis Eta --stream-axis Zeta \\
47 --viscosity 5.0e-05 --wall-model-csv RUN/output/analysis/metrics/wall_model.csv \\
48 --output profile.csv
49@endcode
50"""

61def find_repo_root():
62 """!
63 @brief Locate the PICurv checkout from the shipped example or from a case `picurv init` made.
64 @details In the repository this file sits four levels below the root. A copy inside an
65 initialized case does not, so the case's `.picurv-origin.json`, found by
66 walking up from this file, supplies the checkout it was made from.
67 @return Absolute path to the checkout.
68 """
69 here = os.path.dirname(os.path.abspath(__file__))
70 shipped = os.path.abspath(os.path.join(here, "..", "..", "..", ".."))
71 if os.path.isfile(os.path.join(shipped, "generators", "spectra.gen")):
72 return shipped
73 current = here
74 while True:
75 origin = os.path.join(current, ".picurv-origin.json")
76 if os.path.isfile(origin):
77 with open(origin, encoding="utf-8") as handle:
78 return json.load(handle).get("source_repo_root", shipped)
79 parent = os.path.dirname(current)
80 if parent == current:
81 return shipped
82 current = parent

111def read_window(checkpoint_dir, window):
112 """!
113 @brief Read a statistics window's scalars from one committed checkpoint bundle.
114 @details Only `checkpoint.meta` is read: the window's state, sample count and
115 effective time bounds, which the friction-velocity sources average over.
116 @param[in] checkpoint_dir Checkpoint bundle directory holding `checkpoint.meta`.
117 @param[in] window Window name from monitor.yml.
118 @return Dict of the window's checkpoint scalars.
119 """
120 sys.path.insert(0, REPO_ROOT)

122
123 metadata = os.path.join(checkpoint_dir, "checkpoint.meta")
124 if not os.path.isfile(metadata):
125 raise SystemExit(f"{checkpoint_dir} is not a committed checkpoint bundle: no checkpoint.meta.")
126 options = _read_checkpoint_options(metadata)
127 names = []
128 for index in range(int(options.get("checkpoint_statistics_window_count", 0) or 0)):
129 prefix = f"checkpoint_statistics_window_{index}_"

136def profile_statistics(profile_csv, window_name, grid_path, block, retained):
137 """!
138 @brief Read a window's velocity statistics merged over its homogeneous directions.
139 @details The reduction is the post-processor's: a recipe with
140 `field_statistics.homogeneous` naming the homogeneous directions and
141 `outputs` including `mean` and `reynolds_stress` writes, per retained
142 cell, the plane mean and the covariance over the plane's pooled samples
143 (the per-cell temporal covariance plus the spread of the cell means). This
144 reads that CSV back into arrays indexed by the retained cells in (k, j, i)
145 order, so the folding and wall units below work on them directly. The
146 values are taken as they were written, so the recipe must leave them in
147 solver units: `dimensionalize` off, or unit reference scales.
148 @param[in] profile_csv The `<prefix>_profile_<window>_<step>.csv` the recipe wrote.
149 @param[in] window_name Window the profile was written for.
150 @param[in] grid_path Canonical PICGRID path for the run.
151 @param[in] block Block index.
152 @param[in] retained Axes the profile must retain, as indices in (k, j, i) order.
153 @return Dict with `numpy`, `nodes` ((KM, JM, IM, 3) node coordinates), `mean`
154 (retained axes + (3,)) and `covariance` (retained axes + (6,), in
155 M2_PAIRS order).
156 """
157 helpers = load_spectra_helpers()
158 numpy = helpers.require_numpy()
159 blocks = helpers.read_picgrid_blocks(grid_path)
160 if block >= len(blocks):
161 raise SystemExit(f"block {block} is out of range; the grid holds {len(blocks)}.")
162 node_dims = blocks[block]["dims"]
163 cells_kji = (node_dims[2] - 1, node_dims[1] - 1, node_dims[0] - 1)
164 index_names = ("k", "j", "i")
165 with open(profile_csv, encoding="utf-8") as handle:
166 rows = [line.strip().split(",") for line in handle if line.strip()]
167 header, rows = rows[0], rows[1:]
168 present = tuple(axis for axis in range(3) if index_names[axis] in header)
169 if present != tuple(sorted(retained)):
170 raise SystemExit(f"{profile_csv} retains {[index_names[a] for a in present]}, but this "
171 f"reduction needs {[index_names[a] for a in sorted(retained)]}; set the "
172 "recipe's field_statistics.homogeneous to the other directions.")
173 labels = "xyz"
174 columns = [f"{window_name}_Ucat_mean_{labels[c]}" for c in range(3)]
175 columns += [f"{window_name}_Ucat_R_{labels[a]}{labels[b]}" for a, b in M2_PAIRS]
176 missing = [name for name in columns if name not in header]
177 if missing:
178 raise SystemExit(f"{profile_csv} has no {missing}; the recipe's outputs must include "
179 "mean and reynolds_stress for this window.")
180 shape = tuple(cells_kji[axis] for axis in present)
181 values = numpy.full(shape + (9,), numpy.nan)
182 for row in rows:
183 cell = tuple(int(row[header.index(index_names[axis])]) for axis in present)
184 values[cell] = [float(row[header.index(name)]) for name in columns]
185 if numpy.isnan(values).any():
186 raise SystemExit(f"{profile_csv} does not cover every retained cell; the window must "
187 "have sampled every cell.")
188 return {"numpy": numpy, "nodes": blocks[block]["coords"],
189 "mean": values[..., :3], "covariance": values[..., 3:]}

192def read_metrics_rows(csv_path):
193 """!
194 @brief Rows of a runtime metrics CSV, one per time, as a continued run leaves them.
195 @details A continued run writes a "# Continuation from step N" line into the file,
196 and one continued from an earlier checkpoint than the step it stopped at
197 writes the intervening steps a second time. Comment lines are skipped and,
198 for a repeated time, the later row - the one the continued run wrote - is kept.
199 @param[in] csv_path Path to the metrics CSV.
200 @return List of row dictionaries in time order.
201 """
202 rows = {}
203 with open(csv_path, newline="", encoding="utf-8") as handle:
204 for row in csv.DictReader(line for line in handle if not line.startswith("#")):
205 rows[float(row["time"])] = row
206 return [rows[t] for t in sorted(rows)]

209def friction_velocity_from_wall_model(csv_path, start, end):
210 """!
211 @brief Mean wall shear the wall model applied over a time interval, as a velocity.
212 @details Each wall_model.csv row carries the wall-face mean and standard deviation
213 of u_tau, so the row's mean of u_tau^2 is mean^2 + rms^2.
214 @param[in] csv_path Path to the run's wall_model.csv.
215 @param[in] start Window effective start, in solver time.
216 @param[in] end Window effective end, in solver time.
217 @return Tuple of (sqrt(<u_tau^2>), number of rows used).
218 """
219 total, rows = 0.0, 0
220 for row in read_metrics_rows(csv_path):
221 if start <= float(row["time"]) <= end:
222 total += float(row["u_tau_mean"]) ** 2 + float(row["u_tau_rms"]) ** 2
223 rows += 1
224 if rows == 0:
225 raise SystemExit(f"{csv_path} has no rows with time in the window [{start}, {end}].")
226 return math.sqrt(total / rows), rows

229def mean_driving_acceleration(csv_path, start, end):
230 """!
231 @brief Time-averaged driving acceleration a driven-periodic controller applied.
232 @details In a statistically stationary driven flow the mean force balances the mean
233 wall stress: <a> h = u_tau^2 for a channel of half-height h, and <a> A = tau_w P
234 for a duct of cross-section area A and wetted perimeter P.
235 @param[in] csv_path Path to the run's driven_flow.csv.
236 @param[in] start Window effective start, in solver time.
237 @param[in] end Window effective end, in solver time.
238 @return Tuple of (mean driving acceleration, number of rows used).
239 """
240 total, rows = 0.0, 0
241 for row in read_metrics_rows(csv_path):
242 if start <= float(row["time"]) <= end:
243 total += float(row["driving_acceleration"])
244 rows += 1
245 if rows == 0:
246 raise SystemExit(f"{csv_path} has no rows with time in the window [{start}, {end}].")
247 return total / rows, rows

250def main(argv=None):
251 """!
252 @brief Entry point.
253 @param[in] argv Command-line style argument list supplied to the function.
254 @return Process exit status.
255 """
256 parser = argparse.ArgumentParser(description=__doc__,
257 formatter_class=argparse.RawDescriptionHelpFormatter)
258 parser.add_argument("--profile", required=True,
259 help="Profile CSV a post recipe wrote for the window with homogeneous set "
260 "to the two directions other than the wall-normal one.")
261 parser.add_argument("--checkpoint", required=True,
262 help="Committed checkpoint bundle the profile was derived from; only its "
263 "checkpoint.meta is read, for the window's time bounds.")
264 parser.add_argument("--grid", required=True, help="Canonical PICGRID path for the run.")
265 parser.add_argument("--window", default="stationary", help="Statistics window name.")
266 parser.add_argument("--block", type=int, default=0, help="Block index.")
267 parser.add_argument("--wall-axis", default="Eta", choices=sorted(AXIS_TO_KJI),
268 help="Wall-normal axis token.")
269 parser.add_argument("--stream-axis", default="Zeta", choices=sorted(AXIS_TO_KJI),
270 help="Streamwise (driven) axis token.")
271 parser.add_argument("--viscosity", type=float, required=True,
272 help="Kinematic viscosity in solver units (1/Re for length_ref=velocity_ref=1).")
273 parser.add_argument("--half-height", type=float, default=1.0,
274 help="Channel half-height h in solver units.")
275 source = parser.add_mutually_exclusive_group()
276 source.add_argument("--body-force", type=float,
277 help="Converged driving body force f; u_tau = sqrt(f*h), the exact mean "
278 "force balance.")
279 source.add_argument("--wall-model-csv",
280 help="The run's wall_model.csv; u_tau = sqrt(<u_tau^2>) over the window.")
281 source.add_argument("--driven-flow-csv",
282 help="The run's driven_flow.csv; u_tau = sqrt(<a> h) from the driving "
283 "acceleration averaged over the window.")
284 source.add_argument("--u-tau", type=float, help="Friction velocity supplied directly.")
285 parser.add_argument("--output", required=True, help="Output CSV path.")
286 args = parser.parse_args(argv)
287
288 wall_kji = AXIS_TO_KJI[args.wall_axis]
289 stream_c = AXIS_TO_COMPONENT[args.stream_axis]
290 wall_c = AXIS_TO_COMPONENT[args.wall_axis]
291 span_c = ({0, 1, 2} - {stream_c, wall_c}).pop()
292
293 stats = profile_statistics(args.profile, args.window, args.grid, args.block, (wall_kji,))
294 numpy, nodes = stats["numpy"], stats["nodes"]
295 window = read_window(args.checkpoint, args.window)
296 plane_mean = stats["mean"]
297
298 def covariance(a, b):
299 return stats["covariance"][:, M2_PAIRS.index(tuple(sorted((a, b))))]
300
301 # Wall-normal cell-centre coordinates, taken from the node coordinates of the
302 # wall-normal axis. nodes is (KM, JM, IM, 3) in the same (k, j, i) order.
303 axis_nodes = {0: nodes[:, 0, 0, :], 1: nodes[0, :, 0, :], 2: nodes[0, 0, :, :]}[wall_kji]
304 axis_coord = axis_nodes[:, wall_c]
305 y_centres = 0.5 * (axis_coord[:-1] + axis_coord[1:])
306 if y_centres.size != plane_mean.shape[0]:
307 raise SystemExit(f"wall-normal cell count mismatch: grid gives {y_centres.size}, "
308 f"statistics give {plane_mean.shape[0]}.")
309
310 # Fold the upper half onto the lower one. Reflection reverses the wall-normal
311 # velocity, so <u'v'> changes sign; the orientation term makes the reported
312 # value the shear stress relative to the wall the point is measured from.
313 orientation = 1.0 if axis_coord[-1] > axis_coord[0] else -1.0
314 half = (y_centres.size + 1) // 2
315
316 def fold(values, sign=1.0):
317 return 0.5 * (values[:half] + sign * values[::-1][:half])
318
319 wall_distance = fold(numpy.minimum(numpy.abs(y_centres - axis_coord[0]),
320 numpy.abs(axis_coord[-1] - y_centres)))
321 U = fold(plane_mean[:, stream_c])
322 uu = fold(covariance(stream_c, stream_c))
323 vv = fold(covariance(wall_c, wall_c))
324 ww = fold(covariance(span_c, span_c))
325 uv = orientation * fold(covariance(stream_c, wall_c), -1.0)
326
327 nu, h = args.viscosity, args.half_height
328 if args.body_force is not None:
329 u_tau = math.sqrt(args.body_force * h)
330 source_note = f"sqrt(f*h) with f={args.body_force:.8e}"
331 elif args.driven_flow_csv is not None:
332 start, end = float(window["effective_start"]), float(window["effective_end"])
333 accel, rows = mean_driving_acceleration(args.driven_flow_csv, start, end)
334 u_tau = math.sqrt(accel * h)
335 source_note = f"sqrt(<a>*h) with <a>={accel:.8e} over {rows} driven_flow.csv rows"
336 elif args.wall_model_csv is not None:
337 start, end = float(window["effective_start"]), float(window["effective_end"])
338 u_tau, rows = friction_velocity_from_wall_model(args.wall_model_csv, start, end)
339 source_note = f"sqrt(<u_tau^2>) over {rows} wall_model.csv rows, t in [{start:g}, {end:g}]"
340 elif args.u_tau is not None:
341 u_tau = args.u_tau
342 source_note = "supplied with --u-tau"
343 else:
344 u_tau = math.sqrt(nu * abs(U[0]) / wall_distance[0])
345 source_note = "sqrt(nu*U/y) from the first cell (approximate; needs a resolved sublayer)"
346 if not u_tau > 0.0:
347 raise SystemExit("computed a non-positive friction velocity; check the inputs.")
348
349 with open(args.output, "w", encoding="utf-8") as handle:
350 handle.write(f"# u_tau = {u_tau:.10e} ({source_note})\n")
351 handle.write(f"# Re_tau = u_tau*h/nu = {u_tau * h / nu:.6f}\n")
352 handle.write(f"# nu = {nu:.10e}, h = {h:.10e}, window = {args.window} "
353 f"({window.get('state')}, {window.get('sample_count')} samples, "
354 f"represented time {float(window.get('represented_time', 'nan')):g})\n")
355 handle.write("y,y_plus,U_plus,u_rms_plus,v_rms_plus,w_rms_plus,"
356 "minus_uv_plus,U_plus_loglaw,U_plus_sublayer\n")
357 for index in range(half):
358 y_plus = wall_distance[index] * u_tau / nu
359 loglaw = (1.0 / 0.41) * math.log(y_plus) + 5.2 if y_plus > 0.0 else float("nan")
360 handle.write(
361 f"{wall_distance[index]:.10e},{y_plus:.10e},"
362 f"{U[index] / u_tau:.10e},"
363 f"{math.sqrt(max(uu[index], 0.0)) / u_tau:.10e},"
364 f"{math.sqrt(max(vv[index], 0.0)) / u_tau:.10e},"
365 f"{math.sqrt(max(ww[index], 0.0)) / u_tau:.10e},"
366 f"{-uv[index] / (u_tau ** 2):.10e},"
367 f"{loglaw:.10e},{y_plus:.10e}\n")
368
369 print(f"u_tau = {u_tau:.10e} ({source_note})")
370 print(f"Re_tau = {u_tau * h / nu:.4f}")
371 print(f"wrote {half} wall-normal stations to {args.output}")
372 return 0
