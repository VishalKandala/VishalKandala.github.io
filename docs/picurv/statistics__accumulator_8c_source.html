24/**
25 * @brief Upper-triangular row-major component pairs for a three-vector self-product.
26 *
27 * This pair table is the single definition of the symmetric component order. The
28 * diagonal positions and the two-axis component labels are both derived from it
29 * rather than restated, because accumulation reads the table while derivation reads
30 * what follows from it: a hand-written copy that drifted would mislabel every
31 * Reynolds stress and take the root of the wrong component, with nothing failing.
32 */
33static const PetscInt kProductFirst[6] = {0, 0, 0, 1, 1, 2};
34static const PetscInt kProductSecond[6] = {0, 1, 2, 1, 2, 2};
35
36/* That pair order is (xx, xy, xz, yy, yz, zz), which is exactly the order `SymTensor`
37 * declares its members in. The correspondence is not incidental: it is what lets a
38 * six-component accumulator and a `SymTensor` name the same entry by the same index,
39 * the way a three-component vector field and `Cmpnts` already do. Nothing in the type
40 * system enforces it, so it is asserted here; if either order is ever changed
41 * independently, this fails at compile time rather than silently mislabelling every
42 * Reynolds stress. */
43_Static_assert(offsetof(SymTensor, xx) == 0 * sizeof(PetscReal), "SymTensor order: xx is component 0");
44_Static_assert(offsetof(SymTensor, xy) == 1 * sizeof(PetscReal), "SymTensor order: xy is component 1");
45_Static_assert(offsetof(SymTensor, xz) == 2 * sizeof(PetscReal), "SymTensor order: xz is component 2");
46_Static_assert(offsetof(SymTensor, yy) == 3 * sizeof(PetscReal), "SymTensor order: yy is component 3");
47_Static_assert(offsetof(SymTensor, yz) == 4 * sizeof(PetscReal), "SymTensor order: yz is component 4");
48_Static_assert(offsetof(SymTensor, zz) == 5 * sizeof(PetscReal), "SymTensor order: zz is component 5");

53/**
54 * @brief Internal helper: the product index carrying one component's own variance.
55 * @details Local to this translation unit. Found by searching the pair table for the
56 * entry pairing a component with itself, so the diagonal cannot be stated
57 * separately from the order it belongs to.
58 */
59static PetscInt ProductDiagonalIndex(PetscInt component)
60{
61 for (PetscInt c = 0; c < 6; ++c) {
62 if (kProductFirst[c] == component && kProductSecond[c] == component) return c;
63 }
64 return -1;
65}

158/**
159 * @brief Internal helper: the sample set one field slot, or one covariance pair, uses.
160 * @details Local to this translation unit. A pair is measured where both members are,
161 * so it is occupied when either member is particle-sampled.

163 * @param[in] slot Field slot, or -1 to select @p pair instead.
164 * @param[in] pair Covariance pair, when @p slot is -1.

172 FieldIsParticleSampled(definition->covariances[pair].second)) ? SAMPLE_SET_OCCUPIED : SAMPLE_SET_ALL;

204 * @brief Implementation of \ref PicurvWindowStorageCreate().
205 * @see PicurvWindowStorageCreate()
206 */
207PetscErrorCode PicurvWindowStorageCreate(UserCtx *user, const PicurvWindowDefinition *definition,
208 PicurvWindowStorage *storage)
209{
210 PetscFunctionBeginUser;
211 PROFILE_FUNCTION_BEGIN;
212 PetscCheck(user != NULL && definition != NULL && storage != NULL,
213 PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "Context, definition, and storage are required.");
214 PetscCall(PetscMemzero(storage, sizeof(*storage)));
215 storage->field_count = definition->field_count;
216 storage->covariance_count = definition->covariance_count;
217
218 /* Occupancy is per point and scalar regardless of what is accumulated. */
219 PetscCall(DMCreateGlobalVector(user->da, &storage->count));
220 PetscCall(VecSet(storage->count, 0.0));
221 PetscCall(DMCreateGlobalVector(user->da, &storage->weight));
222 PetscCall(VecSet(storage->weight, 0.0));
223 PetscCall(DMCreateGlobalVector(user->da, &storage->weight_sq));
224 PetscCall(VecSet(storage->weight_sq, 0.0));
225
226 if (definition->field_count > 0) {
227 PetscCall(PetscCalloc1((size_t)definition->field_count, &storage->mean));
228 PetscCall(PetscCalloc1((size_t)definition->field_count, &storage->m2));
229 }
230 for (PetscInt field_index = 0; field_index < definition->field_count; ++field_index) {
231 FieldView view;
232 PetscInt components = 0;
233 DM product_dm = NULL;
234
235 PetscCall(FieldGetView(user, (FieldId)definition->fields[field_index].field_id, &view));
236 /* The mean matches the source field's own layout, so it comes from that
237 * field's DM and inherits its decomposition. */
238 PetscCall(DMCreateGlobalVector(view.dm, &storage->mean[field_index]));
239 PetscCall(VecSet(storage->mean[field_index], 0.0));
240 if (!definition->fields[field_index].want_second) continue;
241 PetscCall(PicurvProductComponentCount(view.descriptor->dof, &components));
242 PetscCall(PicurvStatisticsComponentDM(user, components, &product_dm));
243 PetscCall(DMCreateGlobalVector(product_dm, &storage->m2[field_index]));
244 PetscCall(VecSet(storage->m2[field_index], 0.0));
245 }
246
247 if (definition->covariance_count > 0) {
248 PetscCall(PetscCalloc1((size_t)definition->covariance_count, &storage->cm));
249 }
250 /* The occupied sample set exists only when something is measured over it. */
251 for (PetscInt field_index = 0; field_index < definition->field_count; ++field_index) {
252 if (RequestSampleSet(definition, field_index, -1) != SAMPLE_SET_OCCUPIED) continue;
253 PetscCall(DMCreateGlobalVector(user->da, &storage->occupied_count));
254 PetscCall(VecSet(storage->occupied_count, 0.0));
255 PetscCall(VecDuplicate(storage->occupied_count, &storage->occupied_weight));
256 PetscCall(VecSet(storage->occupied_weight, 0.0));
257 PetscCall(VecDuplicate(storage->occupied_count, &storage->occupied_weight_sq));
258 PetscCall(VecSet(storage->occupied_weight_sq, 0.0));
259 PetscCall(PetscCalloc1((size_t)definition->field_count, &storage->occupied_mean));
260 break;
261 }
262 for (PetscInt pair_index = 0; pair_index < definition->covariance_count; ++pair_index) {
263 FieldView view_a, view_b;
264 PetscInt components = 0;
265 DM pair_dm = NULL;
266
267 PetscCall(FieldGetView(user, (FieldId)definition->covariances[pair_index].first, &view_a));
268 PetscCall(FieldGetView(user, (FieldId)definition->covariances[pair_index].second, &view_b));
269 PetscCall(PicurvCovarianceComponentCount(view_a.descriptor->dof, view_b.descriptor->dof, &components));
270 PetscCall(PicurvStatisticsComponentDM(user, components, &pair_dm));
271 PetscCall(DMCreateGlobalVector(pair_dm, &storage->cm[pair_index]));
272 PetscCall(VecSet(storage->cm[pair_index], 0.0));
273
274 /* An ordinary member of an occupied pair is centred on its occupied-state mean. */
275 for (PetscInt member = 0; member < 2 && RequestSampleSet(definition, -1, pair_index) == SAMPLE_SET_OCCUPIED; ++member) {
276 const PetscInt field_id = member ? definition->covariances[pair_index].second
277 : definition->covariances[pair_index].first;
278 PetscInt slot = 0;
279
280 if (FieldIsParticleSampled(field_id)) continue;
281 PetscCall(FindFieldSlot(definition, field_id, &slot));
282 if (storage->occupied_mean[slot]) continue;
283 PetscCall(VecDuplicate(storage->mean[slot], &storage->occupied_mean[slot]));
284 PetscCall(VecSet(storage->occupied_mean[slot], 0.0));
285 }
286 }

384/**
385 * @brief Implementation of \ref PicurvWindowStoragePayload().
386 * @see PicurvWindowStoragePayload()
387 */
388PetscErrorCode PicurvWindowStoragePayload(UserCtx *user, const PicurvWindowDefinition *definition,
389 const PicurvWindowStorage *storage, PetscInt index,
390 PicurvStatisticsPayload *payload)
391{
392 PetscInt total = 0;
393 PetscInt cursor = index;
394
395 PetscFunctionBeginUser;
396 PetscCheck(user != NULL && definition != NULL && storage != NULL && payload != NULL,
397 PETSC_COMM_SELF, PETSC_ERR_ARG_NULL,
398 "Context, definition, storage, and payload output are required.");
399 PetscCall(PicurvWindowStoragePayloadCount(storage, &total));
400 PetscCheck(index >= 0 && index < total, PETSC_COMM_SELF, PETSC_ERR_ARG_OUTOFRANGE,
401 "Payload index %" PetscInt_FMT " is outside [0, %" PetscInt_FMT ").", index, total);
402 PetscCall(PetscMemzero(payload, sizeof(*payload)));
403
404 /* Occupancy first, then means, then products, then co-moments. The order is the
405 * persistence contract: the manifest inventory, the writer, and the reader all
406 * walk it identically. The occupied set's occupancy follows the shared one and its
407 * means follow the field means, so a window without it keeps its layout. */
408 {
409 static const char *const occupancy_name[6] = {"count", "weight", "weight_sq", "occupied_count",
410 "occupied_weight", "occupied_weight_sq"};
411 const Vec occupancy[6] = {storage->count, storage->weight, storage->weight_sq, storage->occupied_count,
412 storage->occupied_weight, storage->occupied_weight_sq};
413 const PetscInt occupancy_count = storage->occupied_count ? 6 : 3;
414
415 if (cursor < occupancy_count) {
416 PetscCall(PetscStrncpy(payload->name, occupancy_name[cursor], sizeof(payload->name)));
417 payload->vec = occupancy[cursor];
418 payload->components = 1;
419 payload->role = "occupancy";
420 payload->layout = FieldLayoutName(FIELD_LAYOUT_CELL_CENTERED);
421 PetscFunctionReturn(0);
422 }
423 cursor -= occupancy_count;
424 }
425
426 if (cursor < storage->field_count) {
427 const FieldDescriptor *descriptor = NULL;
428
429 PetscCall(FieldGetDescriptor((FieldId)definition->fields[cursor].field_id, &descriptor));
430 PetscCall(PetscSNPrintf(payload->name, sizeof(payload->name), "%s_mean",
431 descriptor->canonical_name));
432 payload->vec = storage->mean[cursor];
433 payload->components = descriptor->dof;
434 payload->role = "mean";
435 payload->layout = FieldLayoutName(descriptor->layout);
436 PetscFunctionReturn(0);
437 }
438 cursor -= storage->field_count;
439 for (PetscInt field_index = 0; storage->occupied_mean && field_index < storage->field_count; ++field_index) {
440 const FieldDescriptor *descriptor = NULL;
441
442 if (!storage->occupied_mean[field_index]) continue;
443 if (cursor-- != 0) continue;
444 PetscCall(FieldGetDescriptor((FieldId)definition->fields[field_index].field_id, &descriptor));
445 PetscCall(PetscSNPrintf(payload->name, sizeof(payload->name), "%s_occupied_mean",
446 descriptor->canonical_name));
447 payload->vec = storage->occupied_mean[field_index];
448 payload->components = descriptor->dof;
449 payload->role = "mean";
450 payload->layout = FieldLayoutName(descriptor->layout);
451 PetscFunctionReturn(0);
452 }
453
454 {
455 PetscInt product_count = 0;
456 PetscInt seen = 0;
457
458 for (PetscInt field_index = 0; storage->m2 && field_index < storage->field_count; ++field_index) {
459 if (storage->m2[field_index]) ++product_count;
460 }
461 if (cursor < product_count) {
462 for (PetscInt field_index = 0; field_index < storage->field_count; ++field_index) {
463 const FieldDescriptor *descriptor = NULL;
464
465 if (!storage->m2[field_index]) continue;
466 if (seen++ != cursor) continue;
467 PetscCall(FieldGetDescriptor((FieldId)definition->fields[field_index].field_id, &descriptor));
468 PetscCall(PetscSNPrintf(payload->name, sizeof(payload->name), "%s_m2",
469 descriptor->canonical_name));
470 payload->vec = storage->m2[field_index];
471 PetscCall(PicurvProductComponentCount(descriptor->dof, &payload->components));
472 payload->role = "second_moment";
473 payload->layout = FieldLayoutName(descriptor->layout);
474 PetscFunctionReturn(0);
475 }
476 }
477 cursor -= product_count;
478 }
479
480 {
481 const FieldDescriptor *first = NULL;
482 const FieldDescriptor *second = NULL;
483
484 PetscCheck(cursor >= 0 && cursor < storage->covariance_count, PETSC_COMM_SELF, PETSC_ERR_PLIB,
485 "Payload index %" PetscInt_FMT " fell through the storage enumeration.", index);
486 PetscCall(FieldGetDescriptor((FieldId)definition->covariances[cursor].first, &first));
487 PetscCall(FieldGetDescriptor((FieldId)definition->covariances[cursor].second, &second));
488 PetscCall(PetscSNPrintf(payload->name, sizeof(payload->name), "%s_%s_cm",
489 first->canonical_name, second->canonical_name));
490 payload->vec = storage->cm[cursor];
491 PetscCall(PicurvCovarianceComponentCount(first->dof, second->dof, &payload->components));
492 payload->role = "co_moment";
493 payload->layout = FieldLayoutName(first->layout);
494 }
495 PetscFunctionReturn(0);
496}

514/**
515 * @brief Power of the source field's reference scale each derived kind carries.
516 *
517 * A first moment and a standard deviation have the field's own units; a covariance,
518 * its trace, and a co-moment flux are quadratic in it. This is what the per-field
519 * scaling table alone cannot express, and why derived statistics were previously left
520 * non-dimensional rather than scaled by a velocity that would have been wrong for
521 * three of the five kinds.
522 */
523static const PetscInt kDerivedKindScaleExponent[DERIVED_KIND_COUNT] = {
524 1, /* mean - first moment, the field's own units */
525 2, /* reynolds_stress - covariance of one field with itself */
526 1, /* rms - standard deviation */
527 2, /* tke - half the trace of that covariance */
528 0, /* flux - a co-moment of two possibly different fields, so its
529 factor is their product and cannot be an exponent of
530 one scale; DERIVED_FLUX resolves it directly. */
531 2 /* point_variance - a variance of one field */
532};

534/**
535 * @brief Internal helper: reports which output kinds a recipe requested.
536 * @details Local to this translation unit.
537 */
538static PetscErrorCode ParseDerivedKinds(const char *outputs, PetscBool wanted[DERIVED_KIND_COUNT])
539{
540 char buffer[STATISTICS_DERIVED_OUTPUT_LENGTH];
541 char *cursor = buffer;
542
543 PetscFunctionBeginUser;
544 for (PetscInt k = 0; k < DERIVED_KIND_COUNT; ++k) wanted[k] = PETSC_FALSE;
545 PetscCheck(outputs != NULL, PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "Output list is required.");
546 PetscCall(PetscStrncpy(buffer, outputs, sizeof(buffer)));
547
548 while (cursor && *cursor) {
549 char *comma = strchr(cursor, ',');
550 PetscBool matched = PETSC_FALSE;
551
552 if (comma) *comma = '\0';
553 TrimWhitespace(cursor);
554 if (cursor[0] != '\0') {
555 for (PetscInt k = 0; k < DERIVED_KIND_COUNT; ++k) {
556 if (strcmp(cursor, kDerivedKindName[k])) continue;
557 wanted[k] = PETSC_TRUE;
558 matched = PETSC_TRUE;
559 break;
560 }
561 PetscCheck(matched, PETSC_COMM_SELF, PETSC_ERR_ARG_WRONG,
562 "Unknown field-statistics output '%s'. Available outputs are "
563 "mean, reynolds_stress, rms, tke, flux, and point_variance.", cursor);
564 }
565 cursor = comma ? comma + 1 : NULL;
566 }
567 PetscFunctionReturn(0);
568}

570/**
571 * @brief Internal helper: counts the derived fields each kind contributes.
572 * @details Local to this translation unit. A kind contributes nothing when the state
573 * it needs was never accumulated, so a recipe may name every output without
574 * having to know which window carries which moment.
575 */
576static PetscErrorCode DerivedKindExtent(const PicurvWindowDefinition *definition,
577 const PicurvWindowStorage *storage,
578 DerivedKind kind, PetscInt *count)
579{
580 PetscFunctionBeginUser;
581 *count = 0;
582 switch (kind) {
583 case DERIVED_MEAN:
584 *count = storage->field_count;
585 break;
586 case DERIVED_REYNOLDS_STRESS:
587 case DERIVED_RMS:
588 case DERIVED_POINT_VARIANCE:
589 for (PetscInt field_index = 0; storage->m2 && field_index < storage->field_count; ++field_index) {
590 const FieldDescriptor *descriptor = NULL;
591 PetscInt components = 0;
592
593 if (!storage->m2[field_index]) continue;
594 PetscCall(FieldGetDescriptor((FieldId)definition->fields[field_index].field_id, &descriptor));
595 PetscCall(PicurvProductComponentCount(descriptor->dof, &components));
596 /* A stress tensor emits every component; an RMS and a point variance
597 * emit only the diagonal, one value per component. */
598 *count += (kind == DERIVED_REYNOLDS_STRESS) ? components : descriptor->dof;
599 }
600 break;
601 case DERIVED_TKE:
602 /* Turbulent kinetic energy is the trace of a three-vector's stress tensor,
603 * so it exists only for a vector field carrying a second moment. */
604 for (PetscInt field_index = 0; storage->m2 && field_index < storage->field_count; ++field_index) {
605 const FieldDescriptor *descriptor = NULL;
606
607 if (!storage->m2[field_index]) continue;
608 PetscCall(FieldGetDescriptor((FieldId)definition->fields[field_index].field_id, &descriptor));
609 if (descriptor->dof == 3) *count += 1;
610 }
611 break;
612 default:
613 *count = storage->covariance_count;
614 break;
615 }
616 PetscFunctionReturn(0);
617}

644/**
645 * @brief Internal helper: takes a square root of a variance that may be barely negative.
646 * @details Local to this translation unit. Centered accumulation can leave a variance
647 * a few ulps below zero when the signal is nearly constant. Clamping is
648 * confined to this one place, applies only under a root, and never touches
649 * stored state; a genuinely negative variance is a defect and is reported.
650 */
651static PetscErrorCode SafeStandardDeviation(PetscReal variance, const char *label, PetscReal *result)
652{
653 PetscFunctionBeginUser;
654 if (variance >= 0.0) {
655 *result = PetscSqrtReal(variance);
656 PetscFunctionReturn(0);
657 }
658 PetscCheck(variance >= -PICURV_STATISTICS_VARIANCE_FLOOR, PETSC_COMM_SELF, PETSC_ERR_FP,
659 "Derived variance for '%s' is %g, which is too negative to be floating-point "
660 "cancellation; the accumulated state is inconsistent.", label, (double)variance);
661 *result = 0.0;
662 PetscFunctionReturn(0);
663}

667 * @details Local to this translation unit. Walks the same order `DerivedKindExtent`

670static PetscErrorCode ResolveDerivedIndex(const PicurvWindowDefinition *definition,
671 const PicurvWindowStorage *storage,
672 const PetscBool wanted[DERIVED_KIND_COUNT],
673 PetscInt index, DerivedKind *kind, PetscInt *offset)

677 PetscInt extent = 0;

681 if (index < extent) {

684 PetscFunctionReturn(0);
685 }
686 index -= extent;
687 }
688 SETERRQ(PETSC_COMM_SELF, PETSC_ERR_ARG_OUTOFRANGE,
689 "Derived index is past the end of the requested output set.");

692#undef __FUNCT__
693#define __FUNCT__ "PicurvWindowDerive"
694/**
695 * @brief Implementation of \ref PicurvWindowDerive().
696 * @see PicurvWindowDerive()
697 */
698PetscErrorCode PicurvWindowDerive(UserCtx *user, const PicurvWindowDefinition *definition,
699 const PicurvWindowStorage *storage, const char *outputs,
700 PetscInt index, Vec scalar_target, Vec vector_target,
701 PicurvDerivedField *field)
702{
703 PetscBool wanted[DERIVED_KIND_COUNT];
704 DerivedKind kind = DERIVED_MEAN;
705 SpatialTargetPlan plan;
706 PetscReal ***weight_arr = NULL, ***weight_sq_arr = NULL, ***count_arr = NULL;
707 PetscScalar ****source = NULL, ****target = NULL;
708 const FieldDescriptor *descriptor = NULL;
709 DM source_dm = NULL;
710 Vec source_vec = NULL;
711 PetscInt offset = 0, slot = 0, member = 0;
712 SampleSet set = SAMPLE_SET_ALL;
713 Vec count_vec = NULL, weight_vec = NULL, weight_sq_vec = NULL;
714 /* Resolved where the branch knows which fields it read; 1.0 leaves the result in
715 * solver units, which is also what an undeclared scale falls back to. */
716 PetscReal dimensional_factor = 1.0;
717 /* The source's component count is not the output's: a scalar RMS reads a
718 * six-component tensor, so each needs its own DM. */
719 PetscInt source_components = 0;
720
721 PetscFunctionBeginUser;
722 PROFILE_FUNCTION_BEGIN;
723 PetscCheck(user != NULL && definition != NULL && storage != NULL && field != NULL,
724 PETSC_COMM_SELF, PETSC_ERR_ARG_NULL,
725 "Context, definition, storage, and output are required.");
726 PetscCheck(index >= 0, PETSC_COMM_SELF, PETSC_ERR_ARG_OUTOFRANGE,
727 "Derived index %" PetscInt_FMT " is negative.", index);
728 PetscCall(PetscMemzero(field, sizeof(*field)));
729 /* The output list is parsed once and the extents walked once. Resolving the
730 * index is what bounds-checks it, so counting first would repeat both. */
731 PetscCall(ParseDerivedKinds(outputs, wanted));
732 PetscCall(ResolveDerivedIndex(definition, storage, wanted, index, &kind, &offset));
733
734 /* Locate the accumulator slot and component this index selects, and the sample
735 * set that slot or pair was measured over. */
736 switch (kind) {
737 case DERIVED_MEAN:
738 slot = offset;
739 PetscCall(FieldGetDescriptor((FieldId)definition->fields[slot].field_id, &descriptor));
740 source_vec = storage->mean[slot];
741 set = RequestSampleSet(definition, slot, -1);
742 field->components = descriptor->dof;
743 source_components = descriptor->dof;
744 PetscCall(PetscSNPrintf(field->name, sizeof(field->name), "%s_%s_mean",
745 definition->name, descriptor->canonical_name));
746 break;
747 case DERIVED_TKE:
748 for (slot = 0; slot < storage->field_count; ++slot) {
749 if (!storage->m2[slot]) continue;
750 PetscCall(FieldGetDescriptor((FieldId)definition->fields[slot].field_id, &descriptor));
751 if (descriptor->dof != 3) continue;
752 if (offset-- == 0) break;
753 }
754 source_vec = storage->m2[slot];
755 set = RequestSampleSet(definition, slot, -1);
756 field->components = 1;
757 PetscCall(PicurvProductComponentCount(descriptor->dof, &source_components));
758 PetscCall(PetscSNPrintf(field->name, sizeof(field->name), "%s_%s_tke",
759 definition->name, descriptor->canonical_name));
760 break;
761 case DERIVED_FLUX:
762 slot = offset;
763 {
764 const FieldDescriptor *first = NULL;
765 const FieldDescriptor *second = NULL;
766
767 PetscCall(FieldGetDescriptor((FieldId)definition->covariances[slot].first, &first));
768 PetscCall(FieldGetDescriptor((FieldId)definition->covariances[slot].second, &second));
769 PetscCall(PicurvCovarianceComponentCount(first->dof, second->dof, &field->components));
770 source_components = field->components;
771 source_vec = storage->cm[slot];
772 set = RequestSampleSet(definition, -1, slot);
773 if (user->simCtx->pps && user->simCtx->pps->dimensionalize) {
774 PetscReal first_scale = 1.0, second_scale = 1.0;
775
776 /* A co-moment carries the product of its two fields' dimensions. */
777 PetscCall(PicurvFieldReferenceScale(user->simCtx, first->canonical_name,
778 &first_scale, NULL, 0));
779 PetscCall(PicurvFieldReferenceScale(user->simCtx, second->canonical_name,
780 &second_scale, NULL, 0));
781 dimensional_factor = first_scale * second_scale;
782 }
783 PetscCall(PetscSNPrintf(field->name, sizeof(field->name), "%s_%s_%s_flux",
784 definition->name, first->canonical_name, second->canonical_name));
785 }
786 break;
787 default:
788 /* Stress and RMS both walk the fields carrying a product; stress emits every
789 * symmetric component, RMS only the diagonal. */
790 for (slot = 0; slot < storage->field_count; ++slot) {
791 PetscInt extent = 0;
792
793 if (!storage->m2[slot]) continue;
794 PetscCall(FieldGetDescriptor((FieldId)definition->fields[slot].field_id, &descriptor));
795 PetscCall(PicurvProductComponentCount(descriptor->dof, &extent));
796 if (kind == DERIVED_RMS || kind == DERIVED_POINT_VARIANCE) extent = descriptor->dof;
797 if (offset < extent) { member = offset; break; }
798 offset -= extent;
799 }
800 source_vec = storage->m2[slot];
801 set = RequestSampleSet(definition, slot, -1);
802 field->components = 1;
803 PetscCall(PicurvProductComponentCount(descriptor->dof, &source_components));
804 if (kind == DERIVED_RMS || kind == DERIVED_POINT_VARIANCE) {
805 PetscCall(PetscSNPrintf(field->name, sizeof(field->name), "%s_%s_%s%s",
806 definition->name, descriptor->canonical_name,
807 kind == DERIVED_RMS ? "rms" : "point_variance",
808 descriptor->dof == 1 ? "" : kAxisName[member]));
809 field->point_averaged = (PetscBool)(kind == DERIVED_POINT_VARIANCE);
810 } else if (descriptor->dof == 1) {
811 PetscCall(PetscSNPrintf(field->name, sizeof(field->name), "%s_%s_variance",
812 definition->name, descriptor->canonical_name));
813 } else {
814 char label[8];
815
816 PetscCall(ProductComponentLabel(member, label, sizeof(label)));
817 PetscCall(PetscSNPrintf(field->name, sizeof(field->name), "%s_%s_R_%s",
818 definition->name, descriptor->canonical_name, label));
819 }
820 break;
821 }
822
823 PetscCall(SpatialTargetPlanCreate(user, (FieldId)definition->fields[0].field_id,
824 PICURV_STATISTICS_MASK_FLUID, &plan));
825 PetscCall(PicurvStatisticsComponentDM(user, source_components, &source_dm));
826 {
827 Vec destination = (field->components == 1) ? scalar_target : vector_target;
828 DM destination_dm = NULL;
829
830 PetscCheck(destination != NULL, PETSC_COMM_SELF, PETSC_ERR_ARG_NULL,
831 "Derived field '%s' needs a %d-component destination.",
832 field->name, (int)field->components);
833 PetscCall(PicurvStatisticsComponentDM(user, field->components, &destination_dm));
834 PetscCall(VecZeroEntries(destination));
835
836 field->occupied = (PetscBool)(set == SAMPLE_SET_OCCUPIED);
837 SampleSetOccupancy(storage, set, &count_vec, &weight_vec, &weight_sq_vec);
838 PetscCall(DMDAVecGetArrayRead(user->da, weight_vec, &weight_arr));
839 PetscCall(DMDAVecGetArrayRead(user->da, weight_sq_vec, &weight_sq_arr));
840 PetscCall(DMDAVecGetArrayRead(user->da, count_vec, &count_arr));
841 PetscCall(DMDAVecGetArrayDOFRead(source_dm, source_vec, &source));
842 PetscCall(DMDAVecGetArrayDOF(destination_dm, destination, &target));
843
844 for (PetscInt k = plan.start[2]; k < plan.end[2]; ++k) {
845 for (PetscInt j = plan.start[1]; j < plan.end[1]; ++j) {
846 for (PetscInt i = plan.start[0]; i < plan.end[0]; ++i) {
847 PicurvCoMomentState pair;
848
849 /* A point the window never sampled has no average at all, so it
850 * is left at zero rather than divided by a zero weight. */
851 if (weight_arr[k][j][i] <= 0.0) continue;
852 pair.count = count_arr[k][j][i];
853 pair.weight = weight_arr[k][j][i];
854 pair.weight_sq = weight_sq_arr[k][j][i];
855 pair.mean_x = 0.0;
856 pair.mean_y = 0.0;
857
858 if (kind == DERIVED_MEAN) {
859 for (PetscInt c = 0; c < field->components; ++c) {
860 target[k][j][i][c] = source[k][j][i][c];
861 }
862 } else if (kind == DERIVED_TKE) {
863 /* The trace of the stress tensor, halved: components 0, 3
864 * and 5 are xx, yy and zz in the stored symmetric order. */
865 PetscReal trace = 0.0;
866
867 for (PetscInt c = 0; c < 3; ++c) {
868 pair.cm = source[k][j][i][ProductDiagonalIndex(c)];
869 trace += PicurvCoMomentStateCovariance(&pair);
870 }
871 target[k][j][i][0] = 0.5 * trace;
872 } else if (kind == DERIVED_RMS || kind == DERIVED_POINT_VARIANCE) {
873 PetscReal deviation = 0.0;
874
875 pair.cm = source[k][j][i][(descriptor->dof == 1)
876 ? 0 : ProductDiagonalIndex(member)];
877 PetscCall(SafeStandardDeviation(PicurvCoMomentStateCovariance(&pair),
878 field->name, &deviation));
879 target[k][j][i][0] = (kind == DERIVED_RMS) ? deviation : deviation * deviation;
880 } else if (kind == DERIVED_FLUX) {
881 for (PetscInt c = 0; c < field->components; ++c) {
882 pair.cm = source[k][j][i][c];
883 target[k][j][i][c] = PicurvCoMomentStateCovariance(&pair);
884 }
885 } else {
886 pair.cm = source[k][j][i][member];
887 target[k][j][i][0] = PicurvCoMomentStateCovariance(&pair);
888 }
889 }
890 }
891 }
892
893 PetscCall(DMDAVecRestoreArrayDOF(destination_dm, destination, &target));
894 PetscCall(DMDAVecRestoreArrayDOFRead(source_dm, source_vec, &source));
895 PetscCall(DMDAVecRestoreArrayRead(user->da, count_vec, &count_arr));
896 PetscCall(DMDAVecRestoreArrayRead(user->da, weight_sq_vec, &weight_sq_arr));
897 PetscCall(DMDAVecRestoreArrayRead(user->da, weight_vec, &weight_arr));
898
899 /* Applied once, here, so the VTK field and the convergence-history CSV - which
900 * both read this staging vector - cannot end up in different unit systems. */
901 if (user->simCtx->pps && user->simCtx->pps->dimensionalize) {
902 if (kind != DERIVED_FLUX && descriptor) {
903 PetscReal base = 1.0;
904
905 PetscCall(PicurvFieldReferenceScale(user->simCtx, descriptor->canonical_name,
906 &base, NULL, 0));
907 dimensional_factor = PetscPowRealInt(base, kDerivedKindScaleExponent[kind]);
908 }
909 if (PetscAbsReal(dimensional_factor - 1.0) > PETSC_MACHINE_EPSILON) {
910 PetscCall(VecScale(destination, dimensional_factor));
911 LOG_ALLOW(GLOBAL, LOG_DEBUG, "Scaled derived '%s' by %.4e.\n",
912 kDerivedKindName[kind], (double)dimensional_factor);
913 }
914 }
915 }
916 PROFILE_FUNCTION_END;
917 PetscFunctionReturn(0);
918}

957/**
958 * @brief Implementation of \ref PicurvWindowValidFractionRange().
959 * @see PicurvWindowValidFractionRange()
960 */
961PetscErrorCode PicurvWindowValidFractionRange(UserCtx *user, const PicurvWindowDefinition *definition,
962 const PicurvWindowStorage *storage, PetscInt sample_count,
963 PetscReal *minimum, PetscReal *maximum)
964{
965 SpatialTargetPlan plan;
966 PetscReal ***nvert = NULL;
967 PetscReal ***count_arr = NULL;
968 PetscReal local_min = PETSC_MAX_REAL;
969 PetscReal local_max = 0.0;
970 PetscReal reduced_min = 0.0, reduced_max = 0.0;
971
972 PetscFunctionBeginUser;
973 PROFILE_FUNCTION_BEGIN;
974 PetscCheck(user != NULL && definition != NULL && storage != NULL &&
975 minimum != NULL && maximum != NULL,
976 PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "Context, definition, storage, and outputs are required.");
977 *minimum = 1.0;
978 *maximum = 0.0;
979 if (definition->field_count == 0 || sample_count <= 0) { PROFILE_FUNCTION_END; PetscFunctionReturn(0); }
980
981 PetscCall(SpatialTargetPlanCreate(user, (FieldId)definition->fields[0].field_id,
982 PICURV_STATISTICS_MASK_FLUID, &plan));
983 PetscCall(DMDAVecGetArrayRead(user->da, user->Nvert, &nvert));
984 PetscCall(DMDAVecGetArrayRead(user->da, storage->count, &count_arr));
985 for (PetscInt k = plan.start[2]; k < plan.end[2]; ++k) {
986 for (PetscInt j = plan.start[1]; j < plan.end[1]; ++j) {
987 for (PetscInt i = plan.start[0]; i < plan.end[0]; ++i) {
988 /* The mask is evaluated at the current state, but a point excluded
989 * now may have contributed earlier, so its stored count is what
990 * decides its fraction rather than its present eligibility. */
991 const PetscReal fraction = count_arr[k][j][i] / (PetscReal)sample_count;
992
993 local_min = PetscMin(local_min, fraction);
994 local_max = PetscMax(local_max, fraction);
995 }
996 }
997 }
998 PetscCall(DMDAVecRestoreArrayRead(user->da, storage->count, &count_arr));
999 PetscCall(DMDAVecRestoreArrayRead(user->da, user->Nvert, &nvert));
1000
1001 /* A rank owning no targeted point must not drag the minimum down, so its
1002 * sentinel is neutral under the reduction rather than zero. */
1003 PetscCallMPI(MPI_Allreduce(&local_min, &reduced_min, 1, MPIU_REAL, MPIU_MIN, PETSC_COMM_WORLD));
1004 PetscCallMPI(MPI_Allreduce(&local_max, &reduced_max, 1, MPIU_REAL, MPIU_MAX, PETSC_COMM_WORLD));
1005 *minimum = (reduced_min == PETSC_MAX_REAL) ? 1.0 : reduced_min;
1006 *maximum = reduced_max;
1007 PROFILE_FUNCTION_END;
1008 PetscFunctionReturn(0);
1009}

1013/**
1014 * @brief Implementation of \ref PicurvWindowAccumulate().
1015 * @see PicurvWindowAccumulate()
1016 */
1017PetscErrorCode PicurvWindowAccumulate(UserCtx *user, const PicurvWindowDefinition *definition,
1018 PicurvWindowStorage *storage, PetscReal weight)
1019{
1020 SpatialTargetPlan plan;
1021 PetscReal ***nvert = NULL, ***particles = NULL;
1022 /* Occupancy of each sample set, indexed by SampleSet. */
1023 PetscReal ***count_arr[2] = {NULL, NULL}, ***weight_arr[2] = {NULL, NULL}, ***weight_sq_arr[2] = {NULL, NULL};
1024 const PetscBool occupied_set = (PetscBool)(storage != NULL && storage->occupied_count != NULL);
1025
1026 PetscFunctionBeginUser;
1027 PROFILE_FUNCTION_BEGIN;
1028 PetscCheck(user != NULL && definition != NULL && storage != NULL,
1029 PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "Context, definition, and storage are required.");
1030 PetscCheck(weight > 0.0, PETSC_COMM_SELF, PETSC_ERR_ARG_OUTOFRANGE,
1031 "Accepted states carry a positive weight, got %g.", (double)weight);
1032 if (definition->field_count == 0) { PROFILE_FUNCTION_END; PetscFunctionReturn(0); }
1033
1034 /* Every requested field shares the pointwise cell-centered domain, so the plan
1035 * is resolved once rather than per field. */
1036 PetscCall(SpatialTargetPlanCreate(user, (FieldId)definition->fields[0].field_id,
1037 PICURV_STATISTICS_MASK_FLUID, &plan));
1038
1039 PetscCall(DMDAVecGetArrayRead(user->da, user->Nvert, &nvert));
1040 for (PetscInt set = 0; set < (occupied_set ? 2 : 1); ++set) {
1041 Vec count = NULL, sum = NULL, sum_sq = NULL;
1042
1043 SampleSetOccupancy(storage, (SampleSet)set, &count, &sum, &sum_sq);
1044 PetscCall(DMDAVecGetArray(user->da, count, &count_arr[set]));
1045 PetscCall(DMDAVecGetArray(user->da, sum, &weight_arr[set]));
1046 PetscCall(DMDAVecGetArray(user->da, sum_sq, &weight_sq_arr[set]));
1047 }
1048 /* A cell is occupied in this state when it holds a particle now; the count is the
1049 * one the scatter just computed for this state. */
1050 if (occupied_set) PetscCall(DMDAVecGetArrayRead(user->da, user->ParticleCount, &particles));
1051
1052 /* Pass one advances per-point occupancy, so every field pass afterwards can
1053 * recover the pre-state weight uniformly as (stored weight - this weight).
1054 * Occupancy is per point and per accepted state, never per field. */
1055 for (PetscInt k = plan.start[2]; k < plan.end[2]; ++k) {
1056 for (PetscInt j = plan.start[1]; j < plan.end[1]; ++j) {
1057 for (PetscInt i = plan.start[0]; i < plan.end[0]; ++i) {
1058 if (!SpatialTargetPlanMaskAllows(&plan, nvert[k][j][i])) continue;
1059 for (PetscInt set = 0; set < (occupied_set ? 2 : 1); ++set) {
1060 if (set == SAMPLE_SET_OCCUPIED && particles[k][j][i] <= 0.0) continue;
1061 count_arr[set][k][j][i] += 1.0;
1062 weight_arr[set][k][j][i] += weight;
1063 weight_sq_arr[set][k][j][i] += weight * weight;
1064 }
1065 }
1066 }
1067 }
1068
1069 /* Pass two updates every cross-field co-moment. It runs before any mean is
1070 * written back, because a co-moment needs the pre-update mean of *both*
1071 * members, and pass three overwrites them field by field. */
1072 for (PetscInt pair = 0; pair < definition->covariance_count; ++pair) {
1073 const SampleSet set = RequestSampleSet(definition, -1, pair);
1074 FieldView view_a, view_b;
1075 PetscScalar ****src_a = NULL, ****mean_a = NULL;
1076 PetscScalar ****src_b = NULL, ****mean_b = NULL;
1077 PetscScalar ****co_moment = NULL;
1078 DM pair_dm = NULL;
1079 PetscInt slot_a = 0, slot_b = 0, dof_a = 0, dof_b = 0, components = 0;
1080
1081 PetscCall(FindFieldSlot(definition, definition->covariances[pair].first, &slot_a));
1082 PetscCall(FindFieldSlot(definition, definition->covariances[pair].second, &slot_b));
1083 PetscCall(FieldGetView(user, (FieldId)definition->covariances[pair].first, &view_a));
1084 PetscCall(FieldGetView(user, (FieldId)definition->covariances[pair].second, &view_b));
1085 dof_a = view_a.descriptor->dof;
1086 dof_b = view_b.descriptor->dof;
1087 PetscCall(PicurvCovarianceComponentCount(dof_a, dof_b, &components));
1088 PetscCall(PicurvStatisticsComponentDM(user, components, &pair_dm));
1089
1090 /* A component-indexed view serves every degree of freedom, so the loop
1091 * below needs no scalar-versus-vector branching. Each member is centred on
1092 * its mean over the pair's own sample set. */
1093 PetscCall(DMDAVecGetArrayDOFRead(view_a.dm, view_a.global_vec, &src_a));
1094 PetscCall(DMDAVecGetArrayDOFRead(view_a.dm, SampleSetMean(storage, set, slot_a), &mean_a));
1095 PetscCall(DMDAVecGetArrayDOFRead(view_b.dm, view_b.global_vec, &src_b));
1096 PetscCall(DMDAVecGetArrayDOFRead(view_b.dm, SampleSetMean(storage, set, slot_b), &mean_b));
1097 PetscCall(DMDAVecGetArrayDOF(pair_dm, storage->cm[pair], &co_moment));
1098
1099 for (PetscInt k = plan.start[2]; k < plan.end[2]; ++k) {
1100 for (PetscInt j = plan.start[1]; j < plan.end[1]; ++j) {
1101 for (PetscInt i = plan.start[0]; i < plan.end[0]; ++i) {
1102 if (!SpatialTargetPlanMaskAllows(&plan, nvert[k][j][i])) continue;
1103 if (set == SAMPLE_SET_OCCUPIED && particles[k][j][i] <= 0.0) continue;
1104 for (PetscInt c = 0; c < components; ++c) {
1105 /* A scalar member contributes its single value against every
1106 * component of a vector member, which is what makes a
1107 * vector-scalar covariance a three-component object. */
1108 const PetscInt component_a = (dof_a == 3) ? c : 0;
1109 const PetscInt component_b = (dof_b == 3) ? c : 0;
1110 PicurvCoMomentState state;
1111
1112 state.count = count_arr[set][k][j][i] - 1.0;
1113 state.weight = weight_arr[set][k][j][i] - weight;
1114 state.weight_sq = weight_sq_arr[set][k][j][i] - weight * weight;
1115 state.mean_x = mean_a[k][j][i][component_a];
1116 state.mean_y = mean_b[k][j][i][component_b];
1117 state.cm = co_moment[k][j][i][c];
1118 PetscCall(PicurvCoMomentStateUpdate(&state,
1119 src_a[k][j][i][component_a],
1120 src_b[k][j][i][component_b], weight));
1121 co_moment[k][j][i][c] = state.cm;
1122 }
1123 }
1124 }
1125 }
1126
1127 PetscCall(DMDAVecRestoreArrayDOF(pair_dm, storage->cm[pair], &co_moment));
1128 PetscCall(DMDAVecRestoreArrayDOFRead(view_b.dm, SampleSetMean(storage, set, slot_b), &mean_b));
1129 PetscCall(DMDAVecRestoreArrayDOFRead(view_b.dm, view_b.global_vec, &src_b));
1130 PetscCall(DMDAVecRestoreArrayDOFRead(view_a.dm, SampleSetMean(storage, set, slot_a), &mean_a));
1131 PetscCall(DMDAVecRestoreArrayDOFRead(view_a.dm, view_a.global_vec, &src_a));
1132 }
1133
1134 /* Pass three updates each requested field's mean and, when asked, its centered
1135 * self-product. A three-vector's self-product is the six symmetric co-moments
1136 * between component pairs, not three per-component variances, so the co-moment
1137 * kernel drives every product component including the diagonal. An ordinary field
1138 * paired with a particle-sampled one also advances its occupied-state mean. */
1139 for (PetscInt field_index = 0; field_index < definition->field_count; ++field_index) {
1140 const SampleSet set = RequestSampleSet(definition, field_index, -1);
1141 const PetscBool second = definition->fields[field_index].want_second;
1142 Vec occupied_mean = (storage->occupied_mean) ? storage->occupied_mean[field_index] : NULL;
1143 FieldView view;
1144 PetscScalar ****src = NULL, ****mean = NULL, ****product = NULL, ****occupied = NULL;
1145 DM product_dm = NULL;
1146 PetscInt dof = 0, components = 0;
1147
1148 PetscCall(FieldGetView(user, (FieldId)definition->fields[field_index].field_id, &view));
1149 dof = view.descriptor->dof;
1150 PetscCall(DMDAVecGetArrayDOFRead(view.dm, view.global_vec, &src));
1151 PetscCall(DMDAVecGetArrayDOF(view.dm, storage->mean[field_index], &mean));
1152 if (occupied_mean) PetscCall(DMDAVecGetArrayDOF(view.dm, occupied_mean, &occupied));
1153 if (second) {
1154 PetscCall(PicurvProductComponentCount(dof, &components));
1155 PetscCall(PicurvStatisticsComponentDM(user, components, &product_dm));
1156 PetscCall(DMDAVecGetArrayDOF(product_dm, storage->m2[field_index], &product));
1157 }
1158
1159 for (PetscInt k = plan.start[2]; k < plan.end[2]; ++k) {
1160 for (PetscInt j = plan.start[1]; j < plan.end[1]; ++j) {
1161 for (PetscInt i = plan.start[0]; i < plan.end[0]; ++i) {
1162 PetscReal prior_weight = 0.0;
1163 PetscReal prior_weight_sq = 0.0;
1164 PetscReal prior_count = 0.0;
1165
1166 if (!SpatialTargetPlanMaskAllows(&plan, nvert[k][j][i])) continue;
1167 if (occupied && particles[k][j][i] > 0.0) {
1168 for (PetscInt c = 0; c < dof; ++c) {
1169 PicurvMomentState moment;
1170
1171 moment.count = count_arr[SAMPLE_SET_OCCUPIED][k][j][i] - 1.0;
1172 moment.weight = weight_arr[SAMPLE_SET_OCCUPIED][k][j][i] - weight;
1173 moment.weight_sq = weight_sq_arr[SAMPLE_SET_OCCUPIED][k][j][i] - weight * weight;
1174 moment.mean = occupied[k][j][i][c];
1175 moment.m2 = 0.0;
1176 PetscCall(PicurvMomentStateUpdate(&moment, src[k][j][i][c], weight));
1177 occupied[k][j][i][c] = moment.mean;
1178 }
1179 }
1180 if (set == SAMPLE_SET_OCCUPIED && particles[k][j][i] <= 0.0) continue;
1181
1182 prior_weight = weight_arr[set][k][j][i] - weight;
1183 prior_weight_sq = weight_sq_arr[set][k][j][i] - weight * weight;
1184 prior_count = count_arr[set][k][j][i] - 1.0;
1185
1186 /* Products are updated before the means are written back,
1187 * because the co-moment kernel needs the pre-update means. */
1188 for (PetscInt c = 0; c < components; ++c) {
1189 const PetscInt a = (dof == 1) ? 0 : kProductFirst[c];
1190 const PetscInt b = (dof == 1) ? 0 : kProductSecond[c];
1191 PicurvCoMomentState state;
1192
1193 state.count = prior_count;
1194 state.weight = prior_weight;
1195 state.weight_sq = prior_weight_sq;
1196 state.mean_x = mean[k][j][i][a];
1197 state.mean_y = mean[k][j][i][b];
1198 state.cm = product[k][j][i][c];
1199 PetscCall(PicurvCoMomentStateUpdate(&state, src[k][j][i][a],
1200 src[k][j][i][b], weight));
1201 product[k][j][i][c] = state.cm;
1202 }
1203
1204 for (PetscInt c = 0; c < dof; ++c) {
1205 PicurvMomentState moment;
1206
1207 moment.count = prior_count;
1208 moment.weight = prior_weight;
1209 moment.weight_sq = prior_weight_sq;
1210 moment.mean = mean[k][j][i][c];
1211 moment.m2 = 0.0;
1212 PetscCall(PicurvMomentStateUpdate(&moment, src[k][j][i][c], weight));
1213 mean[k][j][i][c] = moment.mean;
1214 }
1215 }
1216 }
1217 }
1218
1219 if (second) PetscCall(DMDAVecRestoreArrayDOF(product_dm, storage->m2[field_index], &product));
1220 if (occupied_mean) PetscCall(DMDAVecRestoreArrayDOF(view.dm, occupied_mean, &occupied));
1221 PetscCall(DMDAVecRestoreArrayDOF(view.dm, storage->mean[field_index], &mean));
1222 PetscCall(DMDAVecRestoreArrayDOFRead(view.dm, view.global_vec, &src));
1223 }
1224
1225 if (occupied_set) PetscCall(DMDAVecRestoreArrayRead(user->da, user->ParticleCount, &particles));
1226 for (PetscInt set = (occupied_set ? 1 : 0); set >= 0; --set) {
1227 Vec count = NULL, sum = NULL, sum_sq = NULL;
1228
1229 SampleSetOccupancy(storage, (SampleSet)set, &count, &sum, &sum_sq);
1230 PetscCall(DMDAVecRestoreArray(user->da, sum_sq, &weight_sq_arr[set]));
1231 PetscCall(DMDAVecRestoreArray(user->da, sum, &weight_arr[set]));
1232 PetscCall(DMDAVecRestoreArray(user->da, count, &count_arr[set]));
1233 }
1234 PetscCall(DMDAVecRestoreArrayRead(user->da, user->Nvert, &nvert));
1235 PROFILE_FUNCTION_END;
1236 PetscFunctionReturn(0);
1237}

1241 * @details Local to this translation unit. A self-product walks the symmetric pair
1242 * table; a vector-scalar covariance pairs each vector component with the
1243 * scalar's only one, as `PicurvWindowAccumulate` does.
1244 */

1246 PetscInt first[6], PetscInt second[6])
1247{

1249 first[c] = self ? ((dof_a == 1) ? 0 : kProductFirst[c]) : ((dof_a == 3) ? c : 0);
1250 second[c] = self ? ((dof_a == 1) ? 0 : kProductSecond[c]) : ((dof_b == 3) ? c : 0);
1251 }

1255#define __FUNCT__ "AccumulateParticleCentredProducts"
1256/**
1257 * @brief Internal helper: adds every particle's centred product into its cell.
1258 * @details Local to this translation unit. The second pass of the two-pass ensemble:
1259 * each particle contributes `(a - mean_a)(b - mean_b)` for every component pair,
1260 * against its own cell's mean read from the ghosted copies. Sums go to the
1261 * ghosted patch and are then added onto their owners into @p product, the way
1262 * `AccumulateParticleField` sums go.
1263 */
1264static PetscErrorCode AccumulateParticleCentredProducts(UserCtx *user, ParticleFieldId field_a,
1265 ParticleFieldId field_b, DM dm_a, Vec mean_a,
1266 DM dm_b, Vec mean_b, PetscInt components,
1267 const PetscInt first[6], const PetscInt second[6],
1268 DM product_dm, Vec product)
1269{
1270 const ParticleFieldDescriptor *descriptor_a = NULL, *descriptor_b = NULL;
1271 const PetscReal *values_a = NULL, *values_b = NULL;
1272 const PetscInt *cell = NULL;
1273 PetscScalar ****centre_a = NULL, ****centre_b = NULL, ****sum = NULL;
1274 PetscInt nlocal = 0, gxs, gys, gzs, gxm, gym, gzm;
1275 Vec local_sum = NULL;
1276
1277 PetscFunctionBeginUser;
1278 PROFILE_FUNCTION_BEGIN;
1279 PetscCall(ParticleFieldGetDescriptor(field_a, &descriptor_a));
1280 PetscCall(ParticleFieldGetDescriptor(field_b, &descriptor_b));
1281 PetscCall(DMDAGetGhostCorners(product_dm, &gxs, &gys, &gzs, &gxm, &gym, &gzm));
1282 PetscCall(DMGetLocalVector(product_dm, &local_sum));
1283 PetscCall(VecZeroEntries(local_sum));
1284
1285 PetscCall(DMSwarmGetLocalSize(user->swarm, &nlocal));
1286 PetscCall(DMSwarmGetField(user->swarm, descriptor_a->canonical_name, NULL, NULL, (void **)&values_a));
1287 /* A self-product reads one field twice, and a swarm field cannot be held twice. */
1288 if (field_b == field_a) values_b = values_a;
1289 else PetscCall(DMSwarmGetField(user->swarm, descriptor_b->canonical_name, NULL, NULL, (void **)&values_b));
1290 PetscCall(DMSwarmGetField(user->swarm, ParticleFieldName(PARTICLE_FIELD_ID_CELL_ID), NULL, NULL, (void **)&cell));
1291 PetscCall(DMDAVecGetArrayDOFRead(dm_a, mean_a, &centre_a));
1292 PetscCall(DMDAVecGetArrayDOFRead(dm_b, mean_b, &centre_b));
1293 PetscCall(DMDAVecGetArrayDOF(product_dm, local_sum, &sum));
1294
1295 for (PetscInt p = 0; p < nlocal; ++p) {
1296 /* Cell indices are geometric; storage is shifted by the boundary layer. */
1297 const PetscInt i = cell[3 * p + 0] + 1, j = cell[3 * p + 1] + 1, k = cell[3 * p + 2] + 1;
1298 const PetscReal *a = &values_a[p * descriptor_a->components];
1299 const PetscReal *b = &values_b[p * descriptor_b->components];
1300
1301 if (i < gxs || i >= gxs + gxm || j < gys || j >= gys + gym || k < gzs || k >= gzs + gzm) continue;
1302 for (PetscInt c = 0; c < components; ++c) {
1303 sum[k][j][i][c] += (a[first[c]] - centre_a[k][j][i][first[c]]) *
1304 (b[second[c]] - centre_b[k][j][i][second[c]]);
1305 }
1306 }
1307
1308 PetscCall(DMDAVecRestoreArrayDOF(product_dm, local_sum, &sum));
1309 PetscCall(DMDAVecRestoreArrayDOFRead(dm_b, mean_b, &centre_b));
1310 PetscCall(DMDAVecRestoreArrayDOFRead(dm_a, mean_a, &centre_a));
1311 PetscCall(DMSwarmRestoreField(user->swarm, ParticleFieldName(PARTICLE_FIELD_ID_CELL_ID), NULL, NULL, (void **)&cell));
1312 if (field_b != field_a) PetscCall(DMSwarmRestoreField(user->swarm, descriptor_b->canonical_name, NULL, NULL, (void **)&values_b));
1313 PetscCall(DMSwarmRestoreField(user->swarm, descriptor_a->canonical_name, NULL, NULL, (void **)&values_a));
1314
1315 PetscCall(DMLocalToGlobalBegin(product_dm, local_sum, ADD_VALUES, product));
1316 PetscCall(DMLocalToGlobalEnd(product_dm, local_sum, ADD_VALUES, product));
1317 PetscCall(DMRestoreLocalVector(product_dm, &local_sum));
1318 PROFILE_FUNCTION_END;
1319 PetscFunctionReturn(0);
1320}

1324/**
1325 * @brief Implementation of \ref PicurvWindowStorageFromParticles().
1326 * @see PicurvWindowStorageFromParticles()
1327 */
1328PetscErrorCode PicurvWindowStorageFromParticles(UserCtx *user, const PicurvWindowDefinition *definition,
1329 PicurvWindowStorage *storage)
1330{
1331 ParticleFieldId particle[PICURV_WINDOW_MAX_REQUESTS];
1332 FieldView view[PICURV_WINDOW_MAX_REQUESTS];
1333 Vec centre[PICURV_WINDOW_MAX_REQUESTS] = {NULL};
1334
1335 PetscFunctionBeginUser;
1336 PROFILE_FUNCTION_BEGIN;
1337 PetscCheck(user != NULL && definition != NULL && storage != NULL && user->swarm != NULL,
1338 PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "Context, swarm, definition, and storage are required.");
1339
1340 /* Unit weights: count, weight and squared weight are all the particle count. Every
1341 * sample is a particle, so the occupied set is the whole sample set. */
1342 PetscCall(CalculateParticleCountPerCell(user));
1343 for (PetscInt set = 0; set < (storage->occupied_count ? 2 : 1); ++set) {
1344 Vec count = NULL, weight = NULL, weight_sq = NULL;
1345
1346 SampleSetOccupancy(storage, (SampleSet)set, &count, &weight, &weight_sq);
1347 PetscCall(VecCopy(user->ParticleCount, count));
1348 PetscCall(VecCopy(user->ParticleCount, weight));
1349 PetscCall(VecCopy(user->ParticleCount, weight_sq));
1350 }
1351
1352 /* Pass one: each field's cell mean, the particle sum divided by the count. */
1353 for (PetscInt f = 0; f < definition->field_count; ++f) {
1354 Vec local_sum = NULL, sum = NULL;
1355
1356 PetscCall(ParticleFieldIdFromScatterTarget((FieldId)definition->fields[f].field_id, &particle[f]));
1357 PetscCall(FieldGetView(user, (FieldId)definition->fields[f].field_id, &view[f]));
1358 PetscCall(DMGetLocalVector(view[f].dm, &local_sum));
1359 PetscCall(VecZeroEntries(local_sum));
1360 PetscCall(AccumulateParticleField(user->swarm, particle[f], view[f].dm, local_sum));
1361 PetscCall(VecDuplicate(storage->mean[f], &sum));
1362 PetscCall(VecZeroEntries(sum));
1363 PetscCall(DMLocalToGlobalBegin(view[f].dm, local_sum, ADD_VALUES, sum));
1364 PetscCall(DMLocalToGlobalEnd(view[f].dm, local_sum, ADD_VALUES, sum));
1365 PetscCall(DMRestoreLocalVector(view[f].dm, &local_sum));
1366 PetscCall(NormalizeGridVectorByCount(user->da, storage->count, view[f].dm, sum, storage->mean[f]));
1367 PetscCall(VecDestroy(&sum));
1368 if (storage->occupied_mean && storage->occupied_mean[f]) {
1369 PetscCall(VecCopy(storage->mean[f], storage->occupied_mean[f]));
1370 }
1371 /* The second pass reads each particle's own cell mean, which may be a ghost. */
1372 PetscCall(DMCreateLocalVector(view[f].dm, &centre[f]));
1373 PetscCall(DMGlobalToLocalBegin(view[f].dm, storage->mean[f], INSERT_VALUES, centre[f]));
1374 PetscCall(DMGlobalToLocalEnd(view[f].dm, storage->mean[f], INSERT_VALUES, centre[f]));
1375 }
1376
1377 /* Pass two: centred self-products, then cross-field covariances. */
1378 for (PetscInt f = 0; f < definition->field_count; ++f) {
1379 const PetscInt dof = view[f].descriptor->dof;
1380 PetscInt components = 0, first[6], second[6];
1381 DM product_dm = NULL;
1382
1383 if (!definition->fields[f].want_second) continue;
1384 PetscCall(PicurvProductComponentCount(dof, &components));
1385 PetscCall(PicurvStatisticsComponentDM(user, components, &product_dm));
1386 ProductComponents(PETSC_TRUE, dof, dof, components, first, second);
1387 PetscCall(AccumulateParticleCentredProducts(user, particle[f], particle[f], view[f].dm, centre[f],
1388 view[f].dm, centre[f], components, first, second,
1389 product_dm, storage->m2[f]));
1390 }
1391 for (PetscInt pair = 0; pair < definition->covariance_count; ++pair) {
1392 PetscInt a = 0, b = 0, components = 0, first[6], second[6];
1393 DM pair_dm = NULL;
1394
1395 PetscCall(FindFieldSlot(definition, definition->covariances[pair].first, &a));
1396 PetscCall(FindFieldSlot(definition, definition->covariances[pair].second, &b));
1397 PetscCall(PicurvCovarianceComponentCount(view[a].descriptor->dof, view[b].descriptor->dof, &components));
1398 PetscCall(PicurvStatisticsComponentDM(user, components, &pair_dm));
1399 ProductComponents(PETSC_FALSE, view[a].descriptor->dof, view[b].descriptor->dof, components, first, second);
1400 PetscCall(AccumulateParticleCentredProducts(user, particle[a], particle[b], view[a].dm, centre[a],
1401 view[b].dm, centre[b], components, first, second,
1402 pair_dm, storage->cm[pair]));
1403 }
1404
1405 for (PetscInt f = 0; f < definition->field_count; ++f) PetscCall(VecDestroy(&centre[f]));
1406 PROFILE_FUNCTION_END;
1407 PetscFunctionReturn(0);
1408}

1411 * @brief Internal helper: merges one point-mean field into its slot means over a sample set.
1412 * @details Local to this translation unit. The slot mean is the weighted average of the
1413 * point means, written to every point; a point that set never sampled is
1414 * excluded, never averaged in as a zero.
1415 */
1416static PetscErrorCode ReduceSampleSetMean(UserCtx *user, const SpatialTargetPlan *plan, DM dm, PetscInt dof,
1417 Vec point_mean, Vec weight, const PetscBool average_direction[3],
1418 Vec slot_mean)
1419{

1421 PetscReal ***w = NULL;
1422 Vec scratch = NULL;
1423
1424 PetscFunctionBeginUser;
1425 PetscCall(VecDuplicate(point_mean, &scratch));
1426 PetscCall(DMDAVecGetArrayRead(user->da, weight, &w));
1427 PetscCall(DMDAVecGetArrayDOFRead(dm, point_mean, &mean));

1429 for (PetscInt k = plan->start[2]; k < plan->end[2]; ++k)
1430 for (PetscInt j = plan->start[1]; j < plan->end[1]; ++j)

1445/**
1446 * @brief Implementation of \ref PicurvWindowStorageReduce().

1449PetscErrorCode PicurvWindowStorageReduce(UserCtx *user, const PicurvWindowDefinition *definition,
1450 const PicurvWindowStorage *storage,
1451 const PetscBool average_direction[3],
1452 PicurvWindowStorage *reduced)
1453{
1454 SpatialTargetPlan plan;
1455 const PetscInt set_count = (storage && storage->occupied_count) ? 2 : 1;
1456
1457 PetscFunctionBeginUser;
1458 PROFILE_FUNCTION_BEGIN;
1459 PetscCheck(user != NULL && definition != NULL && storage != NULL && reduced != NULL && average_direction != NULL,
1460 PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "Context, definition, and both storages are required.");

1463 PICURV_STATISTICS_MASK_FLUID, &plan));
1464
1465 /* Slot means, each over its field's own sample set, then the occupied-state means
1466 * of ordinary fields that an occupied pair centres on. */
1467 for (PetscInt f = 0; f < definition->field_count; ++f) {

1469 Vec count = NULL, weight = NULL, weight_sq = NULL;
1470
1471 PetscCall(FieldGetView(user, (FieldId)definition->fields[f].field_id, &view));
1472 SampleSetOccupancy(storage, RequestSampleSet(definition, f, -1), &count, &weight, &weight_sq);
1473 PetscCall(ReduceSampleSetMean(user, &plan, view.dm, view.descriptor->dof, storage->mean[f], weight,
1474 average_direction, reduced->mean[f]));
1475 if (storage->occupied_mean && storage->occupied_mean[f]) {
1476 PetscCall(ReduceSampleSetMean(user, &plan, view.dm, view.descriptor->dof, storage->occupied_mean[f],
1477 storage->occupied_weight, average_direction, reduced->occupied_mean[f]));
1478 }
1479 }
1480
1481 /* Slot co-moments: every point's own co-moment plus the spread of its means about
1482 * the slot means, divided by the slot weight, all over the request's sample set.
1483 * Self-products, then cross pairs. */
1484 for (PetscInt index = 0; index < storage->field_count + storage->covariance_count; ++index) {
1485 const PetscBool self = (PetscBool)(index < storage->field_count);
1486 const PetscInt pair = index - storage->field_count;
1487 const SampleSet set = self ? RequestSampleSet(definition, index, -1) : RequestSampleSet(definition, -1, pair);
1488 PetscInt a = index, b = index, components = 0, first[6], second[6];
1489 FieldView view_a, view_b;
1490 PetscScalar ****point_a = NULL, ****point_b = NULL, ****slot_a = NULL, ****slot_b = NULL;
1491 PetscScalar ****moment = NULL, ****total = NULL;
1492 PetscReal ***weight = NULL;
1493 Vec source = NULL, target = NULL, scratch = NULL, count_vec = NULL, weight_vec = NULL, weight_sq_vec = NULL;
1494 DM product_dm = NULL;
1495
1496 if (self) {
1497 if (!storage->m2[index]) continue;
1498 source = storage->m2[index];
1499 target = reduced->m2[index];
1500 } else {
1501 PetscCall(FindFieldSlot(definition, definition->covariances[pair].first, &a));
1502 PetscCall(FindFieldSlot(definition, definition->covariances[pair].second, &b));
1503 source = storage->cm[pair];
1504 target = reduced->cm[pair];
1505 }
1506 SampleSetOccupancy(storage, set, &count_vec, &weight_vec, &weight_sq_vec);
1507 PetscCall(FieldGetView(user, (FieldId)definition->fields[a].field_id, &view_a));
1508 PetscCall(FieldGetView(user, (FieldId)definition->fields[b].field_id, &view_b));
1509 PetscCall(VecGetDM(source, &product_dm));
1510 PetscCall(DMDAGetInfo(product_dm, NULL, NULL, NULL, NULL, NULL, NULL, NULL, &components,
1511 NULL, NULL, NULL, NULL, NULL));
1512 ProductComponents(self, view_a.descriptor->dof, view_b.descriptor->dof, components, first, second);
1513
1514 PetscCall(VecDuplicate(source, &scratch));
1515 PetscCall(DMDAVecGetArrayRead(user->da, weight_vec, &weight));
1516 PetscCall(DMDAVecGetArrayDOFRead(view_a.dm, SampleSetMean(storage, set, a), &point_a));
1517 PetscCall(DMDAVecGetArrayDOFRead(view_b.dm, SampleSetMean(storage, set, b), &point_b));
1518 PetscCall(DMDAVecGetArrayDOFRead(view_a.dm, SampleSetMean(reduced, set, a), &slot_a));
1519 PetscCall(DMDAVecGetArrayDOFRead(view_b.dm, SampleSetMean(reduced, set, b), &slot_b));
1520 PetscCall(DMDAVecGetArrayDOFRead(product_dm, source, &moment));
1521 PetscCall(DMDAVecGetArrayDOF(product_dm, scratch, &total));
1522 for (PetscInt k = plan.start[2]; k < plan.end[2]; ++k)
1523 for (PetscInt j = plan.start[1]; j < plan.end[1]; ++j)
1524 for (PetscInt i = plan.start[0]; i < plan.end[0]; ++i) {
1525 for (PetscInt c = 0; c < components; ++c) {
1526 total[k][j][i][c] = moment[k][j][i][c] + weight[k][j][i] *
1527 (point_a[k][j][i][first[c]] - slot_a[k][j][i][first[c]]) *
1528 (point_b[k][j][i][second[c]] - slot_b[k][j][i][second[c]]);
1529 }
1530 }
1531 PetscCall(DMDAVecRestoreArrayDOF(product_dm, scratch, &total));
1532 PetscCall(DMDAVecRestoreArrayDOFRead(product_dm, source, &moment));
1533 PetscCall(DMDAVecRestoreArrayDOFRead(view_b.dm, SampleSetMean(reduced, set, b), &slot_b));
1534 PetscCall(DMDAVecRestoreArrayDOFRead(view_a.dm, SampleSetMean(reduced, set, a), &slot_a));
1535 PetscCall(DMDAVecRestoreArrayDOFRead(view_b.dm, SampleSetMean(storage, set, b), &point_b));
1536 PetscCall(DMDAVecRestoreArrayDOFRead(view_a.dm, SampleSetMean(storage, set, a), &point_a));
1537 PetscCall(DMDAVecRestoreArrayRead(user->da, weight_vec, &weight));
1538 PetscCall(PicurvSpatialRatioAverage(user, &plan, scratch, weight_vec, weight_vec,
1539 average_direction, PETSC_COMM_WORLD, target, NULL));
1540 PetscCall(VecDestroy(&scratch));
1541 }
1542
1543 /* Unit weight wherever a set's slot was sampled, so derivation reads the stored
1544 * values as they are. The averaged weight is positive exactly there. */
1545 for (PetscInt set = 0; set < set_count; ++set) {
1546 Vec count = NULL, weight = NULL, weight_sq = NULL, merged_count = NULL, merged = NULL, merged_sq = NULL;
1547 PetscReal ***occupied = NULL;
1548
1549 SampleSetOccupancy(storage, (SampleSet)set, &count, &weight, &weight_sq);
1550 SampleSetOccupancy(reduced, (SampleSet)set, &merged_count, &merged, &merged_sq);
1551 PetscCall(PicurvSpatialRatioAverage(user, &plan, weight, NULL, weight, average_direction,
1552 PETSC_COMM_WORLD, merged, NULL));
1553 PetscCall(DMDAVecGetArray(user->da, merged, &occupied));
1554 for (PetscInt k = plan.start[2]; k < plan.end[2]; ++k)
1555 for (PetscInt j = plan.start[1]; j < plan.end[1]; ++j)
1556 for (PetscInt i = plan.start[0]; i < plan.end[0]; ++i) {
1557 occupied[k][j][i] = (occupied[k][j][i] > 0.0) ? 1.0 : 0.0;
1558 }
1559 PetscCall(DMDAVecRestoreArray(user->da, merged, &occupied));
1560 PetscCall(VecCopy(merged, merged_count));
1561 PetscCall(VecCopy(merged, merged_sq));
1562 }
1563 PROFILE_FUNCTION_END;
1564 PetscFunctionReturn(0);
1565}
