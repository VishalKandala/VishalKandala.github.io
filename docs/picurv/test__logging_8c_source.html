16 * @brief Asserts that one text file contains a required substring.

18
19static PetscErrorCode AssertFileContains(const char *path, const char *needle, const char *context)
20{
21 FILE *fp = NULL;
22 long file_size = 0;
23 char *buffer = NULL;
24
25 PetscFunctionBeginUser;
26 fp = fopen(path, "rb");
27 if (!fp) {
28 SETERRQ(PETSC_COMM_SELF, PETSC_ERR_FILE_OPEN, "Failed to open '%s' for assertion.", path);
29 }
30 if (fseek(fp, 0, SEEK_END) != 0) {
31 fclose(fp);
32 SETERRQ(PETSC_COMM_SELF, PETSC_ERR_FILE_OPEN, "Failed to seek '%s'.", path);
33 }
34 file_size = ftell(fp);
35 if (file_size < 0) {
36 fclose(fp);
37 SETERRQ(PETSC_COMM_SELF, PETSC_ERR_FILE_OPEN, "Failed to measure '%s'.", path);
38 }
39 if (fseek(fp, 0, SEEK_SET) != 0) {
40 fclose(fp);
41 SETERRQ(PETSC_COMM_SELF, PETSC_ERR_FILE_OPEN, "Failed to rewind '%s'.", path);
42 }
43
44 PetscCall(PetscMalloc1((size_t)file_size + 1, &buffer));
45 if (file_size > 0) {
46 size_t bytes_read = fread(buffer, 1, (size_t)file_size, fp);
47 if (bytes_read != (size_t)file_size) {
48 fclose(fp);
49 PetscCall(PetscFree(buffer));
50 SETERRQ(PETSC_COMM_SELF, PETSC_ERR_FILE_OPEN, "Failed to read '%s'.", path);
51 }
52 }
53 buffer[file_size] = '\0';
54 fclose(fp);
55
56 PetscCall(PicurvAssertBool((PetscBool)(strstr(buffer, needle) != NULL), context));
57 PetscCall(PetscFree(buffer));
58 PetscFunctionReturn(0);
59}

113/**
114 * @brief Captures stdout emitted by one logging helper into a temporary file-backed buffer.
115 */
116static PetscErrorCode CaptureLoggingOutput(UserCtx *user,
117 SimCtx *simCtx,
118 CapturedLoggingFn fn,
119 void *ctx,
120 char *captured,
121 size_t captured_len)
122{
123 char tmpdir[PETSC_MAX_PATH_LEN];
124 char capture_path[PETSC_MAX_PATH_LEN];
125 FILE *capture_file = NULL;
126 int saved_stdout = -1;
127 int capture_fd = -1;
128 size_t bytes_read = 0;
129 PetscErrorCode ierr;
130
131 PetscFunctionBeginUser;
132 PetscCheck(fn != NULL, PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "Capture callback cannot be NULL.");
133 PetscCheck(captured != NULL, PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "Capture buffer cannot be NULL.");
134 PetscCheck(captured_len > 0, PETSC_COMM_SELF, PETSC_ERR_ARG_SIZ, "Capture buffer must be non-empty.");
135
136 PetscCall(PicurvMakeTempDir(tmpdir, sizeof(tmpdir)));
137 PetscCall(PetscSNPrintf(capture_path, sizeof(capture_path), "%s/logging.out", tmpdir));
138
139 fflush(stdout);
140 saved_stdout = dup(STDOUT_FILENO);
141 PetscCheck(saved_stdout >= 0, PETSC_COMM_SELF, PETSC_ERR_SYS, "dup(STDOUT_FILENO) failed.");
142 capture_fd = open(capture_path, O_CREAT | O_TRUNC | O_WRONLY, 0600);
143 PetscCheck(capture_fd >= 0, PETSC_COMM_SELF, PETSC_ERR_FILE_OPEN, "Failed to open capture file '%s'.", capture_path);
144 PetscCheck(dup2(capture_fd, STDOUT_FILENO) >= 0, PETSC_COMM_SELF, PETSC_ERR_SYS, "dup2() failed while redirecting stdout.");
145 close(capture_fd);
146 capture_fd = -1;
147
148 ierr = fn(user, simCtx, ctx);
149 fflush(stdout);
150 PetscCheck(dup2(saved_stdout, STDOUT_FILENO) >= 0, PETSC_COMM_SELF, PETSC_ERR_SYS, "dup2() failed while restoring stdout.");
151 close(saved_stdout);
152 saved_stdout = -1;
153 PetscCall(ierr);
154
155 capture_file = fopen(capture_path, "r");
156 PetscCheck(capture_file != NULL, PETSC_COMM_SELF, PETSC_ERR_FILE_OPEN, "Failed to read capture file '%s'.", capture_path);
157 bytes_read = fread(captured, 1, captured_len - 1, capture_file);
158 captured[bytes_read] = '\0';
159 fclose(capture_file);
160 PetscCall(PicurvRemoveTempDir(tmpdir));
161 PetscFunctionReturn(0);
162}
