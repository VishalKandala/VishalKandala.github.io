84/**
85 * @brief The owned cells a seeding mode places particles in, and which particles each gets.
86 * @details Volumetric seeding uses the rank's owned cells and surface seeding the owned cells
87 * that touch the inlet face, so both are a box of cells from `origin` with `n` cells
88 * per axis (none when the rank owns no part of the target). Cell `c`, in i-fastest
89 * order, seeds the particles with IDs `pid_lo[c]` to `pid_hi[c] - 1`.
90 */
91typedef struct {
92 PetscInt face; /**< -1 for the volume, otherwise the BCFace the points lie on */
93 PetscInt origin[3]; /**< Origin node of the box's first cell */
94 PetscInt n[3]; /**< Cells per axis */
95 PetscInt64 *pid_lo; /**< First particle ID of each cell */
96 PetscInt64 *pid_hi; /**< One past the last particle ID of each cell */
97 PetscReal *bound; /**< Largest measure density in each cell (NULL when not built) */
98 PetscInt count; /**< Particles this rank seeds */
99} SeedingCellTable;
100
101/* A seeding point's logical coordinates stay at or below 1 - SEEDING_LOGIC_MARGIN, for
102 MetricLogicalToPhysical(); a point on a face lies SEEDING_FACE_OFFSET of the cell inside it. */
103#define SEEDING_LOGIC_MARGIN 1.0e-7
104#define SEEDING_FACE_OFFSET 1.0e-6
105
106/** @brief Origin node of cell `c` of a seeding table, cells numbered i fastest. */
107static inline void SeedingCellOrigin(const SeedingCellTable *table, PetscInt c, PetscInt origin[3])
108{
109 origin[0] = table->origin[0] + c % table->n[0];
110 origin[1] = table->origin[1] + (c / table->n[0]) % table->n[1];
111 origin[2] = table->origin[2] + c / (table->n[0] * table->n[1]);
112}

114/**
115 * @brief Logical seeding coordinates from `u`: each clamped to [0, 1 - SEEDING_LOGIC_MARGIN], and
116 * for `face` >= 0 the normal one replaced by the point just inside that face.
117 */
118static inline void SeedingLogicalPoint(const PetscReal u[3], PetscInt face, PetscReal logic[3])
119{
120 for (PetscInt d = 0; d < 3; ++d) logic[d] = PetscMax(PetscMin(u[d], 1.0 - SEEDING_LOGIC_MARGIN), 0.0);
121 if (face >= 0) logic[face / 2] = (face % 2) ? 1.0 - SEEDING_FACE_OFFSET : SEEDING_FACE_OFFSET;
122}

124/**
125 * @brief Measure density at a logical point: the Jacobian determinant for the volume, or the
126 * area element |dx/ds x dx/dt| across the face for a face, with the normal coordinate
127 * on the face itself.
128 */
129static PetscErrorCode SeedingMeasureDensity(UserCtx *user, const Cmpnts ***coor, PetscInt face,
130 PetscInt i, PetscInt j, PetscInt k, const PetscReal logic[3],
131 PetscReal *density)
132{
133 PetscReal J[3][3], detJ, l[3] = {logic[0], logic[1], logic[2]};
134
135 PetscFunctionBeginUser;
136 PROFILE_HELPER_BEGIN;
137 if (face < 0) {
138 PetscCall(MetricJacobian(user, coor, i, j, k, l[0], l[1], l[2], J, &detJ));
139 *density = detJ;
140 } else {
141 const PetscInt a = face / 2, b = (a + 1) % 3, c = (a + 2) % 3;
142 Cmpnts cross;
143 l[a] = (face % 2) ? 1.0 : 0.0;
144 PetscCall(MetricJacobian(user, coor, i, j, k, l[0], l[1], l[2], J, NULL));
145 cross.x = J[1][b] * J[2][c] - J[2][b] * J[1][c];
146 cross.y = J[2][b] * J[0][c] - J[0][b] * J[2][c];
147 cross.z = J[0][b] * J[1][c] - J[1][b] * J[0][c];
148 *density = PetscSqrtReal(cross.x * cross.x + cross.y * cross.y + cross.z * cross.z);
149 }
150 PROFILE_HELPER_END;
151 PetscFunctionReturn(0);
152}

154/**
155 * @brief Decide which particles each owned cell seeds, the same way at every rank layout.
156 * @details Collective. The cells are those volumetric (`face` < 0) or surface seeding uses,
157 * and each cell's measure is its volume or its area on the face, integrated with 2x2x2
158 * Gauss points: exact for the volume (the determinant has degree 2 in each
159 * coordinate) and for the area of a planar face, where the points coincide in pairs
160 * and give the 2x2 rule.
161 *
162 * Particles are stratified over the measure in global cell order (i fastest): with
163 * P(c) the measure of all cells before cell c, T the total and N the particle count,
164 * cell c seeds the IDs floor(N P(c) / T) to floor(N (P(c) + m_c) / T) - 1, so each
165 * cell holds its expected share to within one, the IDs are 0..N-1, and both depend
166 * only on the grid. For the result to be identical at any rank layout the measures
167 * are summed as integers, each cell's measure quantised against the largest cell
168 * (an exact maximum) to 2^Q units, with Q chosen so the total fits 63 bits. P(c) is
169 * the sum of the (j,k) rows before the cell's row, from one reduction over the rows,
170 * plus the cells before it in its row, from a scan over the ranks sharing the row.
171 * @param[in] user Grid context.
172 * @param[in] coor Ghosted node-coordinate array.
173 * @param[in] face -1 for the volume, otherwise the BCFace to seed on.
174 * @param[in] numParticles Global particle count.
175 * @param[in] build_bound PETSC_TRUE to also tabulate each cell's largest measure density on
176 * a 3x3x3 lattice, the envelope DrawSeedingPoint samples against.
177 * @param[out] table The owned cells and their particles; free with FreeSeedingCellTable.
178 * @return PetscErrorCode 0 on success; an error for a cell whose measure is not positive.
179 */
180static PetscErrorCode BuildSeedingCellTable(UserCtx *user, const Cmpnts ***coor, PetscInt face,
181 PetscInt numParticles, PetscBool build_bound,
182 SeedingCellTable *table)
183{
184 const PetscReal gauss[2] = {0.5 - 0.5 / PetscSqrtReal(3.0), 0.5 + 0.5 / PetscSqrtReal(3.0)};
185 const PetscInt nodes[3] = {user->info.mx, user->info.my, user->info.mz};
186 PetscInt start, n_cells, n_rows_local, global_cells[3], global_origin[3] = {0, 0, 0}, quantum_bits = 62;
187 PetscInt64 n_global_cells;
188 PetscReal local_max = 0.0, global_max = 0.0;
189 PetscReal *measure = NULL;
190 uint64_t *rows = NULL, *row_before = NULL, *segment = NULL, *segment_before = NULL, *quanta = NULL;
191 uint64_t total = 0;
192 MPI_Comm row_comm = MPI_COMM_NULL;
193 PetscMPIInt row_rank = 0;
194
195 PetscFunctionBeginUser;
196 PROFILE_FUNCTION_BEGIN;
197 PetscCall(PetscMemzero(table, sizeof(*table)));
198 table->face = face;
199 table->origin[0] = user->info.xs; table->origin[1] = user->info.ys; table->origin[2] = user->info.zs;
200 PetscCall(GetOwnedCellRange(&user->info, 0, &start, &table->n[0]));
201 PetscCall(GetOwnedCellRange(&user->info, 1, &start, &table->n[1]));
202 PetscCall(GetOwnedCellRange(&user->info, 2, &start, &table->n[2]));
203 for (PetscInt d = 0; d < 3; ++d) global_cells[d] = nodes[d] - 2;
204 if (face >= 0) {
205 const PetscInt a = face / 2;
206 PetscBool holds_face = PETSC_FALSE;
207 PetscCall(CanRankServiceFace(&user->info, user->IM, user->JM, user->KM, (BCFace)face, &holds_face));
208 global_cells[a] = 1;
209 if (face % 2) global_origin[a] = nodes[a] - 3; /* origin node of the last cell */
210 if (holds_face) {
211 table->n[a] = 1;
212 table->origin[a] = global_origin[a];
213 } else {
214 table->n[0] = table->n[1] = table->n[2] = 0;
215 }
216 }
217 n_cells = table->n[0] * table->n[1] * table->n[2];
218 n_rows_local = table->n[1] * table->n[2];
219 n_global_cells = (PetscInt64)global_cells[0] * global_cells[1] * global_cells[2];
220 while (quantum_bits > 0 && ((PetscInt64)1 << (62 - quantum_bits)) < n_global_cells) --quantum_bits;
221
222 // --- 1. Cell measures, and the density envelope ---
223 PetscCall(PetscMalloc2(n_cells, &measure, n_cells, &quanta));
224 PetscCall(PetscMalloc2(n_cells, &table->pid_lo, n_cells, &table->pid_hi));
225 if (build_bound) PetscCall(PetscMalloc1(n_cells, &table->bound));
226 for (PetscInt c = 0; c < n_cells; ++c) {
227 PetscInt origin[3];
228 PetscReal density;
229 SeedingCellOrigin(table, c, origin);
230 const PetscInt i = origin[0], j = origin[1], k = origin[2];
231 measure[c] = 0.0;
232 for (PetscInt g = 0; g < 8; ++g) {
233 const PetscReal l[3] = {gauss[g & 1], gauss[(g >> 1) & 1], gauss[(g >> 2) & 1]};
234 PetscCall(SeedingMeasureDensity(user, coor, face, i, j, k, l, &density));
235 measure[c] += 0.125 * density;
236 }
237 PetscCheck(measure[c] > 0.0, PETSC_COMM_SELF, PETSC_ERR_ARG_WRONGSTATE,
238 "Cell with origin node (%d,%d,%d) has %s %g; particle seeding needs positive %s.",
239 (int)i, (int)j, (int)k, face < 0 ? "volume" : "inlet area", (double)measure[c],
240 face < 0 ? "volumes" : "areas");
241 local_max = PetscMax(local_max, measure[c]);
242 if (build_bound) {
243 table->bound[c] = 0.0;
244 for (PetscInt q = 0; q < 27; ++q) {
245 const PetscReal l[3] = {0.5 * (q % 3), 0.5 * ((q / 3) % 3), 0.5 * (q / 9)};
246 PetscCall(SeedingMeasureDensity(user, coor, face, i, j, k, l, &density));
247 table->bound[c] = PetscMax(table->bound[c], density);
248 }
249 }
250 }
251 {
252 PetscInt64 local_cells = n_cells, owned_cells = 0;
253 PetscCallMPI(MPI_Allreduce(&local_cells, &owned_cells, 1, MPIU_INT64, MPI_SUM, PETSC_COMM_WORLD));
254 PetscCheck(owned_cells == n_global_cells, PETSC_COMM_WORLD, PETSC_ERR_PLIB,
255 "Seeding cells owned across ranks (%" PetscInt64_FMT ") differ from the global count (%" PetscInt64_FMT ").",
256 owned_cells, n_global_cells);
257 }
258 PetscCallMPI(MPI_Allreduce(&local_max, &global_max, 1, MPIU_REAL, MPI_MAX, PETSC_COMM_WORLD));
259 PetscCheck(global_max > 0.0, PETSC_COMM_WORLD, PETSC_ERR_ARG_WRONGSTATE,
260 face < 0 ? "Volumetric particle seeding found no owned volume on any rank."
261 : "Surface particle seeding found no inlet area on any rank; it needs an INLET face.");
262
263 if (numParticles == 0) {
264 // Nothing to place (an inlet's table): every cell's range is empty. numParticles is the
265 // same on every rank, so all of them skip the reductions below.
266 PetscCall(PetscArrayzero(table->pid_lo, n_cells));
267 PetscCall(PetscArrayzero(table->pid_hi, n_cells));
268 PetscCall(PetscFree2(measure, quanta));
269 PROFILE_FUNCTION_END;
270 PetscFunctionReturn(0);
271 }
272
273 // --- 2. Integer measures, summed over rows and over each rank's segment of a row ---
274 const PetscInt n_rows = global_cells[1] * global_cells[2];
275 PetscCall(PetscCalloc4(n_rows, &rows, n_rows, &row_before, n_rows_local, &segment, n_rows_local, &segment_before));
276 for (PetscInt c = 0; c < n_cells; ++c) {
277 const PetscInt lj = (c / table->n[0]) % table->n[1], lk = c / (table->n[0] * table->n[1]);
278 const PetscInt row = (table->origin[1] + lj - global_origin[1]) + global_cells[1] * (table->origin[2] + lk - global_origin[2]);
279 quanta[c] = (uint64_t)PetscMax(1.0, PetscFloorReal(ldexp((double)(measure[c] / global_max), (int)quantum_bits) + 0.5));
280 rows[row] += quanta[c];
281 segment[lj + table->n[1] * lk] += quanta[c];
282 }
283 PetscCallMPI(MPI_Allreduce(MPI_IN_PLACE, rows, (PetscMPIInt)n_rows, MPI_UINT64_T, MPI_SUM, PETSC_COMM_WORLD));
284 for (PetscInt r = 0; r < n_rows; ++r) { row_before[r] = total; total += rows[r]; }
285 // Ranks with the same j and k ranges share rows; their order along i is the order in the row.
286 PetscCallMPI(MPI_Comm_split(PETSC_COMM_WORLD, n_cells > 0 ? (PetscMPIInt)(table->origin[1] + nodes[1] * table->origin[2]) : MPI_UNDEFINED,
287 (PetscMPIInt)table->origin[0], &row_comm));
288 if (row_comm != MPI_COMM_NULL) {
289 PetscCallMPI(MPI_Exscan(segment, segment_before, (PetscMPIInt)n_rows_local, MPI_UINT64_T, MPI_SUM, row_comm));
290 PetscCallMPI(MPI_Comm_rank(row_comm, &row_rank));
291 if (row_rank == 0) PetscCall(PetscArrayzero(segment_before, n_rows_local)); /* MPI_Exscan leaves rank 0's result undefined */
292 PetscCallMPI(MPI_Comm_free(&row_comm));
293 }
294
295 // --- 3. Stratified particle IDs: floor(N P / T) at each cell boundary ---
296 for (PetscInt lrow = 0; lrow < n_rows_local; ++lrow) {
297 const PetscInt lj = lrow % table->n[1], lk = lrow / table->n[1];
298 const PetscInt row = (table->origin[1] + lj - global_origin[1]) + global_cells[1] * (table->origin[2] + lk - global_origin[2]);
299 uint64_t before = row_before[row] + segment_before[lrow];
300 for (PetscInt li = 0; li < table->n[0]; ++li) {
301 const PetscInt c = li + table->n[0] * lrow;
302 table->pid_lo[c] = (PetscInt64)(((unsigned __int128)numParticles * before) / total);
303 before += quanta[c];
304 table->pid_hi[c] = (PetscInt64)(((unsigned __int128)numParticles * before) / total);
305 table->count += (PetscInt)(table->pid_hi[c] - table->pid_lo[c]);
306 }
307 }
308 PetscCall(PetscFree4(rows, row_before, segment, segment_before));
309 PetscCall(PetscFree2(measure, quanta));
310 PROFILE_FUNCTION_END;
311 PetscFunctionReturn(0);
312}

325/**
326 * @brief Draw one particle's seeding point inside its cell, uniform in the cell's measure, or
327 * on a face in proportion to the flux through it.
328 * @details Uniform logical coordinates are accepted with probability density / `bound`, times
329 * the inward normal velocity / `flux_bound` when `velocity` is given. Every draw is
330 * keyed on the run seed, the particle ID and the attempt, so the point depends only on
331 * the particle, not on the rank that seeds it. A surface point is put just inside the
332 * cell, SEEDING_FACE_OFFSET of the cell from the face, where its velocity is taken.
333 * @param[in] user Grid context.
334 * @param[in] coor Ghosted node-coordinate array.
335 * @param[in] table A table built with `build_bound` set.
336 * @param[in] c The cell, an index into the table.
337 * @param[in] pid The particle's ID.
338 * @param[in] velocity NULL, or for a face table the velocity the swarm would give the point.
339 * @param[in] normal The face's inward unit normal, read with `velocity`.
340 * @param[in] flux_bound An upper bound of the inward normal velocity over the face cell, read
341 * with `velocity`.
342 * @param[out] ci_metric_lnode_out,cj_metric_lnode_out,ck_metric_lnode_out Origin node of the cell.
343 * @param[out] xi_metric_logic_out,eta_metric_logic_out,zta_metric_logic_out Logical point in it.
344 * @param[out] velocity_out The point's velocity, set with `velocity`; or NULL.
345 * @return PetscErrorCode 0 on success.
346 */
347static PetscErrorCode DrawSeedingPoint(
348 UserCtx *user, const Cmpnts ***coor, const SeedingCellTable *table, PetscInt c, PetscInt64 pid,
349 const VelocityPointInterpolator *velocity, Cmpnts normal, PetscReal flux_bound,
350 PetscInt *ci_metric_lnode_out, PetscInt *cj_metric_lnode_out, PetscInt *ck_metric_lnode_out,
351 PetscReal *xi_metric_logic_out, PetscReal *eta_metric_logic_out, PetscReal *zta_metric_logic_out,
352 Cmpnts *velocity_out)
353{
354 PetscReal logic[3], u[4];
355
356 PetscFunctionBeginUser;
357 PROFILE_HELPER_BEGIN;
358 {
359 PetscInt origin[3];
360 SeedingCellOrigin(table, c, origin);
361 *ci_metric_lnode_out = origin[0]; *cj_metric_lnode_out = origin[1]; *ck_metric_lnode_out = origin[2];
362 }
363
364 // Accept uniform logicals with probability density / bound; a cell whose density varies
365 // little accepts almost every draw. The cap only guards a degenerate envelope.
366 for (PetscInt attempt = 0; attempt < 1000; ++attempt) {
367 PetscReal density;
368 ParticleUniformsFromHash(PicurvDrawHash(user->simCtx->particleRandomSeed, PICURV_DRAW_SPACE_PLACEMENT, 0, pid, attempt), u);
369 SeedingLogicalPoint(u, table->face, logic);
370 PetscCall(SeedingMeasureDensity(user, coor, table->face, *ci_metric_lnode_out, *cj_metric_lnode_out,
371 *ck_metric_lnode_out, logic, &density));
372 if (velocity) {
373 const PetscInt cell[3] = {*ci_metric_lnode_out, *cj_metric_lnode_out, *ck_metric_lnode_out};
374 Cmpnts point;
375 PetscCall(MetricLogicalToPhysical(user, coor, cell[0], cell[1], cell[2], logic[0], logic[1], logic[2], &point));
376 PetscCall(InterpolateVelocityAtPoint(velocity, cell, logic, point, velocity_out));
377 const PetscReal u_n = PetscMax(velocity_out->x * normal.x + velocity_out->y * normal.y + velocity_out->z * normal.z, 0.0);
378 if (u_n > flux_bound) {
379 LOG_ALLOW(LOCAL, LOG_WARNING, "Rank %d: inward velocity %g exceeds the bound %g in face cell (%" PetscInt_FMT ",%" PetscInt_FMT ",%" PetscInt_FMT "); placement there is biased.\n",
380 user->simCtx->rank, (double)u_n, (double)flux_bound, cell[0], cell[1], cell[2]);
381 }
382 density *= u_n / flux_bound;
383 }
384 if (u[3] * table->bound[c] <= density) break;
385 if (attempt == 999) {
386 LOG_ALLOW(LOCAL, LOG_WARNING, "Rank %d: 1000 rejected draws in cell (%" PetscInt_FMT ",%" PetscInt_FMT ",%" PetscInt_FMT "); accepting the last one.\n",
387 user->simCtx->rank, *ci_metric_lnode_out, *cj_metric_lnode_out, *ck_metric_lnode_out);
388 }
389 }
390 *xi_metric_logic_out = logic[0]; *eta_metric_logic_out = logic[1]; *zta_metric_logic_out = logic[2];
391
392 PROFILE_HELPER_END;
393 PetscFunctionReturn(0);
394}

397/**
398 * @brief Initialize position-independent particle fields after a particle is created.
399 */
400static PetscErrorCode InitializeParticleBasicProperties(UserCtx *user,
401 PetscInt particlesPerProcess)
402{
403 PetscErrorCode ierr;
404 DM swarm = user->swarm;
405 PetscReal *positions_field = NULL; // Pointer to swarm field for physical positions (x,y,z)
406 PetscInt64 *particleIDs = NULL; // Pointer to swarm field for Particle IDs
407 PetscInt *cellIDs_petsc = NULL; // Pointer to swarm field for DMSwarm_CellID (i,j,k of containing cell)
408 PetscInt *status_field = NULL; // Pointer to swarm field for DMSwarm_location_status(NEEDS_LOCATION etc)
409 PetscMPIInt rank,size; // MPI rank of the current process, and total number of ranks.
410 const Cmpnts ***coor_nodes_local_array; // Read-only access to local node coordinates (from user->da)
411 Vec Coor_local; // Local vector for node coordinates
412 // Owned cells and the particles each seeds, for volumetric and surface seeding
413 SeedingCellTable seed_table;
414
415 PetscFunctionBeginUser;
416
417 PROFILE_FUNCTION_BEGIN;
418
419 // --- 1. Input Validation and Basic Setup ---
420 if (!user) SETERRQ(PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "Null user.");
421 SimCtx *simCtx = user->simCtx;
422 rank = simCtx->rank;
423 size = simCtx->size;
424
425 // Get DMDA information for the node-centered coordinate grid (user->da)
426 Coor_local = user->lCoor;
427 if (!Coor_local) SETERRQ(PETSC_COMM_SELF, PETSC_ERR_PLIB, "user->lCoor is NULL: grid coordinates are not set up.");
428 ierr = DMDAVecGetArrayRead(user->fda, Coor_local, (void*)&coor_nodes_local_array); CHKERRQ(ierr);
429 ierr = PetscMemzero(&seed_table, sizeof(seed_table)); CHKERRQ(ierr);
430
431 LOG_ALLOW(LOCAL, LOG_INFO, "Rank %d: Initializing %" PetscInt_FMT " particles. Mode: %s.\n",
432 rank, particlesPerProcess, ParticleInitializationToString(simCtx->ParticleInitialization));
433
434 // --- 2. Seeding Table for Volumetric and Surface Initialization ---
435 // Volume seeds the owned cells, Surface the owned cells on the inlet face; the table names
436 // each cell's particles, the same split CreateParticleSwarm sized the swarm with.
437 const PetscBool seeds_from_table = (PetscBool)(simCtx->ParticleInitialization == PARTICLE_INIT_VOLUME ||
438 simCtx->ParticleInitialization == PARTICLE_INIT_SURFACE_RANDOM);
439 PetscInt seed_cell = 0; // Cell the next table-seeded particle goes in
440 PetscInt64 seed_pid = 0; // That particle's ID
441 if (seeds_from_table) {
442 const PetscInt face = (simCtx->ParticleInitialization == PARTICLE_INIT_SURFACE_RANDOM) ? (PetscInt)user->identifiedInletBCFace : -1;
443 ierr = BuildSeedingCellTable(user, coor_nodes_local_array, face, simCtx->np, PETSC_TRUE, &seed_table); CHKERRQ(ierr);
444 PetscCheck(seed_table.count == particlesPerProcess, PETSC_COMM_SELF, PETSC_ERR_PLIB,
445 "Seeding table holds %" PetscInt_FMT " particles but the swarm was sized for %" PetscInt_FMT ".",
446 seed_table.count, particlesPerProcess);
447 if (seed_table.n[0] * seed_table.n[1] * seed_table.n[2] > 0) seed_pid = seed_table.pid_lo[0];
448 LOG_ALLOW(LOCAL, LOG_INFO, "Rank %d: Seeding %" PetscInt_FMT " particles stratified over %" PetscInt_FMT " owned cells by %s.\n",
449 rank, particlesPerProcess, seed_table.n[0] * seed_table.n[1] * seed_table.n[2],
450 face < 0 ? "volume" : "inlet area");
451 }
452
453 // --- 3. Get Access to Swarm Fields ---
454 ierr = DMSwarmGetField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_POSITION), NULL, NULL, (void**)&positions_field); CHKERRQ(ierr);
455 ierr = DMSwarmGetField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_PID), NULL, NULL, (void**)&particleIDs); CHKERRQ(ierr);
456 ierr = DMSwarmGetField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_CELL_ID), NULL, NULL, (void**)&cellIDs_petsc); CHKERRQ(ierr);
457 ierr = DMSwarmGetField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_LOCATION_STATUS),NULL,NULL,(void**)&status_field); CHKERRQ(ierr);
458
459 // --- 4. Determine Starting Global PID for this Rank (point source) ---
460 // Point-source PIDs are consecutive across ranks in rank order, so each rank starts where
461 // the ranks before it end. Table seeding takes each particle's ID from its cell instead.
462 PetscInt base_pid_for_rank = 0;
463 {
464 PetscInt local_count = particlesPerProcess;
465 ierr = MPI_Exscan(&local_count, &base_pid_for_rank, 1, MPIU_INT, MPI_SUM, PETSC_COMM_WORLD); CHKERRMPI(ierr);
466 if (rank == 0) base_pid_for_rank = 0; /* MPI_Exscan leaves rank 0's result undefined */
467 }
468
469 // --- 4b. Locate the Point Source Once ---
470 // Every rank places its share at the same point, so its cell is found by one search per rank;
471 // the lowest rank that holds it shares the cell, and settlement then routes the particles there.
472 PetscInt point_cell[3] = {-1, -1, -1};
473 if (simCtx->ParticleInitialization == PARTICLE_INIT_POINT_SOURCE) {
474 Particle probe;
475 ParticleLocationStatus probe_status = NEEDS_LOCATION;
476 PetscMPIInt candidate, owner = size;
477 ierr = PetscMemzero(&probe, sizeof(probe)); CHKERRQ(ierr);
478 probe.PID = -1;
479 probe.cell[0] = -1; probe.cell[1] = -1; probe.cell[2] = -1;
480 probe.loc.x = simCtx->psrc_x; probe.loc.y = simCtx->psrc_y; probe.loc.z = simCtx->psrc_z;
481 ierr = LocateParticleOrFindMigrationTarget(user, NULL, &probe, &probe_status); CHKERRQ(ierr);
482 candidate = (probe_status == ACTIVE_AND_LOCATED) ? rank : size;
483 ierr = MPI_Allreduce(&candidate, &owner, 1, MPI_INT, MPI_MIN, PETSC_COMM_WORLD); CHKERRMPI(ierr);
484 if (owner < size) {
485 if (rank == owner) { point_cell[0] = probe.cell[0]; point_cell[1] = probe.cell[1]; point_cell[2] = probe.cell[2]; }
486 ierr = MPI_Bcast(point_cell, 3, MPIU_INT, owner, PETSC_COMM_WORLD); CHKERRMPI(ierr);
487 LOG_ALLOW(GLOBAL, LOG_INFO, "Point source (%.6e, %.6e, %.6e) lies in cell (%" PetscInt_FMT ",%" PetscInt_FMT ",%" PetscInt_FMT ") on rank %d.\n",
488 simCtx->psrc_x, simCtx->psrc_y, simCtx->psrc_z, point_cell[0], point_cell[1], point_cell[2], owner);
489 } else {
490 LOG_ALLOW(GLOBAL, LOG_WARNING, "Point source (%.6e, %.6e, %.6e) was not located on any rank; its particles will be searched for.\n",
491 simCtx->psrc_x, simCtx->psrc_y, simCtx->psrc_z);
492 }
493 }
494
495 // --- 5. Loop Over Particles to Initialize ---
496 for (PetscInt p = 0; p < particlesPerProcess; p++) {
497 PetscInt ci_metric_lnode, cj_metric_lnode, ck_metric_lnode;
498 PetscReal xi_metric_logic = 0.0, eta_metric_logic = 0.0, zta_metric_logic = 0.0;
499 Cmpnts phys_coords = {0.0, 0.0, 0.0};
500 PetscBool particle_placed_by_this_rank = PETSC_FALSE;
501 PetscInt64 pid = (PetscInt64)base_pid_for_rank + p;
502
503 if (seeds_from_table) { // --- 5.a. Volumetric or Surface Initialization ---
504 while (seed_pid == seed_table.pid_hi[seed_cell]) seed_pid = seed_table.pid_lo[++seed_cell]; // skip filled (or empty) cells
505 pid = seed_pid++;
506 ierr = DrawSeedingPoint(user, coor_nodes_local_array, &seed_table, seed_cell, pid, NULL, (Cmpnts){0.0, 0.0, 0.0}, 0.0,
507 &ci_metric_lnode, &cj_metric_lnode, &ck_metric_lnode,
508 &xi_metric_logic, &eta_metric_logic, &zta_metric_logic, NULL); CHKERRQ(ierr);
509 ierr = MetricLogicalToPhysical(user, coor_nodes_local_array,
510 ci_metric_lnode, cj_metric_lnode, ck_metric_lnode,
511 xi_metric_logic, eta_metric_logic, zta_metric_logic,
512 &phys_coords); CHKERRQ(ierr);
513 particle_placed_by_this_rank = PETSC_TRUE;
514 }else if(simCtx->ParticleInitialization == PARTICLE_INIT_POINT_SOURCE){ // --- 5.c. Point Source Initialization ---
515 // All particles placed at the user-specified fixed point (psrc_x, psrc_y, psrc_z).
516 // No random number generation or logical-to-physical conversion needed.
517 phys_coords.x = simCtx->psrc_x;
518 phys_coords.y = simCtx->psrc_y;
519 phys_coords.z = simCtx->psrc_z;
520 ci_metric_lnode = point_cell[0]; cj_metric_lnode = point_cell[1]; ck_metric_lnode = point_cell[2];
521 particle_placed_by_this_rank = (PetscBool)(point_cell[0] >= 0);
522 }else {
523 SETERRQ(PETSC_COMM_SELF, PETSC_ERR_ARG_OUTOFRANGE, "Unknown ParticleInitialization mode %d.", simCtx->ParticleInitialization);
524 }
525
526 // --- 5.c. Store Particle Properties ---
527 positions_field[3*p+0] = phys_coords.x;
528 positions_field[3*p+1] = phys_coords.y;
529 positions_field[3*p+2] = phys_coords.z;
530
531 particleIDs[p] = pid;
532 // Start the first settlement's walk in the cell the particle was placed in; a point-source
533 // particle no rank located has no cell and is located by search.
534 if (particle_placed_by_this_rank) {
535 cellIDs_petsc[3*p+0] = ci_metric_lnode; cellIDs_petsc[3*p+1] = cj_metric_lnode; cellIDs_petsc[3*p+2] = ck_metric_lnode;
536 } else {
537 cellIDs_petsc[3*p+0] = -1; cellIDs_petsc[3*p+1] = -1; cellIDs_petsc[3*p+2] = -1;
538 }
539 status_field[p] = UNINITIALIZED;
540
541 // --- 5.d. The particle's seed line ---
542 if (seeds_from_table) {
543 LOG_PARTICLE(simCtx, particleIDs[p], "seed",
544 "%s placement on rank %d at (%.6e, %.6e, %.6e): cell (%" PetscInt_FMT ",%" PetscInt_FMT ",%" PetscInt_FMT ") logical (%.4f, %.4f, %.4f)\n",
545 ParticleInitializationToString(simCtx->ParticleInitialization), (int)rank,
546 phys_coords.x, phys_coords.y, phys_coords.z, ci_metric_lnode, cj_metric_lnode, ck_metric_lnode,
547 xi_metric_logic, eta_metric_logic, zta_metric_logic);
548 } else if (particle_placed_by_this_rank) {
549 LOG_PARTICLE(simCtx, particleIDs[p], "seed",
550 "%s placement on rank %d at (%.6e, %.6e, %.6e): cell (%" PetscInt_FMT ",%" PetscInt_FMT ",%" PetscInt_FMT ")\n",
551 ParticleInitializationToString(simCtx->ParticleInitialization), (int)rank,
552 phys_coords.x, phys_coords.y, phys_coords.z, ci_metric_lnode, cj_metric_lnode, ck_metric_lnode);
553 } else {
554 // A point source no rank located.
555 LOG_PARTICLE(simCtx, particleIDs[p], "seed",
556 "%s: not placed by rank %d; parked at (%.6e, %.6e, %.6e) for the settlement to locate\n",
557 ParticleInitializationToString(simCtx->ParticleInitialization), (int)rank,
558 phys_coords.x, phys_coords.y, phys_coords.z);
559 }
560 }
561
562 // --- 6. Restore Pointers and Cleanup ---
563 ierr = DMSwarmRestoreField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_POSITION), NULL, NULL, (void**)&positions_field); CHKERRQ(ierr);
564 ierr = DMSwarmRestoreField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_PID), NULL, NULL, (void**)&particleIDs); CHKERRQ(ierr);
565 ierr = DMSwarmRestoreField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_CELL_ID), NULL, NULL, (void**)&cellIDs_petsc); CHKERRQ(ierr);
566 ierr = DMSwarmRestoreField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_LOCATION_STATUS), NULL, NULL, (void**)&status_field); CHKERRQ(ierr);
567 ierr = DMDAVecRestoreArrayRead(user->fda, Coor_local, (void*)&coor_nodes_local_array); CHKERRQ(ierr);
568 ierr = FreeSeedingCellTable(&seed_table); CHKERRQ(ierr);
569
570 LOG_ALLOW(LOCAL, LOG_INFO, "Rank %d: Completed processing for %" PetscInt_FMT " particles.\n",
571 rank, particlesPerProcess);
572
573 PROFILE_FUNCTION_END;
574
575 PetscFunctionReturn(0);
576}

578/**
579 * @brief Set local particles `first` to `end` - 1 of every field the particle catalog marks
580 * PARTICLE_FIELD_CAPABILITY_DEFAULT_INITIALIZE to the field's catalog default.
581 */
582static PetscErrorCode InitializeDefaultParticleFields(DM swarm, PetscInt first, PetscInt end)
583{
584 PetscFunctionBeginUser;
585 PROFILE_FUNCTION_BEGIN;
586 for (PetscInt raw_id = 0; raw_id < PARTICLE_FIELD_ID_COUNT; ++raw_id) {
587 const ParticleFieldDescriptor *descriptor = NULL;
588 PetscReal *field = NULL;
589
590 PetscCall(ParticleFieldGetDescriptor((ParticleFieldId)raw_id, &descriptor));
591 if (!(descriptor->capabilities & PARTICLE_FIELD_CAPABILITY_DEFAULT_INITIALIZE)) continue;
592 PetscCheck(descriptor->data_type == PETSC_REAL, PETSC_COMM_SELF, PETSC_ERR_ARG_WRONG,
593 "Default initialization supports only PETSC_REAL particle fields; '%s' uses %s.",
594 descriptor->canonical_name, PetscDataTypes[descriptor->data_type]);
595 PetscCall(DMSwarmGetField(swarm, descriptor->canonical_name, NULL, NULL, (void **)&field));
596 for (PetscInt p = first; p < end; ++p) {
597 for (PetscInt d = 0; d < descriptor->components; ++d) field[descriptor->components * p + d] = descriptor->default_real_value;
598 }
599 PetscCall(DMSwarmRestoreField(swarm, descriptor->canonical_name, NULL, NULL, (void **)&field));
600 }
601 PROFILE_FUNCTION_END;
602 PetscFunctionReturn(0);
603}

686 * BuildSeedingCellTable makes, so a rank seeds in proportion to the volume or inlet
687 * area it owns, none if it owns no part of the inlet, and the counts sum to

707 PROFILE_FUNCTION_END;

711/**
712 * @brief Implementation of \ref CreateParticleSwarm().
713 * @see CreateParticleSwarm()
714 */
715PetscErrorCode CreateParticleSwarm(UserCtx *user, PetscInt numParticles, PetscInt *particlesPerProcess) {
716 PetscErrorCode ierr; // PETSc error handling variable
717 PetscMPIInt rank, size; // Variables to store MPI rank and size
718 PetscInt remainder = 0; // Remainder of particles after division
719
720 PetscFunctionBeginUser;
721 PROFILE_FUNCTION_BEGIN;
722 // Validate input parameters
723 if (numParticles < 0) {
724 LOG_ALLOW(GLOBAL,LOG_DEBUG, "Number of particles must not be negative. Given: %" PetscInt_FMT "\n", numParticles);
725 PROFILE_FUNCTION_END;
726 return PETSC_ERR_ARG_OUTOFRANGE;
727 }
728
729 // Retrieve MPI rank and size
730 rank = user->simCtx->rank;
731 size = user->simCtx->size;
732 LOG_ALLOW(GLOBAL,LOG_INFO," Domain dimensions: [%.2f,%.2f],[%.2f,%.2f],[%.2f,%.2f] \n",
733 user->Min_X,user->Max_X,user->Min_Y,user->Max_Y, user->Min_Z,user->Max_Z);
734 LOG_ALLOW_SYNC(GLOBAL,LOG_DEBUG, "[Rank %d] Local Bounding Box: [%.2f,%.2f],[%.2f,%.2f],[%.2f,%.2f] \n",
735 rank,user->bbox.min_coords.x,user->bbox.max_coords.x,
736 user->bbox.min_coords.y,user->bbox.max_coords.y,
737 user->bbox.min_coords.z,user->bbox.max_coords.z);
738 // Distribute particles among MPI processes
739 if (user->simCtx->ParticleInitialization == PARTICLE_INIT_VOLUME) {
740 ierr = DistributeParticlesByOwnedMeasure(user, -1, numParticles, particlesPerProcess); CHKERRQ(ierr);
741 } else if (user->simCtx->ParticleInitialization == PARTICLE_INIT_SURFACE_RANDOM) {
742 PetscCheck(user->inletFaceDefined, PETSC_COMM_WORLD, PETSC_ERR_ARG_WRONGSTATE,
743 "Surface particle seeding needs an INLET face.");
744 ierr = DistributeParticlesByOwnedMeasure(user, (PetscInt)user->identifiedInletBCFace, numParticles, particlesPerProcess); CHKERRQ(ierr);
745 } else {
746 ierr = DistributeParticles(numParticles, rank, size, particlesPerProcess, &remainder); CHKERRQ(ierr);
747 }
748
749 // Initialize the DMSwarm - creates the swarm, sets the type and dimension
750 ierr = InitializeSwarm(user); CHKERRQ(ierr);
751
752 if (user->da) {
753 ierr = DMSwarmSetCellDM(user->swarm, user->da); CHKERRQ(ierr);
754 LOG_ALLOW(LOCAL,LOG_INFO,"Associated DMSwarm with Cell DM (user->da).\n");
755 } else {
756 // If user->da is essential for your simulation logic with particles, this should be a fatal error.
757 LOG_ALLOW(GLOBAL, LOG_WARNING, "user->da (Cell DM for Swarm) is NULL. Cell-based swarm operations might fail.\n");
758 // SETERRQ(PETSC_COMM_WORLD, PETSC_ERR_ARG_WRONGSTATE, "user->da (Cell DM) is NULL but required.");
759 }
760
761 // Register particle fields (position, velocity, CellID, weight, etc.)
762 ierr = RegisterParticleFields(user->swarm); CHKERRQ(ierr);
763
764 // Set the local number of particles for this rank and spare capacity for arrivals (10% of the
765 // local count; PETSc reallocates beyond it). Every resize zeroes the whole spare region, so a
766 // buffer sized to the global count would cost each rank memory and work of order numParticles.
767 ierr = DMSwarmSetLocalSizes(user->swarm, *particlesPerProcess, *particlesPerProcess / 10); CHKERRQ(ierr);
768 LOG_ALLOW(GLOBAL,LOG_INFO, "Set local swarm size: %" PetscInt_FMT " particles.\n", *particlesPerProcess);
769
770 // Optionally, LOG_ALLOW detailed DM info in debug mode
771 if (get_log_level() >= LOG_DEBUG && is_function_allowed(__func__)) {
772 LOG_ALLOW(GLOBAL,LOG_DEBUG,"Viewing DMSwarm:\n");
773 ierr = DMView(user->swarm, PETSC_VIEWER_STDOUT_WORLD); CHKERRQ(ierr);
774 }
775
776 LOG_ALLOW(GLOBAL,LOG_INFO, "Particle swarm creation and initialization complete.\n");
777
778 PROFILE_FUNCTION_END;
779 PetscFunctionReturn(0);
780}

782// NOTE: The following two functions are helpers for unpacking and updating particle data
783// between DMSwarm fields and the Particle struct used in simulation logic.
784// While Swarm fields store data in arrays, the Particle struct provides a convenient
785// way to manipulate individual particle properties during simulation steps.

788/**
789 * @brief Implementation of \ref UnpackSwarmFields().
790 * @details Full API contract (arguments, ownership, side effects) is documented with
791 * the header declaration in `include/ParticleSwarm.h`.
792 * @see UnpackSwarmFields()
793 */
794PetscErrorCode UnpackSwarmFields(PetscInt i, const PetscInt64 *PIDs, const PetscReal *weights,
795 const PetscReal *positions, const PetscInt *cellIndices,
796 PetscReal *velocities,PetscInt *LocStatus,PetscReal *diffusivity, Cmpnts *diffusivitygradient, PetscReal *psi, Particle *particle) {
797 PetscFunctionBeginUser;
798
799 PROFILE_HELPER_BEGIN;
800
801 // The particle's values are reported by the stage that uses them, not here: this runs
802 // several times per particle per step.
803 if (particle == NULL) {
804 SETERRQ(PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "Output Particle pointer is NULL. \n");
805 }
806 if(PIDs == NULL){
807 SETERRQ(PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "Input PIDs pointer is NULL.\n");
808 }
809 if(positions == NULL){
810 SETERRQ(PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "Input positions pointer is NULL.\n");
811 }
812 if(cellIndices == NULL){
813 SETERRQ(PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "Input cellIndices pointer is NULL.\n");
814 }
815 if(LocStatus == NULL){
816 SETERRQ(PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "Input LocStatus pointer is NULL.\n");
817 }
818
819 particle->PID = PIDs[i];
820
821 // A NULL optional field is one the caller does not need; it takes the default shown.
822 if(weights == NULL){
823 particle->weights.x = 1.0;
824 particle->weights.y = 1.0;
825 particle->weights.z = 1.0;
826 }else{
827 particle->weights.x = weights[3 * i];
828 particle->weights.y = weights[3 * i + 1];
829 particle->weights.z = weights[3 * i + 2];
830 }
831
832 particle->loc.x = positions[3 * i];
833 particle->loc.y = positions[3 * i + 1];
834 particle->loc.z = positions[3 * i + 2];
835
836 if(velocities == NULL){
837 particle->vel.x = 0.0;
838 particle->vel.y = 0.0;
839 particle->vel.z = 0.0;
840 }else{
841 particle->vel.x = velocities[3 * i];
842 particle->vel.y = velocities[3 * i + 1];
843 particle->vel.z = velocities[3 * i + 2];
844 }
845
846 particle->diffusivity = (diffusivity == NULL) ? 1.0 : diffusivity[i];
847
848 if(diffusivitygradient == NULL){
849 particle->diffusivitygradient.x = 0.0;
850 particle->diffusivitygradient.y = 0.0;
851 particle->diffusivitygradient.z = 0.0;
852 }else{
853 particle->diffusivitygradient = diffusivitygradient[i];
854 }
855
856 particle->psi = (psi == NULL) ? 0.0 : psi[i];
857
858 particle->cell[0] = cellIndices[3 * i];
859 particle->cell[1] = cellIndices[3 * i + 1];
860 particle->cell[2] = cellIndices[3 * i + 2];
861
862 particle->location_status = (ParticleLocationStatus)LocStatus[i];
863
864 // The destination_rank is only set by the location search, not read from the swarm,
865 // so we initialize it to a known invalid state.
866 particle->destination_rank = MPI_PROC_NULL;
867
868 PROFILE_HELPER_END;
869
870 PetscFunctionReturn(0);
871}

877PetscErrorCode UpdateSwarmFields(PetscInt i, const Particle *particle,
878 PetscReal *positions,
879 PetscReal *velocities,
880 PetscReal *weights,
881 PetscInt *cellIndices,
882 PetscInt *status,
883 PetscReal *diffusivity,
884 Cmpnts *diffusivitygradient,
885 PetscReal *psi)
886{
887 PetscFunctionBeginUser;
888 PROFILE_HELPER_BEGIN;
889
890 if (!particle) {
891 SETERRQ(PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "Input Particle pointer is NULL.\n");
892 }
893
894 // --- 1. Position (x, y, z) ---
895 if (positions) {
896 positions[3 * i + 0] = particle->loc.x;
897 positions[3 * i + 1] = particle->loc.y;
898 positions[3 * i + 2] = particle->loc.z;
899 }
900
901 // --- 2. Velocity (u, v, w) ---
902 if (velocities) {
903 velocities[3 * i + 0] = particle->vel.x;
904 velocities[3 * i + 1] = particle->vel.y;
905 velocities[3 * i + 2] = particle->vel.z;
906 }
907
908 // --- 3. Weights (i, j, k) ---
909 if (weights) {
910 weights[3 * i + 0] = particle->weights.x;
911 weights[3 * i + 1] = particle->weights.y;
912 weights[3 * i + 2] = particle->weights.z;
913 }
914
915 // --- 4. Cell Indices (i, j, k) ---
916 if (cellIndices) {
917 cellIndices[3 * i + 0] = particle->cell[0];
918 cellIndices[3 * i + 1] = particle->cell[1];
919 cellIndices[3 * i + 2] = particle->cell[2];
920 }
921
922 // --- 5. Status ---
923 if (status) {
924 status[i] = (PetscInt)particle->location_status;
925 }
926
927 // --- 6. Diffusivity ---
928 if (diffusivity) {
929 diffusivity[i] = particle->diffusivity;
930 }
931
932 if(diffusivitygradient){
933 diffusivitygradient[i].x = particle->diffusivitygradient.x;
934 diffusivitygradient[i].y = particle->diffusivitygradient.y;
935 diffusivitygradient[i].z = particle->diffusivitygradient.z;
936 }
937 // --- 7. Psi ---
938 if (psi) {
939 psi[i] = particle->psi;
940 }
941
942 PROFILE_HELPER_END;
943 PetscFunctionReturn(0);
944}

948 * @see UpdateParticleWeights()

950PetscErrorCode UpdateParticleWeights(PetscReal *d, Particle *particle) {
951
952 PetscFunctionBeginUser;
953 PROFILE_HELPER_BEGIN;
954
955 // Validate input pointers
956 if (!d || !particle) {
957 SETERRQ(PETSC_COMM_SELF, PETSC_ERR_ARG_NULL,
958 "Null pointer argument (d or particle).");
959 }
960
961
962 // Validate distances
963 for (PetscInt i = LEFT; i < NUM_FACES; i++) {
964 if (d[i] <= INTERPOLATION_DISTANCE_TOLERANCE) {
965 // A particle on a face: benign, so reported only to a VERBOSE run of this function.
966 LOG_ALLOW(LOCAL, LOG_VERBOSE,

972 }

975 particle->weights.x = d[LEFT] / (d[LEFT] + d[RIGHT]);
976 particle->weights.y = d[BOTTOM] / (d[BOTTOM] + d[TOP]);
977 particle->weights.z = d[BACK] / (d[FRONT] + d[BACK]);
978
979
980
981 PROFILE_HELPER_END;
982 PetscFunctionReturn(0);
983}

1049/* ------------------------------------------------------------------------- */
1050/* Injection at inlets */
1051/* ------------------------------------------------------------------------- */
1052
1053/* A checkpoint stores a particle ID as a double, so every ID stays below 2^53. Seeded IDs stay
1054 below the base; an injected ID is base + (step * faces + face cell) * per_cell + k, which
1055 depends only on the grid, and so stays unique when a restart changes the rate or the inlets. */
1056#define PARTICLE_INLET_PID_BASE ((PetscInt64)1 << 40)
1057#define PARTICLE_INLET_PIDS_PER_CELL ((PetscInt64)1 << 10)
1058#define PARTICLE_PID_LIMIT ((PetscInt64)1 << 53)
1059#define PARTICLE_INLET_FACE_COUNT 6
1060#define PARTICLE_INLET_SUMS 4
1061
1062/** @brief One inlet face that injects particles. */
1063struct ParticleInlet {
1064 BCFace face;
1065 PicurvExpression *particles_per_cell; /**< Mean particles per cell the entering fluid carries, in `t`. */
1066 ParticleFieldPlan *plan; /**< This inlet's field values; NULL uses the case's. */
1067 SeedingCellTable table; /**< The owned cells on the face; built at the first injection. */
1068 PetscReal cells_per_volume; /**< The grid's cells over its volume, set with the table. */
1069 PetscBool table_built;
1070};

1072/**
1073 * @brief Implementation of \ref ParticleInletsCreate().
1074 * @see ParticleInletsCreate()
1075 */
1076PetscErrorCode ParticleInletsCreate(SimCtx *simCtx)
1077{
1078 static const char *const names[] = {"t"};
1079 PetscInt count = 0;
1080 PetscBool found = PETSC_FALSE;
1081
1082 PetscFunctionBeginUser;
1083 PROFILE_FUNCTION_BEGIN;
1084 PetscCall(PetscOptionsGetInt(NULL, NULL, "-particle_inlet_count", &count, &found));
1085 if (!found || count == 0) { PROFILE_FUNCTION_END; PetscFunctionReturn(0); }
1086 PetscCheck(count > 0 && count <= PARTICLE_INLET_FACE_COUNT, PETSC_COMM_WORLD, PETSC_ERR_ARG_OUTOFRANGE,
1087 "-particle_inlet_count must be between 0 and %d (got %" PetscInt_FMT ").",
1088 PARTICLE_INLET_FACE_COUNT, count);
1089 PetscCheck((PetscInt64)simCtx->np < PARTICLE_INLET_PID_BASE, PETSC_COMM_WORLD, PETSC_ERR_ARG_OUTOFRANGE,
1090 "Injected particle IDs start at %" PetscInt64_FMT ", so a run with particle inlets seeds fewer "
1091 "particles than that (got %" PetscInt_FMT ").", PARTICLE_INLET_PID_BASE, simCtx->np);
1092 PetscCall(PetscCalloc1(count, &simCtx->particleInlets));
1093 simCtx->particleInletCount = count;
1094 for (PetscInt n = 0; n < count; ++n) {
1095 struct ParticleInlet *inlet = &simCtx->particleInlets[n];
1096 char option[256], text[4096];
1097 PetscInt face = -1;
1098
1099 PetscCall(PetscSNPrintf(option, sizeof(option), "-particle_inlet_%" PetscInt_FMT "_face", n));
1100 PetscCall(PetscOptionsGetInt(NULL, NULL, option, &face, &found));
1101 PetscCheck(found && face >= 0 && face < PARTICLE_INLET_FACE_COUNT, PETSC_COMM_WORLD, PETSC_ERR_ARG_WRONG,
1102 "%s must name a face from 0 to %d.", option, PARTICLE_INLET_FACE_COUNT - 1);
1103 inlet->face = (BCFace)face;
1104 PetscCall(PetscSNPrintf(option, sizeof(option), "-particle_inlet_%" PetscInt_FMT "_particles_per_cell", n));
1105 PetscCall(PetscOptionsGetString(NULL, NULL, option, text, sizeof(text), &found));
1106 PetscCheck(found, PETSC_COMM_WORLD, PETSC_ERR_ARG_WRONG, "%s is missing.", option);
1107 PetscCall(PicurvExpressionCompile(text, names, 1, PETSC_FALSE, &inlet->particles_per_cell));
1108 PetscCall(PetscSNPrintf(option, sizeof(option), "particle_inlet_%" PetscInt_FMT "_fields", n));
1109 PetscCall(ParticleFieldPlanCreate(option, &inlet->plan));
1110 PetscCheck(!inlet->plan || !VerificationScalarOverrideActive(simCtx), PETSC_COMM_WORLD, PETSC_ERR_ARG_WRONG,
1111 "An inlet's particle field values cannot be combined with the verification scalar source, "
1112 "which prescribes Psi at every step.");
1113 }
1114 PROFILE_FUNCTION_END;
1115 PetscFunctionReturn(0);
1116}

1138/**
1139 * @brief Add one inlet's particles for this step.
1140 * @param[in,out] user Grid context holding the swarm.
1141 * @param[in,out] inlet The inlet.
1142 * @param[in] event Time of the step, and the identity keying field draws.
1143 * @param[in] first_id This step's first face-cell identity.
1144 * @param[in] velocity The velocity the swarm gives a point, which places the particles.
1145 * @param[in,out] sums Adds this rank's flow rate [m^3/s], expected and added particles, and Psi added.
1146 * @return PetscErrorCode 0 on success.
1147 */
1148static PetscErrorCode InjectAtInlet(UserCtx *user, struct ParticleInlet *inlet, const ParticleFieldEvent *event,
1149 PetscInt64 first_id, const VelocityPointInterpolator *velocity,
1150 PetscReal sums[PARTICLE_INLET_SUMS])
1151{
1152 SimCtx *simCtx = user->simCtx;
1153 DM swarm = user->swarm;
1154 const SeedingCellTable *table = &inlet->table;
1155 const PetscInt nodes[3] = {user->info.mx, user->info.my, user->info.mz};
1156 const PetscInt normal = (PetscInt)inlet->face / 2, a = (normal == 0) ? 1 : 0, b = (normal == 2) ? 1 : 2;
1157 const PetscBool negative = (PetscBool)((PetscInt)inlet->face % 2 == 0);
1158 const PetscReal sign = negative ? 1.0 : -1.0, L_ref = simCtx->scaling.L_ref;
1159 const PetscInt64 seed = (PetscInt64)simCtx->particleRandomSeed;
1160 const PetscInt n_cells = table->n[0] * table->n[1] * table->n[2];
1161 const Vec metric_vec = (normal == 0) ? user->lCsi : (normal == 1) ? user->lEta : user->lZet;
1162 const Cmpnts ***coor = NULL, ***ucont = NULL, ***metric = NULL;
1163 const PetscReal ***diffusivity = NULL;
1164 PetscInt64 *cell_id = NULL;
1165 PetscInt *count = NULL;
1166 Cmpnts *travel = NULL, *inward = NULL, *normal_of = NULL;
1167 PetscReal *flux_bound = NULL;
1168 PetscInt first = 0, n_new = 0;
1169 PetscReal density = 0.0;
1170 ParticleFieldPlan *plan = inlet->plan ? inlet->plan : simCtx->particleFieldPlan;
1171
1172 PetscFunctionBeginUser;
1173 PROFILE_FUNCTION_BEGIN;
1174 PetscCall(PicurvExpressionEvaluate(inlet->particles_per_cell, &event->physical_time, NULL, &density));
1175 PetscCheck(!PetscIsInfOrNanReal(density) && density >= 0.0, PETSC_COMM_WORLD, PETSC_ERR_ARG_OUTOFRANGE,
1176 "The particles per cell of inlet %s is %g at t = %g; it must be finite and not negative.",
1177 BCFaceToString(inlet->face), (double)density, (double)event->physical_time);
1178 density *= inlet->cells_per_volume; /* particles per unit of the solver's volume */
1179
1180 // --- 1. How many enter each face cell: the mean's integer part, and one more by chance ---
1181 PetscCall(PetscMalloc5(n_cells, &count, n_cells, &cell_id, n_cells, &travel, n_cells, &normal_of, n_cells, &flux_bound));
1182 PetscCall(DMDAVecGetArrayRead(user->fda, user->lCoor, (void *)&coor));
1183 PetscCall(DMDAVecGetArrayRead(user->fda, user->lUcont, (void *)&ucont));
1184 PetscCall(DMDAVecGetArrayRead(user->fda, metric_vec, (void *)&metric));
1185 for (PetscInt c = 0; c < n_cells; ++c) {
1186 PetscInt origin[3];
1187 SeedingCellOrigin(table, c, origin);
1188 PetscInt f[3] = {origin[0] + 1, origin[1] + 1, origin[2] + 1};
1189 f[normal] = negative ? 0 : nodes[normal] - 2;
1190 const Cmpnts u = ucont[f[2]][f[1]][f[0]], m = metric[f[2]][f[1]][f[0]];
1191 const PetscReal flux = PetscMax(sign * ((normal == 0) ? u.x : (normal == 1) ? u.y : u.z), 0.0);
1192 const PetscReal expected = density * flux * simCtx->dt;
1193 const PetscReal whole = PetscFloorReal(expected);
1194 /* u_n dt along the inward unit normal: (flux / |m|) dt (sign m / |m|). */
1195 const PetscReal along_m = sign * flux * simCtx->dt / (m.x * m.x + m.y * m.y + m.z * m.z);
1196
1197 PetscCheck(whole < (PetscReal)PARTICLE_INLET_PIDS_PER_CELL, PETSC_COMM_SELF, PETSC_ERR_ARG_OUTOFRANGE,
1198 "Inlet %s would inject %g particles through one face cell in one step; the limit is %" PetscInt64_FMT
1199 ". Lower particles_per_cell or the time step.", BCFaceToString(inlet->face), (double)expected,
1200 PARTICLE_INLET_PIDS_PER_CELL - 1);
1201 cell_id[c] = first_id + PicurvFaceCellId(nodes, inlet->face, origin[a], origin[b]);
1202 count[c] = (PetscInt)whole;
1203 if (PicurvUnitFromHash(PicurvDrawHash(seed, PICURV_DRAW_SPACE_INLET, 0, cell_id[c], 0)) < expected - whole) count[c]++;
1204 travel[c].x = along_m * m.x; travel[c].y = along_m * m.y; travel[c].z = along_m * m.z;
1205 n_new += count[c];
1206 /* The flux through a face cell varies across it as the swarm's velocity does, and the
1207 * particles leaving the inlet layer leave a gap that varies the same way, so the new ones
1208 * are placed in proportion to the inward velocity. Each method's velocity on a face cell
1209 * is bilinear between the corners, edge midpoints and centre, so its largest value there
1210 * bounds it; the margin covers curved cells. */
1211 {
1212 const PetscReal norm_m = PetscSqrtReal(m.x * m.x + m.y * m.y + m.z * m.z);
1213 normal_of[c].x = sign * m.x / norm_m; normal_of[c].y = sign * m.y / norm_m; normal_of[c].z = sign * m.z / norm_m;
1214 flux_bound[c] = 0.0;
1215 for (PetscInt q = 0; count[c] > 0 && q < 9; ++q) {
1216 PetscReal lattice[3], logic[3];
1217 Cmpnts point, u_q;
1218 lattice[a] = 0.5 * (q % 3); lattice[b] = 0.5 * (q / 3); lattice[normal] = 0.0;
1219 SeedingLogicalPoint(lattice, (PetscInt)inlet->face, logic);
1220 PetscCall(MetricLogicalToPhysical(user, coor, origin[0], origin[1], origin[2], logic[0], logic[1], logic[2], &point));
1221 PetscCall(InterpolateVelocityAtPoint(velocity, origin, logic, point, &u_q));
1222 flux_bound[c] = PetscMax(flux_bound[c], u_q.x * normal_of[c].x + u_q.y * normal_of[c].y + u_q.z * normal_of[c].z);
1223 }
1224 flux_bound[c] *= 1.1;
1225 }
1226 sums[0] += flux * simCtx->scaling.U_ref * L_ref * L_ref;
1227 sums[1] += expected;
1228 }
1229 sums[2] += (PetscReal)n_new;
1230 PetscCall(DMDAVecRestoreArrayRead(user->fda, metric_vec, (void *)&metric));
1231 PetscCall(DMDAVecRestoreArrayRead(user->fda, user->lUcont, (void *)&ucont));
1232
1233 // --- 2. Where: on the face, by area and inward velocity, keyed on the particle's ID ---
1234 PetscCall(DMSwarmGetLocalSize(swarm, &first));
1235 PetscCall(PetscMalloc1(n_new, &inward));
1236 if (n_new > 0) {
1237 PetscReal *positions = NULL, *weights = NULL;
1238 PetscInt *cell = NULL, *status = NULL;
1239 PetscInt64 *pids = NULL;
1240 PetscInt p = first;
1241
1242 PetscCall(DMSwarmSetLocalSizes(swarm, first + n_new, -1));
1243 PetscCall(InitializeDefaultParticleFields(swarm, first, first + n_new));
1244 PetscCall(DMDAVecGetArrayRead(user->da, user->lDiffusivity, (void *)&diffusivity));
1245 PetscCall(DMSwarmGetField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_POSITION), NULL, NULL, (void **)&positions));
1246 PetscCall(DMSwarmGetField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_WEIGHT), NULL, NULL, (void **)&weights));
1247 PetscCall(DMSwarmGetField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_CELL_ID), NULL, NULL, (void **)&cell));
1248 PetscCall(DMSwarmGetField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_PID), NULL, NULL, (void **)&pids));
1249 PetscCall(DMSwarmGetField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_LOCATION_STATUS), NULL, NULL, (void **)&status));
1250 for (PetscInt c = 0; c < n_cells; ++c) {
1251 for (PetscInt k = 0; k < count[c]; ++k, ++p) {
1252 const PetscInt64 pid = PARTICLE_INLET_PID_BASE + cell_id[c] * PARTICLE_INLET_PIDS_PER_CELL + k;
1253 const PetscReal share = PicurvUnitFromHash(PicurvDrawHash(seed, PICURV_DRAW_SPACE_INLET, 1, pid, 0));
1254 const PetscBool weighted = (PetscBool)(flux_bound[c] > 0.0);
1255 PetscInt ci, cj, ck;
1256 PetscReal xi, eta, zta;
1257 Cmpnts point, kick, u_p;
1258
1259 /* Without any inward velocity on the face cell, its count still came from the
1260 * flux: those particles are spread by area, and travel at the cell's mean. */
1261 PetscCall(DrawSeedingPoint(user, coor, table, c, pid, weighted ? velocity : NULL, normal_of[c], flux_bound[c],
1262 &ci, &cj, &ck, &xi, &eta, &zta, &u_p));
1263 if (!weighted) { u_p.x = travel[c].x / simCtx->dt; u_p.y = travel[c].y / simCtx->dt; u_p.z = travel[c].z / simCtx->dt; }
1264 PetscCall(MetricLogicalToPhysical(user, coor, ci, cj, ck, xi, eta, zta, &point));
1265 positions[3 * p + 0] = point.x; positions[3 * p + 1] = point.y; positions[3 * p + 2] = point.z;
1266 weights[3 * p + 0] = xi; weights[3 * p + 1] = eta; weights[3 * p + 2] = zta;
1267 cell[3 * p + 0] = ci; cell[3 * p + 1] = cj; cell[3 * p + 2] = ck;
1268 pids[p] = pid;
1269 status[p] = NEEDS_LOCATION;
1270 /* This step's Brownian kick, as every other particle had. Without it they refill
1271 * the inlet layer that this share also fills, and the first cell runs dense. */
1272 PetscCall(CalculateBrownianDisplacement(user, pid, diffusivity[ck + 1][cj + 1][ci + 1], &kick));
1273 inward[p - first].x = share * u_p.x * simCtx->dt + kick.x;
1274 inward[p - first].y = share * u_p.y * simCtx->dt + kick.y;
1275 inward[p - first].z = share * u_p.z * simCtx->dt + kick.z;
1276 LOG_PARTICLE(simCtx, pid, "inject", "through %s on rank %d at (%.6e, %.6e, %.6e): cell (%" PetscInt_FMT ",%" PetscInt_FMT ",%" PetscInt_FMT ")\n",
1277 BCFaceToString(inlet->face), (int)simCtx->rank, point.x, point.y, point.z, ci, cj, ck);
1278 }
1279 }
1280 PetscCall(DMSwarmRestoreField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_LOCATION_STATUS), NULL, NULL, (void **)&status));
1281 PetscCall(DMSwarmRestoreField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_PID), NULL, NULL, (void **)&pids));
1282 PetscCall(DMSwarmRestoreField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_CELL_ID), NULL, NULL, (void **)&cell));
1283 PetscCall(DMSwarmRestoreField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_WEIGHT), NULL, NULL, (void **)&weights));
1284 PetscCall(DMSwarmRestoreField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_POSITION), NULL, NULL, (void **)&positions));
1285 PetscCall(DMDAVecRestoreArrayRead(user->da, user->lDiffusivity, (void *)&diffusivity));
1286 }
1287 PetscCall(DMDAVecRestoreArrayRead(user->fda, user->lCoor, (void *)&coor));
1288
1289 // --- 3. What it carries, set where it crossed the face. Every rank calls this: see the plan. ---
1290 PetscCall(ParticleFieldPlanApply(user, plan, event, first, first + n_new));
1291
1292 // --- 4. How far it has come since; the search that follows finds its cell ---
1293 if (n_new > 0) {
1294 PetscReal *positions = NULL, *psi = NULL;
1295
1296 PetscCall(DMSwarmGetField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_POSITION), NULL, NULL, (void **)&positions));
1297 PetscCall(DMSwarmGetField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_PSI), NULL, NULL, (void **)&psi));
1298 for (PetscInt q = 0; q < n_new; ++q) {
1299 positions[3 * (first + q) + 0] += inward[q].x;
1300 positions[3 * (first + q) + 1] += inward[q].y;
1301 positions[3 * (first + q) + 2] += inward[q].z;

1315 * @see InjectParticlesAtInlets()
1316 */
1317PetscErrorCode InjectParticlesAtInlets(UserCtx *user)
1318{
1319 SimCtx *simCtx = user->simCtx;
1320 PetscReal local[PARTICLE_INLET_FACE_COUNT][PARTICLE_INLET_SUMS] = {{0.0}};
1321 PetscReal global[PARTICLE_INLET_FACE_COUNT][PARTICLE_INLET_SUMS];
1322 ParticleFieldEvent event = {0.0, 0};
1323 PetscInt64 faces = 0, first_id;
1324 VelocityPointInterpolator *velocity = NULL;
1325
1326 PetscFunctionBeginUser;
1327 PROFILE_FUNCTION_BEGIN;
1328 if (simCtx->particleInletCount == 0) { PROFILE_FUNCTION_END; PetscFunctionReturn(0); }
1329 {
1330 const PetscInt64 cells[3] = {user->info.mx - 2, user->info.my - 2, user->info.mz - 2};
1331 faces = 2 * (cells[0] * cells[1] + cells[1] * cells[2] + cells[2] * cells[0]);
1332 }
1333 first_id = (PetscInt64)simCtx->step * faces;
1334 PetscCheck((__int128)(simCtx->step + 1) * faces * PARTICLE_INLET_PIDS_PER_CELL + PARTICLE_INLET_PID_BASE < (__int128)PARTICLE_PID_LIMIT,
1335 PETSC_COMM_WORLD, PETSC_ERR_ARG_OUTOFRANGE,
1336 "Step %" PetscInt_FMT " on a grid with %" PetscInt64_FMT " boundary face cells passes the last particle ID a "
1337 "checkpoint can store (2^53).", simCtx->step, faces);
1338 PetscCall(PicurvPhysicalTime(simCtx, simCtx->ti, &event.physical_time));
1339 PetscCall(InterpolateVelocityAtPointBegin(user, &velocity));
1340
1341 PetscReal cells_per_volume = 0.0;
1342 PetscBool have_cells_per_volume = PETSC_FALSE; /* computed when an inlet first needs it */
1343 for (PetscInt n = 0; n < simCtx->particleInletCount; ++n) {
1344 struct ParticleInlet *inlet = &simCtx->particleInlets[n];
1345
1346 if (!inlet->table_built) {
1347 const Cmpnts ***coor = NULL;
1348 PetscCheck(user->boundary_faces[inlet->face].mathematical_type == INLET, PETSC_COMM_WORLD, PETSC_ERR_ARG_WRONG,
1349 "Face %s injects particles but is not an INLET.", BCFaceToString(inlet->face));
1350 PetscCall(DMDAVecGetArrayRead(user->fda, user->lCoor, (void *)&coor));
1351 PetscCall(BuildSeedingCellTable(user, coor, (PetscInt)inlet->face, 0, PETSC_TRUE, &inlet->table));
1352 PetscCall(DMDAVecRestoreArrayRead(user->fda, user->lCoor, (void *)&coor));
1353 if (!have_cells_per_volume) {
1354 /* particles_per_cell is a mean over the grid: cells over volume makes it a density.
1355 The volume is the sum of the cells' 1/Aj, not the Jacobian integral the
1356 seeding table uses. table_built is the same on every rank, so all reduce. */
1357 const DMDALocalInfo *info = &user->info;
1358 const PetscReal ***aj = NULL;
1359 PetscReal volume = 0.0;
1360 PetscCall(DMDAVecGetArrayRead(user->da, user->Aj, (void *)&aj));
1361 for (PetscInt k = PetscMax(info->zs, 1); k < PetscMin(info->zs + info->zm, info->mz - 1); ++k)
1362 for (PetscInt j = PetscMax(info->ys, 1); j < PetscMin(info->ys + info->ym, info->my - 1); ++j)
1363 for (PetscInt i = PetscMax(info->xs, 1); i < PetscMin(info->xs + info->xm, info->mx - 1); ++i)
1364 volume += 1.0 / aj[k][j][i];
1365 PetscCall(DMDAVecRestoreArrayRead(user->da, user->Aj, (void *)&aj));
1366 PetscCallMPI(MPI_Allreduce(MPI_IN_PLACE, &volume, 1, MPIU_REAL, MPI_SUM, PETSC_COMM_WORLD));
1367 cells_per_volume = (PetscReal)(info->mx - 2) * (info->my - 2) * (info->mz - 2) / volume;
1368 have_cells_per_volume = PETSC_TRUE;
1369 }
1370 inlet->cells_per_volume = cells_per_volume;
1371 inlet->table_built = PETSC_TRUE;
1372 }
1373 PetscCall(InjectAtInlet(user, inlet, &event, first_id, velocity, local[n]));
1374 }
1375 PetscCall(InterpolateVelocityAtPointEnd(&velocity));
1376
1377 PetscCallMPI(MPI_Allreduce(local, global, PARTICLE_INLET_FACE_COUNT * PARTICLE_INLET_SUMS, MPIU_REAL, MPI_SUM, PETSC_COMM_WORLD));
1378 if (simCtx->rank == 0) {
1379 FILE *file = NULL;
1380
1381 PetscCall(PicurvOpenDiagnosticsCsv(simCtx, simCtx->analysis_dir, "particle_injection.csv",
1382 "step,physical_time,face,flow_rate,expected,injected,psi", PETSC_TRUE, &file));
1383 for (PetscInt n = 0; n < simCtx->particleInletCount; ++n) {
1384 fprintf(file, "%d,%.10e,%s,%.10e,%.10e,%.0f,%.10e\n", (int)simCtx->step, (double)event.physical_time,
1385 BCFaceToString(simCtx->particleInlets[n].face), (double)global[n][0], (double)global[n][1],
1386 (double)global[n][2], (double)global[n][3]);
1387 }
1388 PetscCheck(fclose(file) == 0, PETSC_COMM_SELF, PETSC_ERR_FILE_WRITE, "Unable to close the particle-injection diagnostics file.");
1389 }
1390 {
1391 PetscInt n_local = 0;
1392 PetscInt64 local_total, total = 0;
1393 char object[128];
1394
1395 PetscCall(DMSwarmGetLocalSize(user->swarm, &n_local));
1396 local_total = n_local;
1397 PetscCallMPI(MPI_Allreduce(&local_total, &total, 1, MPIU_INT64, MPI_SUM, PETSC_COMM_WORLD));
1398 PetscCall(PetscSNPrintf(object, sizeof(object), "the %" PetscInt64_FMT " particles after injection (three components each)", total));
1399 PetscCall(PicurvCheckPetscIntRange(PETSC_COMM_WORLD, 3 * total, object));
1400 }
1401 PROFILE_FUNCTION_END;
1402 PetscFunctionReturn(0);
1403}
