82/**
83 * @brief The owned cells a seeding mode places particles in, and which particles each gets.
84 * @details Volumetric seeding uses the rank's owned cells and surface seeding the owned cells
85 * that touch the inlet face, so both are a box of cells from `origin` with `n` cells
86 * per axis (none when the rank owns no part of the target). Cell `c`, in i-fastest
87 * order, seeds the particles with IDs `pid_lo[c]` to `pid_hi[c] - 1`.
88 */
89typedef struct {
90 PetscInt face; /**< -1 for the volume, otherwise the BCFace the points lie on */
91 PetscInt origin[3]; /**< Origin node of the box's first cell */
92 PetscInt n[3]; /**< Cells per axis */
93 PetscInt64 *pid_lo; /**< First particle ID of each cell */
94 PetscInt64 *pid_hi; /**< One past the last particle ID of each cell */
95 PetscReal *bound; /**< Largest measure density in each cell (NULL when not built) */
96 PetscInt count; /**< Particles this rank seeds */
97} SeedingCellTable;

99/**
100 * @brief Measure density at a logical point: the Jacobian determinant for the volume, or the
101 * area element |dx/ds x dx/dt| across the face for a face, with the normal coordinate
102 * on the face itself.
103 */
104static PetscErrorCode SeedingMeasureDensity(UserCtx *user, const Cmpnts ***coor, PetscInt face,
105 PetscInt i, PetscInt j, PetscInt k, const PetscReal logic[3],
106 PetscReal *density)
107{
108 PetscReal J[3][3], detJ, l[3] = {logic[0], logic[1], logic[2]};
109
110 PetscFunctionBeginUser;
111 PROFILE_HELPER_BEGIN;
112 if (face < 0) {
113 PetscCall(MetricJacobian(user, coor, i, j, k, l[0], l[1], l[2], J, &detJ));
114 *density = detJ;
115 } else {
116 const PetscInt a = face / 2, b = (a + 1) % 3, c = (a + 2) % 3;
117 Cmpnts cross;
118 l[a] = (face % 2) ? 1.0 : 0.0;
119 PetscCall(MetricJacobian(user, coor, i, j, k, l[0], l[1], l[2], J, NULL));
120 cross.x = J[1][b] * J[2][c] - J[2][b] * J[1][c];
121 cross.y = J[2][b] * J[0][c] - J[0][b] * J[2][c];
122 cross.z = J[0][b] * J[1][c] - J[1][b] * J[0][c];
123 *density = PetscSqrtReal(cross.x * cross.x + cross.y * cross.y + cross.z * cross.z);
124 }
125 PROFILE_FUNCTION_END;
126 PetscFunctionReturn(0);
127}

129/**
130 * @brief Decide which particles each owned cell seeds, the same way at every rank layout.
131 * @details Collective. The cells are those volumetric (`face` < 0) or surface seeding uses,
132 * and each cell's measure is its volume or its area on the face, integrated with 2x2x2
133 * Gauss points: exact for the volume (the determinant has degree 2 in each
134 * coordinate) and for the area of a planar face, where the points coincide in pairs
135 * and give the 2x2 rule.
136 *
137 * Particles are stratified over the measure in global cell order (i fastest): with
138 * P(c) the measure of all cells before cell c, T the total and N the particle count,
139 * cell c seeds the IDs floor(N P(c) / T) to floor(N (P(c) + m_c) / T) - 1, so each
140 * cell holds its expected share to within one, the IDs are 0..N-1, and both depend
141 * only on the grid. For the result to be identical at any rank layout the measures
142 * are summed as integers, each cell's measure quantised against the largest cell
143 * (an exact maximum) to 2^Q units, with Q chosen so the total fits 63 bits. P(c) is
144 * the sum of the (j,k) rows before the cell's row, from one reduction over the rows,
145 * plus the cells before it in its row, from a scan over the ranks sharing the row.
146 * @param[in] user Grid context.
147 * @param[in] coor Ghosted node-coordinate array.
148 * @param[in] face -1 for the volume, otherwise the BCFace to seed on.
149 * @param[in] numParticles Global particle count.
150 * @param[in] build_bound PETSC_TRUE to also tabulate each cell's largest measure density on
151 * a 3x3x3 lattice, the envelope DrawSeedingPoint samples against.
152 * @param[out] table The owned cells and their particles; free with FreeSeedingCellTable.
153 * @return PetscErrorCode 0 on success; an error for a cell whose measure is not positive.
154 */
155static PetscErrorCode BuildSeedingCellTable(UserCtx *user, const Cmpnts ***coor, PetscInt face,
156 PetscInt numParticles, PetscBool build_bound,
157 SeedingCellTable *table)
158{
159 const PetscReal gauss[2] = {0.5 - 0.5 / PetscSqrtReal(3.0), 0.5 + 0.5 / PetscSqrtReal(3.0)};
160 const PetscInt nodes[3] = {user->info.mx, user->info.my, user->info.mz};
161 PetscInt start, n_cells, n_rows_local, global_cells[3], global_origin[3] = {0, 0, 0}, quantum_bits = 62;
162 PetscInt64 n_global_cells;
163 PetscReal local_max = 0.0, global_max = 0.0;
164 PetscReal *measure = NULL;
165 uint64_t *rows = NULL, *row_before = NULL, *segment = NULL, *segment_before = NULL, *quanta = NULL;
166 uint64_t total = 0;
167 MPI_Comm row_comm = MPI_COMM_NULL;
168 PetscMPIInt row_rank = 0;
169
170 PetscFunctionBeginUser;
171 PROFILE_FUNCTION_BEGIN;
172 PetscCall(PetscMemzero(table, sizeof(*table)));
173 table->face = face;
174 table->origin[0] = user->info.xs; table->origin[1] = user->info.ys; table->origin[2] = user->info.zs;
175 PetscCall(GetOwnedCellRange(&user->info, 0, &start, &table->n[0]));
176 PetscCall(GetOwnedCellRange(&user->info, 1, &start, &table->n[1]));
177 PetscCall(GetOwnedCellRange(&user->info, 2, &start, &table->n[2]));
178 for (PetscInt d = 0; d < 3; ++d) global_cells[d] = nodes[d] - 2;
179 if (face >= 0) {
180 const PetscInt a = face / 2;
181 PetscBool holds_face = PETSC_FALSE;
182 PetscCall(CanRankServiceInletFace(user, &user->info, nodes[0] - 1, nodes[1] - 1, nodes[2] - 1, &holds_face));
183 global_cells[a] = 1;
184 if (face % 2) global_origin[a] = nodes[a] - 3; /* origin node of the last cell */
185 if (holds_face) {
186 table->n[a] = 1;
187 table->origin[a] = global_origin[a];
188 } else {
189 table->n[0] = table->n[1] = table->n[2] = 0;
190 }
191 }
192 n_cells = table->n[0] * table->n[1] * table->n[2];
193 n_rows_local = table->n[1] * table->n[2];
194 n_global_cells = (PetscInt64)global_cells[0] * global_cells[1] * global_cells[2];
195 while (quantum_bits > 0 && ((PetscInt64)1 << (62 - quantum_bits)) < n_global_cells) --quantum_bits;
196
197 // --- 1. Cell measures, and the density envelope ---
198 PetscCall(PetscMalloc2(n_cells, &measure, n_cells, &quanta));
199 PetscCall(PetscMalloc2(n_cells, &table->pid_lo, n_cells, &table->pid_hi));
200 if (build_bound) PetscCall(PetscMalloc1(n_cells, &table->bound));
201 for (PetscInt c = 0; c < n_cells; ++c) {
202 const PetscInt i = table->origin[0] + c % table->n[0];
203 const PetscInt j = table->origin[1] + (c / table->n[0]) % table->n[1];
204 const PetscInt k = table->origin[2] + c / (table->n[0] * table->n[1]);
205 PetscReal density;
206 measure[c] = 0.0;
207 for (PetscInt g = 0; g < 8; ++g) {
208 const PetscReal l[3] = {gauss[g & 1], gauss[(g >> 1) & 1], gauss[(g >> 2) & 1]};
209 PetscCall(SeedingMeasureDensity(user, coor, face, i, j, k, l, &density));
210 measure[c] += 0.125 * density;
211 }
212 PetscCheck(measure[c] > 0.0, PETSC_COMM_SELF, PETSC_ERR_ARG_WRONGSTATE,
213 "Cell with origin node (%d,%d,%d) has %s %g; particle seeding needs positive %s.",
214 (int)i, (int)j, (int)k, face < 0 ? "volume" : "inlet area", (double)measure[c],
215 face < 0 ? "volumes" : "areas");
216 local_max = PetscMax(local_max, measure[c]);
217 if (build_bound) {
218 table->bound[c] = 0.0;
219 for (PetscInt q = 0; q < 27; ++q) {
220 const PetscReal l[3] = {0.5 * (q % 3), 0.5 * ((q / 3) % 3), 0.5 * (q / 9)};
221 PetscCall(SeedingMeasureDensity(user, coor, face, i, j, k, l, &density));
222 table->bound[c] = PetscMax(table->bound[c], density);
223 }
224 }
225 }
226 {
227 PetscInt64 local_cells = n_cells, owned_cells = 0;
228 PetscCallMPI(MPI_Allreduce(&local_cells, &owned_cells, 1, MPIU_INT64, MPI_SUM, PETSC_COMM_WORLD));
229 PetscCheck(owned_cells == n_global_cells, PETSC_COMM_WORLD, PETSC_ERR_PLIB,
230 "Seeding cells owned across ranks (%" PetscInt64_FMT ") differ from the global count (%" PetscInt64_FMT ").",
231 owned_cells, n_global_cells);
232 }
233 PetscCallMPI(MPI_Allreduce(&local_max, &global_max, 1, MPIU_REAL, MPI_MAX, PETSC_COMM_WORLD));
234 PetscCheck(global_max > 0.0, PETSC_COMM_WORLD, PETSC_ERR_ARG_WRONGSTATE,
235 face < 0 ? "Volumetric particle seeding found no owned volume on any rank."
236 : "Surface particle seeding found no inlet area on any rank; it needs an INLET face.");
237
238 // --- 2. Integer measures, summed over rows and over each rank's segment of a row ---
239 const PetscInt n_rows = global_cells[1] * global_cells[2];
240 PetscCall(PetscCalloc4(n_rows, &rows, n_rows, &row_before, n_rows_local, &segment, n_rows_local, &segment_before));
241 for (PetscInt c = 0; c < n_cells; ++c) {
242 const PetscInt lj = (c / table->n[0]) % table->n[1], lk = c / (table->n[0] * table->n[1]);
243 const PetscInt row = (table->origin[1] + lj - global_origin[1]) + global_cells[1] * (table->origin[2] + lk - global_origin[2]);
244 quanta[c] = (uint64_t)PetscMax(1.0, PetscFloorReal(ldexp((double)(measure[c] / global_max), (int)quantum_bits) + 0.5));
245 rows[row] += quanta[c];
246 segment[lj + table->n[1] * lk] += quanta[c];
247 }
248 PetscCallMPI(MPI_Allreduce(MPI_IN_PLACE, rows, (PetscMPIInt)n_rows, MPI_UINT64_T, MPI_SUM, PETSC_COMM_WORLD));
249 for (PetscInt r = 0; r < n_rows; ++r) { row_before[r] = total; total += rows[r]; }
250 // Ranks with the same j and k ranges share rows; their order along i is the order in the row.
251 PetscCallMPI(MPI_Comm_split(PETSC_COMM_WORLD, n_cells > 0 ? (PetscMPIInt)(table->origin[1] + nodes[1] * table->origin[2]) : MPI_UNDEFINED,
252 (PetscMPIInt)table->origin[0], &row_comm));
253 if (row_comm != MPI_COMM_NULL) {
254 PetscCallMPI(MPI_Exscan(segment, segment_before, (PetscMPIInt)n_rows_local, MPI_UINT64_T, MPI_SUM, row_comm));
255 PetscCallMPI(MPI_Comm_rank(row_comm, &row_rank));
256 if (row_rank == 0) PetscCall(PetscArrayzero(segment_before, n_rows_local)); /* MPI_Exscan leaves rank 0's result undefined */
257 PetscCallMPI(MPI_Comm_free(&row_comm));
258 }
259
260 // --- 3. Stratified particle IDs: floor(N P / T) at each cell boundary ---
261 for (PetscInt lrow = 0; lrow < n_rows_local; ++lrow) {
262 const PetscInt lj = lrow % table->n[1], lk = lrow / table->n[1];
263 const PetscInt row = (table->origin[1] + lj - global_origin[1]) + global_cells[1] * (table->origin[2] + lk - global_origin[2]);
264 uint64_t before = row_before[row] + segment_before[lrow];
265 for (PetscInt li = 0; li < table->n[0]; ++li) {
266 const PetscInt c = li + table->n[0] * lrow;
267 table->pid_lo[c] = (PetscInt64)(((unsigned __int128)numParticles * before) / total);
268 before += quanta[c];
269 table->pid_hi[c] = (PetscInt64)(((unsigned __int128)numParticles * before) / total);
270 table->count += (PetscInt)(table->pid_hi[c] - table->pid_lo[c]);
271 }
272 }
273 PetscCall(PetscFree4(rows, row_before, segment, segment_before));
274 PetscCall(PetscFree2(measure, quanta));
275 PROFILE_FUNCTION_END;
276 PetscFunctionReturn(0);
277}

290/**
291 * @brief Draw one particle's seeding point inside its cell, uniform in the cell's measure.
292 * @details Uniform logical coordinates are accepted with probability density / `bound`. Every
293 * draw is keyed on the run seed, the particle ID and the attempt, so the point depends
294 * only on the particle, not on the rank that seeds it. A surface point is then put just
295 * inside the cell, 1e-6 of the cell from the face.
296 * @param[in] user Grid context.
297 * @param[in] coor Ghosted node-coordinate array.
298 * @param[in] table A table built with `build_bound` set.
299 * @param[in] c The cell, an index into the table.
300 * @param[in] pid The particle's ID.
301 * @param[out] ci_metric_lnode_out,cj_metric_lnode_out,ck_metric_lnode_out Origin node of the cell.
302 * @param[out] xi_metric_logic_out,eta_metric_logic_out,zta_metric_logic_out Logical point in it.
303 * @return PetscErrorCode 0 on success.
304 */
305static PetscErrorCode DrawSeedingPoint(
306 UserCtx *user, const Cmpnts ***coor, const SeedingCellTable *table, PetscInt c, PetscInt64 pid,
307 PetscInt *ci_metric_lnode_out, PetscInt *cj_metric_lnode_out, PetscInt *ck_metric_lnode_out,
308 PetscReal *xi_metric_logic_out, PetscReal *eta_metric_logic_out, PetscReal *zta_metric_logic_out)
309{
310 PetscReal logic[3], u[4];
311
312 PetscFunctionBeginUser;
313 PROFILE_HELPER_BEGIN;
314 *ci_metric_lnode_out = table->origin[0] + c % table->n[0];
315 *cj_metric_lnode_out = table->origin[1] + (c / table->n[0]) % table->n[1];
316 *ck_metric_lnode_out = table->origin[2] + c / (table->n[0] * table->n[1]);
317
318 // Accept uniform logicals with probability density / bound; a cell whose density varies
319 // little accepts almost every draw. The cap only guards a degenerate envelope.
320 for (PetscInt attempt = 0; attempt < 1000; ++attempt) {
321 PetscReal density;
322 uint64_t hash = PicurvParticleDrawHash(user->simCtx->particleRandomSeed, PICURV_DRAW_SPACE_PLACEMENT, 0, pid, attempt);
323 for (int w = 0; w < 4; ++w) u[w] = PicurvUnitFromHash(hash = PicurvMix64(hash ^ (uint64_t)w));
324 // Ensure logical coordinates are strictly within [0,1) for robustness with MetricLogicalToPhysical
325 for (PetscInt d = 0; d < 3; ++d) logic[d] = PetscMax(PetscMin(u[d], 1.0 - 1.0e-7), 0.0);
326 PetscCall(SeedingMeasureDensity(user, coor, table->face, *ci_metric_lnode_out, *cj_metric_lnode_out,
327 *ck_metric_lnode_out, logic, &density));
328 if (u[3] * table->bound[c] <= density) break;
329 if (attempt == 999) {
330 LOG_ALLOW(LOCAL, LOG_WARNING, "Rank %d: 1000 rejected draws in cell (%" PetscInt_FMT ",%" PetscInt_FMT ",%" PetscInt_FMT "); accepting the last one.\n",
331 user->simCtx->rank, *ci_metric_lnode_out, *cj_metric_lnode_out, *ck_metric_lnode_out);
332 }
333 }
334 if (table->face >= 0) logic[table->face / 2] = (table->face % 2) ? 1.0 - 1.0e-6 : 1.0e-6;
335 *xi_metric_logic_out = logic[0]; *eta_metric_logic_out = logic[1]; *zta_metric_logic_out = logic[2];
336
337 PROFILE_FUNCTION_END;
338 PetscFunctionReturn(0);
339}

342/**
343 * @brief Initialize position-independent particle fields after a particle is created.
344 */
345static PetscErrorCode InitializeParticleBasicProperties(UserCtx *user,
346 PetscInt particlesPerProcess,
347 BoundingBox *bboxlist) // bboxlist unused for placement
348{
349 PetscErrorCode ierr;
350 (void)bboxlist;
351 DM swarm = user->swarm;
352 PetscReal *positions_field = NULL; // Pointer to swarm field for physical positions (x,y,z)
353 PetscInt64 *particleIDs = NULL; // Pointer to swarm field for Particle IDs
354 PetscInt *cellIDs_petsc = NULL; // Pointer to swarm field for DMSwarm_CellID (i,j,k of containing cell)
355 PetscInt *status_field = NULL; // Pointer to swarm field for DMSwarm_location_status(NEEDS_LOCATION etc)
356 PetscMPIInt rank,size; // MPI rank of the current process, and total number of ranks.
357 const Cmpnts ***coor_nodes_local_array; // Read-only access to local node coordinates (from user->da)
358 Vec Coor_local; // Local vector for node coordinates
359 // Owned cells and the particles each seeds, for volumetric and surface seeding
360 SeedingCellTable seed_table;
361
362 PetscFunctionBeginUser;
363
364 PROFILE_FUNCTION_BEGIN;
365
366 SimCtx *simCtx = user->simCtx;
367
368 // --- 1. Input Validation and Basic Setup ---
369 if (!user) SETERRQ(PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "Null user.");
370 rank = simCtx->rank;
371 size = simCtx->size;
372
373 // Get DMDA information for the node-centered coordinate grid (user->da)
374 Coor_local = user->lCoor;
375 if (!Coor_local) SETERRQ(PETSC_COMM_SELF, PETSC_ERR_PLIB, "user->lCoor is NULL: grid coordinates are not set up.");
376 ierr = DMDAVecGetArrayRead(user->fda, Coor_local, (void*)&coor_nodes_local_array); CHKERRQ(ierr);
377 ierr = PetscMemzero(&seed_table, sizeof(seed_table)); CHKERRQ(ierr);
378
379 LOG_ALLOW(LOCAL, LOG_INFO, "Rank %d: Initializing %" PetscInt_FMT " particles. Mode: %s.\n",
380 rank, particlesPerProcess, ParticleInitializationToString(simCtx->ParticleInitialization));
381
382 // --- 2. Seeding Table for Volumetric and Surface Initialization ---
383 // Volume seeds the owned cells, Surface the owned cells on the inlet face; the table names
384 // each cell's particles, the same split CreateParticleSwarm sized the swarm with.
385 const PetscBool seeds_from_table = (PetscBool)(simCtx->ParticleInitialization == PARTICLE_INIT_VOLUME ||
386 simCtx->ParticleInitialization == PARTICLE_INIT_SURFACE_RANDOM);
387 PetscInt seed_cell = 0; // Cell the next table-seeded particle goes in
388 PetscInt64 seed_pid = 0; // That particle's ID
389 if (seeds_from_table) {
390 const PetscInt face = (simCtx->ParticleInitialization == PARTICLE_INIT_SURFACE_RANDOM) ? (PetscInt)user->identifiedInletBCFace : -1;
391 ierr = BuildSeedingCellTable(user, coor_nodes_local_array, face, simCtx->np, PETSC_TRUE, &seed_table); CHKERRQ(ierr);
392 PetscCheck(seed_table.count == particlesPerProcess, PETSC_COMM_SELF, PETSC_ERR_PLIB,
393 "Seeding table holds %" PetscInt_FMT " particles but the swarm was sized for %" PetscInt_FMT ".",
394 seed_table.count, particlesPerProcess);
395 if (seed_table.n[0] * seed_table.n[1] * seed_table.n[2] > 0) seed_pid = seed_table.pid_lo[0];
396 LOG_ALLOW(LOCAL, LOG_INFO, "Rank %d: Seeding %" PetscInt_FMT " particles stratified over %" PetscInt_FMT " owned cells by %s.\n",
397 rank, particlesPerProcess, seed_table.n[0] * seed_table.n[1] * seed_table.n[2],
398 face < 0 ? "volume" : "inlet area");
399 }
400
401 // --- 3. Get Access to Swarm Fields ---
402 ierr = DMSwarmGetField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_POSITION), NULL, NULL, (void**)&positions_field); CHKERRQ(ierr);
403 ierr = DMSwarmGetField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_PID), NULL, NULL, (void**)&particleIDs); CHKERRQ(ierr);
404 ierr = DMSwarmGetField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_CELL_ID), NULL, NULL, (void**)&cellIDs_petsc); CHKERRQ(ierr);
405 ierr = DMSwarmGetField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_LOCATION_STATUS),NULL,NULL,(void**)&status_field); CHKERRQ(ierr);
406
407 // --- 4. Determine Starting Global PID for this Rank (point source) ---
408 // Point-source PIDs are consecutive across ranks in rank order, so each rank starts where
409 // the ranks before it end. Table seeding takes each particle's ID from its cell instead.
410 PetscInt base_pid_for_rank = 0;
411 {
412 PetscInt local_count = particlesPerProcess;
413 ierr = MPI_Exscan(&local_count, &base_pid_for_rank, 1, MPIU_INT, MPI_SUM, PETSC_COMM_WORLD); CHKERRMPI(ierr);
414 if (rank == 0) base_pid_for_rank = 0; /* MPI_Exscan leaves rank 0's result undefined */
415 }
416
417 // --- 4b. Locate the Point Source Once ---
418 // Every rank places its share at the same point, so its cell is found by one search per rank;
419 // the lowest rank that holds it shares the cell, and settlement then routes the particles there.
420 PetscInt point_cell[3] = {-1, -1, -1};
421 if (simCtx->ParticleInitialization == PARTICLE_INIT_POINT_SOURCE) {
422 Particle probe;
423 ParticleLocationStatus probe_status = NEEDS_LOCATION;
424 PetscMPIInt candidate, owner = size;
425 ierr = PetscMemzero(&probe, sizeof(probe)); CHKERRQ(ierr);
426 probe.PID = -1;
427 probe.cell[0] = -1; probe.cell[1] = -1; probe.cell[2] = -1;
428 probe.loc.x = simCtx->psrc_x; probe.loc.y = simCtx->psrc_y; probe.loc.z = simCtx->psrc_z;
429 ierr = LocateParticleOrFindMigrationTarget(user, NULL, &probe, &probe_status); CHKERRQ(ierr);
430 candidate = (probe_status == ACTIVE_AND_LOCATED) ? rank : size;
431 ierr = MPI_Allreduce(&candidate, &owner, 1, MPI_INT, MPI_MIN, PETSC_COMM_WORLD); CHKERRMPI(ierr);
432 if (owner < size) {
433 if (rank == owner) { point_cell[0] = probe.cell[0]; point_cell[1] = probe.cell[1]; point_cell[2] = probe.cell[2]; }
434 ierr = MPI_Bcast(point_cell, 3, MPIU_INT, owner, PETSC_COMM_WORLD); CHKERRMPI(ierr);
435 LOG_ALLOW(GLOBAL, LOG_INFO, "Point source (%.6e, %.6e, %.6e) lies in cell (%" PetscInt_FMT ",%" PetscInt_FMT ",%" PetscInt_FMT ") on rank %d.\n",
436 simCtx->psrc_x, simCtx->psrc_y, simCtx->psrc_z, point_cell[0], point_cell[1], point_cell[2], owner);
437 } else {
438 LOG_ALLOW(GLOBAL, LOG_WARNING, "Point source (%.6e, %.6e, %.6e) was not located on any rank; its particles will be searched for.\n",
439 simCtx->psrc_x, simCtx->psrc_y, simCtx->psrc_z);
440 }
441 }
442
443 // --- 5. Loop Over Particles to Initialize ---
444 for (PetscInt p = 0; p < particlesPerProcess; p++) {
445 PetscInt ci_metric_lnode, cj_metric_lnode, ck_metric_lnode;
446 PetscReal xi_metric_logic = 0.0, eta_metric_logic = 0.0, zta_metric_logic = 0.0;
447 Cmpnts phys_coords = {0.0, 0.0, 0.0};
448 PetscBool particle_placed_by_this_rank = PETSC_FALSE;
449 PetscInt64 pid = (PetscInt64)base_pid_for_rank + p;
450
451 if (seeds_from_table) { // --- 5.a. Volumetric or Surface Initialization ---
452 while (seed_pid == seed_table.pid_hi[seed_cell]) seed_pid = seed_table.pid_lo[++seed_cell]; // skip filled (or empty) cells
453 pid = seed_pid++;
454 ierr = DrawSeedingPoint(user, coor_nodes_local_array, &seed_table, seed_cell, pid,
455 &ci_metric_lnode, &cj_metric_lnode, &ck_metric_lnode,
456 &xi_metric_logic, &eta_metric_logic, &zta_metric_logic); CHKERRQ(ierr);
457 ierr = MetricLogicalToPhysical(user, coor_nodes_local_array,
458 ci_metric_lnode, cj_metric_lnode, ck_metric_lnode,
459 xi_metric_logic, eta_metric_logic, zta_metric_logic,
460 &phys_coords); CHKERRQ(ierr);
461 particle_placed_by_this_rank = PETSC_TRUE;
462 }else if(simCtx->ParticleInitialization == PARTICLE_INIT_POINT_SOURCE){ // --- 5.c. Point Source Initialization ---
463 // All particles placed at the user-specified fixed point (psrc_x, psrc_y, psrc_z).
464 // No random number generation or logical-to-physical conversion needed.
465 phys_coords.x = simCtx->psrc_x;
466 phys_coords.y = simCtx->psrc_y;
467 phys_coords.z = simCtx->psrc_z;
468 ci_metric_lnode = point_cell[0]; cj_metric_lnode = point_cell[1]; ck_metric_lnode = point_cell[2];
469 particle_placed_by_this_rank = (PetscBool)(point_cell[0] >= 0);
470 }else {
471 SETERRQ(PETSC_COMM_SELF, PETSC_ERR_ARG_OUTOFRANGE, "Unknown ParticleInitialization mode %d.", simCtx->ParticleInitialization);
472 }
473
474 // --- 5.c. Store Particle Properties ---
475 positions_field[3*p+0] = phys_coords.x;
476 positions_field[3*p+1] = phys_coords.y;
477 positions_field[3*p+2] = phys_coords.z;
478
479 particleIDs[p] = pid;
480 // Start the first settlement's walk in the cell the particle was placed in; a point-source
481 // particle no rank located has no cell and is located by search.
482 if (particle_placed_by_this_rank) {
483 cellIDs_petsc[3*p+0] = ci_metric_lnode; cellIDs_petsc[3*p+1] = cj_metric_lnode; cellIDs_petsc[3*p+2] = ck_metric_lnode;
484 } else {
485 cellIDs_petsc[3*p+0] = -1; cellIDs_petsc[3*p+1] = -1; cellIDs_petsc[3*p+2] = -1;
486 }
487 status_field[p] = UNINITIALIZED;
488
489 // --- 5.d. The particle's seed line ---
490 if (seeds_from_table) {
491 LOG_PARTICLE(simCtx, particleIDs[p], "seed",
492 "%s placement on rank %d at (%.6e, %.6e, %.6e): cell (%" PetscInt_FMT ",%" PetscInt_FMT ",%" PetscInt_FMT ") logical (%.4f, %.4f, %.4f)\n",
493 ParticleInitializationToString(simCtx->ParticleInitialization), (int)rank,
494 phys_coords.x, phys_coords.y, phys_coords.z, ci_metric_lnode, cj_metric_lnode, ck_metric_lnode,
495 xi_metric_logic, eta_metric_logic, zta_metric_logic);
496 } else if (particle_placed_by_this_rank) {
497 LOG_PARTICLE(simCtx, particleIDs[p], "seed",
498 "%s placement on rank %d at (%.6e, %.6e, %.6e): cell (%" PetscInt_FMT ",%" PetscInt_FMT ",%" PetscInt_FMT ")\n",
499 ParticleInitializationToString(simCtx->ParticleInitialization), (int)rank,
500 phys_coords.x, phys_coords.y, phys_coords.z, ci_metric_lnode, cj_metric_lnode, ck_metric_lnode);
501 } else {
502 // A point source no rank located.
503 LOG_PARTICLE(simCtx, particleIDs[p], "seed",
504 "%s: not placed by rank %d; parked at (%.6e, %.6e, %.6e) for the settlement to locate\n",
505 ParticleInitializationToString(simCtx->ParticleInitialization), (int)rank,
506 phys_coords.x, phys_coords.y, phys_coords.z);
507 }
508 }
509
510 // --- 6. Restore Pointers and Cleanup ---
511 ierr = DMSwarmRestoreField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_POSITION), NULL, NULL, (void**)&positions_field); CHKERRQ(ierr);
512 ierr = DMSwarmRestoreField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_PID), NULL, NULL, (void**)&particleIDs); CHKERRQ(ierr);
513 ierr = DMSwarmRestoreField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_CELL_ID), NULL, NULL, (void**)&cellIDs_petsc); CHKERRQ(ierr);
514 ierr = DMSwarmRestoreField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_LOCATION_STATUS), NULL, NULL, (void**)&status_field); CHKERRQ(ierr);
515 ierr = DMDAVecRestoreArrayRead(user->fda, Coor_local, (void*)&coor_nodes_local_array); CHKERRQ(ierr);
516 ierr = FreeSeedingCellTable(&seed_table); CHKERRQ(ierr);
517
518 LOG_ALLOW(LOCAL, LOG_INFO, "Rank %d: Completed processing for %" PetscInt_FMT " particles.\n",
519 rank, particlesPerProcess);
520
521 PROFILE_FUNCTION_END;
522
523 PetscFunctionReturn(0);
524}

732/**
733 * @brief Internal helper implementation: `CreateParticleSwarm()`.
734 * @details Local to this translation unit.
735 */
736PetscErrorCode CreateParticleSwarm(UserCtx *user, PetscInt numParticles, PetscInt *particlesPerProcess, BoundingBox *bboxlist) {
737 PetscErrorCode ierr; // PETSc error handling variable
738 (void)bboxlist;
739 PetscMPIInt rank, size; // Variables to store MPI rank and size
740 PetscInt remainder = 0; // Remainder of particles after division
741
742 PetscFunctionBeginUser;
743 PROFILE_FUNCTION_BEGIN;
744 // Validate input parameters
745 if (numParticles <= 0) {
746 LOG_ALLOW(GLOBAL,LOG_DEBUG, "Number of particles must be positive. Given: %" PetscInt_FMT "\n", numParticles);
747 PROFILE_FUNCTION_END;
748 return PETSC_ERR_ARG_OUTOFRANGE;
749 }
750
751 // Retrieve MPI rank and size
752 rank = user->simCtx->rank;
753 size = user->simCtx->size;
754 LOG_ALLOW(GLOBAL,LOG_INFO," Domain dimensions: [%.2f,%.2f],[%.2f,%.2f],[%.2f,%.2f] \n",
755 user->Min_X,user->Max_X,user->Min_Y,user->Max_Y, user->Min_Z,user->Max_Z);
756 LOG_ALLOW_SYNC(GLOBAL,LOG_DEBUG, "[Rank %d] Local Bounding Box: [%.2f,%.2f],[%.2f,%.2f],[%.2f,%.2f] \n",
757 rank,user->bbox.min_coords.x,user->bbox.max_coords.x,
758 user->bbox.min_coords.y,user->bbox.max_coords.y,
759 user->bbox.min_coords.z,user->bbox.max_coords.z);
760 // Distribute particles among MPI processes
761 if (user->simCtx->ParticleInitialization == PARTICLE_INIT_VOLUME) {
762 ierr = DistributeParticlesByOwnedMeasure(user, -1, numParticles, particlesPerProcess); CHKERRQ(ierr);
763 } else if (user->simCtx->ParticleInitialization == PARTICLE_INIT_SURFACE_RANDOM) {
764 ierr = DistributeParticlesByOwnedMeasure(user, (PetscInt)user->identifiedInletBCFace, numParticles, particlesPerProcess); CHKERRQ(ierr);
765 } else {
766 ierr = DistributeParticles(numParticles, rank, size, particlesPerProcess, &remainder); CHKERRQ(ierr);
767 }
768
769 // Initialize the DMSwarm - creates the swarm, sets the type and dimension
770 ierr = InitializeSwarm(user); CHKERRQ(ierr);
771
772 if (user->da) {
773 ierr = DMSwarmSetCellDM(user->swarm, user->da); CHKERRQ(ierr);
774 LOG_ALLOW(LOCAL,LOG_INFO,"Associated DMSwarm with Cell DM (user->da).\n");
775 } else {
776 // If user->da is essential for your simulation logic with particles, this should be a fatal error.
777 LOG_ALLOW(GLOBAL, LOG_WARNING, "user->da (Cell DM for Swarm) is NULL. Cell-based swarm operations might fail.\n");
778 // SETERRQ(PETSC_COMM_WORLD, PETSC_ERR_ARG_WRONGSTATE, "user->da (Cell DM) is NULL but required.");
779 }
780
781 // Register particle fields (position, velocity, CellID, weight, etc.)
782 ierr = RegisterParticleFields(user->swarm); CHKERRQ(ierr);
783
784 // Set the local number of particles for this rank and spare capacity for arrivals (10% of the
785 // local count; PETSc reallocates beyond it). Every resize zeroes the whole spare region, so a
786 // buffer sized to the global count would cost each rank memory and work of order numParticles.
787 ierr = DMSwarmSetLocalSizes(user->swarm, *particlesPerProcess, *particlesPerProcess / 10); CHKERRQ(ierr);
788 LOG_ALLOW(GLOBAL,LOG_INFO, "Set local swarm size: %" PetscInt_FMT " particles.\n", *particlesPerProcess);
789
790 // Optionally, LOG_ALLOW detailed DM info in debug mode
791 if (get_log_level() >= LOG_DEBUG && is_function_allowed(__func__)) {
792 LOG_ALLOW(GLOBAL,LOG_DEBUG,"Viewing DMSwarm:\n");
793 ierr = DMView(user->swarm, PETSC_VIEWER_STDOUT_WORLD); CHKERRQ(ierr);
794 }
795
796 LOG_ALLOW(GLOBAL,LOG_INFO, "Particle swarm creation and initialization complete.\n");
797
798 PROFILE_FUNCTION_END;
799 PetscFunctionReturn(0);
800}

802// NOTE: The following two functions are helpers for unpacking and updating particle data
803// between DMSwarm fields and the Particle struct used in simulation logic.
804// While Swarm fields store data in arrays, the Particle struct provides a convenient
805// way to manipulate individual particle properties during simulation steps.
806
807
808/**
809 * @brief Implementation of \ref UnpackSwarmFields().
810 * @details Full API contract (arguments, ownership, side effects) is documented with
811 * the header declaration in `include/ParticleSwarm.h`.
812 * @see UnpackSwarmFields()
813 */
814PetscErrorCode UnpackSwarmFields(PetscInt i, const PetscInt64 *PIDs, const PetscReal *weights,
815 const PetscReal *positions, const PetscInt *cellIndices,
816 PetscReal *velocities,PetscInt *LocStatus,PetscReal *diffusivity, Cmpnts *diffusivitygradient, PetscReal *psi, Particle *particle) {
817 PetscFunctionBeginUser;
818
819 PROFILE_HELPER_BEGIN;
820
821 // The particle's values are reported by the stage that uses them, not here: this runs
822 // several times per particle per step.
823 if (particle == NULL) {
824 SETERRQ(PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "Output Particle pointer is NULL. \n");
825 }
826 if(PIDs == NULL){
827 SETERRQ(PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "Input PIDs pointer is NULL.\n");
828 }
829 if(positions == NULL){
830 SETERRQ(PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "Input positions pointer is NULL.\n");
831 }
832 if(cellIndices == NULL){
833 SETERRQ(PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "Input cellIndices pointer is NULL.\n");
834 }
835 if(LocStatus == NULL){
836 SETERRQ(PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "Input LocStatus pointer is NULL.\n");
837 }
838
839 particle->PID = PIDs[i];
840
841 // A NULL optional field is one the caller does not need; it takes the default shown.
842 if(weights == NULL){
843 particle->weights.x = 1.0;
844 particle->weights.y = 1.0;
845 particle->weights.z = 1.0;
846 }else{
847 particle->weights.x = weights[3 * i];
848 particle->weights.y = weights[3 * i + 1];
849 particle->weights.z = weights[3 * i + 2];
850 }
851
852 particle->loc.x = positions[3 * i];
853 particle->loc.y = positions[3 * i + 1];
854 particle->loc.z = positions[3 * i + 2];
855
856 if(velocities == NULL){
857 particle->vel.x = 0.0;
858 particle->vel.y = 0.0;
859 particle->vel.z = 0.0;
860 }else{
861 particle->vel.x = velocities[3 * i];
862 particle->vel.y = velocities[3 * i + 1];
863 particle->vel.z = velocities[3 * i + 2];
864 }
865
866 particle->diffusivity = (diffusivity == NULL) ? 1.0 : diffusivity[i];
867
868 if(diffusivitygradient == NULL){
869 particle->diffusivitygradient.x = 0.0;
870 particle->diffusivitygradient.y = 0.0;
871 particle->diffusivitygradient.z = 0.0;
872 }else{
873 particle->diffusivitygradient = diffusivitygradient[i];
874 }
875
876 particle->psi = (psi == NULL) ? 0.0 : psi[i];
877
878 particle->cell[0] = cellIndices[3 * i];
879 particle->cell[1] = cellIndices[3 * i + 1];
880 particle->cell[2] = cellIndices[3 * i + 2];
881
882 particle->location_status = (ParticleLocationStatus)LocStatus[i];
