1/**
2 * @file expression.c
3 * @brief The expression language shared by every configured field: a parser, a postfix
4 * compiler, and a pure stack evaluator.
5 *
6 * Expressions are parsed into a small tree, because a chained comparison evaluates its
7 * middle operand twice, and compiled to postfix code that a stack evaluator runs once per
8 * point. A series `sum(k, lo, hi, body)` compiles its body once and loops over it, with the
9 * index in a slot of its nesting level. Evaluation is pure: random draws are hashed from a caller-supplied key, so a value
10 * does not depend on rank count, evaluation order, or how often it is evaluated.
11 */

126/** @brief Consume @p token if it is next, and report whether it was. */
127static PetscBool Accept(ExpressionParser *parser, const char *token)
128{
129 PROFILE_FUNCTION_BEGIN;
130 size_t length = strlen(token);
131
132 SkipSpace(parser);
133 if (strncmp(parser->text + parser->pos, token, length) != 0) { PROFILE_FUNCTION_END; return PETSC_FALSE; }
134 if (isalpha((unsigned char)token[0])) {
135 const char next = parser->text[parser->pos + length];
136 if (isalnum((unsigned char)next) || next == '_') { PROFILE_FUNCTION_END; return PETSC_FALSE; }
137 }
138 /* `*` must not match the first half of `**`, nor `<` the first half of `<=`. */
139 if ((!strcmp(token, "*") && parser->text[parser->pos + 1] == '*') ||
140 ((!strcmp(token, "<") || !strcmp(token, ">")) && parser->text[parser->pos + 1] == '=')) {
141 PROFILE_FUNCTION_END;
142 return PETSC_FALSE;
143 }
144 parser->pos += length;
145 PROFILE_FUNCTION_END;
146 return PETSC_TRUE;
147}

149/** @brief Append a node to the pool and return its index, or -1 once an error is set. */
150static PetscInt NewNode(ExpressionParser *parser, ExpressionOp op, PetscInt nchild,
151 PetscInt a, PetscInt b, PetscInt c)
152{
153 PROFILE_FUNCTION_BEGIN;
154 if (parser->error[0]) { PROFILE_FUNCTION_END; return -1; }
155 if (parser->count == parser->capacity) {
156 ParserFail(parser, "expression is too long");
157 PROFILE_FUNCTION_END;
158 return -1;
159 }
160 ExpressionNode *node = &parser->nodes[parser->count];
161 memset(node, 0, sizeof(*node));
162 node->op = op;
163 node->nchild = nchild;
164 node->child[0] = a;
165 node->child[1] = b;
166 node->child[2] = c;
167 PROFILE_FUNCTION_END;
168 return parser->count++;
169}

173/** @brief The value of a series bound: an integer literal, possibly negated or parenthesized. */
174static PetscBool SeriesBound(const ExpressionParser *parser, PetscInt index, PetscReal *bound)
175{
176 PROFILE_HELPER_BEGIN;
177 const ExpressionNode *node = &parser->nodes[index];
178 PetscReal sign = 1.0;
179 if (node->op == OP_NEG || node->op == OP_POS) {
180 if (node->op == OP_NEG) sign = -1.0;
181 node = &parser->nodes[node->child[0]];
182 }
183 *bound = sign * node->value;
184 PROFILE_HELPER_END;
185 return (PetscBool)(node->op == OP_CONST && node->value == PetscFloorReal(node->value));
186}

188/** @brief `sum(` has been read: index name, integer bounds, then the body with the index in scope. */
189static PetscInt ParseSeries(ExpressionParser *parser)
190{
191 PROFILE_HELPER_BEGIN;
192 char name[64];
193 size_t length = 0;
194 PetscReal lower = 0.0, upper = 0.0;
195
196 SkipSpace(parser);
197 const char *start = parser->text + parser->pos;
198 while (isalnum((unsigned char)start[length]) || start[length] == '_') length++;
199 if (!length || isdigit((unsigned char)start[0]) || length >= sizeof(name)) {
200 ParserFail(parser, "a series needs an index name first");
201 PROFILE_HELPER_END;
202 return -1;
203 }
204 memcpy(name, start, length);
205 name[length] = '\0';
206 parser->pos += length;
207 for (PetscInt v = 0; v < parser->name_count; ++v) {
208 if (!strcmp(name, parser->names[v])) { ParserFail(parser, "a series index cannot reuse a variable's name"); PROFILE_HELPER_END; return -1; }
209 }
210 for (PetscInt level = 0; level < parser->series_depth; ++level) {
211 if (!strcmp(name, parser->series_index[level])) { ParserFail(parser, "a series index cannot reuse an enclosing index"); PROFILE_HELPER_END; return -1; }
212 }
213 if (!strcmp(name, "pi")) { ParserFail(parser, "a series index cannot be pi"); PROFILE_HELPER_END; return -1; }
214 if (parser->series_depth == EXPRESSION_MAX_SERIES_DEPTH) { ParserFail(parser, "series nest too deeply"); PROFILE_HELPER_END; return -1; }
215 if (!Accept(parser, ",")) { ParserFail(parser, "expected ',' after the series index"); PROFILE_HELPER_END; return -1; }
216 const PetscInt lo = ParseOr(parser);
217 if (parser->error[0] || !Accept(parser, ",")) { ParserFail(parser, "expected ',' after the lower bound"); PROFILE_HELPER_END; return -1; }
218 const PetscInt hi = ParseOr(parser);
219 if (parser->error[0] || !Accept(parser, ",")) { ParserFail(parser, "expected ',' after the upper bound"); PROFILE_HELPER_END; return -1; }
220 if (!SeriesBound(parser, lo, &lower) || !SeriesBound(parser, hi, &upper)) {
221 ParserFail(parser, "series bounds must be integer numbers");
222 PROFILE_HELPER_END;
223 return -1;
224 }
225 if (upper - lower + 1.0 > (PetscReal)EXPRESSION_MAX_SERIES_TERMS) {
226 ParserFail(parser, "a series may add at most 1000000 terms");
227 PROFILE_HELPER_END;
228 return -1;
229 }
230 memcpy(parser->series_index[parser->series_depth], name, length + 1);
231 parser->series_depth++;
232 const PetscInt body = ParseOr(parser);
233 parser->series_depth--;
234 if (parser->error[0]) { PROFILE_HELPER_END; return -1; }
235 if (!Accept(parser, ")")) { ParserFail(parser, "expected ')' after the series body"); PROFILE_HELPER_END; return -1; }
236 PetscInt node = NewNode(parser, OP_SUM, 1, body, -1, -1);
237 if (node >= 0) {
238 parser->nodes[node].value = lower;
239 parser->nodes[node].upper = upper;
240 parser->nodes[node].var = parser->series_depth;
241 }
242 PROFILE_HELPER_END;
243 return node;
244}

246/** @brief number | name | name(args) | sum(index, lo, hi, body) | (expression) */
247static PetscInt ParseAtom(ExpressionParser *parser)
248{
249 PROFILE_HELPER_BEGIN;
250 SkipSpace(parser);
251 const char *start = parser->text + parser->pos;
252
253 if (isdigit((unsigned char)*start) || (*start == '.' && isdigit((unsigned char)start[1]))) {
254 /* Decimal literals only, as the Python validator requires. */
255 size_t length = 0;
256 while (isdigit((unsigned char)start[length])) length++;
257 if (start[length] == '.') { length++; while (isdigit((unsigned char)start[length])) length++; }
258 if (start[length] == 'e' || start[length] == 'E') {
259 size_t exponent = length + 1;
260 if (start[exponent] == '+' || start[exponent] == '-') exponent++;
261 if (!isdigit((unsigned char)start[exponent])) { ParserFail(parser, "malformed number"); PROFILE_HELPER_END; return -1; }
262 while (isdigit((unsigned char)start[exponent])) exponent++;
263 length = exponent;
264 }
265 if (isalnum((unsigned char)start[length]) || start[length] == '_' || start[length] == '.') {
266 ParserFail(parser, "malformed number");
267 PROFILE_HELPER_END;
268 return -1;
269 }
270 const PetscReal value = strtod(start, NULL);
271 parser->pos += length;
272 PetscInt node = NewNode(parser, OP_CONST, 0, -1, -1, -1);
273 if (node >= 0) parser->nodes[node].value = value;
274 PROFILE_HELPER_END;
275 return node;
276 }
277 if (isalpha((unsigned char)*start) || *start == '_') {
278 char name[64];
279 size_t length = 0;
280 while (isalnum((unsigned char)start[length]) || start[length] == '_') length++;
281 if (length >= sizeof(name)) { ParserFail(parser, "name is too long"); PROFILE_HELPER_END; return -1; }
282 memcpy(name, start, length);
283 name[length] = '\0';
284 if (!strcmp(name, "and") || !strcmp(name, "or") || !strcmp(name, "not")) {
285 ParserFail(parser, "operator where an operand was expected");
286 PROFILE_HELPER_END;
287 return -1;
288 }
289 parser->pos += length;
290 if (!strcmp(name, "sum") && Accept(parser, "(")) { PROFILE_HELPER_END; return ParseSeries(parser); }
291 if (Accept(parser, "(")) {
292 for (size_t f = 0; f < sizeof(kFunctions) / sizeof(kFunctions[0]); ++f) {
293 if (strcmp(name, kFunctions[f].name)) continue;
294 if (kFunctions[f].random && !parser->allow_random) {
295 ParserFail(parser, "random draws are not available in this expression");
296 PROFILE_HELPER_END;
297 return -1;
298 }
299 PetscInt args[3] = {-1, -1, -1}, nargs = 0;
300 if (!Accept(parser, ")")) {
301 do {
302 if (nargs == 3) { ParserFail(parser, "too many arguments"); PROFILE_HELPER_END; return -1; }
303 args[nargs++] = ParseOr(parser);
304 if (parser->error[0]) { PROFILE_HELPER_END; return -1; }
305 } while (Accept(parser, ","));
306 if (!Accept(parser, ")")) { ParserFail(parser, "expected ')'"); PROFILE_HELPER_END; return -1; }
307 }
308 if (nargs < kFunctions[f].min_args || nargs > kFunctions[f].max_args) {
309 ParserFail(parser, "wrong number of arguments");
310 PROFILE_HELPER_END;
311 return -1;
312 }
313 if (kFunctions[f].random) {
314 /* A draw's stream is a constant so that two expressions naming the
315 same stream draw the same number for the same particle. */
316 PetscInt node = NewNode(parser, kFunctions[f].op, 0, -1, -1, -1);
317 if (node < 0) { PROFILE_HELPER_END; return -1; }
318 if (nargs == 1) {
319 const ExpressionNode *stream = &parser->nodes[args[0]];
320 if (stream->op != OP_CONST || stream->value < 0.0 ||
321 stream->value != PetscFloorReal(stream->value)) {
322 ParserFail(parser, "a draw's stream must be a non-negative integer literal");
323 PROFILE_HELPER_END;
324 return -1;
325 }
326 parser->nodes[node].value = stream->value;
327 parser->nodes[node].var = 1;
328 }
329 PROFILE_HELPER_END;
330 return node;
331 }
332 PROFILE_HELPER_END;
333 return NewNode(parser, kFunctions[f].op, nargs, args[0], args[1], args[2]);
334 }
335 ParserFail(parser, "unknown function");
336 PROFILE_HELPER_END;
337 return -1;
338 }
339 for (PetscInt level = parser->series_depth - 1; level >= 0; --level) {
340 if (strcmp(name, parser->series_index[level])) continue;
341 PetscInt node = NewNode(parser, OP_LOCAL, 0, -1, -1, -1);
342 if (node >= 0) parser->nodes[node].var = level;
343 PROFILE_HELPER_END;
344 return node;
345 }
346 if (!strcmp(name, "pi")) {
347 PetscInt node = NewNode(parser, OP_CONST, 0, -1, -1, -1);
348 if (node >= 0) parser->nodes[node].value = PETSC_PI;
349 PROFILE_HELPER_END;
350 return node;
351 }
352 for (PetscInt v = 0; v < parser->name_count; ++v) {
353 if (!strcmp(name, parser->names[v])) {
354 PetscInt node = NewNode(parser, OP_VAR, 0, -1, -1, -1);
355 if (node >= 0) parser->nodes[node].var = v;
356 PROFILE_HELPER_END;
357 return node;
358 }
359 }
360 ParserFail(parser, "unknown name");
361 PROFILE_HELPER_END;
362 return -1;
363 }
364 if (Accept(parser, "(")) {
365 PetscInt inner = ParseOr(parser);
366 if (!Accept(parser, ")")) { ParserFail(parser, "expected ')'"); PROFILE_HELPER_END; return -1; }
367 PROFILE_HELPER_END;
368 return inner;
369 }
370 ParserFail(parser, "expected a number, name, or '('");
371 PROFILE_HELPER_END;
372 return -1;
373}

408/**
409 * @brief Parses a left-associative chain of `*`, `/`, `%`, folding each operator into a node.
410 * @details `//` is refused explicitly, since the Python side does not accept floor division.
411 * @param[in,out] parser Parser state; its position advances past the term.
412 * @return Node index of the term, or -1 after an error.
413 */
414static PetscInt ParseTerm(ExpressionParser *parser)
415{
416 PROFILE_FUNCTION_BEGIN;
417 PetscInt left = ParseUnary(parser);
418 for (;;) {
419 if (Accept(parser, "*")) left = NewNode(parser, OP_MUL, 2, left, ParseUnary(parser), -1);
420 else if (Accept(parser, "/")) {
421 if (parser->text[parser->pos] == '/') { ParserFail(parser, "'//' is not supported"); PROFILE_FUNCTION_END; return -1; }
422 left = NewNode(parser, OP_DIV, 2, left, ParseUnary(parser), -1);
423 } else if (Accept(parser, "%")) left = NewNode(parser, OP_MOD, 2, left, ParseUnary(parser), -1);
424 else { PROFILE_FUNCTION_END; return left; }
425 }
426 PROFILE_FUNCTION_END;
427}

446/** @brief sum (comparison sum)*; a chain `a < b < c` is `(a < b) and (b < c)`. */
447static PetscInt ParseComparison(ExpressionParser *parser)
448{
449 PROFILE_FUNCTION_BEGIN;
450 static const struct { const char *token; ExpressionOp op; } kComparisons[] = {
451 {"==", OP_EQ}, {"!=", OP_NE}, {"<=", OP_LE}, {">=", OP_GE}, {"<", OP_LT}, {">", OP_GT},
452 };
453 PetscInt left = ParseSum(parser), result = -1;
454
455 for (;;) {
456 ExpressionOp op = OP_CONST;
457 for (size_t c = 0; c < sizeof(kComparisons) / sizeof(kComparisons[0]); ++c) {
458 if (Accept(parser, kComparisons[c].token)) { op = kComparisons[c].op; break; }
459 }
460 if (op == OP_CONST) { PROFILE_FUNCTION_END; return (result < 0) ? left : result; }
461 const PetscInt right = ParseSum(parser);
462 const PetscInt pair = NewNode(parser, op, 2, left, right, -1);
463 result = (result < 0) ? pair : NewNode(parser, OP_AND, 2, result, pair, -1);
464 left = right;
465 }
466 PROFILE_FUNCTION_END;
467}

500/** @brief Emit postfix code for a subtree and track the stack depth it needs. */
501static void EmitNode(const ExpressionParser *parser, PetscInt index, PicurvExpression *expression,
502 PetscInt *depth)
503{
504 PROFILE_FUNCTION_BEGIN;
505 const ExpressionNode *node = &parser->nodes[index];
506
507 if (node->op == OP_SUM) {
508 const PetscInt body = CountCode(parser, node->child[0]);
509 ExpressionInstruction *start = &expression->code[expression->length++];
510 start->op = OP_SUM_START; start->value = node->value; start->arg = node->var;
511 ExpressionInstruction *hi = &expression->code[expression->length++];
512 hi->op = OP_SUM_HI; hi->value = node->upper; hi->arg = body;
513 *depth += 1; /* the running total */
514 if (*depth > expression->max_depth) expression->max_depth = *depth;
515 EmitNode(parser, node->child[0], expression, depth);
516 ExpressionInstruction *acc = &expression->code[expression->length++];
517 acc->op = OP_SUM_ACC; acc->value = (PetscReal)body; acc->arg = node->var;
518 *depth -= 1;
519 PROFILE_FUNCTION_END;
520 return;
521 }
522 for (PetscInt c = 0; c < node->nchild; ++c) EmitNode(parser, node->child[c], expression, depth);
523 ExpressionInstruction *instruction = &expression->code[expression->length++];
524 instruction->op = node->op;
525 instruction->value = node->value;
526 instruction->arg = node->var;
527 *depth += 1 - node->nchild;
528 if (*depth > expression->max_depth) expression->max_depth = *depth;
529 if (node->op == OP_UNIFORM || node->op == OP_NORMAL) expression->draws = PETSC_TRUE;
530 PROFILE_FUNCTION_END;
531}

533/** @brief Count the instructions a subtree emits (shared subtrees count per use). */
534static PetscInt CountCode(const ExpressionParser *parser, PetscInt index)
535{
536 PROFILE_HELPER_BEGIN;
537 const ExpressionNode *node = &parser->nodes[index];
538 PetscInt total = (node->op == OP_SUM) ? 3 : 1; /* a series: START, HI and ACC around its body */
539
540 for (PetscInt c = 0; c < node->nchild; ++c) total += CountCode(parser, node->child[c]);
541 PROFILE_HELPER_END;
542 return total;
543}

545/**
546 * @brief Implementation of \ref PicurvExpressionCompile().
547 * @see PicurvExpressionCompile()
548 */
549PetscErrorCode PicurvExpressionCompile(const char *text, const char *const *names, PetscInt name_count,
550 PetscBool allow_random, PicurvExpression **expression)
551{
552 ExpressionParser parser;
553 PetscInt root, depth = 0;
554
555 PetscFunctionBeginUser;
556 PROFILE_FUNCTION_BEGIN;
557 PetscCheck(text && expression, PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "Expression text and output are required.");
558 PetscCall(PetscMemzero(&parser, sizeof(parser)));
559 parser.text = text;
560 parser.names = names;
561 parser.name_count = name_count;
562 parser.allow_random = allow_random;
563 parser.capacity = 4 * (PetscInt)strlen(text) + 8;
564 PetscCall(PetscMalloc1(parser.capacity, &parser.nodes));
565
566 root = ParseOr(&parser);
567 SkipSpace(&parser);
568 if (!parser.error[0] && parser.text[parser.pos]) ParserFail(&parser, "unexpected text");
569 if (parser.error[0]) {
570 PetscCall(PetscFree(parser.nodes));
571 SETERRQ(PETSC_COMM_SELF, PETSC_ERR_ARG_WRONG, "Invalid expression: %s.", parser.error);
572 }
573
574 PetscCall(PetscNew(expression));
575 (*expression)->length = 0;
576 PetscCall(PetscMalloc1(CountCode(&parser, root), &(*expression)->code));
577 EmitNode(&parser, root, *expression, &depth);
578 PetscCall(PetscFree(parser.nodes));
579 PROFILE_FUNCTION_END;
580 PetscFunctionReturn(0);
581}

583/**
584 * @brief Implementation of \ref PicurvMix64().
585 * @see PicurvMix64()
586 */
587uint64_t PicurvMix64(uint64_t z)
588{
589 z += 0x9E3779B97F4A7C15ULL;
590 z = (z ^ (z >> 30)) * 0xBF58476D1CE4E5B9ULL;
591 z = (z ^ (z >> 27)) * 0x94D049BB133111EBULL;
592 return z ^ (z >> 31);
593}

595/**
596 * @brief Implementation of \ref PicurvUnitFromHash().
597 * @see PicurvUnitFromHash()
598 */
599PetscReal PicurvUnitFromHash(uint64_t hash)
600{
601 return (PetscReal)(hash >> 11) * (1.0 / 9007199254740992.0);
602}

604/**
605 * @brief Implementation of \ref PicurvParticleDrawHash().
606 * @see PicurvParticleDrawHash()
607 */
608uint64_t PicurvParticleDrawHash(PetscInt64 seed, PicurvDrawSpace space, uint64_t stream,
609 PetscInt64 pid, PetscInt64 salt)
610{
611 uint64_t hash = PicurvMix64((uint64_t)seed);
612 hash = PicurvMix64(hash ^ (uint64_t)space);
613 hash = PicurvMix64(hash ^ stream);
614 hash = PicurvMix64(hash ^ (uint64_t)pid);
615 return PicurvMix64(hash ^ (uint64_t)salt);
616}

618/** @brief A draw keyed on the particle, the event, the stream, and the function. */
619static PetscReal Draw(const PicurvExpressionDrawKey *key, const ExpressionInstruction *instruction)
620{
621 PROFILE_HELPER_BEGIN;
622 /* A default stream is the drawing field's own, so two fields draw independently; an
623 explicit stream is shared by every expression that names it. */
624 const PicurvDrawSpace space = instruction->arg ? PICURV_DRAW_SPACE_NAMED_STREAM : PICURV_DRAW_SPACE_FIELD;
625 const uint64_t stream = instruction->arg ? (uint64_t)instruction->value : (uint64_t)key->field;
626 const uint64_t hash = PicurvMix64(PicurvParticleDrawHash(key->seed, space, stream, key->pid, key->salt)
627 ^ (uint64_t)instruction->op);
628 if (instruction->op == OP_UNIFORM) { PROFILE_HELPER_END; return PicurvUnitFromHash(hash); }
629 /* Box-Muller from two independent uniforms; u1 lies in (0, 1]. */
630 const PetscReal u1 = 1.0 - PicurvUnitFromHash(hash);
631 const PetscReal u2 = PicurvUnitFromHash(PicurvMix64(hash ^ 0xD1B54A32D192ED03ULL));
632 PROFILE_HELPER_END;
633 return PetscSqrtReal(-2.0 * PetscLogReal(u1)) * PetscCosReal(2.0 * PETSC_PI * u2);
634}

636/**
637 * @brief Kelvin functions of order zero, `ber(x) + i bei(x) = J0(x e^{3 pi i / 4})`, from
638 * their power series. Terms grow to about `exp(|x| / sqrt(2))` before the series
639 * converges, so about `|x| / 5` of the sixteen digits are lost to cancellation.
640 */
641static PetscReal Kelvin(PetscReal x, PetscBool imaginary)
642{
643 PROFILE_HELPER_BEGIN;
644 const PetscReal q = 0.0625 * x * x * x * x; /* (x/2)^4 */
645 PetscReal term = imaginary ? 0.25 * x * x : 1.0, total = term;
646 /* ber: (-1)^m (x/2)^(4m) / ((2m)!)^2; bei: (-1)^m (x/2)^(4m+2) / ((2m+1)!)^2. */
647 for (PetscInt m = 1; m < 200; ++m) {
648 const PetscReal a = imaginary ? 2.0 * m : 2.0 * m - 1.0, b = a + 1.0;
649 term *= -q / (a * a * b * b);
650 total += term;
651 if (PetscAbsReal(term) <= 1.0e-17 * PetscAbsReal(total)) break;
652 }
653 PROFILE_HELPER_END;
654 return total;
655}
656
657/** @brief Bessel function of the first or second kind of integer order; NaN for another order. */
658static PetscReal Bessel(PetscReal order, PetscReal x, PetscBool second)
659{
660 PROFILE_HELPER_BEGIN;
661 if (order != PetscFloorReal(order) || PetscAbsReal(order) > 1.0e6) { PROFILE_HELPER_END; return NAN; }
662 const int n = (int)order;
663 PROFILE_HELPER_END;
664 return second ? (PetscReal)yn(n, (double)x) : (PetscReal)jn(n, (double)x);
665}

667/** @brief Maximum or minimum that propagates NaN, as numpy does. */
668static PetscReal Extreme(PetscReal a, PetscReal b, PetscBool maximum)
669{
670 PROFILE_HELPER_BEGIN;
671 if (PetscIsNanReal(a) || PetscIsNanReal(b)) { PROFILE_HELPER_END; return NAN; }
672 PROFILE_HELPER_END;
673 return maximum ? PetscMax(a, b) : PetscMin(a, b);
674}

676/**
677 * @brief Implementation of \ref PicurvExpressionEvaluate().
678 * @see PicurvExpressionEvaluate()
679 */
680PetscErrorCode PicurvExpressionEvaluate(const PicurvExpression *expression, const PetscReal *values,
681 const PicurvExpressionDrawKey *key, PetscReal *result)
682{
683 PetscReal stack[expression->max_depth + 1];
684 PetscReal index[EXPRESSION_MAX_SERIES_DEPTH];
685 PetscInt top = 0;
686
687 PetscFunctionBeginUser;
688 PROFILE_HELPER_BEGIN;
689 PetscCheck(!expression->draws || key, PETSC_COMM_SELF, PETSC_ERR_ARG_NULL,
690 "An expression that draws random numbers needs a draw key.");
691 for (PetscInt i = 0; i < expression->length; ++i) {
692 const ExpressionInstruction *instruction = &expression->code[i];
693 PetscReal a, b, c;
694
695 switch (instruction->op) {
696 case OP_CONST: stack[top++] = instruction->value; break;
697 case OP_VAR: stack[top++] = values[instruction->arg]; break;
698 case OP_UNIFORM:
699 case OP_NORMAL: stack[top++] = Draw(key, instruction); break;
700 case OP_NEG: stack[top - 1] = -stack[top - 1]; break;
701 case OP_POS: break;
702 case OP_NOT: stack[top - 1] = (stack[top - 1] == 0.0) ? 1.0 : 0.0; break;
703 case OP_ABS: stack[top - 1] = PetscAbsReal(stack[top - 1]); break;
704 case OP_COS: stack[top - 1] = PetscCosReal(stack[top - 1]); break;
705 case OP_EXP: stack[top - 1] = PetscExpReal(stack[top - 1]); break;
706 case OP_SIN: stack[top - 1] = PetscSinReal(stack[top - 1]); break;
707 case OP_SQRT: stack[top - 1] = PetscSqrtReal(stack[top - 1]); break;
708 case OP_TAN: stack[top - 1] = PetscTanReal(stack[top - 1]); break;
709 case OP_ERF: stack[top - 1] = (PetscReal)erf((double)stack[top - 1]); break;
710 case OP_TANH: stack[top - 1] = PetscTanhReal(stack[top - 1]); break;
711 case OP_COSH: stack[top - 1] = PetscCoshReal(stack[top - 1]); break;
712 case OP_SINH: stack[top - 1] = PetscSinhReal(stack[top - 1]); break;
713 case OP_LOG: stack[top - 1] = PetscLogReal(stack[top - 1]); break;
714 case OP_FLOOR: stack[top - 1] = PetscFloorReal(stack[top - 1]); break;
715 case OP_BER: stack[top - 1] = Kelvin(stack[top - 1], PETSC_FALSE); break;
716 case OP_BEI: stack[top - 1] = Kelvin(stack[top - 1], PETSC_TRUE); break;
717 case OP_LOCAL: stack[top++] = index[instruction->arg]; break;
718 case OP_SUM_START:
719 stack[top++] = 0.0;
720 index[instruction->arg] = instruction->value;
721 /* An empty range skips the body and its ACC. */
722 if (instruction->value > expression->code[i + 1].value) i += 2 + expression->code[i + 1].arg;
723 else i += 1;
724 break;
725 case OP_SUM_ACC: {
726 const PetscInt body = (PetscInt)instruction->value;
727 const PetscReal upper = expression->code[i - body - 1].value;
728 const PetscReal term = stack[--top];
729 stack[top - 1] += term;
730 index[instruction->arg] += 1.0;
731 if (index[instruction->arg] <= upper) i -= body + 1;
732 } break;
733 case OP_WHERE:
734 c = stack[--top]; b = stack[--top]; a = stack[top - 1];
735 stack[top - 1] = (a != 0.0) ? b : c;
736 break;
737 default:
738 b = stack[--top]; a = stack[top - 1];
739 switch (instruction->op) {
740 case OP_ADD: a = a + b; break;
741 case OP_SUB: a = a - b; break;
742 case OP_MUL: a = a * b; break;
743 case OP_DIV: a = a / b; break;
744 /* Python's modulus takes the sign of the divisor. */
745 case OP_MOD: a = (b == 0.0) ? NAN : a - b * PetscFloorReal(a / b); break;
746 case OP_POW: a = PetscPowReal(a, b); break;
747 case OP_EQ: a = (a == b) ? 1.0 : 0.0; break;
748 case OP_NE: a = (a != b) ? 1.0 : 0.0; break;
749 case OP_LT: a = (a < b) ? 1.0 : 0.0; break;
750 case OP_LE: a = (a <= b) ? 1.0 : 0.0; break;
751 case OP_GT: a = (a > b) ? 1.0 : 0.0; break;
752 case OP_GE: a = (a >= b) ? 1.0 : 0.0; break;
753 case OP_AND: a = (a != 0.0 && b != 0.0) ? 1.0 : 0.0; break;
754 case OP_OR: a = (a != 0.0 || b != 0.0) ? 1.0 : 0.0; break;
755 case OP_ATAN2: a = PetscAtan2Real(a, b); break;
756 case OP_MAXIMUM: a = Extreme(a, b, PETSC_TRUE); break;
757 case OP_MINIMUM: a = Extreme(a, b, PETSC_FALSE); break;
758 case OP_BESSELJ: a = Bessel(a, b, PETSC_FALSE); break;
759 case OP_BESSELY: a = Bessel(a, b, PETSC_TRUE); break;
760 default: SETERRQ(PETSC_COMM_SELF, PETSC_ERR_PLIB, "Unknown expression opcode %d.", (int)instruction->op);
761 }
762 stack[top - 1] = a;
763 }
764 }
765 *result = stack[0];
766 PROFILE_HELPER_END;
767 PetscFunctionReturn(0);
768}

842 PetscCall(PicurvExpressionCompile(text, kPicurvSourceVariableNames,
843 on_face ? PICURV_SOURCE_VAR_COUNT : PICURV_SOURCE_VAR_CELL_COUNT, PETSC_FALSE,
844 &source->component[c]));
845 PetscCall(PicurvExpressionUsesVariable(source->component[c], PICURV_SOURCE_VAR_T, &uses_t));
846 source->time_dependent = (PetscBool)(source->time_dependent || uses_t);
847 }

852 PetscCall(PetscOptionsGetString(NULL, NULL, option, text, sizeof(text), &source->bulk_is_flow_rate));

854 }
855 if (found) {
856 PetscCall(PicurvExpressionCompile(text, kTime, 1, PETSC_FALSE, &source->bulk));
857 PetscCall(PicurvExpressionUsesVariable(source->bulk, 0, &source->bulk_time_dependent));
858 }
859 PROFILE_FUNCTION_END;
860 PetscFunctionReturn(0);
861}

865 * @see PicurvVelocitySourceBulkFactor()
866 */
867PetscErrorCode PicurvVelocitySourceBulkFactor(const PicurvVelocitySource *source, PetscReal physical_time,
868 const PetscReal local[3], const char *where, PetscReal *factor)
869{
870 PetscReal global[3], target;
871
872 PetscFunctionBeginUser;
873 PROFILE_HELPER_BEGIN;

880 if (source->bulk_is_flow_rate) target /= global[2];
881 *factor = (target * global[2] - global[1]) / global[0];
882 LOG_ALLOW(GLOBAL, LOG_DEBUG, "%s: bulk velocity %g, source mean %g, data mean %g, factor %g.\n", where,

888/**
889 * @brief Implementation of \ref PicurvVelocitySourceEvaluate().
890 * @see PicurvVelocitySourceEvaluate()
891 */
892PetscErrorCode PicurvVelocitySourceEvaluate(const PicurvVelocitySource *source, const PetscReal *values,
893 const Cmpnts edges[3], Cmpnts *value)
894{
895 PetscReal v[3] = {0.0, 0.0, 0.0};
896
897 PetscFunctionBeginUser;
898 PROFILE_HELPER_BEGIN;
899 for (PetscInt c = 0; c < source->count; ++c) {
900 PetscCall(PicurvExpressionEvaluate(source->component[c], values, NULL, &v[c]));
901 PetscCheck(!PetscIsInfOrNanReal(v[c]), PETSC_COMM_SELF, PETSC_ERR_FP,
902 "Velocity source value %" PetscInt_FMT " is %g at (x, y, z) = (%g, %g, %g), t = %g.",
903 c, (double)v[c], (double)values[PICURV_SOURCE_VAR_X], (double)values[PICURV_SOURCE_VAR_Y],
904 (double)values[PICURV_SOURCE_VAR_Z], (double)values[PICURV_SOURCE_VAR_T]);
905 }
906 if (source->frame == PICURV_SOURCE_FRAME_CURVILINEAR) {
907 /* Values are components along the unit grid-line directions. */
908 value->x = value->y = value->z = 0.0;
909 for (PetscInt a = 0; a < 3; ++a) {
910 const PetscReal norm = PetscSqrtReal(edges[a].x * edges[a].x + edges[a].y * edges[a].y + edges[a].z * edges[a].z);
911 PetscCheck(norm > 0.0, PETSC_COMM_SELF, PETSC_ERR_ARG_OUTOFRANGE, "A cell edge has zero length.");
912 value->x += v[a] * edges[a].x / norm;
913 value->y += v[a] * edges[a].y / norm;
914 value->z += v[a] * edges[a].z / norm;
915 }
916 } else {
917 value->x = v[0];
918 value->y = v[1];
919 value->z = v[2];
920 }
921 PROFILE_HELPER_END;
922 PetscFunctionReturn(0);
923}

925/**
926 * @brief Implementation of \ref PicurvSourceCellBulkFactor().
927 * @see PicurvSourceCellBulkFactor()

929PetscErrorCode PicurvSourceCellBulkFactor(UserCtx *user, const PicurvVelocitySource *source, Vec fraction, Vec length,
930 PetscReal physical_time, PetscReal *factor)
931{
932 const DMDALocalInfo *info = &user->info;
933 const PetscReal L_ref = user->simCtx->scaling.L_ref, U_ref = user->simCtx->scaling.U_ref;
934 const PetscInt xs = info->xs, xe = info->xs + info->xm, mx = info->mx;
935 const PetscInt ys = info->ys, ye = info->ys + info->ym, my = info->my;
936 const PetscInt zs = info->zs, ze = info->zs + info->zm, mz = info->mz;
937 PetscReal values[PICURV_SOURCE_VAR_COUNT], local[3] = {0.0, 0.0, 0.0};
938 const Cmpnts ***cent, ***frac, ***len, ***csi, ***eta, ***zet, ***ucat;
939 const PetscReal ***aj;
940
941 PetscFunctionBeginUser;
942 PROFILE_FUNCTION_BEGIN;
943 PetscCall(DMDAVecGetArrayRead(user->fda, user->Cent, &cent));
944 PetscCall(DMDAVecGetArrayRead(user->fda, fraction, &frac));
945 PetscCall(DMDAVecGetArrayRead(user->fda, length, &len));
946 PetscCall(DMDAVecGetArrayRead(user->fda, user->lCsi, &csi));
947 PetscCall(DMDAVecGetArrayRead(user->fda, user->lEta, &eta));
948 PetscCall(DMDAVecGetArrayRead(user->fda, user->lZet, &zet));
949 PetscCall(DMDAVecGetArrayRead(user->da, user->lAj, &aj));
950 PetscCall(DMDAVecGetArrayRead(user->fda, user->Ucat, &ucat));
951 values[PICURV_SOURCE_VAR_T] = physical_time;
952 for (PetscInt k = (zs == 0) ? 1 : zs; k < ((ze == mz) ? mz - 1 : ze); ++k)
953 for (PetscInt j = (ys == 0) ? 1 : ys; j < ((ye == my) ? my - 1 : ye); ++j)
954 for (PetscInt i = (xs == 0) ? 1 : xs; i < ((xe == mx) ? mx - 1 : xe); ++i) {
955 const PetscReal volume = 1.0 / PetscAbsReal(aj[k][j][i]);
956 Cmpnts edges[3], v;
957 PicurvSourceSetVariables(values, cent[k][j][i], frac[k][j][i], len[k][j][i], L_ref);
958 PetscCall(ComputeCellEdgeVectors(aj[k][j][i], csi[k][j][i], eta[k][j][i], zet[k][j][i], edges));
959 PetscCall(PicurvVelocitySourceEvaluate(source, values, edges, &v));
960 const PetscReal speed = PetscSqrtReal(v.x * v.x + v.y * v.y + v.z * v.z);
961 local[0] += speed * volume;
962 if (speed > 0.0) local[1] += U_ref * (ucat[k][j][i].x * v.x + ucat[k][j][i].y * v.y +
963 ucat[k][j][i].z * v.z) / speed * volume;
964 local[2] += volume;
965 }
966 PetscCall(DMDAVecRestoreArrayRead(user->fda, user->Ucat, &ucat));
967 PetscCall(DMDAVecRestoreArrayRead(user->da, user->lAj, &aj));
968 PetscCall(DMDAVecRestoreArrayRead(user->fda, user->lZet, &zet));
969 PetscCall(DMDAVecRestoreArrayRead(user->fda, user->lEta, &eta));
970 PetscCall(DMDAVecRestoreArrayRead(user->fda, user->lCsi, &csi));
971 PetscCall(DMDAVecRestoreArrayRead(user->fda, length, &len));
972 PetscCall(DMDAVecRestoreArrayRead(user->fda, fraction, &frac));
973 PetscCall(DMDAVecRestoreArrayRead(user->fda, user->Cent, &cent));
974 PetscCall(PicurvVelocitySourceBulkFactor(source, physical_time, local, "The field", factor));
975 PROFILE_FUNCTION_END;
976 PetscFunctionReturn(0);
977}

983PetscErrorCode PicurvSourceAddFluxes(UserCtx *user, const PicurvVelocitySource *source, Vec fraction, Vec length,
