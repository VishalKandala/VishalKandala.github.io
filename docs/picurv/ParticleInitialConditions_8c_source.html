40/**
41 * @brief Implementation of \ref ParticleFieldPlanCreate().
42 * @see ParticleFieldPlanCreate()
43 */
44PetscErrorCode ParticleFieldPlanCreate(ParticleFieldPlan **plan)
45{
46 PetscInt count = 0;
47 PetscBool found = PETSC_FALSE;
48 char option[256];
49
50 PetscFunctionBeginUser;
51 PROFILE_FUNCTION_BEGIN;
52 PetscCheck(plan, PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "Plan output is required.");
53 *plan = NULL;
54 PetscCall(PetscSNPrintf(option, sizeof(option), "-particle_fields_count"));
55 PetscCall(PetscOptionsGetInt(NULL, NULL, option, &count, &found));
56 if (!found || count == 0) { PROFILE_FUNCTION_END; PetscFunctionReturn(0); }
57 PetscCheck(count > 0 && count <= PARTICLE_FIELD_ID_COUNT, PETSC_COMM_WORLD, PETSC_ERR_ARG_OUTOFRANGE,
58 "%s must be between 0 and %d (got %" PetscInt_FMT ").", option, (int)PARTICLE_FIELD_ID_COUNT, count);
59
60 PetscCall(PetscNew(plan));
61 PetscCall(PetscCalloc1(count, &(*plan)->bindings));
62 (*plan)->count = count;
63 for (PetscInt i = 0; i < count; ++i) {
64 ParticleFieldBinding *binding = &(*plan)->bindings[i];
65 const ParticleFieldDescriptor *descriptor = NULL;
66 char name[64];
67
68 PetscCall(PetscSNPrintf(option, sizeof(option), "-particle_fields_%" PetscInt_FMT "_name", i));
69 PetscCall(PetscOptionsGetString(NULL, NULL, option, name, sizeof(name), &found));
70 PetscCheck(found, PETSC_COMM_WORLD, PETSC_ERR_ARG_WRONG, "%s is missing.", option);
71 PetscCall(ParticleFieldIdFromName(name, &binding->field));
72 PetscCall(ParticleFieldGetDescriptor(binding->field, &descriptor));
73 PetscCheck(descriptor->capabilities & PARTICLE_FIELD_CAPABILITY_USER_INITIALIZE, PETSC_COMM_WORLD,
74 PETSC_ERR_ARG_WRONG,
75 "Particle field '%s' cannot be given an initial value: the runtime sets it from the "
76 "Eulerian fields or from particle location.", name);
77 PetscCheck(descriptor->dimension.kind == FIELD_DIMENSION_FIXED && descriptor->data_type == PETSC_REAL,
78 PETSC_COMM_WORLD, PETSC_ERR_PLIB, "Settable particle field '%s' must be a real quantity.", name);
79 for (PetscInt j = 0; j < i; ++j) {
80 PetscCheck((*plan)->bindings[j].field != binding->field, PETSC_COMM_WORLD, PETSC_ERR_ARG_WRONG,
81 "Particle field '%s' is configured twice.", name);
82 }
83 binding->components = descriptor->components;
84 PetscCheck(binding->components <= PARTICLE_FIELD_MAX_COMPONENTS, PETSC_COMM_WORLD, PETSC_ERR_PLIB,
85 "Particle field '%s' has more components than an initial value supports.", name);
86 for (PetscInt c = 0; c < binding->components; ++c) {
87 char text[PARTICLE_FIELD_EXPRESSION_LENGTH];
88
89 PetscCall(PetscSNPrintf(option, sizeof(option), "-particle_fields_%" PetscInt_FMT "_expr_%" PetscInt_FMT, i, c));
90 PetscCall(PetscOptionsGetString(NULL, NULL, option, text, sizeof(text), &found));
91 PetscCheck(found, PETSC_COMM_WORLD, PETSC_ERR_ARG_WRONG, "%s is missing.", option);
92 PetscCall(PicurvExpressionCompile(text, kParticleVariables, PARTICLE_VARIABLE_COUNT, PETSC_TRUE,
93 &binding->component[c]));
94 }
95 }
96 PROFILE_FUNCTION_END;
97 PetscFunctionReturn(0);
98}

100/** @brief Global domain bounds, in solver units, from the replicated rank bounding boxes. */
101static PetscErrorCode DomainBounds(const SimCtx *simCtx, Cmpnts *lower, Cmpnts *upper)
102{
103 PetscMPIInt size = 1;
104
105 PetscFunctionBeginUser;
106 PROFILE_FUNCTION_BEGIN;
107 PetscCheck(simCtx->bboxlist, PETSC_COMM_SELF, PETSC_ERR_ARG_WRONGSTATE,
108 "Rank bounding boxes are needed before a particle initial value is applied.");
109 size = simCtx->size;
110 *lower = simCtx->bboxlist[0].min_coords;
111 *upper = simCtx->bboxlist[0].max_coords;
112 for (PetscInt r = 1; r < (PetscInt)size * simCtx->block_number; ++r) {
113 lower->x = PetscMin(lower->x, simCtx->bboxlist[r].min_coords.x);
114 lower->y = PetscMin(lower->y, simCtx->bboxlist[r].min_coords.y);
115 lower->z = PetscMin(lower->z, simCtx->bboxlist[r].min_coords.z);
116 upper->x = PetscMax(upper->x, simCtx->bboxlist[r].max_coords.x);
117 upper->y = PetscMax(upper->y, simCtx->bboxlist[r].max_coords.y);
118 upper->z = PetscMax(upper->z, simCtx->bboxlist[r].max_coords.z);
119 }
120 PROFILE_FUNCTION_END;
121 PetscFunctionReturn(0);
122}

124/** @brief Position within the domain along one axis, in [0, 1]; 0 across a flat axis. */
125static PetscReal Normalized(PetscReal value, PetscReal lower, PetscReal upper)
126{
127 return (upper > lower) ? (value - lower) / (upper - lower) : 0.0;
128}

130/**
131 * @brief Implementation of \ref ParticleFieldPlanApply().
132 * @see ParticleFieldPlanApply()
133 */
134PetscErrorCode ParticleFieldPlanApply(UserCtx *user, const ParticleFieldPlan *plan,
135 const ParticleFieldEvent *event, PetscInt first, PetscInt end)
136{
137 SimCtx *simCtx = NULL;
138 const PetscReal *positions = NULL;
139 const PetscInt64 *pids = NULL;
140 Cmpnts lower, upper;
141
142 PetscFunctionBeginUser;
143 PROFILE_FUNCTION_BEGIN;
144 if (!plan || end <= first) { PROFILE_FUNCTION_END; PetscFunctionReturn(0); }
145 PetscCheck(user && user->swarm && event, PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "Swarm and event are required.");
146 simCtx = user->simCtx;
147 PetscCall(DomainBounds(simCtx, &lower, &upper));
148
149 PetscCall(DMSwarmGetField(user->swarm, ParticleFieldName(PARTICLE_FIELD_ID_POSITION), NULL, NULL, (void **)&positions));
150 PetscCall(DMSwarmGetField(user->swarm, ParticleFieldName(PARTICLE_FIELD_ID_PID), NULL, NULL, (void **)&pids));
151 for (PetscInt b = 0; b < plan->count; ++b) {
152 const ParticleFieldBinding *binding = &plan->bindings[b];
153 const ParticleFieldDescriptor *descriptor = NULL;
154 PetscReal *field = NULL, scale = 1.0;
155
156 PetscCall(ParticleFieldGetDescriptor(binding->field, &descriptor));
157 /* A value is physical; the swarm stores solver units. */
158 PetscCall(FieldDimensionReferenceScale(&simCtx->scaling, descriptor->dimension, &scale));
159 PetscCall(DMSwarmGetField(user->swarm, descriptor->canonical_name, NULL, NULL, (void **)&field));
160 for (PetscInt p = first; p < end; ++p) {
161 const PetscReal *position = &positions[3 * p];
162 const PetscReal values[] = {
163 position[0] * simCtx->scaling.L_ref, position[1] * simCtx->scaling.L_ref,
164 position[2] * simCtx->scaling.L_ref,
165 Normalized(position[0], lower.x, upper.x), Normalized(position[1], lower.y, upper.y),
166 Normalized(position[2], lower.z, upper.z),
167 (PetscReal)pids[p], event->physical_time,
168 };
169 const PicurvExpressionDrawKey key = {
170 (PetscInt64)simCtx->particleRandomSeed, pids[p], event->salt, (PetscInt64)binding->field,
171 };
172 for (PetscInt c = 0; c < binding->components; ++c) {
173 PetscReal value = 0.0;
174
175 PetscCall(PicurvExpressionEvaluate(binding->component[c], values, &key, &value));
176 PetscCheck(!PetscIsInfOrNanReal(value), PETSC_COMM_SELF, PETSC_ERR_FP,
177 "The initial value of '%s' is not finite for particle %lld at (%g, %g, %g).",
178 descriptor->canonical_name, (long long)pids[p], (double)values[0],
179 (double)values[1], (double)values[2]);
180 field[binding->components * p + c] = value / scale;
181 }
182 }
183 PetscCall(DMSwarmRestoreField(user->swarm, descriptor->canonical_name, NULL, NULL, (void **)&field));
184 }
185 PetscCall(DMSwarmRestoreField(user->swarm, ParticleFieldName(PARTICLE_FIELD_ID_PID), NULL, NULL, (void **)&pids));
186 PetscCall(DMSwarmRestoreField(user->swarm, ParticleFieldName(PARTICLE_FIELD_ID_POSITION), NULL, NULL, (void **)&positions));
187 PROFILE_FUNCTION_END;
188 PetscFunctionReturn(0);
189}

191/**
192 * @brief Implementation of \ref ParticleFieldPlanSummarize().
193 * @see ParticleFieldPlanSummarize()
194 */
195PetscErrorCode ParticleFieldPlanSummarize(UserCtx *user, const ParticleFieldPlan *plan)
196{
197 SimCtx *simCtx = NULL;
198 FILE *csv = NULL;
199 PetscInt nlocal = 0;
200
201 PetscFunctionBeginUser;
202 PROFILE_FUNCTION_BEGIN;
203 if (!plan) { PROFILE_FUNCTION_END; PetscFunctionReturn(0); }
204 simCtx = user->simCtx;
205 PetscCall(DMSwarmGetLocalSize(user->swarm, &nlocal));
206 if (simCtx->rank == 0) {
207 char path[PETSC_MAX_PATH_LEN];
208
209 PetscCall(PetscSNPrintf(path, sizeof(path), "%s/particle_initial_fields.csv", simCtx->analysis_dir));
210 csv = fopen(path, "w");
211 if (!csv) {
212 LOG_ALLOW(GLOBAL, LOG_WARNING, "Could not write '%s'.\n", path);
213 } else {
214 fprintf(csv, "field,component,count,mean,variance,min,max");
215 for (int bin = 0; bin < PARTICLE_FIELD_HISTOGRAM_BINS; ++bin) fprintf(csv, ",bin_%d", bin);
216 fprintf(csv, "\n");
217 }
218 }
219
220 for (PetscInt b = 0; b < plan->count; ++b) {
221 const ParticleFieldBinding *binding = &plan->bindings[b];
222 const ParticleFieldDescriptor *descriptor = NULL;
223 const PetscReal *field = NULL;
224 PetscReal scale = 1.0;
225
226 PetscCall(ParticleFieldGetDescriptor(binding->field, &descriptor));
227 PetscCall(FieldDimensionReferenceScale(&simCtx->scaling, descriptor->dimension, &scale));
228 PetscCall(DMSwarmGetField(user->swarm, descriptor->canonical_name, NULL, NULL, (void **)&field));
229 for (PetscInt c = 0; c < binding->components; ++c) {
230 /* Sums and extremes in one reduction, then a histogram over the global range. */
231 PetscReal sums[3] = {0.0, 0.0, 0.0}, extremes[2] = {PETSC_MAX_REAL, PETSC_MAX_REAL};
232 PetscReal bins_local[PARTICLE_FIELD_HISTOGRAM_BINS] = {0.0}, bins[PARTICLE_FIELD_HISTOGRAM_BINS];
233
234 for (PetscInt p = 0; p < nlocal; ++p) {
235 const PetscReal value = field[binding->components * p + c] * scale;
236 sums[0] += 1.0;
237 sums[1] += value;
238 sums[2] += value * value;
239 extremes[0] = PetscMin(extremes[0], value);
240 extremes[1] = PetscMin(extremes[1], -value);
241 }
242 PetscCallMPI(MPI_Allreduce(MPI_IN_PLACE, sums, 3, MPIU_REAL, MPI_SUM, PETSC_COMM_WORLD));
243 PetscCallMPI(MPI_Allreduce(MPI_IN_PLACE, extremes, 2, MPIU_REAL, MPI_MIN, PETSC_COMM_WORLD));
244 const PetscReal count = sums[0], lowest = extremes[0], highest = -extremes[1];
245 const PetscReal mean = (count > 0.0) ? sums[1] / count : 0.0;
246 const PetscReal variance = (count > 0.0) ? PetscMax(sums[2] / count - mean * mean, 0.0) : 0.0;
247 const PetscReal width = (highest > lowest) ? (highest - lowest) / PARTICLE_FIELD_HISTOGRAM_BINS : 1.0;
248
249 for (PetscInt p = 0; p < nlocal; ++p) {
250 const PetscReal value = field[binding->components * p + c] * scale;
251 PetscInt bin = (PetscInt)((value - lowest) / width);
252 bins_local[PetscMax(0, PetscMin(bin, PARTICLE_FIELD_HISTOGRAM_BINS - 1))] += 1.0;
253 }
254 PetscCallMPI(MPI_Allreduce(bins_local, bins, PARTICLE_FIELD_HISTOGRAM_BINS, MPIU_REAL, MPI_SUM,
255 PETSC_COMM_WORLD));
256 LOG_ALLOW(GLOBAL, LOG_INFO,
257 "[Particle IC] %s[%" PetscInt_FMT "]: n=%.0f mean=%.6g var=%.6g min=%.6g max=%.6g\n",
258 descriptor->canonical_name, c, (double)count, (double)mean, (double)variance,
259 (double)(count > 0.0 ? lowest : 0.0), (double)(count > 0.0 ? highest : 0.0));
260 if (csv) {
261 fprintf(csv, "%s,%d,%.0f,%.10e,%.10e,%.10e,%.10e", descriptor->canonical_name, (int)c,
262 (double)count, (double)mean, (double)variance,
263 (double)(count > 0.0 ? lowest : 0.0), (double)(count > 0.0 ? highest : 0.0));
264 for (int bin = 0; bin < PARTICLE_FIELD_HISTOGRAM_BINS; ++bin) fprintf(csv, ",%.0f", (double)bins[bin]);
265 fprintf(csv, "\n");
266 }
267 }
268 PetscCall(DMSwarmRestoreField(user->swarm, descriptor->canonical_name, NULL, NULL, (void **)&field));
269 }
270 if (csv) fclose(csv);
271 PROFILE_FUNCTION_END;
272 PetscFunctionReturn(0);
273}

280{
281 PetscFunctionBeginUser;
282 PROFILE_HELPER_BEGIN;
283 if (!plan || !*plan) { PROFILE_HELPER_END; PetscFunctionReturn(0); }
284 for (PetscInt b = 0; b < (*plan)->count; ++b) {
285 for (PetscInt c = 0; c < PARTICLE_FIELD_MAX_COMPONENTS; ++c) {

291 PROFILE_HELPER_END;
