21 PetscReal delta2,
22 PetscReal C_model)
23{
24 PetscFunctionBeginUser;
25
26 if (field_id == PARTICLE_FIELD_ID_PSI) {
27 // The IEM model relaxes Psi exponentially toward its target, mean_val, over dt.
28 PetscReal omega = C_model * diffusivity / delta2;
29 PetscReal decay = PetscExpReal(-omega * dt);
30

32 *psi_io = mean_val + (psi_old - mean_val) * decay;
33 }

71 PETSC_COMM_SELF, PETSC_ERR_SUP,
72 "Particle field '%s' has no registered model-update kernel.",
73 descriptor->canonical_name);

76 // Relax toward the mean of the particles where they are now. The Eulerian Psi otherwise
77 // holds the previous step's scatter, taken before the particles moved and migrated; the
78 // scatter recounts the particles per cell itself. Collective, so it comes before the early
79 // return of ranks with no particles.

81 Vec slope = NULL, centroid = NULL;
82
83 // Psi relaxation requires ghosted Eulerian mean and Jacobian fields.
84 PetscCheck(user->lPsi && user->lAj, PETSC_COMM_SELF, PETSC_ERR_ARG_WRONGSTATE, "UserCtx lPsi or lAj not initialized.");
85 ierr = ScatterAllParticleFieldsToEulerFields(user); CHKERRQ(ierr);
86 // The target is the cell's linear reconstruction m + g . (x - xc), with xc the cell's
87 // particle centroid so each cell mean is kept. Relaxing to the stepped mean m alone
88 // adds transport that does not vanish with refinement. A particle's cell is stored one
89 // index past its origin node, so both fields are read through ghosted copies.
90 // The scatter's count holds only this rank's particles in its ghost cells; the empty-cell
91 // test reads neighbours across subdomain seams, so their totals are fetched first.
92 ierr = UpdateLocalGhosts(user, FIELD_ID_PARTICLE_COUNT); CHKERRQ(ierr);
93 ierr = DMGetGlobalVector(user->fda, &slope); CHKERRQ(ierr);
94 ierr = ComputeCellScalarGradient(user, user->lPsi, user->lParticleCount, slope); CHKERRQ(ierr);
95 ierr = DMGetLocalVector(user->fda, &lslope); CHKERRQ(ierr);
96 ierr = DMGlobalToLocal(user->fda, slope, INSERT_VALUES, lslope); CHKERRQ(ierr);
97 // The scatter above has just counted the particles, so the centroid is not recounted.
98 ierr = DMGetGlobalVector(user->fda, &centroid); CHKERRQ(ierr);
99 ierr = ScatterParticleFieldCellMean(user, PARTICLE_FIELD_ID_POSITION, user->fda, user->ParticleCount, PETSC_FALSE,
100 centroid); CHKERRQ(ierr);
101 ierr = DMGetLocalVector(user->fda, &lcentroid); CHKERRQ(ierr);
102 ierr = DMGlobalToLocal(user->fda, centroid, INSERT_VALUES, lcentroid); CHKERRQ(ierr);
103 ierr = DMRestoreGlobalVector(user->fda, &centroid); CHKERRQ(ierr);
104 ierr = DMRestoreGlobalVector(user->fda, &slope); CHKERRQ(ierr);
105 }
106
107 ierr = DMSwarmGetLocalSize(swarm, &n_local); CHKERRQ(ierr);
108 if (n_local == 0) {
109 if (lslope) { ierr = DMRestoreLocalVector(user->fda, &lslope); CHKERRQ(ierr); }
110 if (lcentroid) { ierr = DMRestoreLocalVector(user->fda, &lcentroid); CHKERRQ(ierr); }
111 PROFILE_FUNCTION_END;
112 PetscFunctionReturn(0);
113 }
114

117

121 ierr = DMSwarmGetField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_POSITION), NULL, NULL, (void**)&pos_arr); CHKERRQ(ierr);
122
123 ierr = DMDAVecGetArrayRead(da, user->lPsi, &grid_mean); CHKERRQ(ierr);
124 ierr = DMDAVecGetArrayRead(da, user->lAj, &grid_aj); CHKERRQ(ierr);
125 ierr = DMDAVecGetArrayRead(user->fda, lslope, &grid_slope); CHKERRQ(ierr);
126 ierr = DMDAVecGetArrayRead(user->fda, lcentroid, &grid_centroid); CHKERRQ(ierr);
127 accessed_eulerian = PETSC_TRUE;
128 }
129
130 // The filter width depends only on the cell, and the particles are stored in cell order,
131 // so it is recomputed only when the cell changes.
132 PetscInt width_cell[3] = {-1, -1, -1};
133 PetscReal p_delta2 = 1.0;
134 for (PetscInt p = 0; p < n_local; ++p) {
135 PetscReal p_diff = 0.0;
136 PetscReal p_mean = 0.0;
137
138 if (field_id == PARTICLE_FIELD_ID_PSI) {
139 // Cell IDs are 0-based geometric indices; cell-centred storage starts at index 1
140 // (index 0 is the boundary slot), the shift the scatter applies when it writes lPsi.
141 PetscInt i = cell_arr[3*p + 0] + 1;
142 PetscInt j = cell_arr[3*p + 1] + 1;
143 PetscInt k = cell_arr[3*p + 2] + 1;
144
145 const Cmpnts g = grid_slope[k][j][i], c = grid_centroid[k][j][i];
146 p_diff = diff_arr[p];
147 p_mean = grid_mean[k][j][i] + g.x * (pos_arr[3*p + 0] - c.x) + g.y * (pos_arr[3*p + 1] - c.y)
148 + g.z * (pos_arr[3*p + 2] - c.z);
149
150 if (i != width_cell[0] || j != width_cell[1] || k != width_cell[2]) {
151 // Aj is the reciprocal cell volume in the curvilinear-grid representation.
152 PetscReal jac = grid_aj[k][j][i];
153 PetscReal p_vol = (jac > 1.0e-14) ? (1.0 / jac) : 1.0e-14;
154 // Guard the LES mixing time scale against degenerate or cut-cell volumes.
155 if (p_vol < 1.0e-14) p_vol = 1.0e-14;
156 p_delta2 = PetscPowReal(p_vol, 0.6666667);
157 width_cell[0] = i; width_cell[1] = j; width_cell[2] = k;
158 }
159 }
160
161 const PetscReal psi_old = psi_arr[p];
162 ierr = UpdateParticleField(field_id, dt, &psi_arr[p], p_diff, p_mean, p_delta2, C_IEM);
163 CHKERRQ(ierr);
164 LOG_PARTICLE(user->simCtx, pid_arr[p], "mix",
165 "%s %.10e -> %.10e | target %.10e | diffusivity %.6e mixing length^2 %.6e C %.4f dt %.6e\n",
166 fieldName, psi_old, psi_arr[p], p_mean, p_diff, p_delta2, C_IEM, dt);
167 }
168
169 ierr = DMSwarmRestoreField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_PID), NULL, NULL, (void**)&pid_arr); CHKERRQ(ierr);
170 ierr = DMSwarmRestoreField(swarm, fieldName, NULL, NULL, (void**)&psi_arr); CHKERRQ(ierr);
171
172 if (field_id == PARTICLE_FIELD_ID_PSI) {
173 ierr = DMSwarmRestoreField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_DIFFUSIVITY), NULL, NULL, (void**)&diff_arr); CHKERRQ(ierr);
174 ierr = DMSwarmRestoreField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_CELL_ID), NULL, NULL, (void**)&cell_arr); CHKERRQ(ierr);
175 ierr = DMSwarmRestoreField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_POSITION), NULL, NULL, (void**)&pos_arr); CHKERRQ(ierr);
176 }
177
178 if (accessed_eulerian) {
179 ierr = DMDAVecRestoreArrayRead(da, user->lPsi, &grid_mean); CHKERRQ(ierr);
180 ierr = DMDAVecRestoreArrayRead(da, user->lAj, &grid_aj); CHKERRQ(ierr);
181 ierr = DMDAVecRestoreArrayRead(user->fda, lslope, &grid_slope); CHKERRQ(ierr);
182 ierr = DMDAVecRestoreArrayRead(user->fda, lcentroid, &grid_centroid); CHKERRQ(ierr);
183 ierr = DMRestoreLocalVector(user->fda, &lslope); CHKERRQ(ierr);
184 ierr = DMRestoreLocalVector(user->fda, &lcentroid); CHKERRQ(ierr);
185 }

189 PROFILE_FUNCTION_END;

230/**
231 * @brief Implementation of \ref ComputeEulerianDiffusivity().
232 * @details Full API contract (arguments, ownership, side effects) is documented with
233 * the header declaration in `include/ParticlePhysics.h`.
234 * @see ComputeEulerianDiffusivity()
235 */
236PetscErrorCode ComputeEulerianDiffusivity(UserCtx *user)
237{
238 const SimCtx *simCtx = user->simCtx;
239 const DMDALocalInfo info = user->info;
240 // Grid uses indices 1 to M-2 for physical cells; index 0 and M-1 are boundary holders.
241 const PetscInt lo[3] = {PetscMax(info.xs, 1), PetscMax(info.ys, 1), PetscMax(info.zs, 1)};
242 const PetscInt hi[3] = {PetscMin(info.xs + info.xm, info.mx - 1), PetscMin(info.ys + info.ym, info.my - 1),
243 PetscMin(info.zs + info.zm, info.mz - 1)};
244 PetscReal ***diff_arr = NULL;
245 const PetscReal ***nut_arr = NULL;
246
247 PetscFunctionBeginUser;
248 PROFILE_FUNCTION_BEGIN;
249
250 if (VerificationDiffusivityOverrideActive(user->simCtx)) {
251 PetscCall(ApplyVerificationDiffusivityOverride(user));
252 PROFILE_FUNCTION_END;
253 PetscFunctionReturn(0);
254 }
255
256 // Gamma = nu / Sc + nu_t / Sc_t, with nu = 1/Re (0 when Re is unset or not positive) and the
257 // Schmidt numbers 1 and 0.7 when unset. The eddy viscosity enters only with LES; a dynamic
258 // model can give a slightly negative one, which is clipped to 0 so the Brownian amplitude
259 // sqrt(2 Gamma dt) stays real.
260 const PetscReal nu_molecular = (simCtx->ren > 1.0e-12) ? 1.0 / simCtx->ren : 0.0;
261 const PetscReal Sc = (simCtx->schmidt_number > 1.0e-6) ? simCtx->schmidt_number : 1.0;
262 const PetscReal Sct = (simCtx->Turbulent_schmidt_number > 1.0e-6) ? simCtx->Turbulent_schmidt_number : 0.7;
263 const PetscReal gamma_molecular = nu_molecular / Sc;
264 const PetscBool use_turbulence_model = simCtx->les ? PETSC_TRUE : PETSC_FALSE;
265
266 PetscCall(DMDAVecGetArray(user->da, user->Diffusivity, &diff_arr));
267 if (use_turbulence_model) PetscCall(DMDAVecGetArrayRead(user->da, user->Nu_t, (void *)&nut_arr));
268 for (PetscInt k = lo[2]; k < hi[2]; k++) {
269 for (PetscInt j = lo[1]; j < hi[1]; j++) {
270 for (PetscInt i = lo[0]; i < hi[0]; i++) {
271 PetscReal gamma_turbulent = 0.0;
272 if (use_turbulence_model) gamma_turbulent = PetscMax(nut_arr[k][j][i], 0.0) / Sct;
273 diff_arr[k][j][i] = gamma_molecular + gamma_turbulent;
274 }
275 }
276 }
277 if (use_turbulence_model) PetscCall(DMDAVecRestoreArrayRead(user->da, user->Nu_t, (void *)&nut_arr));
278 PetscCall(DMDAVecRestoreArray(user->da, user->Diffusivity, &diff_arr));
279
280 // The diffusivity gradient and the particle interpolation read the periodic dummy cells and
281 // the ghosts.
282 const FieldId periodic_fields[] = {FIELD_ID_DIFFUSIVITY};
283 PetscCall(SynchronizePeriodicCellFields(user, 1, periodic_fields));
284 PetscCall(UpdateLocalGhosts(user, FIELD_ID_DIFFUSIVITY));
285 PROFILE_FUNCTION_END;
286 PetscFunctionReturn(0);
287}

289/**
290 * @brief Cartesian gradient of a cell-centred scalar at every owned physical cell.
291 * @details Each computational derivative is the central difference, or at a non-periodic
292 * boundary cell the second-order one-sided difference into the domain, mapped to
293 * Cartesian components with the cell's metrics. With `lcount`, a derivative whose
294 * stencil reaches a cell holding no particles is zero, so an empty cell's zero mean
295 * does not enter. Writes owned entries only; local to this translation unit.
296 * @param[in] user Grid context.
297 * @param[in] lscalar Ghosted scalar, its ghosts current.
298 * @param[in] lcount Ghosted particle count, its ghosts current; or NULL.
299 * @param[out] gradient Global vector on `fda`.
300 */
301static PetscErrorCode ComputeCellScalarGradient(UserCtx *user, Vec lscalar, Vec lcount, Vec gradient)
302{
303 DMDALocalInfo info = user->info;
304 const PetscInt n[3] = {info.mx, info.my, info.mz};
305 PetscBool periodic[3];
306 GetPeriodicAxes(user->simCtx, periodic);
307 // Grid uses indices 1 to M-2 for physical cells; index 0 and M-1 are boundary holders.
308 const PetscInt lo[3] = {PetscMax(info.xs, 1), PetscMax(info.ys, 1), PetscMax(info.zs, 1)};
309 const PetscInt hi[3] = {PetscMin(info.xs + info.xm, n[0] - 1), PetscMin(info.ys + info.ym, n[1] - 1),
310 PetscMin(info.zs + info.zm, n[2] - 1)};
311 const PetscReal ***f = NULL, ***count = NULL, ***aj = NULL;
312 const Cmpnts ***csi = NULL, ***eta = NULL, ***zet = NULL;
313 Cmpnts ***grad = NULL;
314
315 PetscFunctionBeginUser;
316 PROFILE_FUNCTION_BEGIN;
317 PetscCall(DMDAVecGetArrayRead(user->da, lscalar, (void *)&f));
318 if (lcount) PetscCall(DMDAVecGetArrayRead(user->da, lcount, (void *)&count));
319 PetscCall(DMDAVecGetArrayRead(user->fda, user->lCsi, (void *)&csi));
320 PetscCall(DMDAVecGetArrayRead(user->fda, user->lEta, (void *)&eta));
321 PetscCall(DMDAVecGetArrayRead(user->fda, user->lZet, (void *)&zet));
322 PetscCall(DMDAVecGetArrayRead(user->da, user->lAj, (void *)&aj));
323 PetscCall(DMDAVecGetArray(user->fda, gradient, &grad));
324#define CELL(a, d, o) ((d) == 0 ? (a)[k][j][i + (o)] : (d) == 1 ? (a)[k][j + (o)][i] : (a)[k + (o)][j][i])
325 for (PetscInt k = lo[2]; k < hi[2]; k++) {
326 for (PetscInt j = lo[1]; j < hi[1]; j++) {
327 for (PetscInt i = lo[0]; i < hi[0]; i++) {
328 const PetscInt idx[3] = {i, j, k};
329 PetscReal dG[3];
330 for (PetscInt d = 0; d < 3; ++d) {
331 PetscInt o[3] = {-1, 1, 0};
332 PetscReal c[3] = {-1.0, 1.0, 0.0}; /* central: [-1, 0, 1] / 2 */
333 if (!periodic[d] && idx[d] == 1) {
334 o[0] = 0; o[1] = 1; o[2] = 2; c[0] = -3.0; c[1] = 4.0; c[2] = -1.0; /* forward: [-3, 4, -1] / 2 */
335 } else if (!periodic[d] && idx[d] == n[d] - 2) {
336 o[0] = 0; o[1] = -1; o[2] = -2; c[0] = 3.0; c[1] = -4.0; c[2] = 1.0; /* backward: [1, -4, 3] / 2 */
337 }
338 dG[d] = (c[0] * CELL(f, d, o[0]) + c[1] * CELL(f, d, o[1]) + c[2] * CELL(f, d, o[2])) * 0.5;
339 if (count) {
340 for (PetscInt q = 0; q < 3; ++q) if (c[q] != 0.0 && CELL(count, d, o[q]) < 0.5) dG[d] = 0.0;
341 }
342 }
343 TransformScalarDerivativesToPhysical(aj[k][j][i], csi[k][j][i], eta[k][j][i], zet[k][j][i],
344 dG[0], dG[1], dG[2], &grad[k][j][i]);
345 }
346 }
347 }
348#undef CELL
349 PetscCall(DMDAVecRestoreArray(user->fda, gradient, &grad));
350 PetscCall(DMDAVecRestoreArrayRead(user->da, user->lAj, (void *)&aj));
351 PetscCall(DMDAVecRestoreArrayRead(user->fda, user->lZet, (void *)&zet));
352 PetscCall(DMDAVecRestoreArrayRead(user->fda, user->lEta, (void *)&eta));
353 PetscCall(DMDAVecRestoreArrayRead(user->fda, user->lCsi, (void *)&csi));
354 if (lcount) PetscCall(DMDAVecRestoreArrayRead(user->da, lcount, (void *)&count));
355 PetscCall(DMDAVecRestoreArrayRead(user->da, lscalar, (void *)&f));
356 PROFILE_FUNCTION_END;
357 PetscFunctionReturn(0);
358}
