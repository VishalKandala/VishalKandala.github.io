11 * @brief Internal helper implementation: `UpdateParticleField()`.
12 * @details Local to this translation unit.
13 */
14PetscErrorCode UpdateParticleField(ParticleFieldId field_id,
15 PetscReal dt,
16 PetscReal *psi_io,
17 PetscReal diffusivity,
18 PetscReal mean_val,
19 PetscReal delta2,
20 PetscReal C_model)
21{
22 PetscFunctionBeginUser;
23
24 if (field_id == PARTICLE_FIELD_ID_PSI) {
25 // The IEM model relaxes Psi exponentially toward the cell mean over dt.
26 PetscReal omega = C_model * diffusivity / delta2;
27 PetscReal decay = PetscExpReal(-omega * dt);
28
29 PetscReal psi_old = *psi_io;
30 *psi_io = mean_val + (psi_old - mean_val) * decay;
31 }
32
33 PetscFunctionReturn(0);
34}

37 * @brief Internal helper implementation: `UpdateFieldForAllParticles()`.
38 * @details Local to this translation unit.
39 */
40PetscErrorCode UpdateFieldForAllParticles(UserCtx *user, ParticleFieldId field_id)
41{
42 PetscErrorCode ierr;
43 DM swarm = user->swarm;
44 DM da = user->da;
45 PetscInt n_local;
46 PetscReal dt = user->simCtx->dt;
47 PetscReal C_IEM = user->simCtx->iem_constant;
48
49 PetscReal *psi_arr = NULL;
50 PetscReal *diff_arr = NULL;
51 PetscInt *cell_arr = NULL;
52 PetscInt64 *pid_arr = NULL;
53
54 PetscReal ***grid_mean = NULL;
55 PetscReal ***grid_aj = NULL;
56
57 PetscBool accessed_eulerian = PETSC_FALSE;
58 const ParticleFieldDescriptor *descriptor = NULL;
59 const char *fieldName = NULL;
60
61 PetscFunctionBeginUser;
62 PROFILE_FUNCTION_BEGIN;
63
64 ierr = ParticleFieldGetDescriptor(field_id, &descriptor); CHKERRQ(ierr);
65 PetscCheck((descriptor->capabilities & PARTICLE_FIELD_CAPABILITY_MODEL_UPDATE) != 0,
66 PETSC_COMM_SELF, PETSC_ERR_SUP,
67 "Particle field '%s' has no registered model-update kernel.",
68 descriptor->canonical_name);
69 fieldName = descriptor->canonical_name;
70
71 // Relax toward the mean of the particles where they are now. The Eulerian Psi otherwise
72 // holds the previous step's scatter, taken before the particles moved and migrated; the
73 // scatter recounts the particles per cell itself. Collective, so it comes before the early
74 // return of ranks with no particles.
75 if (field_id == PARTICLE_FIELD_ID_PSI) {
76 ierr = ScatterAllParticleFieldsToEulerFields(user); CHKERRQ(ierr);
77 }
78
79 ierr = DMSwarmGetLocalSize(swarm, &n_local); CHKERRQ(ierr);
80 if (n_local == 0) {
81 PROFILE_FUNCTION_END;
82 PetscFunctionReturn(0);
83 }
84
85 ierr = DMSwarmGetField(swarm, fieldName, NULL, NULL, (void**)&psi_arr); CHKERRQ(ierr);
86 ierr = DMSwarmGetField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_PID), NULL, NULL, (void**)&pid_arr); CHKERRQ(ierr);
87
88 if (field_id == PARTICLE_FIELD_ID_PSI) {
89 ierr = DMSwarmGetField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_DIFFUSIVITY), NULL, NULL, (void**)&diff_arr); CHKERRQ(ierr);
90 ierr = DMSwarmGetField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_CELL_ID), NULL, NULL, (void**)&cell_arr); CHKERRQ(ierr);
91
92 // Psi relaxation requires ghosted Eulerian mean and Jacobian fields.
93 if (!user->lPsi || !user->lAj) {
94 SETERRQ(PETSC_COMM_SELF, PETSC_ERR_ARG_WRONGSTATE, "UserCtx lPsi or lAj not initialized.");
95 }
96
97 ierr = DMDAVecGetArrayRead(da, user->lPsi, &grid_mean); CHKERRQ(ierr);
98 ierr = DMDAVecGetArrayRead(da, user->lAj, &grid_aj); CHKERRQ(ierr);
99 accessed_eulerian = PETSC_TRUE;
100 }
101
102 // The filter width depends only on the cell, and the particles are stored in cell order,
103 // so it is recomputed only when the cell changes.
104 PetscInt width_cell[3] = {-1, -1, -1};
105 PetscReal p_delta2 = 1.0;
106 for (PetscInt p = 0; p < n_local; ++p) {
107 PetscReal p_diff = 0.0;
108 PetscReal p_mean = 0.0;
109
110 if (field_id == PARTICLE_FIELD_ID_PSI) {
111 // Cell IDs are 0-based geometric indices; cell-centred storage starts at index 1
112 // (index 0 is the boundary slot), the shift the scatter applies when it writes lPsi.
113 PetscInt i = cell_arr[3*p + 0] + 1;
114 PetscInt j = cell_arr[3*p + 1] + 1;
115 PetscInt k = cell_arr[3*p + 2] + 1;
116
117 p_diff = diff_arr[p];
118 p_mean = grid_mean[k][j][i];
119
120 if (i != width_cell[0] || j != width_cell[1] || k != width_cell[2]) {
121 // Aj is the reciprocal cell volume in the curvilinear-grid representation.
122 PetscReal jac = grid_aj[k][j][i];
123 PetscReal p_vol = (jac > 1.0e-14) ? (1.0 / jac) : 1.0e-14;
124 // Guard the LES mixing time scale against degenerate or cut-cell volumes.
125 if (p_vol < 1.0e-14) p_vol = 1.0e-14;
126 p_delta2 = PetscPowReal(p_vol, 0.6666667);
127 width_cell[0] = i; width_cell[1] = j; width_cell[2] = k;
128 }
129 }
130
131 const PetscReal psi_old = psi_arr[p];
132 ierr = UpdateParticleField(field_id, dt, &psi_arr[p], p_diff, p_mean, p_delta2, C_IEM);
133 CHKERRQ(ierr);
134 LOG_PARTICLE(user->simCtx, pid_arr[p], "mix",
135 "%s %.10e -> %.10e | cell mean %.10e | diffusivity %.6e mixing length^2 %.6e C %.4f dt %.6e\n",
136 fieldName, psi_old, psi_arr[p], p_mean, p_diff, p_delta2, C_IEM, dt);
137 }
138
139 ierr = DMSwarmRestoreField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_PID), NULL, NULL, (void**)&pid_arr); CHKERRQ(ierr);
140 ierr = DMSwarmRestoreField(swarm, fieldName, NULL, NULL, (void**)&psi_arr); CHKERRQ(ierr);
141
142 if (field_id == PARTICLE_FIELD_ID_PSI) {
143 ierr = DMSwarmRestoreField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_DIFFUSIVITY), NULL, NULL, (void**)&diff_arr); CHKERRQ(ierr);
144 ierr = DMSwarmRestoreField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_CELL_ID), NULL, NULL, (void**)&cell_arr); CHKERRQ(ierr);
145 }
146
147 if (accessed_eulerian) {
148 ierr = DMDAVecRestoreArrayRead(da, user->lPsi, &grid_mean); CHKERRQ(ierr);
149 ierr = DMDAVecRestoreArrayRead(da, user->lAj, &grid_aj); CHKERRQ(ierr);
150 }
151
152 LOG_ALLOW(GLOBAL, LOG_DEBUG, "Updated particle physics for field '%s'.\n", fieldName);
153
154 PROFILE_FUNCTION_END;
155 PetscFunctionReturn(0);
156}

195/**
196 * @brief Implementation of \ref ComputeEulerianDiffusivity().
197 * @details Full API contract (arguments, ownership, side effects) is documented with
198 * the header declaration in `include/ParticlePhysics.h`.
199 * @see ComputeEulerianDiffusivity()
200 */
201PetscErrorCode ComputeEulerianDiffusivity(UserCtx *user)
202{
203 PetscErrorCode ierr;
204 DM da = user->da;
205 PetscInt i, j, k, xs, ys, zs, xm, ym, zm, xe, ye, ze;
206 PetscInt lxs, lys, lzs, lxe, lye, lze;
207
208 // Pointers for 3D grid access
209 PetscReal ***diff_arr; // Output: Diffusivity field
210 PetscReal ***nut_arr; // Input: Eddy Viscosity field (optional)
211
212 // Physics parameters
213 PetscReal nu_molecular, gamma_molecular;
214 PetscReal nu_turbulent, gamma_turbulent;
215 PetscReal Sc, Sct;
216 PetscBool use_turbulence_model;
217
218 PetscFunctionBeginUser;
219 PROFILE_FUNCTION_BEGIN;
220
221 if (VerificationDiffusivityOverrideActive(user->simCtx)) {
222 ierr = ApplyVerificationDiffusivityOverride(user); CHKERRQ(ierr);
223 PROFILE_FUNCTION_END;
224 PetscFunctionReturn(0);
225 }
226
227 // ------------------------------------------------------------------------
228 // 1. Parameter Setup & Safety Checks
229 // ------------------------------------------------------------------------
230
231 // Determine Molecular Viscosity (nu = 1/Re)
232 // Guard against division by zero if Re is not set or infinite (inviscid)
233 if (user->simCtx->ren > 1.0e-12) {
234 nu_molecular = 1.0 / user->simCtx->ren;
235 } else {
236 nu_molecular = 0.0;
237 }
238
239 // Set Schmidt Numbers (Default to 1.0 if not provided to prevent NaN)
240 Sc = (user->simCtx->schmidt_number > 1.0e-6) ? user->simCtx->schmidt_number : 1.0;
241 Sct = (user->simCtx->Turbulent_schmidt_number > 1.0e-6) ? user->simCtx->Turbulent_schmidt_number : 0.7;
242
243 // Pre-calculate molecular component
244 gamma_molecular = nu_molecular / Sc;
245
246 // Check if a turbulence model is active
247 use_turbulence_model = user->simCtx->les ? PETSC_TRUE : PETSC_FALSE;
248
249 // ------------------------------------------------------------------------
250 // 2. Data Access
251 // ------------------------------------------------------------------------
252
253 // Get local grid boundaries
254 DMDALocalInfo info;
255 info = user->info;
256
257 xs = info.xs; ys = info.ys; zs = info.zs;
258 xm = info.xm; ym = info.ym; zm = info.zm;
259 xe = xs + xm; ye = ys + ym; ze = zs + zm;
260
261 lxs = (xs == 0)? xs + 1 : xs;
262 lys = (ys == 0)? ys + 1 : ys;
263 lzs = (zs == 0)? zs + 1 : zs;
264 lxe = (xe == info.mx)? xe - 1 : xe;
265 lye = (ye == info.my)? ye - 1 : ye;
266 lze = (ze == info.mz)? ze - 1 : ze;
267
268 // Get write access to the output Diffusivity array
269 ierr = DMDAVecGetArray(da, user->Diffusivity, &diff_arr); CHKERRQ(ierr);
270
271 // Get read access to Eddy Viscosity only if turbulence is active
272 if (use_turbulence_model) {
273 ierr = DMDAVecGetArrayRead(da, user->Nu_t, &nut_arr); CHKERRQ(ierr);
274 }
275
276 // ------------------------------------------------------------------------
277 // 3. Calculation Loop
278 // ------------------------------------------------------------------------
279
280 for (k = lzs; k < lze; k++) {
281 for (j = lys; j < lye; j++) {
282 for (i = lxs; i < lxe; i++) {
283
284 gamma_turbulent = 0.0;
285
286 if (use_turbulence_model) {
287 // Fetch local eddy viscosity
288 nu_turbulent = nut_arr[k][j][i];
289
290 // NUMERICAL SAFETY:
291 // Some turbulence models (dynamic SGS) can locally produce
292 // slightly negative viscosity. We clamp this to 0 to prevent
293 // negative diffusivity, which crashes the Langevin sqrt().
294 if (nu_turbulent < 0.0) {
295 nu_turbulent = 0.0;
296 }
297
298 gamma_turbulent = nu_turbulent / Sct;
299 }
300
301 // Sum components
302 diff_arr[k][j][i] = gamma_molecular + gamma_turbulent;
303 }
304 }
305 }
306
307 // ------------------------------------------------------------------------
308 // 4. Cleanup & Synchronization
309 // ------------------------------------------------------------------------
310
311 // Restore arrays
312 if (use_turbulence_model) {
313 ierr = DMDAVecRestoreArrayRead(da, user->Nu_t, &nut_arr); CHKERRQ(ierr);
314 }
315 ierr = DMDAVecRestoreArray(da, user->Diffusivity, &diff_arr); CHKERRQ(ierr);
316
317 // Update Ghost Points
318 // This is required because downstream operations (Drift Gradient Calculation
319 // and Particle Interpolation) will need access to the halo regions of this field.
320 const FieldId periodic_fields[] = {FIELD_ID_DIFFUSIVITY};
321 ierr = SynchronizePeriodicCellFields(user, 1, periodic_fields); CHKERRQ(ierr);
322 ierr = UpdateLocalGhosts(user, FIELD_ID_DIFFUSIVITY); CHKERRQ(ierr);
323 PROFILE_FUNCTION_END;
324 PetscFunctionReturn(0);
325}

327/**
328 * @brief Internal helper implementation: `ComputeEulerianDiffusivityGradient()`.
329 * @details Local to this translation unit.
330 */
331PetscErrorCode ComputeEulerianDiffusivityGradient(UserCtx *user)
332{
333 PetscErrorCode ierr;
334 DM da = user->da, fda = user->fda;
335 DMDALocalInfo info = user->info;
336
337 // 1. Determine Global Dimensions
338 PetscInt mx = info.mx, my = info.my, mz = info.mz;
339
340 // 2. Determine Local Loop Bounds (Skip Ghosts/Unused Indices)
341 // Grid uses indices 1 to M-2 for physical cells.
342 // Index 0 and M-1 are ghost/boundary holders.
343
344 // Start: If we own the global start (0), skip it and start at 1.
345 PetscInt lxs = (info.xs == 0) ? 1 : info.xs;
346 // End: If we own the global end (mx), stop before it (mx-1), so loop covers mx-2.
347 PetscInt lxe = (info.xs + info.xm == mx) ? mx - 1 : info.xs + info.xm;
348
349 PetscInt lys = (info.ys == 0) ? 1 : info.ys;
350 PetscInt lye = (info.ys + info.ym == my) ? my - 1 : info.ys + info.ym;
351
352 PetscInt lzs = (info.zs == 0) ? 1 : info.zs;
353 PetscInt lze = (info.zs + info.zm == mz) ? mz - 1 : info.zs + info.zm;
354
355 PetscInt i, j, k;
356
357 // Pointers
358 PetscReal ***diff; // Input (Scalar)
359 Cmpnts ***grad_diff; // Output (Vector)
360 Cmpnts ***csi, ***eta, ***zet;
361 PetscReal ***aj;
362
363 // Boundary Flags (Check if Periodic)
364 PetscBool p_x = (user->boundary_faces[BC_FACE_NEG_X].mathematical_type == PERIODIC);
365 PetscBool p_y = (user->boundary_faces[BC_FACE_NEG_Y].mathematical_type == PERIODIC);
366 PetscBool p_z = (user->boundary_faces[BC_FACE_NEG_Z].mathematical_type == PERIODIC);
367
368 PetscFunctionBeginUser;
369 PROFILE_FUNCTION_BEGIN;
370
371 // 3. Update Ghosts for Diffusivity
372 // Required so that Central Differences at i=2 can read i=1,
373 // and Central Differences at Periodic Boundaries work correctly.
374 ierr = UpdateLocalGhosts(user, FIELD_ID_DIFFUSIVITY); CHKERRQ(ierr);
375
376 // 4. Get Arrays (Read Only)
377 ierr = DMDAVecGetArrayRead(da, user->lDiffusivity, &diff); CHKERRQ(ierr);
378 ierr = DMDAVecGetArrayRead(fda, user->lCsi, &csi); CHKERRQ(ierr);
379 ierr = DMDAVecGetArrayRead(fda, user->lEta, &eta); CHKERRQ(ierr);
380 ierr = DMDAVecGetArrayRead(fda, user->lZet, &zet); CHKERRQ(ierr);
381 ierr = DMDAVecGetArrayRead(da, user->lAj, &aj); CHKERRQ(ierr);
382
383 LOG_ALLOW(GLOBAL,LOG_DEBUG,"Diffusivity and Metrics arrays accessed successfully! .\n");
384
385 // 5. Get Output Array (Read/Write)
386 ierr = DMDAVecGetArray(fda, user->DiffusivityGradient, &grad_diff); CHKERRQ(ierr);
387
388 LOG_ALLOW(GLOBAL,LOG_DEBUG,"Diffusivity Gradient array accessed successfully! .\n");
389
390 // 6. Loop over Physical Domain
391 for (k = lzs; k < lze; k++) {
392 for (j = lys; j < lye; j++) {
393 for (i = lxs; i < lxe; i++) {
394
395 PetscReal dGdCsi, dGdEta, dGdZet;
396
397 // ---------------------------------------------------------
398 // I-Direction (Csi)
399 // ---------------------------------------------------------
400 if (!p_x && i == 1) {
401 // Physical Start: 2nd Order Forward Difference
402 // Stencil: [-3, 4, -1] / 2
403 dGdCsi = (-3.0 * diff[k][j][i] + 4.0 * diff[k][j][i+1] - diff[k][j][i+2]) * 0.5;
404 }
405 else if (!p_x && i == mx - 2) {
406 // Physical End: 2nd Order Backward Difference
407 // Stencil: [1, -4, 3] / 2
408 dGdCsi = (3.0 * diff[k][j][i] - 4.0 * diff[k][j][i-1] + diff[k][j][i-2]) * 0.5;
409 }
410 else {
411 // Interior / Periodic: Central Difference
412 // Stencil: [-1, 0, 1] / 2
413 dGdCsi = (diff[k][j][i+1] - diff[k][j][i-1]) * 0.5;
414 }
415
416 // ---------------------------------------------------------
417 // J-Direction (Eta)
418 // ---------------------------------------------------------
419 if (!p_y && j == 1) {
420 // Physical Start: Forward
421 dGdEta = (-3.0 * diff[k][j][i] + 4.0 * diff[k][j+1][i] - diff[k][j+2][i]) * 0.5;
422 }
423 else if (!p_y && j == my - 2) {
424 // Physical End: Backward
425 dGdEta = (3.0 * diff[k][j][i] - 4.0 * diff[k][j-1][i] + diff[k][j-2][i]) * 0.5;
426 }
427 else {
428 // Interior: Central
429 dGdEta = (diff[k][j+1][i] - diff[k][j-1][i]) * 0.5;
430 }
431
432 // ---------------------------------------------------------
433 // K-Direction (Zet)
434 // ---------------------------------------------------------
435 if (!p_z && k == 1) {
436 // Physical Start: Forward
437 dGdZet = (-3.0 * diff[k][j][i] + 4.0 * diff[k+1][j][i] - diff[k+2][j][i]) * 0.5;
438 }
439 else if (!p_z && k == mz - 2) {
440 // Physical End: Backward
441 dGdZet = (3.0 * diff[k][j][i] - 4.0 * diff[k-1][j][i] + diff[k-2][j][i]) * 0.5;
442 }
443 else {
444 // Interior: Central
445 dGdZet = (diff[k+1][j][i] - diff[k-1][j][i]) * 0.5;
446 }
447
448 // ---------------------------------------------------------
449 // Transform to Physical Space (Cartesian Gradient)
450 // ---------------------------------------------------------
451 TransformScalarDerivativesToPhysical(aj[k][j][i],
452 csi[k][j][i], eta[k][j][i], zet[k][j][i],
453 dGdCsi, dGdEta, dGdZet,
454 &grad_diff[k][j][i]);
455 }
456 }
457 }
458
459 // 7. Restore Arrays
460 ierr = DMDAVecRestoreArrayRead(da, user->lDiffusivity, &diff); CHKERRQ(ierr);
461 ierr = DMDAVecRestoreArrayRead(fda, user->lCsi, &csi); CHKERRQ(ierr);
462 ierr = DMDAVecRestoreArrayRead(fda, user->lEta, &eta); CHKERRQ(ierr);
463 ierr = DMDAVecRestoreArrayRead(fda, user->lZet, &zet); CHKERRQ(ierr);
464 ierr = DMDAVecRestoreArrayRead(da, user->lAj, &aj); CHKERRQ(ierr);
465 ierr = DMDAVecRestoreArray(fda, user->DiffusivityGradient, &grad_diff); CHKERRQ(ierr);
466
467 // 8. Update Ghosts for the Result
468 // Important: Particles near subdomain boundaries will need to interpolate
469 // this gradient vector, so the ghosts of the vector field must be filled.
470 ierr = UpdateLocalGhosts(user, FIELD_ID_DIFFUSIVITY_GRADIENT); CHKERRQ(ierr);
471
472 PROFILE_FUNCTION_END;
473 PetscFunctionReturn(0);
474}
