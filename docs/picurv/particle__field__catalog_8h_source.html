1/**
2 * @file particle_field_catalog.h
3 * @brief Typed identities and metadata for persistent solver-particle fields.
4 *
5 * Particle fields intentionally use a catalog separate from Eulerian fields.
6 * They are registered on a DMSwarm, may use integer storage, and do not own a
7 * global/local Vec pair in UserCtx. Text is translated only where PETSc or a
8 * user/file interface requires a registered DMSwarm name.
