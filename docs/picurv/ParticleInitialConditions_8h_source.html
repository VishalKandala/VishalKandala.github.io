1/**
2 * @file ParticleInitialConditions.h
3 * @brief Configured initial values of particle-carried fields.
4 *
5 * A plan binds each configured particle field to one compiled expression per component.
6 * Expressions are written in physical units: `x`, `y`, `z` are physical coordinates,
7 * `t` is physical time, and a value is divided by its field's catalog reference scale
8 * when written. The plan is applied to a contiguous local range of swarm entries, so the
9 * t=0 population and, later, particles injected at a boundary share one path.
10 *
11 * The expressions are compiled and evaluated by the shared engine in expression.h.
12 */

34/**
35 * @brief Read a plan from the options database.
36 * @details Reads `-<prefix>_count`, and for each field `i`, `-<prefix>_<i>_name` and one
37 * expression per component, `-<prefix>_<i>_expr_<c>`. Every named field must be a
38 * particle-carried field of the catalog (capability `USER_INITIALIZE`).
39 * @param[in] prefix Option prefix without its dash: `particle_fields` for the t=0 population,
40 * `particle_inlet_<n>_fields` for an inlet.
41 * @param[out] plan Compiled plan, or NULL when no field is configured.
42 * @return Zero on success; `PETSC_ERR_ARG_WRONG` for an unknown or non-settable field or a
43 * malformed expression.
44 */
45PetscErrorCode ParticleFieldPlanCreate(const char *prefix, ParticleFieldPlan **plan);

47/**
48 * @brief Write the plan's values into swarm entries `[first, end)` of this rank.
49 * @details Each rank evaluates its own entries. Every rank calls it even with an empty range: the
50 * first call that reads a grid line's fraction or length is collective, computing them once
51 * for the plan. Those expressions need each particle's cell and weights. A value that is not finite is an error naming the
52 * particle and its position.
53 * @param[in,out] user Block owning the swarm.
54 * @param[in] plan Plan to apply; NULL does nothing.
55 * @param[in] event Time and identity of the event producing these particles.
56 * @param[in] first First local entry.
57 * @param[in] end One past the last local entry.
58 * @return Zero on success.
59 */
60PetscErrorCode ParticleFieldPlanApply(UserCtx *user, ParticleFieldPlan *plan,
61 const ParticleFieldEvent *event, PetscInt first, PetscInt end);
