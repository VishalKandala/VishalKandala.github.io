13static PetscErrorCode TestComputeSpecificKE(void)
14{
15 SimCtx *simCtx = NULL;
16 UserCtx *user = NULL;
17 PetscReal (*vel_arr)[3] = NULL;
18 PetscScalar *ske_arr = NULL;
19
20 PetscFunctionBeginUser;
21 PetscCall(PicurvCreateMinimalContexts(&simCtx, &user, 4, 4, 4));

24 PetscCall(DMSwarmGetField(user->swarm, "velocity", NULL, NULL, (void *)&vel_arr));

28 vel_arr[1][0] = 0.0;

30 vel_arr[1][2] = 4.0;
31 PetscCall(DMSwarmRestoreField(user->swarm, "velocity", NULL, NULL, (void *)&vel_arr));

34 PetscCall(DMSwarmGetField(user->post_swarm, "ske", NULL, NULL, (void *)&ske_arr));

133static PetscErrorCode TestNormalizeRelativeField(void)
134{
135 SimCtx *simCtx = NULL;
136 UserCtx *user = NULL;
137 PetscInt ref_idx = 0;
138 PetscScalar ref_value = 0.0;
139 PetscReal vmin = 0.0, vmax = 0.0;
140
141 PetscFunctionBeginUser;
142 PetscCall(PicurvCreateMinimalContexts(&simCtx, &user, 4, 4, 4));
143 PetscCall(PetscCalloc1(1, &simCtx->pps));
144 simCtx->pps->reference[0] = 1;
145 simCtx->pps->reference[1] = 1;
146 simCtx->pps->reference[2] = 1;
147
148 PetscCall(VecSet(user->P, 10.0));
149 ref_idx = simCtx->pps->reference[2] * (user->info.mx * user->info.my) +
150 simCtx->pps->reference[1] * user->info.mx +
151 simCtx->pps->reference[0];
152 PetscCall(VecSetValue(user->P, ref_idx, 4.0, INSERT_VALUES));
153 PetscCall(VecAssemblyBegin(user->P));
154 PetscCall(VecAssemblyEnd(user->P));
155
156 PetscCall(NormalizeRelativeField(user, "P"));
157
158 PetscCall(VecGetValues(user->P, 1, &ref_idx, &ref_value));
159 PetscCall(PicurvAssertRealNear(0.0, PetscRealPart(ref_value), 1.0e-12, "NormalizeRelativeField reference value"));
160 PetscCall(VecMin(user->P, NULL, &vmin));
161 PetscCall(VecMax(user->P, NULL, &vmax));
162 PetscCall(PicurvAssertRealNear(0.0, vmin, 1.0e-12, "NormalizeRelativeField minimum"));
163 PetscCall(PicurvAssertRealNear(6.0, vmax, 1.0e-12, "NormalizeRelativeField maximum"));
164
165 PetscCall(PicurvDestroyMinimalContexts(&simCtx, &user));
166 PetscFunctionReturn(0);
167}

188/**
189 * @brief Tests that the per-step dimensionalization leaves the persistent grid alone.
190 * @details The postprocessor runs DimensionalizeAllLoadedFields once per processed step.
191 * Scaling the coordinates there multiplied them by L_ref once per step; they are
192 * loaded once and must be scaled once, by the postprocessor loop.
193 */
194static PetscErrorCode TestDimensionalizeAllLoadedFieldsLeavesCoordinatesAlone(void)
195{
196 SimCtx *simCtx = NULL;
197 UserCtx *user = NULL;
198 Vec coordinates = NULL, before = NULL;
199 PetscReal difference = 0.0;
200
201 PetscFunctionBeginUser;
202 PetscCall(PicurvCreateMinimalContexts(&simCtx, &user, 4, 4, 4));
203 simCtx->scaling.L_ref = 2.0;
204 simCtx->scaling.U_ref = 3.0;
205 simCtx->scaling.P_ref = 5.0;
206 PetscCall(DMGetCoordinates(user->da, &coordinates));
207 PetscCall(VecDuplicate(coordinates, &before));
208 PetscCall(VecCopy(coordinates, before));
209 PetscCall(VecSet(user->Ucat, 1.0));
210
211 PetscCall(DimensionalizeAllLoadedFields(user));
212 PetscCall(DimensionalizeAllLoadedFields(user));
213
214 PetscCall(VecAXPY(before, -1.0, coordinates));
215 PetscCall(VecNorm(before, NORM_INFINITY, &difference));
216 PetscCall(PicurvAssertRealNear(0.0, difference, 1.0e-14,
217 "per-step dimensionalization must not rescale the grid coordinates"));
218 PetscCall(PicurvAssertVecConstant(user->Ucat, 9.0, 1.0e-12,
219 "each call scales the reloaded velocity by U_ref"));
220
221 PetscCall(VecDestroy(&before));
222 PetscCall(PicurvDestroyMinimalContexts(&simCtx, &user));
223 PetscFunctionReturn(0);
224}

226/**
227 * @brief Tests that a dimensionalized Q-criterion carries (U_ref/L_ref)^2, not U_ref^2.
228 * @details Ucat is already dimensional when Q is computed, but the metrics are not, so
229 * the result must be divided by L_ref^2.
230 */
231static PetscErrorCode TestComputeQCriterionDimensionalizedScalesByLengthSquared(void)
232{
233 SimCtx *simCtx = NULL;
234 UserCtx *user = NULL;
235 Cmpnts ***ucat = NULL;
236 Vec nondimensional = NULL;
237 PostProcessParams *pps = NULL;
238 PetscReal difference = 0.0, reference = 0.0;
239
240 PetscFunctionBeginUser;
241 PetscCall(PicurvCreateMinimalContexts(&simCtx, &user, 6, 6, 6));
242 PetscCall(PicurvPopulateIdentityMetrics(user));
243 PetscCall(DMDAVecGetArray(user->fda, user->Ucat, &ucat));
244 for (PetscInt k = user->info.zs; k < user->info.zs + user->info.zm; k++)
245 for (PetscInt j = user->info.ys; j < user->info.ys + user->info.ym; j++)
246 for (PetscInt i = user->info.xs; i < user->info.xs + user->info.xm; i++)
247 ucat[k][j][i] = (Cmpnts){-(PetscReal)j, (PetscReal)i, 0.0}; /* solid-body rotation */
248 PetscCall(DMDAVecRestoreArray(user->fda, user->Ucat, &ucat));
249
250 PetscCall(ComputeQCriterion(user));
251 PetscCall(VecDuplicate(user->Qcrit, &nondimensional));
252 PetscCall(VecCopy(user->Qcrit, nondimensional));
253 PetscCall(VecNorm(nondimensional, NORM_INFINITY, &reference));
254 PetscCall(PicurvAssertBool((PetscBool)(reference > 0.1), "a rotating field should have a positive Q"));
255
256 PetscCall(PetscNew(&pps));
257 pps->dimensionalize = PETSC_TRUE;
258 simCtx->pps = pps;
259 simCtx->scaling.L_ref = 2.0;
260 PetscCall(ComputeQCriterion(user));
261 PetscCall(VecAXPY(nondimensional, -4.0, user->Qcrit));
262 PetscCall(VecNorm(nondimensional, NORM_INFINITY, &difference));
263 PetscCall(PicurvAssertRealNear(0.0, difference, 1.0e-12 * reference,
264 "dimensionalized Q must be the nondimensional Q over L_ref^2"));
265
266 simCtx->pps = NULL;
267 PetscCall(PetscFree(pps));
268 PetscCall(VecDestroy(&nondimensional));
269 PetscCall(PicurvDestroyMinimalContexts(&simCtx, &user));
270 PetscFunctionReturn(0);
271}
