64def iter_markdown_files(repo_root: Path, include_readme: bool):
65 """!
66 @brief Yield every repository Markdown file the current commit would carry.
67 @param[in] repo_root Repository root directory.
68 @param[in] include_readme Whether the top-level README participates in the scan.
69 @return Generator of Markdown paths.
70 """
71 candidates = git_markdown(repo_root)
72 if candidates is None:
73 # Fallback for a non-git export: walk the tree and skip generated/scratch trees.
74 candidates = sorted(repo_root.rglob("*.md"))
75 for markdown in candidates:
76 if not markdown.is_file():
77 continue
78 parts = markdown.relative_to(repo_root).parts
79 if SKIP_DIRS.intersection(parts):
80 continue
81 if not include_readme and markdown == repo_root / "README.md":
82 continue
83 yield markdown

86def should_skip_link(target: str) -> bool:
87 """!
88 @brief Perform should skip link.
89 @param[in] target Argument passed to `should_skip_link()`.
90 @return Value returned by `should_skip_link()`.
91 """
92 lower = target.lower()

101def normalize_target(raw_target: str) -> str:
102 """!
103 @brief Remove Markdown angle brackets, fragments, and query strings before resolving a link target.
104 @param[in] raw_target Argument passed to `normalize_target()`.
105 @return Value returned by `normalize_target()`.
106 """
107 cleaned = raw_target.strip().strip("<>").split("#", 1)[0].split("?", 1)[0]

111def main() -> int:
112 """!
113 @brief Entry point for this script.
114 @return Value returned by `main()`.
115 """
116 args = parse_args()
117 repo_root = args.repo_root.resolve()
118 failures = []
119
120 scanned = list(iter_markdown_files(repo_root, include_readme=not args.no_readme))
121 for md_file in scanned:
122 if not md_file.is_file():
123 failures.append((str(md_file), "<file>", "Markdown file not found"))
124 continue
125 content = md_file.read_text(encoding="utf-8", errors="replace")
126 for match in LINK_PATTERN.findall(content):
127 target = match[0] or match[1]
128 if should_skip_link(target):
129 continue
130 normalized = normalize_target(target)
131 if not normalized:
132 continue
133 resolved = (md_file.parent / normalized).resolve()
134 try:
135 exists = resolved.exists()
136 except OSError as exc:
137 failures.append((str(md_file.relative_to(repo_root)), target, f"{resolved} ({exc})"))
138 continue
139 if not exists:
140 failures.append((str(md_file.relative_to(repo_root)), target, str(resolved)))
141
142 if failures:
143 print("Broken markdown links detected:")
144 for src, target, resolved in failures:
145 print(f" - {src}: '{target}' -> missing '{resolved}'")
146 return 1
147
148 print(f"Markdown link check passed for {len(scanned)} repository Markdown files.")
149 return 0
