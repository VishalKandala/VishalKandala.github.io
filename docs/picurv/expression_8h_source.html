97/**
98 * @brief Compile one expression against a list of variable names.
99 * @details The grammar is Python's expression syntax restricted to numbers, the given
100 * names, `pi`, arithmetic (`+ - * / % **`), comparisons (chained ones included),
101 * `and`/`or`/`not`, the functions `abs atan2 bei ber besselj bessely cos cosh erf
102 * exp floor log maximum minimum sin sinh sqrt tan tanh where`, the series
103 * `sum(k, lo, hi, body)` over integer bounds, plus `uniform` and `normal` when
104 * @p allow_random is true. Comparisons and logical operators yield 1 or 0; `%`
105 * takes the sign of the divisor; `**` binds tighter than a unary minus on its left
106 * and groups to the right.
107 * @param[in] text Expression text.
108 * @param[in] names Variable names, in the order evaluation supplies their values.
109 * @param[in] name_count Number of names.
110 * @param[in] allow_random Whether `uniform` and `normal` may be called.
111 * @param[out] expression Compiled expression, owned by the caller.
112 * @return Zero on success; `PETSC_ERR_ARG_WRONG` for a syntax error or an unknown name
113 * or function, with the offending position in the message.
114 */
115PetscErrorCode PicurvExpressionCompile(const char *text, const char *const *names, PetscInt name_count,
116 PetscBool allow_random, PicurvExpression **expression);

244/**
245 * @brief Fill a source's position and grid-line variables at one point.
246 * @param[out] values Array of #PICURV_SOURCE_VAR_COUNT; `t` is left to the caller.
247 * @param[in] point Solver-unit position.
248 * @param[in] fraction Grid-line fractions there, from ComputeGridLineFractions().
249 * @param[in] length Solver-unit grid-line lengths there.
250 * @param[in] L_ref Length reference [m].

252static inline void PicurvSourceSetVariables(PetscReal *values, Cmpnts point, Cmpnts fraction, Cmpnts length,
253 PetscReal L_ref)
254{
255 values[PICURV_SOURCE_VAR_X] = point.x * L_ref;
256 values[PICURV_SOURCE_VAR_Y] = point.y * L_ref;
257 values[PICURV_SOURCE_VAR_Z] = point.z * L_ref;
258 values[PICURV_SOURCE_VAR_XI] = fraction.x;
259 values[PICURV_SOURCE_VAR_ETA] = fraction.y;

261 values[PICURV_SOURCE_VAR_L_XI] = length.x * L_ref;
262 values[PICURV_SOURCE_VAR_L_ETA] = length.y * L_ref;
263 values[PICURV_SOURCE_VAR_L_ZETA] = length.z * L_ref;

266/**
267 * @brief Global identity of a boundary face cell, unique over the six faces of a block and
268 * independent of the decomposition.
269 * @param[in] nodes Global node counts along i, j, k.
270 * @param[in] face Face.
271 * @param[in] p Zero-based cell index along the first tangential axis.
272 * @param[in] q Zero-based cell index along the second.
273 * @return The identity, from 0 below `2 (nx ny + ny nz + nz nx)` for `nx ny nz` cells.
274 */
275static inline PetscInt64 PicurvFaceCellId(const PetscInt nodes[3], BCFace face, PetscInt p, PetscInt q)
276{
277 const PetscInt64 cells[3] = {nodes[0] - 2, nodes[1] - 2, nodes[2] - 2};
278 PetscInt64 id = p + cells[((PetscInt)face / 2 == 0) ? 1 : 0] * q;
279
280 for (PetscInt f = 0; f < (PetscInt)face; ++f) id += cells[(f / 2 + 1) % 3] * cells[(f / 2 + 2) % 3];
281 return id;
282}

284/**
285 * @brief Draw identity of a grid point: its block in the top 16 bits, above an identity unique
286 * within the block (a cell's natural index, or PicurvFaceCellId()).
287 * @param[in] block Block index.
288 * @param[in] id Identity within the block, below 2^48.
289 * @return The identity.
290 */
291static inline PetscInt64 PicurvGridDrawId(PetscInt block, PetscInt64 id)
292{
293 return ((PetscInt64)block << 48) + id;
294}
