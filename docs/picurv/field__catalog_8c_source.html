172/**
173 * @brief Returns the immutable catalog descriptor for one typed field identity.
174 * @details Validates the enum range and the table's ID/index invariant before
175 * exposing the catalog-owned descriptor.
176 * @see FieldGetDescriptor()
177 */
178PetscErrorCode FieldGetDescriptor(FieldId field_id, const FieldDescriptor **descriptor)
179{
180 PetscFunctionBeginUser;
181 PetscCheck(descriptor != NULL, PETSC_COMM_SELF, PETSC_ERR_ARG_NULL,
182 "Field descriptor output cannot be NULL.");
183 PetscCheck(field_id >= 0 && field_id < FIELD_ID_COUNT, PETSC_COMM_SELF, PETSC_ERR_ARG_OUTOFRANGE,
184 "Invalid FieldId value %d.", (int)field_id);
185 PetscCheck(gFieldCatalog[field_id].id == field_id, PETSC_COMM_SELF, PETSC_ERR_PLIB,
186 "Field catalog entry %d is not initialized consistently.", (int)field_id);
187
188 *descriptor = &gFieldCatalog[field_id];
189 PetscFunctionReturn(0);
190}

192/**
193 * @brief Resolves a canonical field name or registered alias to a typed identity.
194 * @details Name comparison is intentionally confined to ingress; numerical
195 * consumers receive the resolved @ref FieldId.
196 * @see FieldIdFromName()
197 */
198PetscErrorCode FieldIdFromName(const char *field_name, FieldId *field_id)
199{
200 PetscFunctionBeginUser;
201 PetscCheck(field_name != NULL, PETSC_COMM_SELF, PETSC_ERR_ARG_NULL,
202 "Field name cannot be NULL.");
203 PetscCheck(field_id != NULL, PETSC_COMM_SELF, PETSC_ERR_ARG_NULL,
204 "FieldId output cannot be NULL.");
205
206 for (PetscInt index = 0; index < FIELD_ID_COUNT; ++index) {
207 const FieldDescriptor *descriptor = &gFieldCatalog[index];
208 PetscBool match = PETSC_FALSE;
209
210 PetscCall(PetscStrcasecmp(field_name, descriptor->canonical_name, &match));
211 if (!match && descriptor->alias_1) PetscCall(PetscStrcasecmp(field_name, descriptor->alias_1, &match));
212 if (!match && descriptor->alias_2) PetscCall(PetscStrcasecmp(field_name, descriptor->alias_2, &match));
213 if (match) {
214 *field_id = descriptor->id;
215 PetscFunctionReturn(0);
216 }
217 }
218
219 *field_id = FIELD_ID_INVALID;
220 SETERRQ(PETSC_COMM_SELF, PETSC_ERR_ARG_UNKNOWN_TYPE,
221 "Field name '%s' is not registered in the Eulerian field catalog.", field_name);
222}

224/**
225 * @brief Returns catalog-owned printable text for a field identity.
226 * @details Invalid enum values return a non-null diagnostic sentinel string.
227 * @see FieldCanonicalName()
228 */
229const char *FieldCanonicalName(FieldId field_id)
230{
231 if (field_id < 0 || field_id >= FIELD_ID_COUNT) return "InvalidField";
232 if (gFieldCatalog[field_id].id != field_id) return "InvalidField";
233 return gFieldCatalog[field_id].canonical_name;

253/**
254 * @brief Binds one descriptor to the PETSc objects already owned by a UserCtx.
255 * @details Resolves DM and Vec handles without allocating, referencing, or
256 * destroying PETSc storage.
257 * @see FieldGetView()
258 */
259PetscErrorCode FieldGetView(UserCtx *user, FieldId field_id, FieldView *view)
260{
261 const FieldDescriptor *descriptor = NULL;
262
263 PetscFunctionBeginUser;
264 PetscCheck(user != NULL, PETSC_COMM_SELF, PETSC_ERR_ARG_NULL,
265 "UserCtx cannot be NULL when resolving a field view.");
266 PetscCheck(view != NULL, PETSC_COMM_SELF, PETSC_ERR_ARG_NULL,
267 "Field view output cannot be NULL.");
268 PetscCall(FieldGetDescriptor(field_id, &descriptor));
269
270 view->descriptor = descriptor;
271 view->dm = NULL;
272 view->global_vec = NULL;
273 view->local_vec = NULL;
274
275 switch (descriptor->dm_kind) {
276 case FIELD_DM_DA:
277 view->dm = user->da;
278 break;
279 case FIELD_DM_FDA:
280 view->dm = user->fda;
281 break;
282 case FIELD_DM_FDA2:
283 view->dm = user->fda2;
284 break;
285 case FIELD_DM_COORDINATES:
286 view->dm = user->fda;
287 PetscCheck(user->da != NULL, PETSC_COMM_SELF, PETSC_ERR_ARG_WRONGSTATE,
288 "Coordinate DM is unavailable for field '%s'.", descriptor->canonical_name);
289 PetscCall(DMGetCoordinates(user->da, &view->global_vec));
290 PetscCall(DMGetCoordinatesLocal(user->da, &view->local_vec));
291 break;
292 default:
293 SETERRQ(PETSC_COMM_SELF, PETSC_ERR_PLIB,
294 "Field '%s' has an invalid DM selector.", descriptor->canonical_name);
295 }
296
297 if (descriptor->dm_kind != FIELD_DM_COORDINATES) {
298 view->global_vec = *(Vec *)((char *)user + descriptor->global_vec_offset);
299 view->local_vec = *(Vec *)((char *)user + descriptor->local_vec_offset);
300 }
301
302 PetscCheck(view->dm != NULL, PETSC_COMM_SELF, PETSC_ERR_ARG_WRONGSTATE,
303 "DM for field '%s' is unavailable in this UserCtx.", descriptor->canonical_name);
304 PetscCheck(view->global_vec != NULL, PETSC_COMM_SELF, PETSC_ERR_ARG_WRONGSTATE,
305 "Global vector for field '%s' is unavailable in this UserCtx.", descriptor->canonical_name);
306 PetscCheck(view->local_vec != NULL, PETSC_COMM_SELF, PETSC_ERR_ARG_WRONGSTATE,
307 "Local vector for field '%s' is unavailable in this UserCtx.", descriptor->canonical_name);
308
309 PetscFunctionReturn(0);
310}
