187 char cmd[512];
188
189 PetscFunctionBeginUser;

191 /* Safety: only remove paths under /tmp/picurv-test- */
192 if (strncmp(path, "/tmp/picurv-test-", 17) != 0) {

197 if (system(cmd) != 0) {

199 }
200 PetscFunctionReturn(0);

703/**
