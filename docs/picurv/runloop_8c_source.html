10 * @brief Internal helper implementation: `UpdateSolverHistoryVectors()`.

13PetscErrorCode UpdateSolverHistoryVectors(UserCtx *user)
14{
15 PetscErrorCode ierr;
16 SimCtx *simCtx = user->simCtx; // Access global settings if needed
17
18 PetscFunctionBeginUser;
19 LOG_ALLOW(LOCAL, LOG_DEBUG, "Rank %d, Block %d: Updating solver history vectors.\n",
20 simCtx->rank, user->_this);
21
22 // --- Primary Contravariant Velocity History ---
23 // The order is critical here.
24 // 1. First, move the n-1 state (Ucont_o) to the n-2 slot (Ucont_rm1).
25 ierr = VecCopy(user->Ucont_o, user->Ucont_rm1); CHKERRQ(ierr);
26 // 2. Then, move the new n state (Ucont) to the n-1 slot (Ucont_o).
27 ierr = VecCopy(user->Ucont, user->Ucont_o); CHKERRQ(ierr);
28
29 LOG_ALLOW(LOCAL,LOG_DEBUG, "Rank %d, Block %d, Ucont history updated.\n",simCtx->rank,user->_this);
30
31 // --- Update History for Other Fields ---
32 // These are typically only needed at the n-1 state.
33 ierr = VecCopy(user->Ucat, user->Ucat_o); CHKERRQ(ierr);
34 ierr = VecCopy(user->P, user->P_o); CHKERRQ(ierr);
35 LOG_ALLOW(LOCAL,LOG_DEBUG, "Rank %d, Block %d, Ucat & P history updated.\n",simCtx->rank,user->_this);
36
37 if (simCtx->immersed) {
38 ierr = VecCopy(user->Nvert, user->Nvert_o); CHKERRQ(ierr);
39 }
40
41 // --- Update History for Turbulence Models (if active) ---
42 if (simCtx->rans) {
43 ierr = VecCopy(user->K_Omega, user->K_Omega_o); CHKERRQ(ierr);
44 }
45
46 // --- Synchronize Local Ghost Regions for the new history vectors ---
47 // This is essential so that stencils in the next time step's calculations
48 // have correct values from neighboring processes.
49 ierr = DMGlobalToLocalBegin(user->fda, user->Ucont_o, INSERT_VALUES, user->lUcont_o); CHKERRQ(ierr);
50 ierr = DMGlobalToLocalEnd(user->fda, user->Ucont_o, INSERT_VALUES, user->lUcont_o); CHKERRQ(ierr);
51
52 ierr = DMGlobalToLocalBegin(user->fda, user->Ucont_rm1, INSERT_VALUES, user->lUcont_rm1); CHKERRQ(ierr);
53 ierr = DMGlobalToLocalEnd(user->fda, user->Ucont_rm1, INSERT_VALUES, user->lUcont_rm1); CHKERRQ(ierr);
54
55 if (simCtx->immersed) {
56 ierr = DMGlobalToLocalBegin(user->da, user->Nvert_o, INSERT_VALUES, user->lNvert_o); CHKERRQ(ierr);
57 ierr = DMGlobalToLocalEnd(user->da, user->Nvert_o, INSERT_VALUES, user->lNvert_o); CHKERRQ(ierr);
58 }
59
60 if (simCtx->rans) {
61 ierr = DMGlobalToLocalBegin(user->fda2, user->K_Omega_o, INSERT_VALUES, user->lK_Omega_o); CHKERRQ(ierr);
62 ierr = DMGlobalToLocalEnd(user->fda2, user->K_Omega_o, INSERT_VALUES, user->lK_Omega_o); CHKERRQ(ierr);
63 }
64
65 PetscFunctionReturn(0);
66}

74PetscErrorCode PerformInitializedParticleSetup(SimCtx *simCtx)
75{
76 PetscErrorCode ierr;
77 // --- Get pointers from SimCtx instead of passing them as arguments ---
78 UserCtx *user = simCtx->usermg.mgctx[simCtx->usermg.mglevels-1].user;
79 BoundingBox *bboxlist = simCtx->bboxlist;
80
81 PetscFunctionBeginUser;
82
83 LOG_ALLOW(GLOBAL, LOG_INFO, "[T=%.4f, Step=%d] Performing initial particle setup procedures.\n", simCtx->ti, simCtx->step);
84
85 // --- 0. Loop over all blocks to compute Eulerian diffusivity.
86 for (PetscInt bi = 0; bi < simCtx->block_number; bi++) {
87 ierr = ComputeEulerianDiffusivity(&user[bi]); CHKERRQ(ierr);
88 ierr = ComputeEulerianDiffusivityGradient(&user[bi]); CHKERRQ(ierr);
89 }
90
91 // --- 1. Initial Particle Settlement (Location and Migration) ---
92 LOG_ALLOW(GLOBAL, LOG_INFO, "[T=%.4f, Step=%d] Initial Settlement: Locating and migrating all particles...\n", simCtx->ti, simCtx->step);
93 ierr = LocateAllParticlesInGrid(user, bboxlist); CHKERRQ(ierr);
94
95 if(get_log_level() >= LOG_DEBUG && is_function_allowed(__FUNCT__)==true){
96 LOG_ALLOW(GLOBAL, LOG_DEBUG, "[T=%.4f, Step=%d] Particle field states after Initial settlement...\n", simCtx->ti, simCtx->step);
97 ierr = LOG_PARTICLE_FIELDS(user,simCtx->LoggingFrequency); CHKERRQ(ierr);
98 }
99
100 // --- 2. Re-initialize Particles on Inlet Surface (if applicable) ---
101 if ((simCtx->ParticleInitialization == PARTICLE_INIT_SURFACE_RANDOM ||
102 simCtx->ParticleInitialization == PARTICLE_INIT_SURFACE_EDGES) && user->inletFaceDefined) {
103 LOG_ALLOW(GLOBAL, LOG_INFO, "[T=%.4f, Step=%d] Re-initializing particles on inlet surface...\n", simCtx->ti, simCtx->step);
104 ierr = ReinitializeParticlesOnInletSurface(user, simCtx->ti, simCtx->step); CHKERRQ(ierr);
105
106 LOG_ALLOW(GLOBAL, LOG_INFO, "[T=%.4f, Step=%d] Resetting statuses for post-reinitialization settlement.\n", simCtx->ti, simCtx->step);
107 ierr = ResetAllParticleStatuses(user); CHKERRQ(ierr);
108
109 LOG_ALLOW(GLOBAL, LOG_INFO, "[T=%.4f, Step=%d] Post-Reinitialization Settlement...\n", simCtx->ti, simCtx->step);
110 ierr = LocateAllParticlesInGrid(user, bboxlist); CHKERRQ(ierr);
111
112 }
113
114 // --- 3. Finalize State for t=0 ---
115 LOG_ALLOW(GLOBAL, LOG_INFO, "[T=%.4f, Step=%d] Interpolating initial fields to settled particles.\n", simCtx->ti, simCtx->step);
116 ierr = InterpolateAllFieldsToSwarm(user); CHKERRQ(ierr);
117 //ierr = ScatterAllParticleFieldsToEulerFields(user); CHKERRQ(ierr);
118
119 // --- 4. Initial History and Output ---
120 // Update solver history vectors with the t=0 state before the first real step
121 for (PetscInt bi = 0; bi < simCtx->block_number; bi++) {
122 ierr = UpdateSolverHistoryVectors(&user[bi]); CHKERRQ(ierr);
123 }
124
125 if (simCtx->tiout > 0 || (simCtx->StepsToRun == 0 && simCtx->StartStep == 0)) {
126 LOG_ALLOW(GLOBAL, LOG_INFO, "[T=%.4f, Step=%d] Writing initial simulation data.\n", simCtx->ti, simCtx->step);
127 ierr = WriteAllSwarmFields(user); CHKERRQ(ierr);
128
129 // --- Eulerian Field Output (MUST loop over all blocks) --- // <<< CHANGED/FIXED
130 for (PetscInt bi = 0; bi < simCtx->block_number; bi++) {
131 ierr = WriteSimulationFields(&user[bi]); CHKERRQ(ierr);
132 }
133 }
134
135 if (IsParticleConsoleSnapshotEnabled(simCtx)) {
136 ierr = EmitParticleConsoleSnapshot(user, simCtx, simCtx->step); CHKERRQ(ierr);
137 }
138
139 LOG_ALLOW(GLOBAL, LOG_INFO, "--- Initial setup complete. Ready for time marching. ---\n");
140 PetscFunctionReturn(0);
141}

214PetscErrorCode FinalizeRestartState(SimCtx *simCtx)
215{
216 PetscErrorCode ierr;
217 UserCtx *user = simCtx->usermg.mgctx[simCtx->usermg.mglevels - 1].user;
218
219 PetscFunctionBeginUser;
220
221 LOG_ALLOW(GLOBAL, LOG_INFO, "--- Finalizing RESTART from state (step=%d, t=%.4f) ---\n", simCtx->StartStep, simCtx->ti);
222
223 // This function only needs to handle the particle finalization logic.
224 // The Eulerian state is assumed to be fully loaded and consistent at this point.
225 if (simCtx->np > 0) {
226
227 // Use the particle restart mode to decide the workflow.
228 if (strcmp(simCtx->particleRestartMode, "load") == 0) {
229 // PARTICLES WERE LOADED: The state is complete, but we must verify
230 // the loaded CellIDs and build the in-memory grid-to-particle links.
231 LOG_ALLOW(GLOBAL, LOG_INFO, "Particle Mode 'load': Verifying particle locations and building grid links...\n");
232 ierr = PerformLoadedParticleSetup(simCtx); CHKERRQ(ierr);
233
234 } else { // Mode must be "init"
235 // PARTICLES WERE RE-INITIALIZED: They need to be fully settled and coupled
236 // to the surrounding (restarted) fluid state.
237 LOG_ALLOW(GLOBAL, LOG_INFO, "Particle Mode 'init': Running full initial setup for new particles in restarted flow.\n");
238 ierr = PerformInitializedParticleSetup(simCtx); CHKERRQ(ierr);
239 }
240 } else {
241 LOG_ALLOW(GLOBAL, LOG_INFO, "No particles in simulation, restart finalization is complete.\n");
242
243 // Write the initial eulerian fields (this is done in PerformInitialSetup if particles exist.)
244 for(PetscInt bi = 0; bi < simCtx->block_number; bi ++){
245 ierr = WriteSimulationFields(&user[bi]); CHKERRQ(ierr);
246 }
247 }
248
249 LOG_ALLOW(GLOBAL, LOG_INFO, "--- Restart state successfully finalized. --\n");
250
251 PetscFunctionReturn(0);

260PetscErrorCode AdvanceSimulation(SimCtx *simCtx)
261{
262 PetscErrorCode ierr;
263 // Get the master context from the first block. All blocks share it.
264 UserCtx *user = simCtx->usermg.mgctx[simCtx->usermg.mglevels-1].user;
265
266 // Retrieve control parameters from SimCtx for clarity
267 const PetscInt StartStep = simCtx->StartStep;
268 const PetscInt StepsToRun = simCtx->StepsToRun;
269 const PetscReal dt = simCtx->dt;
270
271 // Variables for particle removal statistics
272 //PetscInt removed_local_ob, removed_global_ob;
273 PetscInt removed_local_lost, removed_global_lost;
274
275 PetscFunctionBeginUser;
276 PROFILE_FUNCTION_BEGIN;
277 LOG_ALLOW(GLOBAL, LOG_INFO, "Starting main time-marching loop: %d steps from step %d (t=%.4f), dt=%.4f\n",
278 StepsToRun, StartStep, simCtx->StartTime, dt);
279
280 // --- Main Time-Marching Loop ---
281 for (PetscInt step = StartStep; step < StartStep + StepsToRun; step++) {
282
283 // =================================================================
284 // 1. PRE-STEP SETUP
285 // =================================================================
286
287 // Update simulation time and step counters in the master context
288 simCtx->step = step + 1;
289 simCtx->ti += simCtx->dt; //simCtx->StartTime + step * simCtx->dt;
290
291 LOG_ALLOW(GLOBAL, LOG_INFO, "--- Advancing Step %d (To t=%.4f) ---\n", simCtx->step, simCtx->ti);
292
293
294 // For particles, reset their status to prepare for the new advection/location cycle
295 if (simCtx->np > 0) {
296 LOG_ALLOW(GLOBAL, LOG_DEBUG, "Resetting all particle statuses to NEEDS_LOCATION.\n");
297 ierr = ResetAllParticleStatuses(user); CHKERRQ(ierr);
298 }
299
300 // =================================================================
301 // 2. EULERIAN SOLVER STEP
302 // =================================================================
303 if(get_log_level() == LOG_VERBOSE && is_function_allowed(__FUNCT__)==true){
304 ierr = LOG_FIELD_ANATOMY(&user[0],"Coordinates","PreFlowSolver"); CHKERRQ(ierr);
305 ierr = LOG_FIELD_ANATOMY(&user[0],"Csi","PreFlowSolver"); CHKERRQ(ierr);
306 ierr = LOG_FIELD_ANATOMY(&user[0],"Eta","PreFlowSolver"); CHKERRQ(ierr);
307 ierr = LOG_FIELD_ANATOMY(&user[0],"Zet","PreFlowSolver"); CHKERRQ(ierr);
308 ierr = LOG_FIELD_ANATOMY(&user[0],"Center-Coordinates","PreFlowSolver"); CHKERRQ(ierr);
309 ierr = LOG_FIELD_ANATOMY(&user[0],"X-Face-Centers","PreFlowSolver"); CHKERRQ(ierr);
310 ierr = LOG_FIELD_ANATOMY(&user[0],"Y-Face-Centers","PreFlowSolver"); CHKERRQ(ierr);
311 ierr = LOG_FIELD_ANATOMY(&user[0],"Z-Face-Centers","PreFlowSolver"); CHKERRQ(ierr);
312 ierr = LOG_FIELD_ANATOMY(&user[0],"Ucat","PreFlowSolver"); CHKERRQ(ierr);
313 }
314 LOG_ALLOW(GLOBAL, LOG_INFO, "Updating Eulerian Field ...\n");
315 if(strcmp(simCtx->eulerianSource,"load")==0){
316 //LOAD mode: Read pre-computed fields for the current step.
317 LOG_ALLOW(GLOBAL,LOG_INFO,"Eulerian Source 'load': Reading fields (t=%.4f,step=%d)...\n",simCtx->ti,simCtx->step);
318 for(PetscInt bi = 0; bi < simCtx->block_number;bi++){
319 ierr = ReadSimulationFields(&user[bi],simCtx->step); CHKERRQ(ierr);
320 }
321 }else if(strcmp(simCtx->eulerianSource,"analytical")==0){
322 // ANALYTICAL mode:Call the Analytical Solution Prescription Engine to enable a variety of analytical functions
323 LOG_ALLOW(GLOBAL,LOG_INFO,"Eulerian Source 'analytical'. Updating Eulerian field via the Analytical Solution Engine ...\n");
324 ierr = AnalyticalSolutionEngine(simCtx); CHKERRQ(ierr);
325 }else if(strcmp(simCtx->eulerianSource,"solve")==0){
326 // SOLVE mode:Call the refactored, high-level legacy solver. This single function
327 // advances the entire multi-block fluid field from t_n to t_{n+1}.
328 LOG_ALLOW(GLOBAL,LOG_INFO,"Eulerian Source 'solve'. Updating Eulerian field via Solver...\n");
329 ierr = FlowSolver(simCtx); CHKERRQ(ierr);
330 }
331 LOG_ALLOW(GLOBAL, LOG_INFO, "Eulerian Field Updated ...\n");
332 if(get_log_level() == LOG_VERBOSE && is_function_allowed(__FUNCT__)==true){
333 LOG_ALLOW(GLOBAL, LOG_VERBOSE, "Post FlowSolver field states:\n");
334 ierr = LOG_FIELD_ANATOMY(&user[0],"Ucat","PostFlowSolver"); CHKERRQ(ierr);
335 ierr = LOG_FIELD_ANATOMY(&user[0],"P","PostFlowSolver"); CHKERRQ(ierr);
336 ierr = LOG_FIELD_ANATOMY(&user[0],"Ucont","PostFlowSolver"); CHKERRQ(ierr);
337 }
338
339
340 // =================================================================
341 // 3. LAGRANGIAN PARTICLE STEP
342 // =================================================================
343
344 if (simCtx->np > 0) {
345 LOG_ALLOW(GLOBAL, LOG_INFO, "Updating Lagrangian particle system...\n");
346
347 // a. Update Eulerian Transport Properties:
348 // Optimization: Only recalculate if turbulence is active (Nu_t changes).
349 // For Laminar flow, the value calculated at Setup is constant.
350 if (simCtx->les || simCtx->rans) {
351 for (PetscInt bi = 0; bi < simCtx->block_number; bi++) {
352 ierr = ComputeEulerianDiffusivity(&user[bi]); CHKERRQ(ierr);
353 ierr = ComputeEulerianDiffusivityGradient(&user[bi]); CHKERRQ(ierr);
354 }
355 }
356
357 // a.1 (Optional) Log Eulerian Diffusivity min/max and anatomy for debugging.
358 if(get_log_level() == LOG_VERBOSE && is_function_allowed(__FUNCT__)==true){
359 LOG_ALLOW(GLOBAL, LOG_VERBOSE, "Updated Diffusivity Min/Max:\n");
360 ierr = LOG_FIELD_MIN_MAX(&user[0],"Diffusivity"); CHKERRQ(ierr);
361 ierr = LOG_FIELD_MIN_MAX(&user[0],"DiffusivityGradient"); CHKERRQ(ierr);
362 //LOG_ALLOW(GLOBAL, LOG_VERBOSE, "Updated Diffusivity Anatomy:\n");
363 ierr = LOG_FIELD_ANATOMY(&user[0],"Diffusivity","PostDiffusivityUpdate"); CHKERRQ(ierr);
364 }
365 // b. Advect particles using the velocity interpolated from the *previous* step.
366 // P(t_{n+1}) = P(t_n) + V_p(t_n) * dt
367 ierr = UpdateAllParticlePositions(user); CHKERRQ(ierr);
368
369 // c. Settle all particles: find their new host cells and migrate them across ranks.
370 ierr = LocateAllParticlesInGrid(user, simCtx->bboxlist); CHKERRQ(ierr);
371
372 // d. Remove any particles that are now lost or out of the global domain.
373 ierr = CheckAndRemoveLostParticles(user, &removed_local_lost, &removed_global_lost); CHKERRQ(ierr);
374 //ierr = CheckAndRemoveOutOfBoundsParticles(user, &removed_local_ob, &removed_global_ob, simCtx->bboxlist); CHKERRQ(ierr);
375 if (removed_global_lost> 0) { // if(removed_global_lost + removed_global_ob > 0){
376 LOG_ALLOW(GLOBAL, LOG_INFO, "Removed %d particles globally this step.\n", removed_global_lost); // removed_global_lost + removed_global_ob;
377 simCtx->particlesLostLastStep = removed_global_lost;
378 }
379
380 // e. Interpolate the NEW fluid velocity (just computed by FlowSolver) onto the
381 // particles' new positions. This gives them V_p(t_{n+1}) for the *next* advection step.
382 ierr = InterpolateAllFieldsToSwarm(user); CHKERRQ(ierr);
383
384 // f. Update the Particle Fields (e.g., temperature, concentration) if applicable.
385 // This can be extended to include reactions, growth, etc.
386 ierr = UpdateAllParticleFields(user); CHKERRQ(ierr);
387
388 // g. (For Two-Way Coupling) Scatter particle data back to the grid to act as a source term.
389 ierr = CalculateParticleCountPerCell(user); CHKERRQ(ierr);
390 ierr = ScatterAllParticleFieldsToEulerFields(user); CHKERRQ(ierr);
391
392 // h. (Optional) Calculate advanced particle metrics for logging/debugging.
393 ierr = CalculateAdvancedParticleMetrics(user); CHKERRQ(ierr);
394
395 ierr = LOG_PARTICLE_METRICS(user, "Timestep Metrics"); CHKERRQ(ierr);
396
397
398 if(get_log_level() == LOG_VERBOSE && is_function_allowed(__FUNCT__)==true){
399 LOG_ALLOW(GLOBAL, LOG_VERBOSE, "Post Lagrangian update field states:\n");
400 ierr = LOG_FIELD_MIN_MAX(&user[0],"Psi"); CHKERRQ(ierr);
401 }
402 }
403
404 // =================================================================
405 // 4. UPDATE HISTORY & I/O
406 // =================================================================
407
408 // Copy the newly computed fields (Ucont, P, etc.) to the history vectors
409 // (_o, _rm1) to prepare for the next time step.
410 for (PetscInt bi = 0; bi < simCtx->block_number; bi++) {
411 ierr = UpdateSolverHistoryVectors(&user[bi]); CHKERRQ(ierr);
412 }
413
414 //ierr = LOG_UCAT_ANATOMY(&user[0],"Final"); CHKERRQ(ierr);
415
416 // Handle periodic file output
417 if (ShouldWriteDataOutput(simCtx, simCtx->step)) {
418 LOG_ALLOW(GLOBAL, LOG_INFO, "Writing output for step %d.\n",simCtx->step);
419 for (PetscInt bi = 0; bi < simCtx->block_number; bi++) {
420 ierr = WriteSimulationFields(&user[bi]); CHKERRQ(ierr);
421 }
422 if (simCtx->np > 0) {
423 ierr = WriteAllSwarmFields(user); CHKERRQ(ierr);
424 if (get_log_level() >= LOG_INFO && strcmp(simCtx->eulerianSource,"analytical") == 0) {
425 LOG_INTERPOLATION_ERROR(user);
426 }
427 }
428 }
429
430 if (ShouldEmitPeriodicParticleConsoleSnapshot(simCtx, simCtx->step)) {
431 ierr = EmitParticleConsoleSnapshot(user, simCtx, simCtx->step); CHKERRQ(ierr);
432 }
433
434 ProfilingLogTimestepSummary(simCtx, simCtx->step);
435
436 // Update Progress Bar
437 if(simCtx->rank == 0) {
438 PrintProgressBar(step,StartStep,StepsToRun,simCtx->ti);
439 if(get_log_level()>=LOG_WARNING) PetscPrintf(PETSC_COMM_SELF,"\n");
440 }
441 } // --- End of Time-Marching Loop ---
442
443 // After the loop, print the 100% complete bar on rank 0 and add a newline
444 // to ensure subsequent terminal output starts on a fresh line.
445 if (simCtx->rank == 0 && StepsToRun > 0) {
446 PrintProgressBar(StartStep + StepsToRun - 1, StartStep, StepsToRun, simCtx->ti);
447 PetscPrintf(PETSC_COMM_SELF, "\n");
448 fflush(stdout);
449 }
450
451 LOG_ALLOW(GLOBAL, LOG_INFO, "Time marching completed. Final time t=%.4f.\n", simCtx->ti);
452 PROFILE_FUNCTION_END;
453 PetscFunctionReturn(0);
454}
