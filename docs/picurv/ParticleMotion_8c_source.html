14 * @see GenerateGaussianNoise()

16void GenerateGaussianNoise(PetscReal u1, PetscReal u2, PetscReal *n1, PetscReal *n2)
17{
18 const PetscReal magnitude = PetscSqrtReal(-2.0 * PetscLogReal(u1));
19 const PetscReal theta = 2.0 * PETSC_PI * u2;
20
21 *n1 = magnitude * PetscCosReal(theta);
22 *n2 = magnitude * PetscSinReal(theta);

27 * @details Local to this translation unit.

29PetscErrorCode CalculateBrownianDisplacement(UserCtx *user, PetscInt64 pid, PetscReal diff_eff, Cmpnts *displacement)
30{
31 const SimCtx *simCtx = user->simCtx;
32 PetscReal u[4], n[4], sigma;
33 uint64_t hash;
34
35 PetscFunctionBeginUser;
36 displacement->x = displacement->y = displacement->z = 0.0;
37 if (diff_eff <= 1.0e-12) PetscFunctionReturn(0);
38
39 /* Keyed on (seed, PID, step) alone, so the increment is independent of rank layout
40 * and particle order, and a restart reproduces it. */
41 hash = PicurvParticleDrawHash(simCtx->particleRandomSeed, PICURV_DRAW_SPACE_BROWNIAN, 0, pid, simCtx->step);
42 for (int w = 0; w < 4; ++w) u[w] = PicurvUnitFromHash(hash = PicurvMix64(hash ^ (uint64_t)w));
43 GenerateGaussianNoise(1.0 - u[0], u[1], &n[0], &n[1]); /* 1 - u lies in (0, 1]: log finite */
44 GenerateGaussianNoise(1.0 - u[2], u[3], &n[2], &n[3]);
45
46 sigma = PetscSqrtReal(2.0 * diff_eff * simCtx->dt);
47 displacement->x = sigma * n[0];
48 displacement->y = sigma * n[1];
49 displacement->z = sigma * n[2];
50 PetscFunctionReturn(0);
51}

179/**
180 * @brief Internal helper implementation: `CheckAndRemoveOutOfBoundsParticles()`.
181 * @details Local to this translation unit.
182 */
183PetscErrorCode CheckAndRemoveOutOfBoundsParticles(UserCtx *user,
184 PetscInt *removedCountLocal,
185 PetscInt *removedCountGlobal,
186 const BoundingBox *bboxlist)
187{
188 PetscErrorCode ierr;
189 DM swarm = user->swarm;
190 PetscInt nLocalInitial;
191 PetscReal *pos_p = NULL;
192 PetscInt64 *pid_p = NULL; // For better logging
193 PetscBool *remove = NULL;
194 PetscInt local_removed_count = 0;
195 PetscInt global_removed_count_mpi = 0;
196 PetscMPIInt rank, size;
197
198 PetscFunctionBeginUser;
199 PROFILE_HELPER_BEGIN;
200 rank = user->simCtx->rank;
201 size = user->simCtx->size;
202 LOG_ALLOW(LOCAL, LOG_INFO, "[Rank %d] Checking for out-of-bounds particles...", rank);
203
204 // Initialize output parameters to ensure clean state
205 *removedCountLocal = 0;
206 if (removedCountGlobal) *removedCountGlobal = 0;
207
208 ierr = DMSwarmGetLocalSize(swarm, &nLocalInitial); CHKERRQ(ierr);
209
210 // Only proceed if there are particles to check on this rank.
211 // All ranks will still participate in the final collective MPI_Allreduce.
212 if (nLocalInitial > 0) {
213 // Get access to swarm fields once before the loop begins.
214 ierr = DMSwarmGetField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_POSITION), NULL, NULL, (void **)&pos_p); CHKERRQ(ierr);
215 ierr = DMSwarmGetField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_PID), NULL, NULL, (void **)&pid_p); CHKERRQ(ierr);
216 ierr = PetscMalloc1(nLocalInitial, &remove); CHKERRQ(ierr);
217
218 // --- Flag escaped particles BACKWARDS, the order the removal below fills their slots in ---
219 for (PetscInt p = nLocalInitial - 1; p >= 0; p--) {
220 Cmpnts current_pos = {pos_p[3*p + 0], pos_p[3*p + 1], pos_p[3*p + 2]};
221
222 // Remove the particle unless it is inside ANY of the rank bounding boxes
223 remove[p] = PETSC_TRUE;
224 for (PetscMPIInt proc = 0; proc < size; proc++) {
225 if (IsParticleInBox(&bboxlist[proc], &current_pos)) {
226 remove[p] = PETSC_FALSE;
227 break; // Particle is inside a valid domain, stop checking.
228 }
229 }
230
231 if (remove[p]) {
232 LOG_ALLOW(LOCAL, LOG_DEBUG, "Rank %d: Removing out-of-bounds particle [PID %lld] at local index %" PetscInt_FMT ". Pos: (%g, %g, %g)\n",
233 rank, (long long)pid_p[p], p, current_pos.x, current_pos.y, current_pos.z);
234 local_removed_count++;
235 }
236 }
237
238 ierr = DMSwarmRestoreField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_POSITION), NULL, NULL, (void **)&pos_p); CHKERRQ(ierr);
239 ierr = DMSwarmRestoreField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_PID), NULL, NULL, (void **)&pid_p); CHKERRQ(ierr);
240 ierr = RemoveFlaggedParticles(swarm, nLocalInitial, remove, PETSC_TRUE); CHKERRQ(ierr);
241 ierr = PetscFree(remove); CHKERRQ(ierr);
242 } // End of if (nLocalInitial > 0)
243
244 PetscInt nLocalFinal;
245 ierr = DMSwarmGetLocalSize(swarm, &nLocalFinal); CHKERRQ(ierr);
246 LOG_ALLOW(LOCAL, LOG_INFO, "[Rank %d] Finished removing %" PetscInt_FMT " out-of-bounds particles. Final local size: %" PetscInt_FMT ".\n", rank, local_removed_count, nLocalFinal);
247
248 // --- Synchronize counts across all ranks ---
249 *removedCountLocal = local_removed_count;
250 if (removedCountGlobal) {
251 ierr = MPI_Allreduce(&local_removed_count, &global_removed_count_mpi, 1, MPIU_INT, MPI_SUM, PetscObjectComm((PetscObject)swarm)); CHKERRQ(ierr);
252 *removedCountGlobal = global_removed_count_mpi;
253 LOG_ALLOW(GLOBAL, LOG_INFO, "Removed %" PetscInt_FMT " out-of-bounds particles globally.\n", *removedCountGlobal);
254 }
255
256 PROFILE_HELPER_END;
257 PetscFunctionReturn(0);
258}

384/**
385 * @brief Implementation of \ref RemoveFlaggedParticles().
386 * @details Full API contract (arguments, ownership, side effects) is documented with
387 * the header declaration in `include/ParticleMotion.h`.
388 * @see RemoveFlaggedParticles()
389 */
390PetscErrorCode RemoveFlaggedParticles(DM swarm, PetscInt n_scan, PetscBool remove[], PetscBool scan_backward)
391{
392 PetscErrorCode ierr;
393 PetscInt n_local, n_kept = n_scan;
394
395 PetscFunctionBeginUser;
396 PROFILE_FUNCTION_BEGIN;
397 ierr = DMSwarmGetLocalSize(swarm, &n_local); CHKERRQ(ierr);
398
399 // Fill each flagged slot with the last kept point, as DMSwarmRemovePointAtIndex
400 // does, so the surviving order matches removing the points one at a time.
401 for (PetscInt k = 0; k < n_scan; ++k) {
402 PetscInt p = scan_backward ? n_scan - 1 - k : k;
403 while (p < n_kept && remove[p]) {
404 n_kept--;
405 if (p != n_kept) {
406 ierr = DMSwarmCopyPoint(swarm, n_kept, p); CHKERRQ(ierr);
407 remove[p] = remove[n_kept];
408 }
409 }
410 }
411
412 // Points after the scanned range (migration arrivals) move down behind the survivors.
413 for (PetscInt q = n_scan; n_kept < n_scan && q < n_local; ++q) {
414 ierr = DMSwarmCopyPoint(swarm, q, n_kept + q - n_scan); CHKERRQ(ierr);
415 }
416 ierr = DMSwarmSetLocalSizes(swarm, n_kept + n_local - n_scan, -1); CHKERRQ(ierr);
417
418 PROFILE_FUNCTION_END;
419 PetscFunctionReturn(0);
420}

433/**
434 * @brief Implementation of \ref SortParticlesByCell().
435 * @details Full API contract (arguments, ownership, side effects) is documented with
436 * the header declaration in `include/ParticleMotion.h`.
437 * @see SortParticlesByCell()
438 */
439PetscErrorCode SortParticlesByCell(DM swarm)
440{
441 PetscInt n_local, lo[3] = {PETSC_MAX_INT, PETSC_MAX_INT, PETSC_MAX_INT}, hi[3] = {-1, -1, -1};
442 PetscInt *cells = NULL, *order = NULL, *key = NULL;
443 PetscInt64 extent[3], nbuckets, *start = NULL;
444 PetscBool sorted = PETSC_TRUE;
445 char *buffer = NULL;
446 size_t max_bytes = 0;
447
448 PetscFunctionBeginUser;
449 PROFILE_FUNCTION_BEGIN;
450 PetscCall(DMSwarmGetLocalSize(swarm, &n_local));
451 if (n_local < 2) { PROFILE_FUNCTION_END; PetscFunctionReturn(0); }
452
453 // Bucket key: the cell's offset in the bounding box of the local cells, i fastest; a point
454 // without a valid cell takes the bucket after the last.
455 PetscCall(DMSwarmGetField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_CELL_ID), NULL, NULL, (void **)&cells));
456 for (PetscInt p = 0; p < n_local; ++p) {
457 const PetscInt *c = &cells[3 * p];
458 if (c[0] < 0 || c[1] < 0 || c[2] < 0) continue;
459 for (int d = 0; d < 3; ++d) { lo[d] = PetscMin(lo[d], c[d]); hi[d] = PetscMax(hi[d], c[d]); }
460 }
461 for (int d = 0; d < 3; ++d) extent[d] = (hi[d] >= lo[d]) ? (PetscInt64)(hi[d] - lo[d] + 1) : 0;
462 nbuckets = extent[0] * extent[1] * extent[2] + 1;
463 PetscCall(PetscMalloc2(n_local, &key, n_local, &order));
464 PetscCall(PetscCalloc1(nbuckets + 1, &start));
465 for (PetscInt p = 0; p < n_local; ++p) {
466 const PetscInt *c = &cells[3 * p];
467 const PetscInt64 b = (c[0] < 0 || c[1] < 0 || c[2] < 0) ? nbuckets - 1
468 : ((PetscInt64)(c[2] - lo[2]) * extent[1] + (c[1] - lo[1])) * extent[0] + (c[0] - lo[0]);
469 key[p] = (PetscInt)b;
470 if (p > 0 && key[p] < key[p - 1]) sorted = PETSC_FALSE;
471 start[b + 1]++;
472 }
473 PetscCall(DMSwarmRestoreField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_CELL_ID), NULL, NULL, (void **)&cells));
474 if (sorted) {
475 PetscCall(PetscFree(start));
476 PetscCall(PetscFree2(key, order));
477 PROFILE_FUNCTION_END;
478 PetscFunctionReturn(0);
479 }
480
481 // Stable counting sort: order[new position] = old position.
482 for (PetscInt64 b = 0; b < nbuckets; ++b) start[b + 1] += start[b];
483 for (PetscInt p = 0; p < n_local; ++p) order[start[key[p]]++] = p;
484 PetscCall(PetscFree(start));
485
486 // Permute every catalogued field through one buffer sized for the widest.
487 for (PetscInt id = 0; id < PARTICLE_FIELD_ID_COUNT; ++id) {
488 PetscInt bs;
489 PetscDataType type;
490 size_t type_bytes;
491 void *data;
492 PetscCall(DMSwarmGetField(swarm, ParticleFieldName((ParticleFieldId)id), &bs, &type, &data));
493 PetscCall(PetscDataTypeGetSize(type, &type_bytes));
494 max_bytes = PetscMax(max_bytes, (size_t)bs * type_bytes);
495 PetscCall(DMSwarmRestoreField(swarm, ParticleFieldName((ParticleFieldId)id), &bs, &type, &data));
496 }
497 PetscCall(PetscMalloc1((size_t)n_local * max_bytes, &buffer));
498 for (PetscInt id = 0; id < PARTICLE_FIELD_ID_COUNT; ++id) {
499 PetscInt bs;
500 PetscDataType type;
501 size_t type_bytes, bytes;
502 char *data;
503 PetscCall(DMSwarmGetField(swarm, ParticleFieldName((ParticleFieldId)id), &bs, &type, (void **)&data));
504 PetscCall(PetscDataTypeGetSize(type, &type_bytes));
505 bytes = (size_t)bs * type_bytes;
506 // A size known at compile time lets each record copy compile to a few loads and stores
507 // instead of a library call per particle; these are the sizes the particle fields use.
508 switch (bytes) {
509 case 4: GatherSwarmRecords(buffer, data, order, n_local, 4); break;
510 case 8: GatherSwarmRecords(buffer, data, order, n_local, 8); break;
511 case 12: GatherSwarmRecords(buffer, data, order, n_local, 12); break;
512 case 24: GatherSwarmRecords(buffer, data, order, n_local, 24); break;
513 default: GatherSwarmRecords(buffer, data, order, n_local, bytes); break;
514 }
515 memcpy(data, buffer, (size_t)n_local * bytes);
516 PetscCall(DMSwarmRestoreField(swarm, ParticleFieldName((ParticleFieldId)id), &bs, &type, (void **)&data));
517 }
518 PetscCall(PetscFree(buffer));
519 PetscCall(PetscFree2(key, order));
520 PROFILE_FUNCTION_END;
521 PetscFunctionReturn(0);
522}

529 * @see PerformMigration()

532{
533 PetscErrorCode ierr;
534 DM swarm = user->swarm;
535 PetscMPIInt rank;
536 PetscInt nlocal_before;

565 PROFILE_FUNCTION_END;

575 * @details Full API contract (arguments, ownership, side effects) is documented with
576 * the header declaration in `include/logging.h`.

578 */
579PetscErrorCode CalculateParticleCountPerCell(UserCtx *user) {
580 PetscErrorCode ierr;
581 DM da = user->da;
582 DM swarm = user->swarm;
583 Vec countVec = user->ParticleCount;
584 Vec localcountVec = user->lParticleCount;
585 PetscInt nlocal, p;
586 PetscInt *global_cell_id_arr; // Read GLOBAL cell IDs
587 PetscScalar ***count_arr_3d; // Use 3D accessor
588 PetscInt64 *PID_arr;
589 PetscMPIInt rank;
590 char msg[ERROR_MSG_BUFFER_SIZE];
591 PetscInt particles_counted_locally = 0;
592
593 PetscFunctionBeginUser;
594 PROFILE_FUNCTION_BEGIN;
595 rank = user->simCtx->rank;
596
597 // --- Input Validation ---
598 if (!da) SETERRQ(PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "UserCtx->da is NULL.");
599 if (!swarm) SETERRQ(PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "UserCtx->swarm is NULL.");
600 if (!countVec) SETERRQ(PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "UserCtx->ParticleCount is NULL.");
601 // Check DOF of da
602 PetscInt count_dof;
603 count_dof = user->info.dof;
604 if (count_dof != 1) {
605 PetscSNPrintf(msg, sizeof(msg), "countDM must have DOF=1, got %" PetscInt_FMT ".", count_dof);
606 SETERRQ(PETSC_COMM_SELF, PETSC_ERR_ARG_WRONG, "%s", msg);
607 }
608
609 // --- Zero the local count vector ---
610 ierr = VecSet(localcountVec, 0.0); CHKERRQ(ierr);
611
612 // --- Get Particle Data ---
613 LOG_ALLOW(GLOBAL,LOG_DEBUG, "Accessing particle data.\n");
614 ierr = DMSwarmGetLocalSize(swarm, &nlocal); CHKERRQ(ierr);
615 ierr = DMSwarmGetField(swarm,ParticleFieldName(PARTICLE_FIELD_ID_CELL_ID), NULL, NULL, (void **)&global_cell_id_arr); CHKERRQ(ierr);
616 ierr = DMSwarmGetField(swarm,ParticleFieldName(PARTICLE_FIELD_ID_PID),NULL,NULL,(void **)&PID_arr);CHKERRQ(ierr);
617
618 // --- Get Grid Vector Array using DMDA accessor ---
619 LOG_ALLOW(GLOBAL, LOG_DEBUG, "Accessing ParticleCount vector array (using DMDAVecGetArray).\n");
620 ierr = DMDAVecGetArray(da, localcountVec, &count_arr_3d); CHKERRQ(ierr);
621
622 // Get local owned + ghosted range for writing into ghost slots.
623 PetscInt gxs, gys, gzs, gxm, gym, gzm;
624 gxs = user->info.gxs; gys = user->info.gys; gzs = user->info.gzs;
625 gxm = user->info.gxm; gym = user->info.gym; gzm = user->info.gzm;
626
627 // --- Accumulate Counts Locally ---
628 LOG_ALLOW(LOCAL, LOG_DEBUG, "CalculateParticleCountPerCell (Rank %d): Processing %" PetscInt_FMT " local particles using GLOBAL CellIDs.\n",rank,nlocal);
629 for (p = 0; p < nlocal; p++) {
630 // Read the GLOBAL indices stored for this particle
631 PetscInt i_geom = global_cell_id_arr[p * 3 + 0]; // Global i index
632 PetscInt j_geom = global_cell_id_arr[p * 3 + 1]; // Global j index
633 PetscInt k_geom = global_cell_id_arr[p * 3 + 2]; // Global k index
634
635 // Apply the shift to ensure ParticleCount follows the indexing convention for cell-centered data in this codebase.
636 PetscInt i = (PetscInt)i_geom + 1; // Shift for cell-centered
637 PetscInt j = (PetscInt)j_geom + 1; // Shift for cell-centered
638 PetscInt k = (PetscInt)k_geom + 1; // Shift for cell-centered
639
640 // *** Bounds check is implicitly handled by DMDAVecGetArray for owned+ghost region ***
641 // However, accessing outside this region using global indices WILL cause an error.
642 // A preliminary check might still be wise if global IDs could be wild.
643 // We rely on LocateAllParticles to provide valid global indices [0..IM-1] etc.
644
645 LOG_LOOP_ALLOW(LOCAL, LOG_VERBOSE, p, 100,
646 "[Rank %d] Read CellID for p=%" PetscInt_FMT ", PID = %" PetscInt64_FMT ": (%" PetscInt_FMT ", %" PetscInt_FMT ", %" PetscInt_FMT ")\n",
647 rank, p, PID_arr[p], i, j, k);
648
649 // Check if the global index (i,j,k) falls within the local + ghost range
650 if (i >= gxs && i < gxs + gxm &&
651 j >= gys && j < gys + gym && // Adjust based on actual ghost width
652 k >= gzs && k < gzs + gzm ) // This check prevents definite crashes but doesn't guarantee ownership
653 {
654
655 // Increment count at the location corresponding to GLOBAL index (I,J,K)
656 // LOG_ALLOW(LOCAL, LOG_DEBUG, "CalculateParticleCountPerCell (Rank %d): Particle %d with global CellID (%d, %d, %d) incremented with a particle.\n",rank, p, i, j, k);
657 count_arr_3d[k][j][i] += 1.0;
658 particles_counted_locally++;
659 } else {
660 // This particle's global ID is likely outside the range this rank handles (even ghosts)
661 // note: this is not necessarily an error if the particle is legitimately outside the local+ghost region
662 LOG_ALLOW(LOCAL, LOG_VERBOSE,
663 "(Rank %d): Skipping particle %" PetscInt64_FMT " with global CellID (%" PetscInt_FMT ", %" PetscInt_FMT ", %" PetscInt_FMT ") - likely outside local+ghost range.\n",
664 rank, PID_arr[p], i, j, k);
665 }
666 }
667 LOG_ALLOW(LOCAL, LOG_DEBUG, "(Rank %d): Local counting finished. Processed %" PetscInt_FMT " particles locally.\n", rank, particles_counted_locally);
668
669 // --- Restore Access ---
670 ierr = DMDAVecRestoreArray(da, localcountVec, &count_arr_3d); CHKERRQ(ierr);
671 ierr = DMSwarmRestoreField(swarm,ParticleFieldName(PARTICLE_FIELD_ID_CELL_ID), NULL, NULL, (void **)&global_cell_id_arr); CHKERRQ(ierr);
672 ierr = DMSwarmRestoreField(swarm,ParticleFieldName(PARTICLE_FIELD_ID_PID),NULL,NULL,(void **)&PID_arr);CHKERRQ(ierr);
673
674 // --- Assemble Global Vector ---
675 LOG_ALLOW(GLOBAL, LOG_DEBUG, "Assembling global ParticleCount vector.\n");
676 ierr = VecZeroEntries(countVec); CHKERRQ(ierr); // Ensure global vector is zeroed before accumulation
677 ierr = DMLocalToGlobalBegin(da, localcountVec, ADD_VALUES, countVec); CHKERRQ(ierr);
678 ierr = DMLocalToGlobalEnd(da, localcountVec, ADD_VALUES, countVec); CHKERRQ(ierr);
679 /*
680 * OPTIONAL: Synchronize Ghosts for Stencil Operations
681 * If a future function needs to read ParticleCount from neighbor cells (e.g., density smoothing
682 * or gradient calculations), uncomment the following lines to update the ghost slots
683 * in user->lParticleCount with the final summed values.
684 *
685 ierr = UpdateLocalGhosts(user, FIELD_ID_PARTICLE_COUNT); CHKERRQ(ierr);
686 */
687
688 // --- Verification Logging ---
689 PetscReal total_counted_particles = 0.0, max_count_in_cell = 0.0;
690 ierr = VecSum(countVec, &total_counted_particles); CHKERRQ(ierr);
691 PetscInt max_idx_global = -1;
692 ierr = VecMax(countVec, &max_idx_global, &max_count_in_cell); CHKERRQ(ierr);
693 LOG_ALLOW(GLOBAL, LOG_INFO, "Total counted globally = %.0f, Max count in cell = %.0f\n",
694 total_counted_particles, max_count_in_cell);
695
696 // --- ADD THIS DEBUGGING BLOCK ---
697 if (max_idx_global >= 0) { // Check if VecMax found a location
698 // Need to convert the flat global index back to 3D global index (I, J, K)
699 // Get global grid dimensions (Nodes, NOT Cells IM/JM/KM)
700 PetscInt M, N;
701 M = user->info.mx; N = user->info.my;
702 // Note: Assuming DOF=1 for countVec, index mapping uses node dimensions M,N,P from DMDA creation (IM+1, etc)
703 // Re-check if your DMDA uses cell counts (IM) or node counts (IM+1) for Vec layout. Let's assume Node counts M,N,P.
704 PetscInt Kmax = max_idx_global / (M * N);
705 PetscInt Jmax = (max_idx_global % (M * N)) / M;
706 PetscInt Imax = max_idx_global % M;
707 LOG_ALLOW(GLOBAL, LOG_INFO, " -> Max count located at global index (I,J,K) = (%d, %d, %d) [Flat index: %d]\n",
708 (int)Imax, (int)Jmax, (int)Kmax, (int)max_idx_global);
709
710 // Also, let's explicitly check the count at (0,0,0)
711 PetscScalar count_at_origin = 0.0;
712 PetscScalar ***count_arr_for_check;
713 ierr = DMDAVecGetArrayRead(da, countVec, &count_arr_for_check); CHKERRQ(ierr);
714 // Check bounds before accessing - crucial if using global indices
715 PetscInt xs, ys, zs, xm, ym, zm;
716 xs = user->info.xs; ys = user->info.ys; zs = user->info.zs;
717 xm = user->info.xm; ym = user->info.ym; zm = user->info.zm;
718 if (0 >= xs && 0 < xs+xm && 0 >= ys && 0 < ys+ym && 0 >= zs && 0 < zs+zm) {
719 count_at_origin = count_arr_for_check[0][0][0]; // Access using global index (0,0,0)
720 } else {
721 // Origin is not on this rank (relevant for parallel, but check anyway)
722 count_at_origin = -999.0; // Indicate it wasn't accessible locally
723 }
724 ierr = DMDAVecRestoreArrayRead(da, countVec, &count_arr_for_check); CHKERRQ(ierr);
725 LOG_ALLOW(GLOBAL, LOG_INFO, " -> Count at global index (0,0,0) = %.1f\n", count_at_origin);
726
727 } else {
728 LOG_ALLOW(GLOBAL, LOG_WARNING, " -> VecMax did not return a location for the maximum value.\n");
729 }
730 // --- END DEBUGGING BLOCK ---
731
732 LOG_ALLOW(GLOBAL, LOG_INFO, "Particle counting complete.\n");
733
734
735 PROFILE_FUNCTION_END;
736 PetscFunctionReturn(0);
737}

943 * @param[in] first_newcomer Survivor count reported by PerformMigration().
944 * @return 0 when the check is off or passes; an error otherwise.
945 */

947 PetscInt first_newcomer)
948{
949 PetscInt n_after, mismatches = 0;

951
952 PetscFunctionBeginUser;
953 PROFILE_HELPER_BEGIN;
954 if (!user->simCtx->verifyGhostUpdates) { PROFILE_HELPER_END; PetscFunctionReturn(0); }

966 PROFILE_HELPER_END;

1006PetscErrorCode MigrateRestartParticlesUsingCellID(UserCtx *user)
1007{
1008 PetscErrorCode ierr;
1009 DM swarm = user->swarm;
1010 PetscInt nlocal;
1011 PetscInt *cell_p = NULL;
1012 PetscInt64 *pid_p = NULL;
1013 PetscMPIInt rank;
1014
1015 MigrationInfo *migrationList = NULL;
1016 PetscInt local_migration_count = 0;
1017 PetscInt migrationListCapacity = 0;
1018 PetscInt global_migration_count = 0;
1019
1020 PetscFunctionBeginUser;
1021 PROFILE_FUNCTION_BEGIN;
1022 rank = user->simCtx->rank;
1023
1024 ierr = DMSwarmGetLocalSize(swarm, &nlocal); CHKERRQ(ierr);

1027 if (nlocal > 0) {
1028 ierr = DMSwarmGetField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_CELL_ID), NULL, NULL, (void**)&cell_p); CHKERRQ(ierr);
1029 ierr = DMSwarmGetField(swarm, ParticleFieldName(PARTICLE_FIELD_ID_PID), NULL, NULL, (void**)&pid_p); CHKERRQ(ierr);
1030
1031 // Note: We do NOT need to modify the status field here.
1032 // We trust the loaded status (ACTIVE_AND_LOCATED) is correct for the destination rank.
1033
1034 for (PetscInt p_idx = 0; p_idx < nlocal; ++p_idx) {
1035 PetscInt ci = cell_p[3*p_idx + 0];
1036 PetscInt cj = cell_p[3*p_idx + 1];
1037 PetscInt ck = cell_p[3*p_idx + 2];
1038
1039 /* Skip particles with invalid Cell IDs (will be handled by LocateAllParticles) */
1040 if (ci < 0 || cj < 0 || ck < 0) {
1041 continue;
1042 }

1045 ierr = FindOwnerOfCell(user, ci, cj, ck, &owner_rank); CHKERRQ(ierr);
1046
1047 if (owner_rank != -1 && owner_rank != rank) {
1048 /* Particle belongs to another rank - migrate it */
1049 ierr = AddToMigrationList(&migrationList, &migrationListCapacity, &local_migration_count,
1050 p_idx, owner_rank); CHKERRQ(ierr);
1051
1052 LOG_PARTICLE(user->simCtx, pid_p[p_idx], "restart", "saved cell (%" PetscInt_FMT ",%" PetscInt_FMT ",%" PetscInt_FMT ") belongs to rank %d: migrating from rank %d\n",
1053 ci, cj, ck, owner_rank, (int)rank);
1054 }
1055 }
1056

1059 }

1062 ierr = MPI_Allreduce(&local_migration_count, &global_migration_count, 1, MPIU_INT, MPI_SUM, PETSC_COMM_WORLD); CHKERRQ(ierr);
1063
1064 if (global_migration_count > 0) {
1065 LOG_ALLOW(GLOBAL, LOG_INFO, "Fast restart migration: Directly migrating %" PetscInt_FMT " particles using CellIDs.\n", global_migration_count);
1066 ierr = SetMigrationRanks(user, migrationList, local_migration_count); CHKERRQ(ierr);
1067 ierr = PerformMigration(user, NULL); CHKERRQ(ierr);
1068 /* We do NOT flag newcomers here. We trust their loaded status (ACTIVE_AND_LOCATED) */
1069 /* is valid for their destination rank. */
1070 } else {
1071 LOG_ALLOW(GLOBAL, LOG_INFO, "Fast restart migration: All particles are already on correct ranks.\n");
1072 }
1073
1074 ierr = PetscFree(migrationList); CHKERRQ(ierr);
1075
1076 PROFILE_FUNCTION_END;
1077 PetscFunctionReturn(0);
1078}

1080/**
1081 * @brief Select the rank whose gathered bounding box is the best owner candidate for a particle.
1082 * @note Testing status:
1083 * The current direct surface reaches this helper through orchestrator
1084 * tests, but direction-complete immediate-neighbor coverage and the
1085 * explicit "not found in any rank" path are still targeted for future
1086 * bespoke tests.
1087 */
1088static PetscErrorCode GuessParticleOwnerWithBBox(UserCtx *user,
1089 const Particle *particle,
1090 const BoundingBox *bboxlist,
1091 PetscMPIInt *guess_rank_out)
1092{
1093 PetscMPIInt rank, size;
1094 const RankNeighbors *neighbors = &user->neighbors; // Use a direct pointer for clarity
1095 const BoundingBox *localBBox = &user->bbox;
1096
1097 PetscFunctionBeginUser;
1098
1099 PROFILE_HELPER_BEGIN;
1100
1101 // --- 1. Input Validation and Setup ---
1102 if (!user || !particle || !guess_rank_out || !bboxlist) {
1103 SETERRQ(PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "Null pointer provided to GuessParticleOwnerWithBBox.");
1104 }
1105 if (!localBBox|| !neighbors) {
1106 SETERRQ(PETSC_COMM_SELF, PETSC_ERR_ARG_WRONGSTATE, "Required user->bboxl or user->neighbors is not initialized.");
1107 }
1108
1109 rank = user->simCtx->rank;
1110 size = user->simCtx->size;
1111
1112 *guess_rank_out = MPI_PROC_NULL; // Default to "not found"
1113
1114 // The caller reports the guess's result in the particle's locate line.
1115 // --- Step 0: Check if the particle is inside the CURRENT rank's bounding box FIRST. ---
1116 // This handles the common case of initial placement where a particle is "lost" but physically local.
1117 if (IsParticleInBox(localBBox, &particle->loc)) {
1118 *guess_rank_out = rank;
1119
1120 PROFILE_FUNCTION_END;
1121 PetscFunctionReturn(0); // Found it, we're done.
1122 }
1123 // --- 2. Fast Path: Check Immediate Neighbors Based on Exit Direction ---
1124
1125 // Determine likely exit direction(s) to prioritize neighbor check
1126 PetscBool exit_xm = particle->loc.x < localBBox->min_coords.x;
1127 PetscBool exit_xp = particle->loc.x > localBBox->max_coords.x;
1128 PetscBool exit_ym = particle->loc.y < localBBox->min_coords.y;
1129 PetscBool exit_yp = particle->loc.y > localBBox->max_coords.y;
1130 PetscBool exit_zm = particle->loc.z < localBBox->min_coords.z;
1131 PetscBool exit_zp = particle->loc.z > localBBox->max_coords.z;
1132
1133 if (exit_xm && neighbors->rank_xm != MPI_PROC_NULL && IsParticleInBox(&bboxlist[neighbors->rank_xm], &particle->loc)) {
1134 *guess_rank_out = neighbors->rank_xm;
1135 } else if (exit_xp&& neighbors->rank_xp != MPI_PROC_NULL && IsParticleInBox(&bboxlist[neighbors->rank_xp], &particle->loc)) {
1136 *guess_rank_out = neighbors->rank_xp;
1137 } else if (exit_ym && neighbors->rank_ym != MPI_PROC_NULL && IsParticleInBox(&bboxlist[neighbors->rank_ym], &particle->loc)) {
1138 *guess_rank_out = neighbors->rank_ym;
1139 } else if (exit_yp && neighbors->rank_yp != MPI_PROC_NULL && IsParticleInBox(&bboxlist[neighbors->rank_yp], &particle->loc)) {
1140 *guess_rank_out = neighbors->rank_yp;
1141 } else if (exit_zm && neighbors->rank_zm != MPI_PROC_NULL && IsParticleInBox(&bboxlist[neighbors->rank_zm], &particle->loc)) {
1142 *guess_rank_out = neighbors->rank_zm;
1143 } else if (exit_zp && neighbors->rank_zp != MPI_PROC_NULL && IsParticleInBox(&bboxlist[neighbors->rank_zp], &particle->loc)) {
1144 *guess_rank_out = neighbors->rank_zp;
1145 }
1146 // Note: This does not handle corner/edge neighbors, which is why the fallback is essential.
1147
1148 if (*guess_rank_out != MPI_PROC_NULL) {
1149 PROFILE_FUNCTION_END;
1150 PetscFunctionReturn(0); // Found it, we're done.
1151 }
1152
1153 // --- 3. Robust Fallback: Check All Other Ranks ---
1154 // If we get here, the particle was not in any of the immediate face neighbors' boxes.
1155 for (PetscMPIInt r = 0; r < size; ++r) {
1156 if (r == rank) continue; // Don't check ourselves.
1157
1158 if (IsParticleInBox(&bboxlist[r], &particle->loc)) {
1159 *guess_rank_out = r;
1160 PROFILE_FUNCTION_END;
1161 PetscFunctionReturn(0); // Found it, we're done.
1162 }
1163 }
1164
1165 // Not in any rank's bounding box: guess_rank_out stays MPI_PROC_NULL, signaling failure to the caller.
1166 PROFILE_FUNCTION_END;
1167 PetscFunctionReturn(0);
1168}

1170/**
1171 * @brief Implementation of \ref LocateAllParticlesInGrid().
1172 * @details Full API contract (arguments, ownership, side effects) is documented with
1173 * the header declaration in `include/ParticleMotion.h`.
1174 * @see LocateAllParticlesInGrid()
1175 */
1176PetscErrorCode LocateAllParticlesInGrid(UserCtx *user,BoundingBox *bboxlist)
1177{
1178 PetscErrorCode ierr;
1179 PetscInt passes = 0;
1180 const PetscInt MAX_MIGRATION_PASSES = 50; // Safety break for runaway loops
1181 PetscInt global_migrations_this_pass;
1182 PetscMPIInt rank;
1183 PetscInt total_migrated_this_timestep = 0;
1184
1185 PetscFunctionBeginUser;
1186 PROFILE_FUNCTION_BEGIN;
1187 rank = user->simCtx->rank;
1188 ierr = ResetSearchMetrics(user->simCtx); CHKERRQ(ierr);
1189 LOG_ALLOW(GLOBAL, LOG_INFO, "LocateAllParticlesInGrid (Orchestrator) - Beginning particle settlement process.\n");
1190
1191 // This loop ensures that particles that jump across multiple ranks are
1192 // handled correctly in successive, iterative handoffs.
1193 do {
1194 passes++;
1195 user->simCtx->searchMetrics.currentSettlementPass = passes;
1196 LOG_ALLOW(GLOBAL, LOG_INFO, "Starting migration pass %" PetscInt_FMT ".\n", passes);
1197
1198 // --- STAGE 1: PER-PASS INITIALIZATION ---
1199 MigrationInfo *migrationList = NULL;
1200 PetscInt local_migration_count = 0;
1201 PetscInt migrationListCapacity = 0;
1202 PetscInt nlocal_before;
1203 PetscInt64 *pids_before_snapshot = NULL;
1204 PetscInt local_lost_count = 0;
1205
1206 ierr = DMSwarmGetLocalSize(user->swarm, &nlocal_before); CHKERRQ(ierr);
1207 if (passes == 1) {
1208 user->simCtx->searchMetrics.searchPopulation += (PetscInt64)nlocal_before;
1209 }
1210 LOG_ALLOW(LOCAL, LOG_DEBUG, "[Rank %d] Pass %" PetscInt_FMT " begins with %" PetscInt_FMT " local particles.\n", rank, passes, nlocal_before);
1211
1212
1213 // --- STAGE 2: PRE-MIGRATION SNAPSHOT & MAIN PROCESSING LOOP ---
1214 if (nlocal_before > 0) {
1215 // Get pointers to all fields needed for this pass
1216 PetscReal *pos_p, *weights_p, *vel_p;
1217 PetscInt *cell_p, *status_p;
1218 PetscInt64 *pid_p;
1219 ierr = DMSwarmGetField(user->swarm, ParticleFieldName(PARTICLE_FIELD_ID_POSITION), NULL, NULL, (void**)&pos_p); CHKERRQ(ierr);
1220 ierr = DMSwarmGetField(user->swarm, ParticleFieldName(PARTICLE_FIELD_ID_VELOCITY), NULL, NULL, (void**)&vel_p); CHKERRQ(ierr);
1221 ierr = DMSwarmGetField(user->swarm, ParticleFieldName(PARTICLE_FIELD_ID_WEIGHT), NULL, NULL, (void**)&weights_p); CHKERRQ(ierr);
1222 ierr = DMSwarmGetField(user->swarm, ParticleFieldName(PARTICLE_FIELD_ID_CELL_ID), NULL, NULL, (void**)&cell_p); CHKERRQ(ierr);
1223 ierr = DMSwarmGetField(user->swarm, ParticleFieldName(PARTICLE_FIELD_ID_PID), NULL, NULL, (void**)&pid_p); CHKERRQ(ierr);
1224 ierr = DMSwarmGetField(user->swarm, ParticleFieldName(PARTICLE_FIELD_ID_LOCATION_STATUS), NULL, NULL, (void**)&status_p); CHKERRQ(ierr);
1225 // Opened once for the pass rather than by each particle's walk.
1226 FacePlaneArrays plane_arrays;
1227 ierr = MetricGetFacePlaneArrays(user, PETSC_FALSE, &plane_arrays); CHKERRQ(ierr);
1228
1229 // Newcomers are identified by position after migration; the sorted PID snapshot is
1230 // taken only to cross-check that under -picurv_verify_ghost_updates.
1231 if (user->simCtx->verifyGhostUpdates) {
1232 ierr = GetLocalPIDSnapshot(pid_p, nlocal_before, &pids_before_snapshot); CHKERRQ(ierr);
1233 }
1234
1235 for (PetscInt p_idx = 0; p_idx < nlocal_before; p_idx++) {
1236
1237 // LOST and EXITED are final within one settlement: only pass 1 searches them, so a
1238 // particle beyond the boundary is never re-walked and re-migrated pass after pass.
1239 if (status_p[p_idx] == ACTIVE_AND_LOCATED ||
1240 (passes > 1 && (status_p[p_idx] == LOST || status_p[p_idx] == EXITED))) {
1241 continue;
1242 }
1243
1244 // UNPACK: Create a temporary C struct for easier processing using our helper.
1245 Particle current_particle;
1246 ierr = UnpackSwarmFields(p_idx, pid_p, weights_p, pos_p, cell_p, vel_p, status_p,NULL,NULL,NULL,&current_particle); CHKERRQ(ierr);
1247
1248 ParticleLocationStatus final_status = (ParticleLocationStatus)status_p[p_idx];
1249 const PetscBool walk_from_prior_cell = (PetscBool)(current_particle.cell[0] >= 0);
1250
1251 // CASE 1: Particle has a valid prior cell index.
1252 // It has moved, so we only need to run the robust walk from its last known location.
1253 if (walk_from_prior_cell) {
1254 ierr = LocateParticleOrFindMigrationTarget(user, &plane_arrays, &current_particle, &final_status); CHKERRQ(ierr);
1255 }
1256
1257 // CASE 2: Particle has no valid cell (cell = -1), or its walk from the prior cell was
1258 // blocked by the domain boundary and is retried here, in the same pass. Strategy: Guess -> Verify.
1259 if (!walk_from_prior_cell || final_status == NEEDS_LOCATION) {
1260 PetscMPIInt guessed_owner_rank = MPI_PROC_NULL;
1261 ierr = GuessParticleOwnerWithBBox(user, &current_particle, bboxlist, &guessed_owner_rank); CHKERRQ(ierr);
1262
1263 // If the guess finds a DIFFERENT rank, we can mark for migration and skip the walk.
1264 if (guessed_owner_rank != MPI_PROC_NULL && guessed_owner_rank != rank) {
1265 user->simCtx->searchMetrics.bboxGuessSuccessCount++;
1266 LOG_PARTICLE(user->simCtx, current_particle.PID, "locate", "pass %" PetscInt_FMT ": bounding box of rank %d holds (%.6e, %.6e, %.6e): migrating there unwalked\n",
1267 passes, guessed_owner_rank, current_particle.loc.x, current_particle.loc.y, current_particle.loc.z);
1268 final_status = MIGRATING_OUT;
1269 current_particle.destination_rank = guessed_owner_rank;
1270 } else {
1271 // The guess either found no owner or found this rank; either way the walk decides.
1272 user->simCtx->searchMetrics.bboxGuessFallbackCount++;
1273 LOG_PARTICLE(user->simCtx, current_particle.PID, "locate", "pass %" PetscInt_FMT ": bounding-box guess %s; walking\n",
1274 passes, (guessed_owner_rank == rank) ? "found this rank" : "found no rank");
1275 ierr = LocateParticleOrFindMigrationTarget(user, &plane_arrays, &current_particle, &final_status); CHKERRQ(ierr);
1276 }
1277 }
1278
1279 // --- PROCESS THE FINAL, DEFINITIVE STATUS ---
1280 current_particle.location_status = final_status;
1281 ierr = UpdateSwarmFields(p_idx, &current_particle, pos_p, vel_p, weights_p, cell_p, status_p,NULL,NULL,NULL); CHKERRQ(ierr);
1282
1283 if (final_status == MIGRATING_OUT) {
1284 ierr = AddToMigrationList(&migrationList, &migrationListCapacity, &local_migration_count, p_idx, current_particle.destination_rank); CHKERRQ(ierr);
1285 } else if (final_status == LOST) {
1286 local_lost_count++;
1287 user->simCtx->searchMetrics.searchLostCount++;
1288 } else if (final_status == EXITED) {
1289 user->simCtx->searchMetrics.searchExitedCount++;
1290 } else if (final_status == ACTIVE_AND_LOCATED) {
1291 user->simCtx->searchMetrics.searchLocatedCount++;
1292 }
1293
1294 } // End of main particle processing loop
1295
1296 // Restore all the fields acquired for this pass.
1297 ierr = DMSwarmRestoreField(user->swarm, ParticleFieldName(PARTICLE_FIELD_ID_POSITION), NULL, NULL, (void**)&pos_p); CHKERRQ(ierr);
1298 ierr = DMSwarmRestoreField(user->swarm, ParticleFieldName(PARTICLE_FIELD_ID_VELOCITY), NULL, NULL, (void**)&vel_p); CHKERRQ(ierr);
1299 ierr = DMSwarmRestoreField(user->swarm, ParticleFieldName(PARTICLE_FIELD_ID_WEIGHT), NULL, NULL, (void**)&weights_p); CHKERRQ(ierr);
1300 ierr = DMSwarmRestoreField(user->swarm, ParticleFieldName(PARTICLE_FIELD_ID_CELL_ID), NULL, NULL, (void**)&cell_p); CHKERRQ(ierr);
1301 ierr = DMSwarmRestoreField(user->swarm, ParticleFieldName(PARTICLE_FIELD_ID_PID), NULL, NULL, (void**)&pid_p); CHKERRQ(ierr);
1302 ierr = DMSwarmRestoreField(user->swarm, ParticleFieldName(PARTICLE_FIELD_ID_LOCATION_STATUS), NULL, NULL, (void**)&status_p); CHKERRQ(ierr);
1303 ierr = MetricRestoreFacePlaneArrays(user, &plane_arrays); CHKERRQ(ierr);
1304 }
1305
1306 // --- STAGE 3: ACTION & MPI COMMUNICATION ---
1307 LOG_ALLOW(LOCAL, LOG_DEBUG, "[Rank %d] Pass %" PetscInt_FMT ": Identified %" PetscInt_FMT " particles to migrate out.\n", rank, passes, local_migration_count);
1308
1309 // --- STAGE 3: SYNCHRONIZE AND DECIDE ---
1310 // FIRST, determine if any rank wants to migrate. This call is safe because
1311 // all ranks have finished their local work and can participate.
1312 ierr = MPI_Allreduce(&local_migration_count, &global_migrations_this_pass, 1, MPIU_INT, MPI_SUM, PETSC_COMM_WORLD); CHKERRQ(ierr);
1313
1314 total_migrated_this_timestep += global_migrations_this_pass;
1315
1316 if(global_migrations_this_pass > 0 ){
1317
1318 LOG_ALLOW(GLOBAL, LOG_INFO, "Pass %" PetscInt_FMT ": Migrating %" PetscInt_FMT " particles globally.\n", passes, global_migrations_this_pass);
1319
1320 ierr = SetMigrationRanks(user, migrationList, local_migration_count); CHKERRQ(ierr);
1321 PetscInt n_survivors = 0;
1322 ierr = PerformMigration(user, &n_survivors); CHKERRQ(ierr);
1323
1324 // --- STAGE 4: POST-MIGRATION RESET ---
1325 // The received particles follow the survivors; flag them with NEEDS_LOCATION so
1326 // they are processed in the next pass.
1327 ierr = FlagNewcomersForLocation(user->swarm, nlocal_before, n_survivors); CHKERRQ(ierr);
1328 ierr = VerifyNewcomersAgainstPIDs(user, nlocal_before, pids_before_snapshot, n_survivors); CHKERRQ(ierr);
1329 }
1330 // --- STAGE 5: LOOP SYNCHRONIZATION AND CLEANUP ---
1331
1332 ierr = PetscFree(pids_before_snapshot);
1333 ierr = PetscFree(migrationList);
1334
1335 LOG_ALLOW(GLOBAL, LOG_INFO, "End of pass %" PetscInt_FMT ". Total particles migrated globally: %" PetscInt_FMT ".\n", passes, global_migrations_this_pass);
1336
1337 } while (global_migrations_this_pass > 0 && passes < MAX_MIGRATION_PASSES);
1338
1339 // --- FINAL CHECKS ---
1340 if (passes >= MAX_MIGRATION_PASSES) {
1341 SETERRQ(PETSC_COMM_WORLD, PETSC_ERR_CONV_FAILED, "Particle migration failed to converge after %" PetscInt_FMT " passes. Check for particles oscillating between ranks.", MAX_MIGRATION_PASSES);
1342 }
1343
1344 user->simCtx->particlesMigratedLastStep = total_migrated_this_timestep;
1345 user->simCtx->migrationPassesLastStep = passes;
1346 user->simCtx->searchMetrics.maxParticlePassDepth = PetscMax(user->simCtx->searchMetrics.maxParticlePassDepth, (PetscInt64)passes);
1347 user->simCtx->searchMetrics.currentSettlementPass = 0;
1348
1349 LOG_ALLOW(GLOBAL, LOG_INFO, "Particle Location completed in %" PetscInt_FMT " passes.\n", passes);
1350
1351 PROFILE_FUNCTION_END;
1352 PetscFunctionReturn(0);
1353}
