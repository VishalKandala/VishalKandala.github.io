48def resolve_doxygen_file_href(html_dir: Path, rel_path: str) -> str:
49 """Resolve doxygen file href."""
50 name = Path(rel_path).name
51 candidate = doxygen_file_page(name)
52 if (html_dir / candidate).exists():
53 return candidate
54 candidate_path = doxygen_file_page_with_path(rel_path)

65def collect_file_rows(repo_root: Path, html_dir: Path, base_dir: str, suffixes: set[str]) -> list[tuple[str, str, str]]:
66 """Collect file rows."""
67 rows: list[tuple[str, str, str]] = []
68 root = repo_root / base_dir
69 if not root.exists():
70 return rows
71 for path in sorted(root.rglob("*")):
72 if not path.is_file():
73 continue
74 if suffixes and path.suffix.lower() not in suffixes:
75 continue
76 rel = path.relative_to(repo_root).as_posix()
77 href = resolve_doxygen_file_href(html_dir, rel) or make_repo_href(rel)
78 rows.append((path.name, rel, href))
79 return rows

91def collect_struct_rows(repo_root: Path, html_dir: Path) -> list[tuple[str, str, str]]:
92 """Collect struct rows."""
93 struct_to_header: dict[str, str] = {}
94 include_dir = repo_root / "include"
95 if not include_dir.exists():
96 return []
97
98 for header in sorted(include_dir.rglob("*.h")):
99 text = header.read_text(encoding="utf-8", errors="ignore")
100 header_rel = header.relative_to(repo_root).as_posix()
101 names = extract_struct_names(text)
102 for name in names:
103 struct_to_header.setdefault(name, header_rel)
104
105 rows: list[tuple[str, str, str]] = []
106 for name, header_rel in sorted(struct_to_header.items(), key=lambda item: item[0].lower()):
107 if name in IGNORED_STRUCT_NAMES:
108 continue
109 struct_page = f"struct{name}.html"
110 if (html_dir / struct_page).exists():
111 href = struct_page
112 else:
113 href = resolve_doxygen_file_href(html_dir, header_rel) or make_repo_href(header_rel)
114 rows.append((name, header_rel, href))
115 return rows

118def extract_struct_names(text: str) -> set[str]:
119 """Extract struct names."""
120 names: set[str] = set()
121
122 for match in NAMED_STRUCT_RE.finditer(text):
123 names.add(match.group(1))
124
125 in_typedef_struct = False
126 brace_depth = 0
127 typedef_tag_name: str | None = None
128 for line in text.splitlines():
129 if not in_typedef_struct:
130 start = TYPEDEF_START_RE.search(line)
131 if not start:
132 continue
133 in_typedef_struct = True
134 typedef_tag_name = start.group(1)
135 if typedef_tag_name:
136 names.add(typedef_tag_name)
137 brace_depth = line.count("{") - line.count("}")
138 if brace_depth <= 0:
139 in_typedef_struct = False
140 typedef_tag_name = None
141 continue
142
143 brace_depth += line.count("{") - line.count("}")
144 end = TYPEDEF_END_RE.search(line)
145 if end:
146 names.add(end.group(1))
147 if brace_depth <= 0:
148 in_typedef_struct = False
149 typedef_tag_name = None
150
151 return names

154def categorize_struct(name: str) -> str:
155 """Categorize struct."""
156 if name.startswith("BC") or "Boundary" in name or name == "FlowWave":
157 return "Boundary Condition System"
158 if name.startswith("IBM") or name in {"FSInfo", "SurfElmtInfo", "Cstart"}:
159 return "Immersed Boundary and FSI"
160 if name.startswith("Particle") or name in {"MigrationInfo"}:
161 return "Particle Transport and Statistics"
162 if name in {"SimCtx", "UserCtx", "UserMG", "MGCtx", "ScalingCtx", "DualMonitorCtx", "ProfiledFunction"}:
163 return "Runtime Control and Solver Orchestration"
164 if name.startswith("VTK") or name == "PostProcessParams":
165 return "I/O and Postprocessing"
166 if name in {"BoundingBox", "Cell", "Cmpnts", "Cmpnts2", "Cpt2D", "RankCellInfo", "RankNeighbors"}:
167 return "Grid and Geometry"
168 return "Generic Containers and Utilities"

180def render_rows(rows: list[tuple[str, str, str]], empty_msg: str) -> str:
181 """Render rows."""
182 if not rows:
183 return f"<tr><td colspan='2'>{html.escape(empty_msg)}</td></tr>"
184 out: list[str] = []
185 for name, rel, href in rows:
186 out.append(
187 "<tr>"
188 f"<td class='indexkey'>{render_link(name, href)}</td>"
189 f"<td class='indexvalue'><code>{html.escape(rel)}</code></td>"
190 "</tr>"
191 )

232def write_structured_file_index(repo_root: Path, html_dir: Path) -> None:
