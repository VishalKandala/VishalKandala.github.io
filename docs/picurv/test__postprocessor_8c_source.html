164static PetscErrorCode TestSetupSimulationEnvironmentCreatesStatisticsDirectoryWithoutClobbering(void)
165{
166 SimCtx *simCtx = NULL;
167 UserCtx *user = NULL;
168 PostProcessParams *pps = NULL;
169 char tmpdir[PETSC_MAX_PATH_LEN];
170 char post_path[PETSC_MAX_PATH_LEN];
171 char source_dir[PETSC_MAX_PATH_LEN];
172 char stats_dir[PETSC_MAX_PATH_LEN];
173 char stats_prefix[PETSC_MAX_PATH_LEN];
174 char sentinel_path[PETSC_MAX_PATH_LEN];
175 PetscBool exists = PETSC_FALSE;
176 FILE *file = NULL;
177
178 PetscFunctionBeginUser;
179 PetscCall(PicurvCreateMinimalContexts(&simCtx, &user, 4, 4, 4));
180 PetscCall(PetscNew(&simCtx->pps));
181 pps = simCtx->pps;
182 PetscCall(PetscMemzero(pps, sizeof(*pps)));
183 PetscCall(PicurvMakeTempDir(tmpdir, sizeof(tmpdir)));
184
185 PetscCall(PetscSNPrintf(post_path, sizeof(post_path), "%s/post.run", tmpdir));
186 file = fopen(post_path, "w");
187 PetscCheck(file != NULL, PETSC_COMM_SELF, PETSC_ERR_FILE_OPEN, "Failed to create post config %s.", post_path);
188 fputs("startTime = 0\nendTime = 0\ntimeStep = 1\n", file);
189 fclose(file);
190 file = NULL;
191
192 PetscCall(PetscSNPrintf(source_dir, sizeof(source_dir), "%s/output", tmpdir));
193 PetscCall(PicurvEnsureDir(source_dir));
194 PetscCall(PetscSNPrintf(stats_dir, sizeof(stats_dir), "%s/output/statistics", tmpdir));
195 PetscCall(PetscSNPrintf(stats_prefix, sizeof(stats_prefix), "%s/BrownianStats", stats_dir));
196 PetscCall(PetscSNPrintf(sentinel_path, sizeof(sentinel_path), "%s/sentinel.keep", stats_dir));
197
198 simCtx->exec_mode = EXEC_MODE_POSTPROCESSOR;
199 simCtx->generate_grid = PETSC_TRUE;
200 PetscCall(PetscStrncpy(simCtx->PostprocessingControlFile, post_path, sizeof(simCtx->PostprocessingControlFile)));
201 PetscCall(PetscStrncpy(pps->source_dir, source_dir, sizeof(pps->source_dir)));
202 PetscCall(PetscSNPrintf(pps->output_prefix, sizeof(pps->output_prefix), "%s/viz/Field", tmpdir));
203 PetscCall(PetscStrncpy(pps->statistics_pipeline, "ComputeMSD", sizeof(pps->statistics_pipeline)));
204 PetscCall(PetscStrncpy(pps->statistics_output_prefix, stats_prefix, sizeof(pps->statistics_output_prefix)));
205
206 PetscCall(SetupSimulationEnvironment(simCtx));
207 PetscCall(PetscTestDirectory(stats_dir, 'r', &exists));
208 PetscCall(PicurvAssertBool(exists,
209 "SetupSimulationEnvironment should create the statistics directory when it is missing"));
210
211 file = fopen(sentinel_path, "w");
212 PetscCheck(file != NULL, PETSC_COMM_SELF, PETSC_ERR_FILE_OPEN, "Failed to create statistics sentinel %s.", sentinel_path);
213 fputs("keep\n", file);
214 fclose(file);
215 file = NULL;
216
217 PetscCall(SetupSimulationEnvironment(simCtx));
218 PetscCall(PicurvAssertFileExists(sentinel_path,
219 "SetupSimulationEnvironment should preserve existing statistics directory contents on repeat runs"));
220
221 PetscCall(PicurvRemoveTempDir(tmpdir));
222 PetscCall(PicurvDestroyMinimalContexts(&simCtx, &user));
223 PetscFunctionReturn(0);
224}

267
268 PetscFunctionBeginUser;
269 PetscCall(PetscMemzero(&pps, sizeof(pps)));
270 PetscCall(PicurvCreateMinimalContexts(&simCtx, &user, 4, 4, 4));
271 PetscCall(PicurvCreateSwarmPair(user, 3, "ske"));

273 PetscCall(PetscSNPrintf(vtk_path, sizeof(vtk_path), "%s/particles_00004.vtp", tmpdir));
274
275 pps.outputParticles = PETSC_TRUE;
276 pps.particle_output_freq = 1;
277 PetscCall(PetscStrncpy(pps.particle_fields, "position,velocity", sizeof(pps.particle_fields)));

279
280 PetscCall(DMSwarmGetField(user->swarm, "position", NULL, NULL, (void *)&pos_arr));

282 for (PetscInt p = 0; p < 3; ++p) {
283 pos_arr[p][0] = (PetscReal)p;
284 pos_arr[p][1] = (PetscReal)(p + 1);
285 pos_arr[p][2] = (PetscReal)(p + 2);
286 vel_arr[p][0] = 2.0 * (PetscReal)p;
287 vel_arr[p][1] = 3.0 * (PetscReal)p;
288 vel_arr[p][2] = 4.0 * (PetscReal)p;
289 }
290 PetscCall(DMSwarmRestoreField(user->swarm, "position", NULL, NULL, (void *)&pos_arr));
291 PetscCall(DMSwarmRestoreField(user->swarm, "velocity", NULL, NULL, (void *)&vel_arr));
292
293 PetscCall(WriteParticleFile(user, &pps, 4));
294 PetscCall(PicurvAssertFileExists(vtk_path, "WriteParticleFile should emit a .vtp file for requested particle fields"));
295
296 PetscCall(PicurvRemoveTempDir(tmpdir));
297 PetscCall(PicurvDestroyMinimalContexts(&simCtx, &user));
