14/**
15 * @brief Assert that a file path does not exist.
16 * @param[in] path File path to check.
17 * @param[in] context Failure context message.
18 * @return Petsc error code.
19 */
20static PetscErrorCode PicurvAssertFileMissing(const char *path, const char *context)
21{
22 PetscBool exists = PETSC_FALSE;
23
24 PetscFunctionBeginUser;
25 PetscCall(PetscTestFile(path, 'r', &exists));
26 if (exists) {
27 PetscCall(PetscPrintf(PETSC_COMM_WORLD, "[FAIL] %s | unexpected file: %s\n", context, path));
28 SETERRQ(PETSC_COMM_SELF, PETSC_ERR_FILE_OPEN, "Unexpected file is present.");
29 }
30 PetscFunctionReturn(0);
31}

33/**
34 * @brief Count file lines that begin with a given prefix.
35 * @param[in] path File path to scan.
36 * @param[in] prefix Prefix to match at the beginning of each line.
37 * @param[out] count_out Count of matching lines.
38 * @return Petsc error code.
39 */
40static PetscErrorCode CountLinesWithPrefix(const char *path, const char *prefix, PetscInt *count_out)
41{
42 FILE *file = NULL;
43 char line[4096];
44 PetscInt count = 0;
45
46 PetscFunctionBeginUser;
47 PetscCheck(count_out != NULL, PETSC_COMM_SELF, PETSC_ERR_ARG_NULL, "count_out must not be NULL.");
48 file = fopen(path, "r");
49 PetscCheck(file != NULL, PETSC_COMM_SELF, PETSC_ERR_FILE_OPEN, "Failed to open %s.", path);
50 while (fgets(line, sizeof(line), file)) {
51 if (strncmp(line, prefix, strlen(prefix)) == 0) count++;
52 }
53 fclose(file);
54 *count_out = count;
55 PetscFunctionReturn(0);
56}
