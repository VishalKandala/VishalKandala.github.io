1#!/usr/bin/env python3
2"""!
3@file cross_section_profile.py
4@brief Reduce a duct's field-statistics window to cross-section fields and bisector profiles.
5
6@details
7A square duct has one homogeneous direction, the streamwise one, so its statistics
8reduce to a cross-section rather than a wall-normal line. The reduction itself, the
9window lookup in `checkpoint.meta` and the covariance over the homogeneous set, is
10the one `driven_channel/tools/wall_normal_profile.py` performs and is imported from
11it; this script averages along the stream only and then reads off what the case's
12acceptance criteria name (`driven_duct.md` section 5):
13
14- `<prefix>_cross_section.csv`: every cell's mean velocity and the six covariance
15 components, in solver units, for contouring the secondary flow;
16- `<prefix>_wall_bisector.csv`: wall units along the wall bisector, with the
17 wall-normal secondary velocity;
18- `<prefix>_corner_bisector.csv`: wall units along the corner bisector, with the
19 secondary velocity along it (square cross-sections only);
20- `<prefix>_wall_shear.csv`: the local wall shear around the perimeter, normalized
21 by its perimeter mean.
22
23The cross-section is folded onto one octant by default: its four reflections and,
24when the two cross-stream axes carry identical node distributions, the diagonal
25swap. A reflection reverses the velocity component normal to its mirror line, so
26that component's mean and its covariances with the other two change sign. Folding
27averages eight images of one long average, which is what makes a 1-3% secondary
28flow readable; `--no-fold` keeps the raw cross-section, whose asymmetry is a
29measure of the remaining sampling error.
30
31The friction velocity is the perimeter mean of the wall shear: from `--body-force`
32as sqrt(f A / P), the exact mean force balance, or the same with f averaged from the
33run's `driven_flow.csv` (`--driven-flow-csv`); from `--u-tau`; or by default from
34the first cells' resolved gradient, nu U / d, which is meaningful only when the
35first cells lie in the viscous sublayer.
36
37Usage:
38
39@code
40 cross_section_profile.py \\
41 --checkpoint RUN/output/checkpoints/step_000000200000 --window stationary \\
42 --grid RUN/inputs/grid/grid.run --viscosity 4.5351474e-04 \\
43 --output-prefix duct
44@endcode
45"""

70def transform(numpy, mean, covariance, pairs, permutation, sign):
71 """!
72 @brief Apply a component permutation with sign changes to a mean and its covariance.
73 @param[in] numpy The numpy module.
74 @param[in] mean Array ending in 3 velocity components.
75 @param[in] covariance Array ending in 6 covariance components, in `pairs` order.
76 @param[in] pairs Component pairs of the covariance layout.
77 @param[in] permutation New component index of each old component.
78 @param[in] sign Sign applied to each old component.
79 @return Transformed (mean, covariance).
80 """
81 new_mean = numpy.empty_like(mean)
82 new_cov = numpy.empty_like(covariance)
83 for c in range(3):
84 new_mean[..., permutation[c]] = sign[c] * mean[..., c]
85 for index, (a, b) in enumerate(pairs):
86 target = pairs.index(tuple(sorted((permutation[a], permutation[b]))))
87 new_cov[..., target] = sign[a] * sign[b] * covariance[..., index]
88 return new_mean, new_cov

91def main(argv=None):
92 """!
93 @brief Entry point.
94 @param[in] argv Command-line style argument list supplied to the function.
95 @return Process exit status.
96 """
97 parser = argparse.ArgumentParser(description=__doc__,
98 formatter_class=argparse.RawDescriptionHelpFormatter)
99 parser.add_argument("--checkpoint", required=True,
100 help="Committed checkpoint bundle directory (output/checkpoints/step_NNN).")
101 parser.add_argument("--grid", required=True, help="Canonical PICGRID path for the run.")
102 parser.add_argument("--window", default="stationary", help="Statistics window name.")
103 parser.add_argument("--block", type=int, default=0, help="Block index.")
104 parser.add_argument("--stream-axis", default="Zeta", choices=("Xi", "Eta", "Zeta"),
105 help="Streamwise (driven, homogeneous) axis token.")
106 parser.add_argument("--viscosity", type=float, required=True,
107 help="Kinematic viscosity in solver units.")
108 source = parser.add_mutually_exclusive_group()
109 source.add_argument("--body-force", type=float,
110 help="Converged driving body force f; u_tau = sqrt(f*A/P).")
111 source.add_argument("--driven-flow-csv",
112 help="The run's driven_flow.csv; u_tau = sqrt(<a> A / P) from the driving "
113 "acceleration averaged over the window.")
114 source.add_argument("--u-tau", type=float, help="Friction velocity supplied directly.")
115 parser.add_argument("--no-fold", action="store_true",
116 help="Keep the raw cross-section instead of folding it onto one octant.")
117 parser.add_argument("--output-prefix", required=True, help="Prefix for the four output CSVs.")
118 args = parser.parse_args(argv)
119
120 channel = load_channel_tool()
121 pairs = channel.M2_PAIRS
122 stream_kji = channel.AXIS_TO_KJI[args.stream_axis]
123 stream_c = channel.AXIS_TO_COMPONENT[args.stream_axis]
124 stats = channel.homogeneous_statistics(args.checkpoint, args.window, args.grid, args.block,
125 (stream_kji,))
126 numpy, nodes, window = stats["numpy"], stats["nodes"], stats["window"]
127 mean, covariance = stats["mean"], stats["covariance"]
128
129 # The two remaining axes, in (k, j, i) order, and the velocity component normal
130 # to each one's walls. The reduced arrays are indexed (first, second).
131 cross_kji = [axis for axis in (0, 1, 2) if axis != stream_kji]
132 kji_to_component = {0: 2, 1: 1, 2: 0}
133 comp = [kji_to_component[axis] for axis in cross_kji]
134 names = "xyz"
135
136 def axis_nodes(kji, component):
137 line = {0: nodes[:, 0, 0, :], 1: nodes[0, :, 0, :], 2: nodes[0, 0, :, :]}[kji]
138 return line[:, component]
139
140 edges = [axis_nodes(kji, c) for kji, c in zip(cross_kji, comp)]
141 # Bisectors and wall shear need a rectilinear cross-section: each cross-stream
142 # coordinate must depend on its own index alone.
143 span = [abs(e[-1] - e[0]) for e in edges]
144 for kji, c, e, length in zip(cross_kji, comp, edges, span):
145 full = nodes[..., c]
146 reference = {0: e[:, None, None], 1: e[None, :, None], 2: e[None, None, :]}[kji]
147 if numpy.max(numpy.abs(full - reference)) > 1.0e-9 * length:
148 raise SystemExit("the cross-section is not rectilinear; this reduction assumes it is.")
149 centres = [0.5 * (e[:-1] + e[1:]) for e in edges]
150 widths = [numpy.abs(numpy.diff(e)) for e in edges]
151 n0, n1 = mean.shape[0], mean.shape[1]
152 if (centres[0].size, centres[1].size) != (n0, n1):
153 raise SystemExit(f"cross-section cell counts {(centres[0].size, centres[1].size)} "
154 f"do not match the statistics {(n0, n1)}.")
155
156 def symmetric(c, e, length):
157 return numpy.max(numpy.abs((c - e[0]) - (e[-1] - c[::-1]))) <= 1.0e-9 * length
158
159 folded = "raw (--no-fold)"
160 if not args.no_fold:
161 if not all(symmetric(c, e, length) for c, e, length in zip(centres, edges, span)):
162 raise SystemExit("the cross-section grid is not mirror-symmetric; rerun with --no-fold.")
163 identity = (0, 1, 2)
164 images = []
165 for flip0 in (False, True):
166 for flip1 in (False, True):
167 m, v = mean, covariance
168 sign = [1.0, 1.0, 1.0]
169 if flip0:
170 m, v = m[::-1, :], v[::-1, :]
171 sign[comp[0]] = -1.0
172 if flip1:
173 m, v = m[:, ::-1], v[:, ::-1]
174 sign[comp[1]] = -1.0
175 images.append(transform(numpy, m, v, pairs, identity, sign))
176 diagonal = n0 == n1 and numpy.max(
177 numpy.abs((edges[0] - edges[0][0]) - (edges[1] - edges[1][0]))) <= 1.0e-9 * span[0]
178 if diagonal:
179 swap = list(identity)
180 swap[comp[0]], swap[comp[1]] = comp[1], comp[0]
181 images += [transform(numpy, numpy.swapaxes(m, 0, 1), numpy.swapaxes(v, 0, 1),
182 pairs, swap, (1.0, 1.0, 1.0)) for m, v in images]
183 mean = sum(m for m, _ in images) / len(images)
184 covariance = sum(v for _, v in images) / len(images)
185 folded = f"folded over {len(images)} symmetry images"
186
187 nu = args.viscosity
188 U = mean[..., stream_c]
189 area = numpy.outer(widths[0], widths[1])
190 bulk = float(numpy.sum(U * area) / numpy.sum(area))
191
192 # Local wall shear from the first cell on each of the four walls: nu U / d.
193 first = [abs(centres[0][0] - edges[0][0]), abs(edges[0][-1] - centres[0][-1]),
194 abs(centres[1][0] - edges[1][0]), abs(edges[1][-1] - centres[1][-1])]
195 walls = [
196 (f"{names[comp[0]]}_min", centres[1], widths[1], nu * U[0, :] / first[0]),
197 (f"{names[comp[0]]}_max", centres[1], widths[1], nu * U[-1, :] / first[1]),
198 (f"{names[comp[1]]}_min", centres[0], widths[0], nu * U[:, 0] / first[2]),
199 (f"{names[comp[1]]}_max", centres[0], widths[0], nu * U[:, -1] / first[3]),
200 ]
201 perimeter = sum(float(numpy.sum(w)) for _, _, w, _ in walls)
202 tau_mean = sum(float(numpy.sum(t * w)) for _, _, w, t in walls) / perimeter
203
204 if args.body_force is not None:
205 cross_area = span[0] * span[1]
206 u_tau = math.sqrt(args.body_force * cross_area / perimeter)
207 source_note = f"sqrt(f*A/P) with f={args.body_force:.8e}"
208 elif args.driven_flow_csv is not None:
209 accel, rows = channel.mean_driving_acceleration(args.driven_flow_csv, float(window["effective_start"]),
210 float(window["effective_end"]))
211 u_tau = math.sqrt(accel * span[0] * span[1] / perimeter)
212 source_note = f"sqrt(<a>*A/P) with <a>={accel:.8e} over {rows} driven_flow.csv rows"
213 elif args.u_tau is not None:
214 u_tau = args.u_tau
215 source_note = "supplied with --u-tau"
216 else:
217 u_tau = math.sqrt(tau_mean)
218 source_note = "sqrt of the perimeter-mean nu*U/d from the first cells (needs a resolved sublayer)"
219 if not u_tau > 0.0:
220 raise SystemExit("computed a non-positive friction velocity; check the inputs.")
221
222 secondary = numpy.hypot(mean[..., comp[0]], mean[..., comp[1]])
223 peak = numpy.unravel_index(numpy.argmax(secondary), secondary.shape)
224 half_width = 0.5 * min(span)
225 header = (f"# u_tau = {u_tau:.10e} ({source_note})\n"
226 f"# Re_tau = u_tau*a/nu = {u_tau * half_width / nu:.6f} with half-width a = {half_width:.10e}\n"
227 f"# U_b = {bulk:.10e}, max secondary speed / U_b = {float(secondary[peak]) / bulk:.6e} "
228 f"at ({centres[0][peak[0]]:.6e}, {centres[1][peak[1]]:.6e})\n"
229 f"# nu = {nu:.10e}, window = {args.window} ({window.get('state')}, "
230 f"{window.get('sample_count')} samples), {folded}\n")
231 prefix = args.output_prefix
232 pair_names = [names[a] + names[b] for a, b in pairs]
233
234 with open(prefix + "_cross_section.csv", "w", encoding="utf-8") as handle:
235 handle.write(header)
236 handle.write(f"{names[comp[0]]},{names[comp[1]]},u_x,u_y,u_z,"
237 + ",".join("cov_" + p for p in pair_names) + "\n")
238 for a in range(n0):
239 for b in range(n1):
240 handle.write(f"{centres[0][a]:.10e},{centres[1][b]:.10e},"
241 + ",".join(f"{value:.10e}" for value in mean[a, b])
242 + "," + ",".join(f"{value:.10e}" for value in covariance[a, b]) + "\n")
243
244 def cov(a_index, b_index, c0, c1):
245 return covariance[a_index, b_index, pairs.index(tuple(sorted((c0, c1))))]
246
247 # Wall bisector: from the first axis's lower wall, at the centre of the second.
248 middle = [n1 // 2] if n1 % 2 else [n1 // 2 - 1, n1 // 2]
249 orientation = 1.0 if edges[0][-1] > edges[0][0] else -1.0
250 with open(prefix + "_wall_bisector.csv", "w", encoding="utf-8") as handle:
251 handle.write(header)
252 handle.write("d,d_plus,U_plus,u_rms_plus,v_rms_plus,w_rms_plus,minus_uv_plus,V_over_Ub\n")
253 for a in range((n0 + 1) // 2):
254 pick = lambda f: sum(f(a, b) for b in middle) / len(middle)
255 d = abs(centres[0][a] - edges[0][0])
256 values = (
257 pick(lambda i, j: U[i, j]) / u_tau,
258 math.sqrt(max(pick(lambda i, j: cov(i, j, stream_c, stream_c)), 0.0)) / u_tau,
259 math.sqrt(max(pick(lambda i, j: cov(i, j, comp[0], comp[0])), 0.0)) / u_tau,
260 math.sqrt(max(pick(lambda i, j: cov(i, j, comp[1], comp[1])), 0.0)) / u_tau,
261 -orientation * pick(lambda i, j: cov(i, j, stream_c, comp[0])) / u_tau ** 2,
262 orientation * pick(lambda i, j: mean[i, j, comp[0]]) / bulk,
263 )
264 handle.write(f"{d:.10e},{d * u_tau / nu:.10e}," + ",".join(f"{v:.10e}" for v in values) + "\n")
265
266 if n0 == n1:
267 # Corner bisector: the diagonal cells from the (min, min) corner. Positive
268 # along-diagonal velocity points away from the corner.
269 sign0 = 1.0 if edges[0][-1] > edges[0][0] else -1.0
270 sign1 = 1.0 if edges[1][-1] > edges[1][0] else -1.0
271 with open(prefix + "_corner_bisector.csv", "w", encoding="utf-8") as handle:
272 handle.write(header)
273 handle.write("d,d_plus,U_plus,along_diagonal_over_Ub,k_plus\n")
274 for a in range((n0 + 1) // 2):
275 d = math.hypot(centres[0][a] - edges[0][0], centres[1][a] - edges[1][0])
276 along = (sign0 * mean[a, a, comp[0]] + sign1 * mean[a, a, comp[1]]) / math.sqrt(2.0)
277 k = 0.5 * sum(cov(a, a, c, c) for c in range(3))
278 handle.write(f"{d:.10e},{d * u_tau / nu:.10e},{U[a, a] / u_tau:.10e},"
279 f"{along / bulk:.10e},{k / u_tau ** 2:.10e}\n")
280
281 with open(prefix + "_wall_shear.csv", "w", encoding="utf-8") as handle:
282 handle.write(header)
283 handle.write("wall,s,tau_w,tau_w_over_mean\n")
284 for name, coordinate, _width, tau in walls:
285 for s_value, t_value in zip(coordinate, tau):
286 handle.write(f"{name},{s_value:.10e},{t_value:.10e},{t_value / tau_mean:.10e}\n")
287
288 print(header, end="")
289 print(f"wrote {prefix}_cross_section.csv, {prefix}_wall_bisector.csv, "
290 + (f"{prefix}_corner_bisector.csv, " if n0 == n1 else "")
291 + f"{prefix}_wall_shear.csv")
292 return 0
